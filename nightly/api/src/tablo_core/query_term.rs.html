1//! The bounded search term the `?q=` transport and the option search share.
2//!
3//! A leaf: `resource` (the list state's transport parse and the live-search
4//! shard) and `schema` (relationship option search) both clamp through here,
5//! and neither depends on the other for it.
6
7/// Longest search term accepted (`?q=` and the shard's `q`): bounded
8/// echoed state. Applied by [`TableState::from_parts`], so the GET path, the
9/// shard, and the public [`TableState::from_live_args`] all clamp alike
10/// and by the relationship option search so a keystroke burst cannot
11/// grow the pattern past the same bound.
12///
13/// [`TableState::from_parts`]: crate::resource::TableState
14/// [`TableState::from_live_args`]: crate::resource::TableState::from_live_args
15pub(crate) const MAX_QUERY_TERM: usize = 128;
16
17/// Clamp a search term to [`MAX_QUERY_TERM`] chars (chars, not bytes, so a
18/// multibyte term truncates on boundaries).
19pub(crate) fn clamp_query_term(term: &str) -> String {
20    term.trim().chars().take(MAX_QUERY_TERM).collect()
21}
