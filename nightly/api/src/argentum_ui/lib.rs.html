76/// Tailwind build helper for the per-app contract.
77///
78/// Call from the app's `build.rs`:
79///
80/// ```ignore
81/// fn main() {
82///     argentum_ui::tailwind_build().unwrap();
83/// }
84/// ```
85///
86/// This scans `styles.css` (which must `@import "tailwindcss"` and contain
87/// `@source` globs for the app's `src/` and for `argentum-ui/src/`) and
88/// writes `$OUT_DIR/tailwind.css` for `tailwind::stylesheet!()`.
89pub fn tailwind_build() -> Result<std::path::PathBuf, topcoat::tailwind::BuildError> {
90    topcoat::tailwind::BuildConfig::new()
91        .input("styles.css")
92        .render()
93}
