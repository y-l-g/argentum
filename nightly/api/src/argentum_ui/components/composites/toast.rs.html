1//! Toast — the shadcn/Sonner surface, owned by Argentum (GH #151).
2//!
3//! The parts mirror shadcn's Sonner composition: a [`toaster`] stack, the
4//! [`toast`] surface, [`toast_icon`], [`toast_content`] with [`toast_title`] /
5//! [`toast_description`], and [`toast_close`]. The classes follow shadcn's
6//! new-york Sonner theming (`bg-background text-foreground border-border
7//! shadow-lg`, description `text-muted-foreground`) over Sonner's own layout
8//! values (356px stack, 16px padding, 13px text, 14px gap, 20px circular
9//! close button, bottom-right offset).
10//!
11//! The enter/exit motion is Sonner's: the server renders `data-mounted="false"`
12//! (slid down, transparent), `notifications.js` flips it to `data-mounted="true"`
13//! to play the transition, and flips `data-removed="true"` before removing the
14//! toast. A `<noscript>` rule keeps toasts visible without JS, where
15//! auto-dismissal and the close button only work by navigation.
16

25/// The fixed stack Sonner calls the toaster: bottom-right, 356px, 14px gap.
26///
27/// The viewport offset is Sonner's: 16px on small screens, 24px from `sm` up.
28/// The stack takes no pointer events so the empty area never blocks the page;
29/// each toast re-enables them.
30const TOASTER: StaticClass = class!(
31    "pointer-events-none fixed right-4 bottom-4 z-50 flex w-[356px] \
32     max-w-[calc(100vw-2rem)] flex-col gap-3.5 sm:right-6 sm:bottom-6",
33);

35/// Sonner's styled toast surface under shadcn's new-york theming.
36///
37/// `data-mounted` / `data-removed` drive the Sonner transition; the base state
38/// is the settled one so a toast without JS is simply visible.
39const TOAST: StaticClass = class!(
40    "pointer-events-auto relative flex w-full translate-y-0 items-center gap-1.5 rounded-lg \
41     border border-border bg-background p-4 text-[13px] text-foreground shadow-lg \
42     [overflow-wrap:anywhere] \
43     transition-[translate,opacity] duration-[400ms] ease-[cubic-bezier(0.25,0.1,0.25,1)] \
44     focus-visible:ring-2 focus-visible:ring-ring/50 focus-visible:outline-none \
45     data-[mounted=false]:translate-y-full data-[mounted=false]:opacity-0 \
46     data-[removed=true]:translate-y-full data-[removed=true]:opacity-0",
47);
48

68/// The toast stack (shadcn/Sonner's toaster).
69///
70/// Renders Sonner's polite live region around the `<ol data-sonner-toaster>`;
71/// the `<noscript>` rule re-reveals toasts whose enter transition never got its
72/// `data-mounted` flip.
73///
74/// Behavior asset: the enter transition, auto-dismissal, and the close button
75/// need `assets/notifications.js` (`crate::NOTIFICATION_JS`, hooks
76/// `data-sonner-toast` / `data-close-button`), emitted by
77/// `Panel::render_document` on every document with shell assets (see
78/// ADR-0014). Without the script the toast keeps `data-mounted="false"` and
79/// stays hidden; with scripting disabled the `<noscript>` rule keeps it visible
80/// until the next navigation.
81///
82/// ```ignore
83/// view! {
84///     toaster(
85///         toast(attrs: attributes! { data-type="success" },
86///             toast_icon(icon(data: icons::CIRCLE_CHECK))
87///             toast_content(
88///                 toast_title("Created")
89///                 toast_description("The record is live.")
90///             )
91///             toast_close()
92///         )
93///     )
94/// }
95/// ```
96#[component]
97pub async fn toaster(
98    #[default] mut attrs: Attributes,
99    #[default] child: Child<'_>,
100) -> Result<impl View> {
101    Ok(view! {
102        <section
103            aria-label="Notifications"
104            tabindex="-1"
105            aria-live="polite"
106            aria-relevant="additions text"
107            aria-atomic="false"
108        >
109            <noscript>
110                <style>
111                    "[data-sonner-toast]{opacity:1 !important;transform:none !important}"
112                </style>
113            </noscript>
114            <ol
115                class=(class!(TOASTER, attrs.remove("class")))
116                data-sonner-toaster=""
117                data-y-position="bottom"
118                data-x-position="right"
119                (attrs)
120            >
121                (child)
122            </ol>
123        </section>
124    })
125}

127/// A toast surface, Sonner's `[data-sonner-toast]`.
128///
129/// Start from `data-mounted="false"` so the enter transition plays when
130/// `notifications.js` mounts it; a toast without JS stays visible through the
131/// toaster's `<noscript>` rule.
132#[component]
133pub async fn toast(
134    #[default] mut attrs: Attributes,
135    #[default] child: Child<'_>,
136) -> Result<impl View> {
137    Ok(view! {
138        <li
139            tabindex="0"
140            class=(class!(TOAST, attrs.remove("class")))
141            data-sonner-toast=""
142            data-styled="true"
143            data-mounted="false"
144            data-visible="true"
145            data-y-position="bottom"
146            data-x-position="right"
147            data-front="true"
148            (attrs)
149        >
150            (child)
151        </li>
152    })
153}
