7const ERROR_STATE: StaticClass = class!(
8    "flex flex-col items-center gap-3 rounded-lg border border-destructive/30 \
9     bg-background px-6 py-12 text-center [&>svg]:size-5 [&>svg]:text-destructive"

11const ERROR_STATE_TITLE: StaticClass = class!("text-sm font-medium text-destructive");
12const ERROR_STATE_DETAIL: StaticClass = class!("text-sm text-muted-foreground");
13const ERROR_STATE_ACTION: StaticClass = class!("text-sm font-medium text-primary hover:underline");

15/// Failed-load rendering for a content region (CONTEXT.md:`ErrorState`).
16///
17/// A destructive-accented block — icon, title, optional muted detail, and an
18/// optional action (typically a retry link) — rendered *inside* the region
19/// that failed, so the surrounding page (shell, header, toolbar) survives.
20/// Deliberately distinct from the zero-rows EmptyState rendering
21/// (`Table::render_empty_cell`): EmptyState says "no data", ErrorState says
22/// "no answer".
23///
24/// The `detail` line must stay generic: never interpolate error internals
25/// (driver messages, SQL, paths) into the page. Log the error at the call
26/// site and keep operators informed there; the markup stays leak-free.
27///
28/// ```ignore
29/// let action = view! { cx => <a href=(list_url)>"Retry"</a> }.boxed();
30/// Ok(view! { cx =>
31///     error_state(
32///         title: "Couldn't load Users",
33///         detail: "Something went wrong while loading the records.",
34///         action: Some(action.into()),
35///     )
36/// }.boxed())
37/// ```
38#[component]
39pub async fn error_state(
40    /// Short heading naming what failed, e.g. `"Couldn't load Users"`.
41    #[into]
42    title: String,
43    /// Optional muted detail line. Keep it generic — no error internals.
44    #[into]
45    #[default]
46    detail: String,
47    /// Optional action under the text (e.g. a retry link). It inherits the
48    /// component's action styling; pass a plain `<a>` (or button) without
49    /// your own color classes.
50    #[default]
51    action: Option<Child<'_>>,
52    /// Extra attributes for the container. Add `role="alert"` here when the
53    /// region swaps in during a visit and the failure should be announced
54    /// (same trade-off the `alert` primitive documents).
55    #[default]
56    mut attrs: Attributes,
57) -> Result<impl View> {
58    Ok(view! {
59        <div class=(class!(ERROR_STATE, attrs.remove("class"))) (attrs)>
60            icon(data: iconify_icon!("lucide:circle-alert"))
61            <p class=(class!(ERROR_STATE_TITLE))>(title)</p>
62            if !detail.is_empty() {
63                <p class=(class!(ERROR_STATE_DETAIL))>(detail)</p>
64            }
65            if let Some(action) = action {
66                <div class=(class!(ERROR_STATE_ACTION))>(action)</div>
67            }
68        </div>
69    })
70}

73mod tests {
74    use topcoat::{
75        context::CxTestBuilder,

79    use super::*;

81    #[tokio::test]
82    async fn error_state_renders_title_detail_and_styled_action() {
83        let cx = CxTestBuilder::new().build();
84        let cx_ref = &cx;
85        let action = view! { cx_ref => <a href="/admin/users">"Retry"</a> }.boxed();
86        let html = view! {
87            cx_ref =>
88            error_state(
89                title: "Couldn't load Users",
90                detail: "Something went wrong while loading the records.",
91                action: Some(action.into())
92            )
93        }
94        .single()
95        .await
96        .unwrap()
97        .render(&cx);
98
99        assert!(
100            html.contains("Couldn't load Users"),
101            "title missing: {html}"
102        );
103        assert!(
104            html.contains("Something went wrong while loading the records."),
105            "detail missing: {html}"
106        );
107        // The caller's action reaches the output intact (GH #216: the classes
108        // that style it are paint; the href is the caller's own markup).
109        assert!(
110            html.contains("Retry") && html.contains("href=\"/admin/users\""),
111            "caller action missing: {html}"
112        );
113        // Icon present; its colour is paint.
114        assert!(html.contains("<svg"), "icon missing: {html}");
115    }
116
117    #[tokio::test]
118    async fn error_state_detail_is_optional_and_attrs_survive() {
119        let cx = CxTestBuilder::new().build();
120        let cx_ref = &cx;
121        let html = view! {
122            cx_ref =>
123            error_state(
124                title: "Couldn't load Users",
125                attrs: attributes! { id="load-error" class="my-class" }
126            )
127        }
128        .single()
129        .await
130        .unwrap()
131        .render(&cx);
132
133        assert!(
134            html.contains("Couldn't load Users"),
135            "title missing: {html}"
136        );
137        // No detail line: the title is the only `<p>` element the component
138        // renders when `detail` is empty (GH #216: the muted colour that used
139        // to stand in for this is paint). Spelled `<p ` / `<p>` so the icon's
140        // `<path>` cannot count as one.
141        assert_eq!(
142            html.matches("<p ").count() + html.matches("<p>").count(),
143            1,
144            "no detail expected: {html}"
145        );
146        assert!(html.contains("id=\"load-error\""), "attrs missing: {html}");
147        assert!(html.contains("my-class"), "caller class missing: {html}");
148    }
