14/// A keyboard key label rendered as `<kbd>`.
15///
16/// Pass the key name as children and use [`kbd_group`] for a shortcut with several
17/// keys. `attrs` are forwarded to the `<kbd>`, with extra classes added to its classes.
18///
19/// ```ignore
20/// view! {
21///     kbd_group(kbd("Ctrl") kbd("K"))
22/// }
23/// ```
24#[component]
25pub async fn kbd(
26    #[default] mut attrs: Attributes,
27    #[default] child: Child<'_>,
28) -> Result<impl View> {
29    Ok(view! { <kbd class=(class!(KBD, attrs.remove("class"))) (attrs)>(child)</kbd> })
30}
31
32/// A row of key labels for one keyboard shortcut. The keys stay on one line.
33#[component]
34pub async fn kbd_group(
35    #[default] mut attrs: Attributes,
36    #[default] child: Child<'_>,
37) -> Result<impl View> {
38    Ok(view! {
39        <span
40            class=(class!(
41                "inline-flex items-center gap-1 whitespace-nowrap",
42                attrs.remove("class"),
43            ))
44            (attrs)
45        >
46            (child)
47        </span>
48    })
49}
