7/// A tooltip component: a hint that shows while its trigger is hovered or
8/// focused.
9///
10/// Child nodes are the trigger and the [`tooltip_content`] holding the hint.
11/// The hint shows on hover and on keyboard focus, both of which are the
12/// browser's own doing, so nothing here needs scripting. It also never moves
13/// out of the way of the viewport's edge, which does: keep hints short, and
14/// give the content a side that has room.
15///
16/// The hint is only a hint. Say what the trigger does in the trigger itself,
17/// through its text or an `aria-label`, since a reader who never hovers will
18/// not meet the tooltip at all.
19///
20/// ```ignore
21/// view! {
22///     tooltip(
23///         button(
24///             size: ButtonSize::Icon,
25///             variant: ButtonVariant::Outline,
26///             icon(data: iconify_icon!("lucide:copy"), label: "Copy link")
27///         )
28///         tooltip_content("Copy link")
29///     )
30/// }
31/// ```
32#[component]
33pub async fn tooltip(
34    #[default] mut attrs: Attributes,
35    #[default] child: Child<'_>,
36) -> Result<impl View> {
37    Ok(view! {
38        <span
39            class=(class!("group relative inline-flex", attrs.remove("class")))
40            (attrs)
41        >
42            (child)
43        </span>
44    })
45}

47/// The classes for the [`tooltip_content`] bubble.
48///
49/// The bubble is painted in the foreground color on the background one, the
50/// reverse of the page, which is what sets a hint apart from the surface it
51/// covers. It sits above the trigger, centered on it, and takes no pointer
52/// events, so hovering along the way to it never gets in the way of what is
53/// underneath.
54///
55/// It fades in on hover and on focus within the trigger, and the visibility
56/// that keeps it out of the way in between is named in the transition with
57/// `allow-discrete`: it has no in-between values, so without that it would
58/// snap and the fade would play against a hint that is already gone. Naming
59/// both properties one by one is what makes that work; `all` does not carry
60/// the visibility along.
61const BUBBLE: StaticClass = class!(
62    "pointer-events-none invisible absolute bottom-full left-1/2 z-50 mb-2 \
63     -translate-x-1/2 rounded-md bg-foreground px-2.5 py-1 text-xs font-medium text-background \
64     opacity-0 shadow-sm whitespace-nowrap \
65     [transition:opacity_150ms_ease-out,visibility_150ms_allow-discrete] \
66     group-hover:visible group-hover:opacity-100 \
67     group-focus-within:visible group-focus-within:opacity-100",
68);
69
70/// The hint a [`tooltip`] shows, in a bubble above its trigger.
71#[component]
72pub async fn tooltip_content(
73    #[default] mut attrs: Attributes,
74    #[default] child: Child<'_>,
75) -> Result<impl View> {
76    Ok(view! {
77        <span role="tooltip" class=(class!(BUBBLE, attrs.remove("class"))) (attrs)>
78            (child)
79        </span>
80    })
