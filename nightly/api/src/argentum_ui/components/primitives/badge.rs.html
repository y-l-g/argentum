7/// The visual style of a [`badge`].
8///
9/// [`Default`] is `BadgeVariant::Primary`, used when no variant is given.
10#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
11#[allow(dead_code)]
12pub enum BadgeVariant {
13    /// The primary-filled badge for highlighted statuses.
14    #[default]
15    Primary,
16    /// A muted, tinted fill for neutral statuses.
17    Secondary,
18    /// A hairline-bordered badge on the page background.
19    Outline,
20    /// A destructive-filled badge for errors and warnings.
21    Destructive,
22}
23
24impl BadgeVariant {
25    /// The Tailwind classes for this variant.
26    ///
27    /// Each variant sets its own border color rather than inheriting a
28    /// transparent one from [`BASE`]: with two border-color classes on the
29    /// same element, stylesheet order (not class order) would decide the
30    /// winner.
31    fn classes(self) -> StaticClass {
32        match self {
33            Self::Primary => class!("border-transparent bg-primary text-primary-foreground"),
34            Self::Secondary => class!("border-transparent bg-foreground/5 text-foreground"),
35            Self::Outline => class!("border-border text-foreground"),
36            Self::Destructive => {
37                class!("border-transparent bg-destructive text-destructive-foreground")
38            }
39        }
40    }
41}
42
43/// The classes shared by every badge, regardless of variant.
44///
45/// Every badge carries a border (colored per variant) so that the `Outline`
46/// variant, which only recolors it, does not change the badge's dimensions.
47const BASE: StaticClass = class!(
48    "inline-flex w-fit shrink-0 items-center justify-center gap-1 rounded-md \
49     border px-2 py-0.5 text-xs font-medium whitespace-nowrap",
50);
51
52/// Builds the full class list for a badge of the given `variant`.
53///
54/// Use it to give badge styling to another element, such as a link:
55///
56/// ```ignore
57/// view! {
58///     <a href="/releases/v2" class=(badge_variants(BadgeVariant::Outline))>"v2.0"</a>
59/// }
60/// ```
61#[must_use]
62pub fn badge_variants(variant: BadgeVariant) -> Class<(StaticClass, StaticClass)> {
63    class!(BASE, variant.classes())
64}
65
66/// A badge component: a small inline pill for statuses, counts, and tags.
67///
68/// The `variant` parameter selects the styling, defaulting to `Primary`. The
69/// `attrs` (such as `class` or `title`) are forwarded to the underlying
70/// `<span>`; a `class` among them is appended to the computed classes. Child
71/// nodes become the badge's content.
72///
73/// ```ignore
74/// view! {
75///     badge(variant: BadgeVariant::Destructive, "Failed")
76/// }
77/// ```
78///
79/// To style another element like a badge, use [`badge_variants`] directly.
80#[component]
81pub async fn badge(
82    #[default] variant: BadgeVariant,
83    #[default] mut attrs: Attributes,
84    #[default] child: Child<'_>,
85) -> Result<impl View> {
86    Ok(view! {
87        <span class=(class!(BASE, variant.classes(), attrs.remove("class"))) (attrs)>
88            (child)
89        </span>
90    })
91}
