8/// The classes for the [`sheet`] overlay: a layer covering the viewport,
9/// veiling the page behind it and holding the panel against one edge.
10///
11/// It is the dialog's overlay without the padding and the centering, since a
12/// sheet lies flush against the side it comes from; which side that is, is
13/// the panel's business. As with the dialog, only the open state sets a
14/// display, so a closed sheet stays hidden by the browser's own rule.
15const OVERLAY: StaticClass = class!(
16    "fixed inset-0 z-50 size-full max-h-none max-w-none overflow-hidden \
17     bg-background/80 text-foreground backdrop-blur-sm open:flex",
18);
19
20/// The classes fading the veil in and out.
21///
22/// A sheet goes from not being rendered at all to covering the page, which
23/// takes two things beyond the fade itself. `display` is named in the
24/// transition with `allow-discrete`, which holds the layer on the page for as
25/// long as the fade out lasts instead of taking it away at once. And
26/// `@starting-style` gives the layer the value to come from: an element that
27/// was not rendered a moment ago has no previous style to leave behind, so
28/// without it the fade in has nothing to run from.
29const FADE: StaticClass = class!(
30    "opacity-0 open:opacity-100 starting:open:opacity-0 \
31     [transition:opacity_200ms_ease-out,display_200ms_allow-discrete]",
32);

34/// A sheet component: a panel that comes in from an edge of the page.
35///
36/// It is a [`dialog`](super::dialog::dialog) laid against a side rather than
37/// centered, for the things a dialog is too small for: filters, a form, a
38/// detail view of the row being read. Everything else is the dialog's: its
39/// open state is the `open` parameter, which accepts a boolean for a fixed
40/// state or a runtime expression to open and close it in the browser.
41///
42/// Child nodes are the sheet's content, normally a single [`sheet_content`]
43/// panel, whose `side` decides which edge it lies against. Build the inside
44/// out of [`dialog_header`](super::dialog::dialog_header),
45/// [`dialog_title`](super::dialog::dialog_title) and the rest. The `attrs`
46/// are forwarded to the `<dialog>`; a `class` among them is appended to the
47/// computed classes.
48///
49/// ```ignore
50/// view! {
51///     sheet(
52///         open: filtering,
53///         sheet_content(
54///             dialog_header(
55///                 dialog_title("Filters")
56///                 dialog_description("Narrow the deployments below.")
57///             )
58///             (fields)
59///         )
60///     )
61/// }
62/// ```
63#[component]
64pub async fn sheet(
65    /// Whether the sheet shows.
66    #[into]
67    open: Expr<bool>,
68    /// Extra attributes for the `<dialog>` element.
69    #[default]
70    mut attrs: Attributes,
71    /// The sheet's content.
72    #[default]
73    child: Child<'_>,
74) -> Result<impl View> {
75    Ok(view! {
76        <dialog
77            :open=(open)
78            class=(class!(OVERLAY, FADE, attrs.remove("class")))
79            (attrs)
80        >
81            (child)
82        </dialog>
83    })
84}

86/// The edge a [`sheet_content`] lies against.
87///
88/// [`Default`] is `SheetSide::Right`, used when no side is given.
89#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
90#[allow(dead_code)]
91pub enum SheetSide {
92    /// Along the left edge, full height.
93    Left,
94    /// Along the right edge, full height. This is where a sheet usually comes
95    /// from, since it leaves the start of every line on the page in view.
96    #[default]
97    Right,
98    /// Across the top edge, full width.
99    Top,
100    /// Across the bottom edge, full width.
101    Bottom,
102}
103
104impl SheetSide {
105    /// The Tailwind classes for this side.
106    ///
107    /// The panel takes the whole of one axis and is pushed against its edge
108    /// by an automatic margin on the other, which is what leaves the veil
109    /// showing on the remaining side. It is bordered only along the edge that
110    /// faces the page, the one edge of it that is not against the viewport.
111    fn classes(self) -> StaticClass {
112        match self {
113            Self::Left => class!("mr-auto h-full w-full max-w-sm border-r"),
114            Self::Right => class!("ml-auto h-full w-full max-w-sm border-l"),
115            Self::Top => class!("mb-auto max-h-full w-full border-b"),
116            Self::Bottom => class!("mt-auto max-h-full w-full border-t"),
117        }
118    }
119
120    /// The Tailwind classes sliding the panel in from this side.
121    ///
122    /// The panel rests off the edge it comes from and is brought in while the
123    /// sheet around it is open, which is what makes it slide both ways: in as
124    /// the sheet opens, and back out as it closes, for as long as the veil's
125    /// fade holds the sheet on the page. `@starting-style` gives it the place
126    /// to come from the first time, since a panel that was not rendered a
127    /// moment ago has no previous position to leave.
128    fn motion(self) -> StaticClass {
129        match self {
130            Self::Left => class!(
131                "-translate-x-full in-[[open]]:translate-x-0 \
132                 starting:in-[[open]]:-translate-x-full",
133            ),
134            Self::Right => class!(
135                "translate-x-full in-[[open]]:translate-x-0 \
136                 starting:in-[[open]]:translate-x-full",
137            ),
138            Self::Top => class!(
139                "-translate-y-full in-[[open]]:translate-y-0 \
140                 starting:in-[[open]]:-translate-y-full",
141            ),
142            Self::Bottom => class!(
143                "translate-y-full in-[[open]]:translate-y-0 \
144                 starting:in-[[open]]:translate-y-full",
145            ),
146        }
147    }
148}

150/// The classes shared by every sheet panel, regardless of side.
151///
152/// The panel is the dialog's, squared off and stretched to its edge: it sets
153/// its own background and text color, stacks its sections in a column, and
154/// scrolls within itself once there is more in it than the edge it lies
155/// against is long.
156const CONTENT: StaticClass = class!(
157    "flex flex-col gap-4 overflow-y-auto border-border bg-card p-6 \
158     text-card-foreground shadow-sm [transition:translate_200ms_ease-out]",
159);

161/// The panel of a [`sheet`], holding its sections.
162///
163/// The `side` parameter decides which edge it lies against, defaulting to
164/// `Right`. The `attrs` (such as `class`) are forwarded to the underlying
165/// `<div>`; a `class` among them is appended to the computed classes, so a
166/// wider or narrower sheet is a `max-w-*` class among them.
167#[component]
168pub async fn sheet_content(
169    /// The edge the panel lies against.
170    #[default]
171    side: SheetSide,
172    /// Extra attributes for the `<div>` element.
173    #[default]
174    mut attrs: Attributes,
175    /// The panel's sections.
176    #[default]
177    child: Child<'_>,
178) -> Result<impl View> {
179    Ok(view! {
180        <div
181            class=(class!(CONTENT, side.classes(), side.motion(), attrs.remove("class")))
182            (attrs)
183        >
184            (child)
185        </div>
186    })
