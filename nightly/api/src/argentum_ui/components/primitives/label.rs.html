7/// The classes for the [`label`] element.
8///
9/// The label lays out its content in a centered row, so an inline icon or a
10/// wrapped control lines up with the text. It dims and stops receiving
11/// pointer events when its control is disabled: a wrapped control is matched
12/// with `has-[:disabled]`, a preceding sibling control marked `peer` with
13/// `peer-disabled`, and a control inside a preceding `peer` wrapper with
14/// `peer-has-[:disabled]`. A control immediately after the label is matched
15/// with `has-[+:disabled]`.
16const LABEL: StaticClass = class!(
17    "flex items-center gap-2 text-sm leading-none font-medium select-none \
18     peer-disabled:pointer-events-none peer-disabled:opacity-50 \
19     peer-has-[:disabled]:pointer-events-none peer-has-[:disabled]:opacity-50 \
20     has-[+:disabled]:pointer-events-none has-[+:disabled]:opacity-50 \
21     has-[:disabled]:pointer-events-none has-[:disabled]:opacity-50",
22);

24/// A caption for a form control, rendered as a `<label>`.
25///
26/// Associate it with a control either by wrapping the control or by pointing
27/// a `for` attribute at the control's `id`. The `attrs` (such as `class` or
28/// `for`) are forwarded to the underlying `<label>`; a `class` among them is
29/// appended to the computed classes. Child nodes become the label's content.
30///
31/// ```ignore
32/// view! {
33///     <div class="flex flex-col gap-2">
34///         label(attrs: attributes! { for="email" }, "Email")
35///         input(attrs: attributes! { id="email" type="email" })
36///     </div>
37/// }
38/// ```
39#[component]
40pub async fn label(
41    #[default] mut attrs: Attributes,
42    #[default] child: Child<'_>,
43) -> Result<impl View> {
44    Ok(view! {
45        <label class=(class!(LABEL, attrs.remove("class"))) (attrs)>(child)</label>
46    })
47}
