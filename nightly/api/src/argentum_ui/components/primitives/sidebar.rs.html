126/// A desktop panel that becomes a sheet over the page below `md` (48rem).
127///
128/// The `--sidebar-*` theme tokens control its colors independently of cards,
129/// sheets and page content. Nested controls inherit the sidebar palette.
130///
131/// `open` controls desktop expansion. `mobile_open` independently controls
132/// the mobile sheet, so it can start closed while desktop starts expanded.
133/// Pass runtime expressions reading the caller's signals to make both live.
134/// Header, content and footer children are rendered once and shared between
135/// viewport sizes, preserving their inputs and signal identities.
136///
137/// `attrs` apply to the outer wrapper. `sheet_attrs` apply to the sheet and
138/// accept a label, an ID, and event handlers for Escape or backdrop dismissal.
139/// Include a close button in the mobile header. Like [`sheet`], this is a
140/// visual overlay; focus trapping requires additional application scripting.
141#[component]
142pub async fn sidebar(
143    #[into]
144    #[default(true.into())]
145    open: Expr<bool>,
146    #[into]
147    #[default(false.into())]
148    mobile_open: Expr<bool>,
149    #[default] side: SidebarSide,
150    #[default] variant: SidebarVariant,
151    #[default] collapsible: SidebarCollapsible,
152    #[default] mut attrs: Attributes,
153    #[default] mut sheet_attrs: Attributes,
154    #[default] child: Child<'_>,
155) -> Result<impl View> {
156    let open = if collapsible == SidebarCollapsible::None {
157        true.into()
158    } else {
159        open
160    };
161    let collapse = collapsible.name();
162
163    Ok(view! {
164        <aside
165            data-sidebar="sidebar"
166            data-side=(side.name())
167            data-variant=(variant.name())
168            :data-state=$(if open { "expanded" } else { "collapsed" })
169            :data-collapsible=$(if open { "" } else { collapse })
170            class=(class!(
171                "group/sidebar relative w-0 shrink-0 text-sidebar-foreground md:sticky md:top-0 md:h-svh md:w-(--sidebar-width) md:self-start md:transition-[width] md:duration-200 md:data-[collapsible=offcanvas]:w-0 motion-reduce:transition-none",
172                if variant == SidebarVariant::Floating {
173                    "md:p-2 md:data-[collapsible=icon]:w-[calc(var(--sidebar-width-icon)+1rem+2px)] md:data-[collapsible=offcanvas]:px-0"
174                } else {
175                    "md:data-[collapsible=icon]:w-(--sidebar-width-icon)"
176                },
177                "md:py-2" if variant == SidebarVariant::Inset,
178                attrs.remove("class"),
179            ))
180            (attrs)
181        >
182            sheet(
183                open: mobile_open,
184                attrs: attributes! {
185                    role="navigation"
186                    aria-label="Sidebar"
187                    class=(class!(
188                        "md:relative md:inset-auto md:flex md:overflow-visible md:bg-transparent md:opacity-100 md:backdrop-blur-none md:starting:open:opacity-100 md:transition-none md:group-data-[collapsible=offcanvas]/sidebar:invisible motion-reduce:transition-none",
189                        sheet_attrs.remove("class"),
190                    ))
191                    (sheet_attrs)
192                },
193                sheet_content(
194                    side: side.sheet(),
195                    attrs: attributes! {
196                        data-sidebar="panel"
197                        class=(class!(
198                            PANEL_THEME,
199                            "relative [&]:w-(--sidebar-width-mobile) [&]:max-w-[calc(100vw-3rem)] [&]:gap-0 [&]:overflow-hidden [&]:p-0 md:[&]:w-full md:[&]:max-w-none md:[&]:translate-x-0 md:[&]:transition-none motion-reduce:transition-none",
200                            match variant {
201                                SidebarVariant::Sidebar => "md:shadow-none",
202                                SidebarVariant::Floating => "md:rounded-xl md:border md:shadow-sm",
203                                SidebarVariant::Inset => "md:border-0 md:bg-transparent md:shadow-none",
204                            },
205                        ))
206                    },
207                    (child)
208                )
209            )
210        </aside>
211    })
212}

542/// A menu action, or a navigation link when `href` is supplied.
543///
544/// Put an icon before a `<span>` containing the label. `tooltip` supplies a
545/// native title hint for the icon-only state. `active` accepts a boolean or
546/// runtime expression. Event handlers and other attributes are forwarded.
547#[component]
548pub async fn sidebar_menu_button(
549    #[default] variant: SidebarMenuButtonVariant,
550    #[default] size: SidebarMenuButtonSize,
551    #[into]
552    #[default(false.into())]
553    active: Expr<bool>,
554    #[default] href: Option<&str>,
555    #[default] tooltip: Option<&str>,
556    #[default] mut attrs: Attributes,
557    #[default] child: Child<'_>,
558) -> Result<impl View> {
559    let attrs = attributes! {
560        data-sidebar="menu-button"
561        :data-active=$(if active { "true" } else { "false" })
562        :aria-current=$(active.then_some("page"))
563        title=(tooltip)
564        class=(class!(
565            sidebar_menu_button_variants(variant, size),
566            attrs.remove("class"),
567        ))
568        (attrs)
569    };
570
571    Ok(view! {
572        if let Some(href) = href {
573            <a href=(href) (attrs)>(child)</a>
574        } else {
575            <button type="button" (attrs)>(child)</button>
576        }
577    })
578}
