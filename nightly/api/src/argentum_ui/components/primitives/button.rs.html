7/// The visual style of a [`button`].
8///
9/// [`Default`] is `ButtonVariant::Primary`, used when no variant is given.
10#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
11#[allow(dead_code)]
12pub enum ButtonVariant {
13    /// The primary-filled button for the main action.
14    #[default]
15    Primary,
16    /// A muted, tinted fill for secondary actions.
17    Secondary,
18    /// A hairline-bordered button on the page background.
19    Outline,
20    /// No fill until hovered, for toolbars and inline actions.
21    Ghost,
22    /// A destructive-filled button for actions such as deleting data.
23    Destructive,
24}
25
26impl ButtonVariant {
27    /// The Tailwind classes for this variant.
28    ///
29    /// Hover and press states apply the fill or foreground color at reduced
30    /// opacity, so they hold up in both color schemes without `dark:`
31    /// overrides. Variants with a resting fill cast the theme's control
32    /// shadow. Outline and ghost buttons have no shadow.
33    ///
34    /// Each variant sets its own border color rather than inheriting a
35    /// transparent one from [`BASE`]: with two border-color classes on the
36    /// same element, stylesheet order (not class order) would decide the
37    /// winner.
38    fn classes(self) -> StaticClass {
39        match self {
40            Self::Primary => class!(
41                "border-transparent bg-primary text-primary-foreground shadow-xs \
42                 hover:bg-primary/90 active:bg-primary/80",
43            ),
44            Self::Secondary => class!(
45                "border-transparent bg-foreground/5 text-foreground shadow-xs \
46                 hover:bg-foreground/10 active:bg-foreground/15",
47            ),
48            Self::Outline => class!(
49                "border-border text-foreground hover:bg-foreground/5 \
50                 active:bg-foreground/10",
51            ),
52            Self::Ghost => class!(
53                "border-transparent text-foreground hover:bg-foreground/5 active:bg-foreground/10",
54            ),
55            Self::Destructive => class!(
56                "border-transparent bg-destructive text-destructive-foreground shadow-xs \
57                 hover:bg-destructive/90 active:bg-destructive/80",
58            ),
59        }
60    }
61}

63/// The size of a [`button`].
64///
65/// [`Default`] is `ButtonSize::Md`, used when no size is given.
66#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
67#[allow(dead_code)]
68pub enum ButtonSize {
69    /// A compact button.
70    Sm,
71    /// The standard button size.
72    #[default]
73    Md,
74    /// A prominent button.
75    Lg,
76    /// A square button sized for a single icon.
77    Icon,
78}
79
80impl ButtonSize {
81    /// The Tailwind classes for this size.
82    ///
83    /// Sizes change the control's dimensions while keeping the text size
84    /// consistent.
85    fn classes(self) -> StaticClass {
86        match self {
87            Self::Sm => class!("h-8 gap-1.5 rounded-md px-3"),
88            Self::Md => class!("h-9 gap-2 rounded-lg px-4"),
89            Self::Lg => class!("h-10 gap-2 rounded-lg px-5"),
90            Self::Icon => class!("size-9 rounded-lg"),
91        }
92    }
93}

95/// The classes shared by every button, regardless of variant or size.
96///
97/// Every button carries a border (colored per variant) so that the `Outline`
98/// variant, which only recolors it, does not change the button's dimensions.
99const BASE: StaticClass = class!(
100    "inline-flex shrink-0 items-center justify-center border \
101     text-sm font-medium whitespace-nowrap transition-colors outline-none select-none \
102     focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 \
103     focus-visible:ring-offset-background disabled:pointer-events-none disabled:opacity-50",
104);
105
106/// Builds the full class list for a button of the given `variant` and `size`.
107///
108/// Use it to give button styling to an element that is not a `<button>`, such
109/// as a link styled as a button:
110///
111/// ```ignore
112/// view! {
113///     <a href="/login" class=(button_variants(ButtonVariant::Outline, ButtonSize::Md))>
114///         "Sign in"
115///     </a>
116/// }
117/// ```
118#[must_use]
119pub fn button_variants(
120    variant: ButtonVariant,
121    size: ButtonSize,
122) -> Class<(StaticClass, StaticClass, StaticClass)> {
123    class!(BASE, variant.classes(), size.classes())
124}
125
126/// A button component.
127///
128/// The `variant` and `size` parameters select the styling, defaulting to
129/// `Primary` and `Md`. The `attrs` (such as `class`, `type`, `disabled`, or
130/// event handlers) are forwarded to the underlying `<button>`; a `class` among
131/// them is appended to the computed classes. Child nodes become the button's
132/// content.
133///
134/// ```ignore
135/// view! {
136///     button(
137///         variant: ButtonVariant::Destructive,
138///         attrs: attributes! { type="submit" },
139///         "Delete"
140///     )
141/// }
142/// ```
143///
144/// To style a non-`<button>` element like a button, use [`button_variants`]
145/// directly.
146#[component]
147pub async fn button(
148    #[default] variant: ButtonVariant,
149    #[default] size: ButtonSize,
150    #[default] mut attrs: Attributes,
151    #[default] child: Child<'_>,
152) -> Result<impl View> {
153    Ok(view! {
154        <button
155            class=(class!(
156                BASE,
157                variant.classes(),
158                size.classes(),
159                attrs.remove("class"),
160            ))
161            (attrs)
162        >
163            (child)
164        </button>
165    })
166}
