8/// The classes for the native `<input type="checkbox">` inside the
9/// [`checkbox`] component.
10///
11/// The native glyph is suppressed with `appearance-none` so the component can
12/// draw its own checkmark, which keeps the control looking the same across
13/// browsers. The unchecked box matches the input control's border;
14/// checking it fills the box with the primary color.
15const CHECKBOX: StaticClass = class!(
16    "peer size-4 shrink-0 appearance-none rounded-[4px] border border-border \
17     bg-background transition-colors outline-none \
18     checked:border-primary checked:bg-primary \
19     focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 \
20     focus-visible:ring-offset-background disabled:pointer-events-none",
21);
22
23/// A checkbox component: a themed native `<input type="checkbox">`.
24///
25/// The `attrs` (such as `name`, `checked`, `disabled`, or event handlers) are
26/// forwarded to the `<input>`; a `class` among them is appended to the
27/// wrapping element's classes. Set the checked state with a plain `checked`
28/// attribute. The indeterminate state is not styled: it is only reachable
29/// through the DOM property, so setting it takes a script to begin with.
30///
31/// ```ignore
32/// view! {
33///     <div class="flex items-center gap-2">
34///         checkbox(attrs: attributes! { id="terms" name="terms" checked="" })
35///         label(attrs: attributes! { for="terms" }, "Accept terms")
36///     </div>
37/// }
38/// ```
39#[component]
40pub async fn checkbox(#[default] mut attrs: Attributes) -> Result<impl View> {
41    // The checkmark cannot be drawn by the `<input>` itself, which renders no
42    // children or pseudo-elements: it is a sibling icon overlaid on the
43    // control, revealed by the input's `peer` state while checked.
44    Ok(view! {
45        <span
46            class=(class!(
47                "peer relative inline-flex shrink-0 has-[:disabled]:opacity-50",
48                attrs.remove("class"),
49            ))
50        >
51            <input type="checkbox" class=(CHECKBOX) (attrs)>
52            icon(
53                data: iconify_icon!("lucide:check"),
54                attrs: attributes! {
55                    class="pointer-events-none absolute inset-0 m-auto size-3.5 \
56                        text-primary-foreground opacity-0 peer-checked:opacity-100"
57                }
58            )
59        </span>
60    })
61}
