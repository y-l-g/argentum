9/// The classes for the native `<select>` inside the [`select`] component.
10///
11/// Sized to match the input control. The native dropdown arrow is suppressed
12/// so the component can draw its own chevron, which keeps the control looking
13/// the same across browsers; the extra right padding reserves the chevron's
14/// space.

16    "h-9 w-full appearance-none items-center rounded-lg border border-border \

18     focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 \
19     aria-invalid:border-destructive aria-invalid:focus-visible:ring-destructive \
20     focus-visible:ring-offset-background disabled:pointer-events-none",

23/// The classes restyling the drop-down picker, for browsers that support
24/// customizable selects (`appearance: base-select`, set on the `<select>` by
25/// the component's wrapper).
26///
27/// The panel and its option rows take after the dropdown menu's content and
28/// items: the same raised surface, the same ghost-tinted hover and focus
29/// states, and the checked option marked by a checkmark on the row's right
30/// edge: the [`CHECKMARK`] icon, masked over the theme's muted foreground
31/// (see [`checkmark_style`]). The browser's own picker icon is hidden in
32/// favor of the component's chevron. On browsers without support every rule
33/// here is inert and the operating system's picker shows instead.
34const PICKER: StaticClass = class!(
35    "[&::picker(select)]:[appearance:base-select] \
36     [&::picker(select)]:mt-1 [&::picker(select)]:rounded-lg \

38     [&::picker(select)]:bg-popover [&::picker(select)]:p-1 \
39     [&::picker(select)]:text-popover-foreground [&::picker(select)]:shadow-sm \
40     [&::picker-icon]:hidden \
41     [&_optgroup>legend]:px-2 [&_optgroup>legend]:py-1.5 \
42     [&_optgroup>legend]:text-xs [&_optgroup>legend]:font-medium \
43     [&_optgroup>legend]:text-muted-foreground [&_optgroup>legend]:cursor-default \
44     [&_optgroup>legend]:select-none \
45     [&_option]:flex [&_option]:items-center [&_option]:gap-2 [&_option]:rounded-md \

56/// The icon marking the picker's checked option.
57const CHECKMARK: IconData = iconify_icon!("lucide:check");
58
59/// The inline style for the [`select`] wrapper, carrying [`CHECKMARK`] as a
60/// data URI in the `--select-checkmark` custom property. The indirection
61/// exists because the `::checkmark` pseudo-element can only take the icon
62/// through a stylesheet, as a mask image, while the icon's markup is only
63/// available here.
64fn checkmark_style(cx: &Cx) -> String {
65    let svg = format!(
66        r#"<svg xmlns="http://www.w3.org/2000/svg" viewBox="{}">{}</svg>"#,
67        CHECKMARK.view_box(),
68        CHECKMARK.into_body().render(cx),
69    );
70    let mut style = String::from(r#"--select-checkmark: url("data:image/svg+xml,"#);
71    // Percent-encode the characters that cannot appear in a double-quoted
72    // CSS url().
73    for char in svg.chars() {
74        match char {
75            '%' => style.push_str("%25"),
76            '"' => style.push_str("%22"),
77            '#' => style.push_str("%23"),
78            _ => style.push(char),
79        }
80    }
81    style.push_str(r#"")"#);
82    style
83}

85/// A select component: a themed native `<select>`.
86///
87/// Child nodes become the `<select>`'s content, typically `<option>` and
88/// `<optgroup>` elements. The `attrs` (such as `name`, `disabled`, or event
89/// handlers) are forwarded to the `<select>`; a `class` among them is appended
90/// to the wrapping element's classes, so width utilities size the whole
91/// control. Like the input, it fills its container by default.
92/// Set `aria-invalid="true"` to use the error border and focus ring.
93/// For a styled group heading, add a `<legend>` as the first child of the
94/// `<optgroup>`. Keep its `label` attribute for native picker fallbacks.
95///
96/// On browsers with customizable select support the drop-down picker is
97/// restyled to match the dropdown menu component, and the chevron flips while
98/// it is open; other browsers keep the operating system's picker. The control
99/// itself looks the same everywhere.
100///
101/// ```ignore
102/// view! {
103///     select(
104///         attrs: attributes! { name="region" },
105///         <option>"eu-central-1"</option>
106///         <option>"us-east-1"</option>
107///     )
108/// }
109/// ```
110#[component]
111pub async fn select(
112    cx: &Cx,
113    #[default] mut attrs: Attributes,
114    #[default] child: Child<'_>,
115) -> Result<impl View> {
116    // `appearance: base-select` opts into the customizable picker. It is set
117    // from the wrapper because the descendant selector outranks the
118    // `appearance-none` fallback in specificity, making the outcome
119    // independent of stylesheet order; browsers without support drop the
120    // invalid declaration and keep the fallback.
121    Ok(view! {
122        <span
123            class=(class!(
124                "relative block has-[:disabled]:opacity-50 \
125                 [&>select]:[appearance:base-select] \
126                 [&:has(select:open)>svg]:rotate-180",
127                attrs.remove("class"),
128            ))
129            style=(checkmark_style(cx))
130        >
131            <select class=(class!(SELECT, PICKER)) (attrs)>(child)</select>
132            icon(
133                data: iconify_icon!("lucide:chevron-down"),
134                attrs: attributes! {
135                    class="pointer-events-none absolute top-1/2 right-3 size-4 \
136                        -translate-y-1/2 text-muted-foreground transition-transform"
137                }
138            )
139        </span>
140    })
141}
