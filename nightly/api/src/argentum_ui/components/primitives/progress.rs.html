7/// The classes for the [`progress`] bar.
8///
9/// The element is a native `<progress>`, restyled through its
10/// vendor-prefixed pseudo-elements so it looks the same across browsers: the
11/// track is the element itself (the `::-webkit-progress-bar` layer is
12/// cleared to let it show through), and the filled portion is rounded and
13/// painted with the primary color. Browsers styled through the
14/// `::-webkit-*` pseudo-elements only expose the filled portion while a
15/// value is set, so the indeterminate state renders as a static empty track
16/// there; Firefox keeps its animated bar.
17const PROGRESS: StaticClass = class!(
18    "h-2 w-full appearance-none overflow-hidden rounded-full \
19     bg-foreground/10 [&::-webkit-progress-bar]:bg-transparent \
20     [&::-webkit-progress-value]:rounded-full [&::-webkit-progress-value]:bg-primary \
21     [&::-webkit-progress-value]:transition-all \
22     [&::-moz-progress-bar]:rounded-full [&::-moz-progress-bar]:bg-primary",
23);

25/// A progress component: a themed native `<progress>` bar.
26///
27/// `value` is the completed amount, out of `max` (defaulting to 100, so a
28/// plain value reads as a percentage). Omitting `value` renders the
29/// indeterminate state, for work whose extent is unknown. The `attrs` (such
30/// as `class` or `aria-label`) are forwarded to the `<progress>`; a `class`
31/// among them is appended to the computed classes. The bar fills its
32/// container, so size it through the container or with a width class.
33///
34/// ```ignore
35/// view! {
36///     progress(value: 62.0)
37/// }
38/// ```
39#[component]
40pub async fn progress(
41    /// The completed amount, out of `max`.
42    #[into]
43    #[default]
44    value: Option<f32>,
45    /// The amount that counts as complete.
46    #[default(100.0)]
47    max: f32,
48    /// Extra attributes for the `<progress>` element.
49    #[default]
50    mut attrs: Attributes,
51) -> Result<impl View> {
52    Ok(view! {
53        <progress
54            class=(class!(PROGRESS, attrs.remove("class")))
55            value=(value)
56            max=(max)
57            (attrs)
58        ></progress>
59    })
60}
