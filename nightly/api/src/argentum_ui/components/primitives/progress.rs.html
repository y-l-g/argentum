7/// Classes for the native progress track and fill. Indeterminate animation depends on
8/// the browser and may appear as an empty track.
9const PROGRESS: StaticClass = class!(
10    "h-2 w-full appearance-none overflow-hidden rounded-full \
11     bg-foreground/10 [&::-webkit-progress-bar]:bg-transparent \
12     [&::-webkit-progress-value]:rounded-full [&::-webkit-progress-value]:bg-primary \
13     [&::-webkit-progress-value]:transition-all \
14     [&::-moz-progress-bar]:rounded-full [&::-moz-progress-bar]:bg-primary",

17/// A native progress bar.
18///
19/// `value` is the completed amount out of `max`, which defaults to 100. Omit `value`
20/// when the total work is unknown. The indeterminate appearance depends on the browser.
21///
22/// `attrs` are forwarded to the `<progress>`, with extra classes added to its classes.
23/// Give it an accessible label through `aria-label` or an associated label element. The
24/// bar fills its container by default.
25///
26/// ```ignore
27/// view! {
28///     progress(value: 62.0)
29/// }
30/// ```
31#[component]
32pub async fn progress(
33    /// The completed amount, out of `max`.
34    #[into]
35    #[default]
36    value: Option<f32>,
37    /// The amount that counts as complete.
38    #[default(100.0)]
39    max: f32,
40    /// Extra attributes for the `<progress>` element.
41    #[default]
42    mut attrs: Attributes,
43) -> Result<impl View> {
44    Ok(view! {
45        <progress
46            class=(class!(PROGRESS, attrs.remove("class")))
47            value=(value)
48            max=(max)
49            (attrs)
50        ></progress>
51    })
52}
