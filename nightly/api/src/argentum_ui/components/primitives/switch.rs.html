7/// Classes for the checkbox input that forms the switch track.
8const SWITCH: StaticClass = class!(
9    "peer h-4.5 w-8 shrink-0 appearance-none rounded-full \
10     bg-foreground/20 shadow-xs transition-colors outline-none checked:bg-primary \
11     focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 \
12     focus-visible:ring-offset-background disabled:pointer-events-none",
13);

15/// Classes that position the thumb at either end of the track according to the checked

22/// An on/off control for a setting.
23///
24/// Uses a native checkbox with the `switch` role. Pass `checked` in `attrs` for the
25/// initial state. Classes apply to the wrapper, while other attributes and event
26/// handlers go on the `<input>`.
27///
28/// ```ignore
29/// view! {
30///     <div class="flex items-center gap-2">
31///         switch(attrs: attributes! { id="airplane-mode" checked="" })
32///         label(attrs: attributes! { for="airplane-mode" }, "Airplane mode")
33///     </div>
34/// }
35/// ```
36#[component]
37pub async fn switch(#[default] mut attrs: Attributes) -> Result<impl View> {
38    // The thumb cannot be drawn by the `<input>` itself, which renders no
39    // children or pseudo-elements: it is a sibling overlaid on the track,
40    // slid to the far end by the input's `peer` state while checked.
41    Ok(view! {
42        <span
43            class=(class!(
44                "peer relative inline-flex shrink-0 has-[:disabled]:opacity-50",
45                attrs.remove("class"),
46            ))
47        >
48            <input type="checkbox" role="switch" class=(SWITCH) (attrs)>
49            <span class=(THUMB)></span>
50        </span>
51    })
52}
