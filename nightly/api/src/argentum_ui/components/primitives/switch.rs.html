7/// The classes for the native `<input type="checkbox">` serving as the
8/// [`switch`] track.
9///
10/// The native glyph is suppressed with `appearance-none` and the input is
11/// stretched into a pill-shaped track, which keeps the control looking the
12/// same across browsers. Checking it fills the track with the primary color.
13const SWITCH: StaticClass = class!(
14    "peer h-4.5 w-8 shrink-0 appearance-none rounded-full \
15     bg-foreground/20 shadow-xs transition-colors outline-none checked:bg-primary \
16     focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 \
17     focus-visible:ring-offset-background disabled:pointer-events-none",
18);
19
20/// The classes for the thumb sliding along the track.
21///
22/// The thumb is two spacing units smaller than the track, inset by one unit
23/// at rest, and travels to the mirrored position when checked, so the rim
24/// around it is uniform at both ends of the track.
25const THUMB: StaticClass = class!(
26    "pointer-events-none absolute top-1/2 left-0.5 size-3.5 -translate-y-1/2 \
27     rounded-full bg-background shadow-xs transition-transform peer-checked:translate-x-3.5",
28);

30/// A switch component: an on/off toggle for a setting that applies
31/// immediately.
32///
33/// The control is a native `<input type="checkbox">` carrying the `switch`
34/// role, so assistive technology announces it as a switch. The `attrs` (such
35/// as `name`, `checked`, `disabled`, or event handlers) are forwarded to the
36/// `<input>`; a `class` among them is appended to the wrapping element's
37/// classes. Set the on state with a plain `checked` attribute.
38///
39/// ```ignore
40/// view! {
41///     <div class="flex items-center gap-2">
42///         switch(attrs: attributes! { id="airplane-mode" checked="" })
43///         label(attrs: attributes! { for="airplane-mode" }, "Airplane mode")
44///     </div>
45/// }
46/// ```
47#[component]
48pub async fn switch(#[default] mut attrs: Attributes) -> Result<impl View> {
49    // The thumb cannot be drawn by the `<input>` itself, which renders no
50    // children or pseudo-elements: it is a sibling overlaid on the track,
51    // slid to the far end by the input's `peer` state while checked.
52    Ok(view! {
53        <span
54            class=(class!(
55                "peer relative inline-flex shrink-0 has-[:disabled]:opacity-50",
56                attrs.remove("class"),
57            ))
58        >
59            <input type="checkbox" role="switch" class=(SWITCH) (attrs)>
60            <span class=(THUMB)></span>
61        </span>
62    })
63}
