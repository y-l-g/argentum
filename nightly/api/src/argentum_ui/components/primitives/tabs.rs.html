8/// A tabs component: one panel at a time out of several, picked by the row of
9/// triggers above it.
10///
11/// Triggers can navigate to a server-rendered panel or update a signal in a
12/// click handler. For browser-side switching, bind each trigger's `active`
13/// prop and each panel's `hidden` attribute to the selected-tab signal.
14/// Triggers are ordinary links, without the ARIA tab pattern's arrow-key
15/// navigation.
16///
17/// The `attrs` (such as `class`) are forwarded to the underlying `<div>`; a
18/// `class` among them is appended to the computed classes.
19///
20/// ```ignore
21/// view! {
22///     tabs(
23///         tabs_list(
24///             for (value, text) in TABS {
25///                 tabs_trigger(
26///                     active: value == tab,
27///                     attrs: attributes! { href=(format!("?tab={value}")) },
28///                     (text)
29///                 )
30///             }
31///         )
32///         tabs_content((panel))
33///     )
34/// }
35/// ```
36#[component]
37pub async fn tabs(
38    #[default] mut attrs: Attributes,
39    #[default] child: Child<'_>,
40) -> Result<impl View> {
41    Ok(view! {
42        <div class=(class!("flex flex-col gap-4", attrs.remove("class"))) (attrs)>
43            (child)
44        </div>
45    })
46}

48/// The row of triggers at the top of a [`tabs`].
49///
50/// The triggers sit in a rail, the same one a toggle group uses, so a set of
51/// tabs reads as one control rather than as loose links.
52#[component]
53pub async fn tabs_list(
54    #[default] mut attrs: Attributes,
55    #[default] child: Child<'_>,
56) -> Result<impl View> {
57    Ok(view! {
58        <div
59            class=(class!(
60                "inline-flex w-fit items-center gap-1 rounded-lg border border-border p-1",
61                attrs.remove("class"),
62            ))
63            (attrs)
64        >
65            (child)
66        </div>
67    })
68}

70/// The classes for a [`tabs_trigger`].
71///
72/// The trigger for the panel on show is tinted and takes the full foreground
73/// color, which is what tells it apart from the rest; the others light up on
74/// hover.
75const TRIGGER: StaticClass = class!(
76    "inline-flex shrink-0 cursor-pointer items-center justify-center gap-2 \
77     rounded-md px-3 py-1.5 text-sm font-medium whitespace-nowrap transition-colors outline-none \
78     focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 \
79     focus-visible:ring-offset-background text-muted-foreground \
80     hover:bg-foreground/5 hover:text-foreground \
81     aria-[current=page]:bg-foreground/10 aria-[current=page]:text-foreground \
82     aria-[current=page]:hover:bg-foreground/10",
83);

85/// One trigger of a [`tabs_list`]: a link to the page showing its panel.
86///
87/// `active` accepts a boolean or a runtime expression and controls the
88/// selected styling and `aria-current="page"`. Pass the `href` among the
89/// `attrs`. A client-side click handler can prevent navigation and select a
90/// panel by updating a signal instead.
91#[component]
92pub async fn tabs_trigger(
93    /// Whether this trigger's panel is the one on show.
94    #[into]
95    #[default(false.into())]
96    active: Expr<bool>,
97    /// Extra attributes for the `<a>` element.
98    #[default]
99    mut attrs: Attributes,
100    /// The trigger's label.
101    #[default]
102    child: Child<'_>,
103) -> Result<impl View> {
104    Ok(view! {
105        <a
106            :aria-current=$(active.then_some("page"))
107            class=(class!(TRIGGER, attrs.remove("class")))
108            (attrs)
109        >
110            (child)
111        </a>
112    })
113}

115/// The panel of the [`tabs_trigger`] on show.
116///
117/// Render only the selected panel on the server, or render all panels with
118/// `:hidden` bindings for browser-side switching.
119#[component]
120pub async fn tabs_content(
121    #[default] mut attrs: Attributes,
122    #[default] child: Child<'_>,
123) -> Result<impl View> {
124    Ok(view! { <div class=(attrs.remove("class")) (attrs)>(child)</div> })
