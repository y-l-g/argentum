7/// A hover card component: a card of detail about its trigger, shown while
8/// the trigger is hovered or focused.
9///
10/// Where a tooltip carries a few words, a hover card carries a view: the
11/// person behind a mention, the project behind a link. Child nodes are the
12/// trigger and the [`hover_card_content`] holding that view.
13///
14/// The card waits before it appears, so that passing the cursor over the
15/// trigger on the way somewhere else does not summon it, and waits again
16/// before it goes, so that the way to it is not a race. What is in it is
17/// detail, not the only copy of anything: a reader who never hovers, and one
18/// on a touch screen, will not see it.
19///
20/// ```ignore
21/// view! {
22///     hover_card(
23///         <a href="/people/ada" class="font-medium underline">"@ada"</a>
24///         hover_card_content(
25///             <p class="text-sm font-medium">"Ada Lovelace"</p>
26///             <p class="text-sm text-muted-foreground">"Owner, joined in 2024."</p>
27///         )
28///     )
29/// }
30/// ```
31#[component]
32pub async fn hover_card(
33    #[default] mut attrs: Attributes,
34    #[default] child: Child<'_>,
35) -> Result<impl View> {
36    Ok(view! {
37        <span
38            class=(class!("group relative inline-flex", attrs.remove("class")))
39            (attrs)
40        >
41            (child)
42        </span>
43    })
44}

46/// The classes for the [`hover_card_content`] panel.
47///
48/// The panel is a raised surface like the card's, dropped below the trigger
49/// and aligned to its left edge. It sets its own background and text color,
50/// so it reads the same over anything it covers, and it takes pointer events,
51/// unlike a tooltip's bubble: a hover card can hold a link worth following.
52///
53/// The delay before it fades in and out is what keeps it from flickering as
54/// the cursor passes by. The visibility that keeps it out of the way in
55/// between is named in the transition with `allow-discrete`, since it has no
56/// in-between values; naming both properties one by one is what makes that
57/// work, as `all` does not carry the visibility along.
58const PANEL: StaticClass = class!(
59    "invisible absolute top-full left-0 z-50 mt-2 w-64 rounded-lg border \
60     border-border bg-popover p-4 text-popover-foreground opacity-0 shadow-sm \
61     [transition:opacity_150ms_ease-out_300ms,visibility_150ms_allow-discrete_300ms] \
62     group-hover:visible group-hover:opacity-100 \
63     group-focus-within:visible group-focus-within:opacity-100",
64);
65
66/// The view a [`hover_card`] shows, in a panel below its trigger.
67#[component]
68pub async fn hover_card_content(
69    #[default] mut attrs: Attributes,
70    #[default] child: Child<'_>,
71) -> Result<impl View> {
72    Ok(view! {
73        <span
74            class=(class!("flex flex-col gap-2", PANEL, attrs.remove("class")))
75            (attrs)
76        >
77            (child)
78        </span>
79    })
