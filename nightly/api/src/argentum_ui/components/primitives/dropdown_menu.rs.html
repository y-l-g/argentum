8/// A dropdown menu: a trigger that toggles a floating panel of actions.
9///
10/// Built on `<details>`, so it opens and closes without scripting: clicking
11/// the [`dropdown_menu_trigger`] toggles the [`dropdown_menu_content`] panel.
12/// Clicking outside does not close it; that behavior needs scripting. The
13/// `attrs` (such as `class` or `open`) are forwarded to the underlying
14/// `<details>`; a `class` among them is appended to the computed classes.
15///
16/// ```ignore
17/// view! {
18///     dropdown_menu(
19///         dropdown_menu_trigger("Options")
20///         dropdown_menu_content(
21///             dropdown_menu_item("Rename")
22///             dropdown_menu_item("Duplicate")
23///             dropdown_menu_separator()
24///             dropdown_menu_item(
25///                 attrs: attributes! { class="text-destructive" },
26///                 "Delete"
27///             )
28///         )
29///     )
30/// }
31/// ```
32#[component]
33pub async fn dropdown_menu(
34    #[default] mut attrs: Attributes,
35    #[default] child: Child<'_>,
36) -> Result<impl View> {
37    Ok(view! {
38        <details
39            class=(class!("group relative inline-block", attrs.remove("class")))
40            (attrs)
41        >
42            (child)
43        </details>
44    })
45}

47/// The classes making a `<summary>` a plain clickable trigger: the default
48/// disclosure marker is hidden and the cursor marks it as interactive.
49const TRIGGER: StaticClass = class!("cursor-pointer list-none [&::-webkit-details-marker]:hidden",);
50
51/// The trigger of a [`dropdown_menu`]: a `<summary>` that toggles the menu.
52///
53/// Child nodes become the trigger's content; any view works. The trigger
54/// carries no styling of its own: dress it as a button by passing the classes
55/// from [`button_variants`](super::button::button_variants), or leave it bare
56/// for a custom look. While the menu is open the `group-open:` variant
57/// applies within it, so a chevron with `group-open:rotate-180` flips along.
58/// The `attrs` are forwarded to the `<summary>`; a `class` among them is
59/// appended to the computed classes.
60///
61/// ```ignore
62/// view! {
63///     dropdown_menu_trigger(
64///         attrs: attributes! {
65///             class=(button_variants(ButtonVariant::Outline, ButtonSize::Md))
66///         },
67///         "Options"
68///         icon(
69///             data: iconify_icon!("lucide:chevron-down"),
70///             attrs: attributes! { class="group-open:rotate-180" }
71///         )
72///     )
73/// }
74/// ```
75#[component]
76pub async fn dropdown_menu_trigger(
77    #[default] mut attrs: Attributes,
78    #[default] child: Child<'_>,
79) -> Result<impl View> {
80    Ok(view! {
81        <summary class=(class!(TRIGGER, attrs.remove("class"))) (attrs)>
82            (child)
83        </summary>
84    })
85}

87/// The classes shared by the [`dropdown_menu_content`] and
88/// [`dropdown_menu_sub_content`] panels: a raised surface styled like a card;
89/// `z-50` lifts it over later content. It sets its own background and text
90/// color, so it reads the same on any ancestor.
91const PANEL: StaticClass = class!(
92    "absolute z-50 min-w-40 rounded-lg border border-border bg-popover p-1 \
93     text-popover-foreground shadow-sm",
94);

96/// The floating panel of a [`dropdown_menu`], holding the menu's items.
97///
98/// The panel drops directly below the trigger, aligned to its left edge.
99#[component]
100pub async fn dropdown_menu_content(
101    #[default] mut attrs: Attributes,
102    #[default] child: Child<'_>,
103) -> Result<impl View> {
104    Ok(view! {
105        <div
106            class=(class!(PANEL, "top-full left-0 mt-1", attrs.remove("class")))
107            (attrs)
108        >
109            (child)
110        </div>
111    })
112}

114/// The classes for a [`dropdown_menu_item`] row.
115///
116/// Hover, focus, and press tint the row like a ghost button, deriving the
117/// states from the foreground color so they hold up in both color schemes.
118const ITEM: StaticClass = class!(
119    "flex w-full items-center gap-2 rounded-md px-2 py-1.5 text-left text-sm \
120     whitespace-nowrap outline-none hover:bg-foreground/5 focus-visible:bg-foreground/5 \
121     active:bg-foreground/10 disabled:pointer-events-none disabled:opacity-50",
122);

124/// One action in a [`dropdown_menu_content`], rendered as a `<button>`.

126pub async fn dropdown_menu_item(

131        <button class=(class!(ITEM, attrs.remove("class"))) (attrs)>(child)</button>

135/// A nested submenu placed among the items of a [`dropdown_menu_content`].
136///
137/// Like the [`dropdown_menu`] itself it is built on `<details>`, so clicking
138/// the [`dropdown_menu_sub_trigger`] toggles its [`dropdown_menu_sub_content`]
139/// panel without scripting. The submenu tracks its own open state under the
140/// `group/sub` name, so the `group-open/sub:` variant targets it without
141/// disturbing the enclosing menu's `group`. Closing the enclosing menu hides
142/// an open submenu but does not close it, so it is open again the next time
143/// the menu opens; resetting it needs scripting. The `attrs` are forwarded to
144/// the underlying `<details>`; a `class` among them is appended to the
145/// computed classes.
146///
147/// ```ignore
148/// view! {
149///     dropdown_menu_content(
150///         dropdown_menu_item("Back")
151///         dropdown_menu_sub(
152///             dropdown_menu_sub_trigger("Move to")
153///             dropdown_menu_sub_content(
154///                 dropdown_menu_item("Inbox")
155///                 dropdown_menu_item("Archive")
156///             )
157///         )
158///     )
159/// }
160/// ```
161#[component]
162pub async fn dropdown_menu_sub(
163    #[default] mut attrs: Attributes,
164    #[default] child: Child<'_>,
165) -> Result<impl View> {
166    Ok(view! {
167        <details class=(class!("group/sub relative", attrs.remove("class"))) (attrs)>
168            (child)
169        </details>
170    })
171}

173/// The trigger row of a [`dropdown_menu_sub`]: a `<summary>` styled as a
174/// [`dropdown_menu_item`] that toggles the submenu.
175///
176/// Child nodes become the row's label; a chevron pointing toward the submenu
177/// is appended automatically, and the row stays tinted while the submenu is
178/// open. The `attrs` are forwarded to the `<summary>`; a `class` among them is
179/// appended to the computed classes.
180#[component]
181pub async fn dropdown_menu_sub_trigger(
182    #[default] mut attrs: Attributes,
183    #[default] child: Child<'_>,
184) -> Result<impl View> {
185    Ok(view! {
186        <summary
187            class=(class!(
188                ITEM,
189                TRIGGER,
190                "group-open/sub:bg-foreground/5",
191                attrs.remove("class"),
192            ))
193            (attrs)
194        >
195            (child)
196            icon(
197                data: iconify_icon!("lucide:chevron-right"),
198                attrs: attributes! { class="ml-auto size-4" }
199            )
200        </summary>
201    })
202}

204/// The floating panel of a [`dropdown_menu_sub`], holding the submenu's items.
205///
206/// A submenu opens beside its trigger rather than below it: `left-full` places
207/// it against the right edge of the parent panel, `top-0` lines its top up with
208/// the trigger row, and `ml-1` leaves the same gap the menu keeps from its own
209/// trigger.
210#[component]
211pub async fn dropdown_menu_sub_content(
212    #[default] mut attrs: Attributes,
213    #[default] child: Child<'_>,
214) -> Result<impl View> {
215    Ok(view! {
216        <div
217            class=(class!(PANEL, "top-0 left-full ml-1", attrs.remove("class")))
218            (attrs)
219        >
220            (child)
221        </div>
222    })
223}
