1//! Authentication — credentials, server-side sessions, and the panel gate
2//! (ADR-0013, spec #127).
3//!
4//! A [`Panel`] is gated by default. The shipped [`PasswordAuth`]
5//! verifies Argon2id PHC hashes against the [`AdminUser`] model, a session
6//! cookie issued by Topcoat's token transport identifies one server-side
7//! [`AuthSession`] row, and the resolved [`CurrentUser`] travels in request
8//! `Cx` for pages, shards, and app code. An app with its own user table
9//! implements [`Authenticator`] and swaps it in with
10//! [`Panel::auth`](crate::Panel::auth); [`Auth::disabled`] is the explicit,
11//! greppable opt-out for public demos.
12//!
13//! Sessions are always the framework's: the shipped [`AuthSession`] table maps
14//! a token hash to a user id, so an app registers it whatever authenticator it
15//! uses. A custom user model does not have to be an `AdminUser`.
16

47/// The one error every failed login renders, so accounts cannot be
48/// enumerated and panel membership stays private (ADR-0013).
49const GENERIC_ERROR: &str = "Invalid email or password.";

51/// What a failed login attempt renders when the database behind it could not
52/// answer (GH #230).
53///
54/// [`GENERIC_ERROR`] is deliberately non-specific about *why* credentials were
55/// refused, so reusing it for an outage would tell a user their password was
56/// wrong while the database was down — the one place they would retry
57/// uselessly. This copy names the machinery that is down instead, and like the
58/// generic one it never carries driver text: that goes to the log.
59const UNAVAILABLE_ERROR: &str = "Sign-in is unavailable right now. Try again shortly.";

61/// A dummy Argon2id PHC string verified against when the account does not
62/// exist, so unknown emails pay the same work as known ones (ADR-0013).
63/// Generated with `Argon2::default()` parameters (`m=19456,t=2,p=1`).
64const DUMMY_PASSWORD_HASH: &str = "$argon2id$v=19$m=19456,t=2,p=1$h3oXdPBVwhcgZ1OTO/PuzQ$zLrHLgIkwhqu4ZlLTfSyB8mPuL6mAtaswv/eXJ5ADO8";

66/// The shipped credential model (ADR-0013): unique email, Argon2id PHC hash,
67/// display name, active flag, and an optional tenant.
68///
69/// Register it (plus [`AuthSession`]) in the app's `Db` model list, seed one
70/// row, and the default [`PasswordAuth`] works with no further wiring:
71/// `toasty::models!(…, argentum_core::auth::AdminUser, argentum_core::auth::AuthSession)`.
72#[derive(Debug, Clone, toasty::Model)]
73pub struct AdminUser {
74    #[key]
75    #[auto]
76    pub id: Uuid,
77    #[unique]
78    pub email: String,
79    /// Argon2id password hash in PHC string format.
80    pub password_hash: String,
81    pub display_name: String,
82    /// `false` denies login and panel access immediately.
83    pub active: bool,
84    pub tenant_id: Option<Uuid>,
85    pub created_at: Timestamp,
86}

88/// The shipped server-side session record (ADR-0013): the SHA-256 hash of the
89/// client token, the user it authenticates (opaque id), and its expiry.
90///
91/// The raw token is never stored; a leaked session table contains nothing a
92/// client could present. Register this model alongside the app's user model.
93#[derive(Debug, Clone, toasty::Model)]
94pub struct AuthSession {
95    /// Hex-encoded SHA-256 of the session token (the raw token stays client-side).
96    #[key]
97    pub token_hash: String,
98    /// [`CurrentUser::id`] of the authenticated user.
99    #[index]
100    pub user_id: String,
101    pub expires_at: Timestamp,
102    pub created_at: Timestamp,
103}

105/// The erased identity resolution places in request `Cx` (ADR-0013).
106///
107/// `id` is an opaque string so non-UUID keys fit; read it only through
108/// [`current_user`] / [`require_authenticated`].
109#[derive(Debug, Clone, PartialEq, Eq)]
110pub struct CurrentUser {
111    /// Stable, opaque user key; custom user tables may use any string.
112    pub id: String,
113    /// The identifier the user logged in with (the shipped default: email).
114    pub login: String,
115    pub display_name: String,
116    /// Optional tenant carried from the user; the gate injects [`Tenant`] only
117    /// when present.
118    ///
119    /// [`Tenant`]: crate::Tenant
120    pub tenant_id: Option<Uuid>,
121    /// Whether the user may enter the panel. Denied users answer 403,
122    /// indistinguishable from bad credentials at login (ADR-0013).
123    pub can_access_panel: bool,
124}

129/// The one authentication seam (ADR-0013).
130///
131/// The default [`PasswordAuth`] implements it against the shipped
132/// [`AdminUser`] model; an app with an existing user table implements it and
133/// passes the value to [`Panel::auth`](crate::Panel::auth) via
134/// [`Auth::custom`]. Sessions are the framework's, so a custom implementation
135/// only maps credentials to a [`CurrentUser`] and back.
136///
137/// Every *credential* failure must return `Ok(None)`, never a distinguishable
138/// error: the login response is one generic message for all of them. An
139/// infrastructure failure is not a credential verdict, so an implementation
140/// that cannot reach its store may return the driver's error instead — the
141/// login handler maps it to the opaque outage page (GH #230), which is what
142/// keeps a database outage from rendering as a rejected password. An
143/// implementation's own error keeps its own mapping.
144pub trait Authenticator: Send + Sync + 'static {
145    /// Verify `login`/`password`, returning the user on success.
146    ///
147    /// Implementations must run comparable work for unknown accounts (the
148    /// shipped [`PasswordAuth`] verifies a dummy hash) so timing does not leak
149    /// account existence.
150    fn verify<'a>(
151        &'a self,
152        cx: &'a Cx,
153        login: &'a str,
154        password: &'a str,
155    ) -> AuthFuture<'a, Option<CurrentUser>>;
156
157    /// Resolve the live session user by [`CurrentUser::id`].
158    ///
159    /// Loading the row each request is what makes deactivation and revocation
160    /// take effect immediately; return `None` for a user that no longer
161    /// authenticates.
162    fn find_by_id<'a>(&'a self, cx: &'a Cx, id: &'a str) -> AuthFuture<'a, Option<CurrentUser>>;
163}

165/// The opaque error an infrastructure failure on an auth or session path
166/// carries (GH #230).
167///
168/// Its `Display` is [`UNAVAILABLE_ERROR`], so nothing driver-shaped can travel
169/// inside it, and its concrete type is what lets the login handler recognise
170/// the one case it answers with the outage page rather than a 500.
171#[derive(Debug)]
172struct Unavailable;
173
174impl std::fmt::Display for Unavailable {
175    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
176        f.write_str(UNAVAILABLE_ERROR)
177    }
178}
179
180impl std::error::Error for Unavailable {}

182/// Map a failed auth or session operation to the error the response carries
183/// (GH #230) — the counterpart of [`crate::db::hook_failure`] for this module.
184///
185/// An error that is the driver's is an infrastructure failure: it becomes
186/// [`Unavailable`], and the driver's own text goes to the log under this
187/// event, never to the page. Anything else is app-authored (a custom
188/// [`Authenticator`]'s own error) and keeps its mapping, so the seam does not
189/// swallow it. Every auth and session path maps through here, which is what
190/// keeps a database outage from reading as a rejected password — and what
191/// makes the two distinguishable in the logs: a rejection is `Ok(None)`, the
192/// generic 403 and no error line, while an outage logs the driver's text.
193fn infrastructure_failure(error: impl Into<topcoat::Error>) -> topcoat::Error {
194    let error = error.into();
195    if error.is::<toasty::Error>() {
196        tracing::error!(error = %error, "auth infrastructure failure");
197        Unavailable.into()
198    } else {
199        error
200    }
201}

203/// The shipped default authenticator: Argon2id verification against
204/// [`AdminUser`].
205#[derive(Debug, Default, Clone, Copy)]
206pub struct PasswordAuth;
207
208impl PasswordAuth {
209    /// Creates the shipped authenticator.
210    #[must_use]
211    pub fn new() -> Self {
212        Self
213    }
214}
215
216impl Authenticator for PasswordAuth {
217    fn verify<'a>(
218        &'a self,
219        cx: &'a Cx,
220        login: &'a str,
221        password: &'a str,
222    ) -> AuthFuture<'a, Option<CurrentUser>> {
223        Box::pin(async move {
224            let mut db = crate::db::db(cx);
225            let found = AdminUser::filter(AdminUser::fields().email().eq(login.trim().to_string()))
226                .first()
227                .exec(&mut db)
228                .await
229                .map_err(infrastructure_failure)?;
230            let Some(user) = found else {
231                // Unknown account: pay a verification anyway (ADR-0013).
232                let _ = verify_password(password, DUMMY_PASSWORD_HASH);
233                return Ok(None);
234            };
235            if !verify_password(password, &user.password_hash) {
236                return Ok(None);
237            }
238            Ok(Some(current_user_from(&user)))
239        })
240    }
241
242    fn find_by_id<'a>(&'a self, cx: &'a Cx, id: &'a str) -> AuthFuture<'a, Option<CurrentUser>> {
243        Box::pin(async move {
244            let Ok(id) = Uuid::parse_str(id) else {
245                return Ok(None);
246            };
247            let mut db = crate::db::db(cx);
248            let user = AdminUser::filter(AdminUser::fields().id().eq(id))
249                .first()
250                .exec(&mut db)
251                .await
252                .map_err(infrastructure_failure)?;
253            // A deactivated account stops resolving: its live sessions are
254            // purged and the next request redirects to login (spec #127 US11).
255            Ok(user
256                .filter(|user| user.active)
257                .map(|user| current_user_from(&user)))
258        })
259    }
260}

262/// Hash a password with Argon2id into a PHC string (the shipped storage
263/// format). Seeds and record fns call this; it never stores the plaintext.
264pub fn hash_password(password: &str) -> topcoat::Result<String> {
265    use argon2::password_hash::PasswordHasher;
266
267    argon2::Argon2::default()
268        .hash_password(password.as_bytes())
269        .map(|hash| hash.to_string())
270        .map_err(topcoat::Error::from)
271}
272
273/// Verify a password against a PHC hash; `false` on any malformed input.
274fn verify_password(password: &str, phc: &str) -> bool {
275    use argon2::password_hash::{PasswordVerifier, phc::PasswordHash};
276
277    let Ok(parsed) = PasswordHash::new(phc) else {
278        return false;
279    };
280    argon2::Argon2::default()
281        .verify_password(password.as_bytes(), &parsed)
282        .is_ok()
283}

296/// The panel's authentication configuration (ADR-0013).
297///
298/// The default is [`Auth::password`]; [`Auth::custom`] swaps in an app-owned
299/// [`Authenticator`]; [`Auth::disabled`] is the explicit fail-open opt-out.
300pub enum Auth {
301    /// The shipped Argon2id + [`AdminUser`] authenticator.
302    Password(PasswordAuth),
303    /// An app-owned authenticator over its own user table.
304    Custom(Box<dyn Authenticator>),
305    /// Explicit opt-out: no gate, no login routes, sessions unused.
306    Disabled,
307}
308
309impl Auth {
310    /// The shipped default.
311    #[must_use]
312    pub fn password() -> Self {
313        Self::Password(PasswordAuth::new())
314    }
315
316    /// An app-owned authenticator, erased to the trait object the panel keeps.
317    #[must_use]
318    pub fn custom(authenticator: impl Authenticator) -> Self {
319        Self::Custom(Box::new(authenticator))
320    }
321
322    /// Explicit fail-open opt-out for public demos (ADR-0013).
323    #[must_use]
324    pub fn disabled() -> Self {
325        Self::Disabled
326    }
327
328    /// The authenticator, or `None` when auth is explicitly disabled.
329    #[must_use]
330    pub fn authenticator(&self) -> Option<&dyn Authenticator> {
331        match self {
332            Self::Password(password) => Some(password),
333            Self::Custom(authenticator) => Some(&**authenticator),
334            Self::Disabled => None,
335        }
336    }
337
338    /// Whether the explicit opt-out is set.
339    #[must_use]
340    pub fn is_disabled(&self) -> bool {
341        matches!(self, Self::Disabled)
342    }
343}

345impl Default for Auth {
346    fn default() -> Self {
347        Self::password()

364/// Require an authenticated, panel-permitted user.
365///
366/// Answers per request kind (ADR-0013): pages redirect to the login route
367/// with a same-origin-relative `next`, runtime endpoints, non-GET requests,
368/// and page re-runs (marked POSTs the runtime layer rewrites into GETs)
369/// answer 401, and an authenticated user without panel access answers 403.
370pub fn require_authenticated(cx: &Cx) -> topcoat::Result<CurrentUser> {
371    if let Some(user) = current_user(cx) {
372        if user.can_access_panel {
373            return Ok(user);
374        }
375        return Err(forbidden().into());
376    }
377    Err(unauthenticated_error(cx))
378}

380/// The error an unauthenticated request answers with, by request kind.
381fn unauthenticated_error(cx: &Cx) -> topcoat::Error {
382    let path = uri(cx).path();
383    let page_method = matches!(*method(cx), http::Method::GET | http::Method::HEAD);
384    // A page re-run reaches the gate as a rewritten GET: Topcoat's runtime
385    // layer rewrites the browser's marked POST into a GET for the page's own
386    // URL. The original request tells it apart from a plain page load, so a
387    // logged-out re-run answers 401 like any other non-page request instead
388    // of redirecting into the login page.
389    let rerun = !matches!(*original_method(cx), http::Method::GET | http::Method::HEAD)
390        && original_headers(cx).get(&topcoat::runtime::RUNTIME_HEADER) == Some(&RERUN_MARKER);
391    if path.starts_with(RUNTIME_PREFIX) || !page_method || rerun {
392        unauthorized().into()
393    } else {
394        redirect(login_url_with_next(cx)).into()
395    }
396}

408/// The login URL with a validated `next` back to the requested page.
409fn login_url_with_next(cx: &Cx) -> String {
410    let mut url = login_url(cx);
411    let request = uri(cx);
412    let requested = match request.query() {
413        Some(query) => format!("{}?{query}", request.path()),
414        None => request.path().to_string(),
415    };
416    if let Some(next) = safe_next(&requested) {
417        let mut serializer = form_urlencoded::Serializer::new(String::new());
418        serializer.append_pair(NEXT_FIELD, next);
419        url.push('?');
420        url.push_str(&serializer.finish());
421    }
422    url
423}

437/// Accept only same-origin relative paths as a post-login destination
438/// (ADR-0013): absolute URLs, scheme-relative `//host` targets, backslash
439/// tricks, and control characters are rejected.
440pub(crate) fn safe_next(next: &str) -> Option<&str> {
441    let next = next.trim();
442    if !next.starts_with('/') || next.starts_with("//") {
443        return None;
444    }
445    if next.contains('\\') || next.chars().any(|c| c.is_control()) {
446        return None;
447    }
448    Some(next)
449}

451/// The validated `?next=` the login page embeds as a hidden field.
452fn next_from_query(cx: &Cx) -> Option<String> {
453    let query = uri(cx).query()?;
454    form_urlencoded::parse(query.as_bytes())
455        .find(|(key, _)| key == NEXT_FIELD)
456        .map(|(_, value)| value.into_owned())
457        .filter(|value| safe_next(value).is_some())
458}

460/// Hex-encode a token hash into its storage key.
461fn token_key(hash: &TokenHash) -> String {
462    use std::fmt::Write as _;
463
464    let mut key = String::with_capacity(64);
465    for byte in hash.iter() {
466        write!(key, "{byte:02x}").expect("writing to a String cannot fail");
467    }
468    key
469}

476/// Delete one stored session row by its hex token-hash key.

487/// Revoke every live session of `user_id` (ADR-0013).
488///
489/// Call this whenever a credential changes out from under live sessions —
490/// password reset/change and deactivation alike. Nothing in-core calls it
491/// (there is no password-change flow in the framework); sessions otherwise
492/// stay valid for their full fixed lifetime, so a reset that skips this
493/// leaves a stolen session usable. A password-reset flow must call it,
494/// and custom `Authenticator` apps own the same obligation.
495pub async fn revoke_sessions_for_user(cx: &Cx, user_id: &str) -> topcoat::Result<()> {
496    let mut db = crate::db::db(cx);
497    AuthSession::filter(AuthSession::fields().user_id().eq(user_id.to_string()))
498        .delete()
499        .exec(&mut db)
500        .await
501        .map_err(infrastructure_failure)?;
502    Ok(())
503}

505/// Resolve the request's session into a user, lazily and without touching the
506/// database when no session cookie is present.
507pub(crate) async fn resolve(
508    cx: &Cx,
509    authenticator: &dyn Authenticator,
510) -> topcoat::Result<Option<CurrentUser>> {
511    let Some(hash) = session::token_hash(cx).await? else {
512        return Ok(None);
513    };
514    let key = token_key(&hash);
515    let mut db = crate::db::db(cx);
516    let row = AuthSession::filter(AuthSession::fields().token_hash().eq(key))
517        .first()
518        .exec(&mut db)
519        .await
520        .map_err(infrastructure_failure)?;
521    let Some(row) = row else {
522        return Ok(None);
523    };
524    if row.expires_at <= Timestamp::now() {
525        // Expired sessions do not resolve; purge the row on the way out.
526        delete_session_row(cx, &row.token_hash).await?;
527        return Ok(None);
528    }
529    match authenticator
530        .find_by_id(cx, &row.user_id)
531        .await
532        .map_err(infrastructure_failure)?
533    {
534        Some(user) => Ok(Some(user)),
535        None => {
536            // The session names a user who no longer authenticates (deleted
537            // or deactivated): purge it so removal is real (US11).
538            delete_session_row(cx, &row.token_hash).await?;
539            Ok(None)
540        }
541    }
542}

544/// The layer that gates the panel and runtime prefixes (ADR-0013): resolves
545/// the session into request `Cx` when present and answers fail-closed when a
546/// route inside its prefix has no permitted user.
547pub(crate) struct AuthGate {
548    path: PathBuf,
549}
550
551impl AuthGate {
552    /// Guards requests under `path`.
553    pub(crate) fn new(path: &str) -> Self {
554        Self {
555            path: route_path(path),
556        }
557    }
558}
559
560impl Layer for AuthGate {
561    fn path(&self) -> Option<&Path> {
562        Some(&self.path)
563    }
564
565    fn handle<'a>(&'a self, cx: &'a Cx, body: Body, next: Next<'a>) -> LayerFuture<'a> {
566        Box::pin(async move {
567            // The login page must answer while logged out.
568            if uri(cx).path() == login_url(cx) {
569                return next.run(cx, body).await;
570            }
571            let auth = app_context::<Auth>(cx);
572            let Some(authenticator) = auth.authenticator() else {
573                return next.run(cx, body).await;
574            };
575            // The logout route must answer for any resolved user, even one
576            // whose panel access was revoked after login (GH #146) — clearing
577            // the session row + cookie must not require panel permission, or
578            // the session lingers to expiry. The bypass is POST-only at the
579            // exact logout path: the route table registers nothing else
580            // there, and a non-POST method must not smuggle a resolved
581            // identity to any handler an app might mount at the same path.
582            let logout_route =
583                uri(cx).path() == logout_url(cx) && matches!(*method(cx), http::Method::POST);
584            match resolve(cx, authenticator).await? {
585                Some(user) if user.can_access_panel || logout_route => {
586                    // The logged-in user's optional tenant becomes the request
587                    // tenant; auth never requires one (ADR-0013).
588                    let tenant_id = user.tenant_id;
589                    let mut child = cx.with(user);
590                    if let Some(tenant_id) = tenant_id {
591                        child = child.with(crate::tenancy::Tenant(tenant_id));
592                    }
593                    next.run(&child, body).await
594                }
595                // Authenticated but not permitted: 403, indistinguishable
596                // from bad credentials at login (ADR-0013). The logout route
597                // is answered above (GH #146).
598                Some(_) => Err(forbidden().into()),
599                // Pages redirect to the login route with a validated `next`;
600                // runtime endpoints, non-GET requests, and page re-runs
601                // answer 401.
602                None => Err(unauthenticated_error(cx)),
603            }
604        })
605    }
606}

608/// Install session support and the gate layers on a panel router.
609pub(crate) fn install(
610    builder: topcoat::router::RouterBuilder,
611    prefix: &str,
612) -> topcoat::router::RouterBuilder {
613    builder
614        .sessions(SessionConfig::builder().lifetime(SESSION_LIFETIME).build())
615        .layer(AuthGate::new(prefix))
616        .layer(AuthGate::new(RUNTIME_PREFIX))

619/// What a failed login attempt renders: the generic credential rejection, or
620/// the sign-in outage (GH #230).
621///
622/// Two variants and no more, each with its own copy and status, so a failed
623/// attempt is always a deliberate answer: a driver failure is never rendered
624/// as a credential verdict, and a credential verdict never borrows the outage
625/// copy. Neither variant carries driver text.
626#[derive(Debug, Clone, Copy)]
627enum LoginError {
628    /// Wrong password, unknown account, empty fields, or valid credentials
629    /// without panel access: one 403 with one message (ADR-0013).
630    Credentials,
631    /// The database behind sign-in could not answer: a 503 that says so,
632    /// because the user must not read an outage as a rejected password.
633    Unavailable,
634}
635
636impl LoginError {
637    /// The message the login page's alert carries.
638    fn message(self) -> &'static str {
639        match self {
640            Self::Credentials => GENERIC_ERROR,
641            Self::Unavailable => UNAVAILABLE_ERROR,
642        }
643    }
644
645    /// The status a failed attempt answers with.
646    fn status(self) -> http::StatusCode {
647        match self {
648            Self::Credentials => http::StatusCode::FORBIDDEN,
649            Self::Unavailable => http::StatusCode::SERVICE_UNAVAILABLE,
650        }
651    }
652}
653
654/// The status and route a login attempt renders: a [`LoginError`] page or a
655/// redirect back to `next`.
656///
657/// The login page is a settled view, so [`ViewExt::single`] resolves it into
658/// an owned handle before the response is built — no borrowed view escapes.
659async fn login_response(
660    cx: &Cx,
661    error: Option<LoginError>,
662    next: String,
663) -> topcoat::Result<topcoat::router::response::Response> {
664    let page = render_login_page(cx, error, next).await?;
665    page.single().await?.into_response(cx)
666}

668/// `GET {prefix}/login` — the standalone login page.
669pub(crate) fn login_page(cx: &Cx, _body: Body) -> RouteFuture<'_> {
670    let next = next_from_query(cx).unwrap_or_default();
671    Box::pin(login_response(cx, None, next))
672}

674/// `POST {prefix}/login` — verify, rotate the session, redirect to `next`.
675pub(crate) fn login_post(cx: &Cx, body: Body) -> RouteFuture<'_> {
676    Box::pin(async move {
677        // Login/logout carry no file parts: the values half is all they read.
678        let values = crate::panel::parse_form_body(cx, body).await?.values;
679        crate::csrf::verify(cx, &values)?;
680        // Keep a validated destination across a failed attempt so the retry
681        // form still returns where the visitor was headed (US6).
682        let next = values
683            .get(NEXT_FIELD)
684            .and_then(|value| safe_next(value))
685            .map(str::to_string)
686            .or_else(|| next_from_query(cx))
687            .unwrap_or_default();
688        let email = values.get(LOGIN_FIELD).map(|value| value.trim());
689        let password = values.get(PASSWORD_FIELD).map(String::as_str);
690        let auth = app_context::<Auth>(cx);
691        let verified = match (auth.authenticator(), email, password) {
692            (Some(authenticator), Some(email), Some(password))
693                if !email.is_empty() && !password.is_empty() =>
694            {
695                match authenticator.verify(cx, email, password).await {
696                    Ok(user) => user,
697                    // A driver failure is not a credential verdict: the page
698                    // says sign-in is unavailable instead of rendering a
699                    // rejection (GH #230). The seam logs the driver's text and
700                    // hands back an app-authored error untouched.
701                    Err(error) => {
702                        let error = infrastructure_failure(error);
703                        if error.is::<Unavailable>() {
704                            return login_response(cx, Some(LoginError::Unavailable), next).await;
705                        }
706                        return Err(error);
707                    }
708                }
709            }
710            _ => None,
711        };
712        // One path for every failure: wrong password, unknown account, empty
713        // fields, or valid credentials without panel access (ADR-0013).
714        let Some(user) = verified.filter(|user| user.can_access_panel) else {
715            return login_response(cx, Some(LoginError::Credentials), next).await;
716        };
717        // Rotate on login: a token this request presented cannot be replayed.
718        if let Some(hash) = session::token_hash(cx).await? {
719            delete_session(cx, &hash).await?;
720        }
721        let session = session::start(cx).await?;
722        let mut db = crate::db::db(cx);
723        let recorded = toasty::create!(AuthSession {
724            token_hash: token_key(&session.token_hash),
725            user_id: user.id.clone(),
726            expires_at: Timestamp::try_from(session.expires_at).map_err(topcoat::Error::from)?,
727            created_at: Timestamp::now(),
728        })
729        .exec(&mut db)
730        .await;
731        if let Err(error) = recorded {
732            // The credentials were right; the session row could not be
733            // recorded. Same outage page as a failed verification — never the
734            // driver's text (GH #230).
735            let error = infrastructure_failure(error);
736            if error.is::<Unavailable>() {
737                return login_response(cx, Some(LoginError::Unavailable), next).await;
738            }
739            return Err(error);
740        }
741        let target = if next.is_empty() {
742            panel_root(cx)
743        } else {
744            next
745        };
746        // Success stays on the `Ok` path so `Set-Cookie` flushes
747        // (upstream topcoat#126).
748        topcoat::router::error::see_other(target).into_response(cx)
749    })
750}
751
752/// `POST {prefix}/logout` — delete the session row and clear the cookie.
753pub(crate) fn logout_post(cx: &Cx, body: Body) -> RouteFuture<'_> {
754    Box::pin(async move {
755        // Defense in depth: the route only exists on gated panels, but it
756        // re-checks so a missing layer cannot leave logout ungated. Any
757        // resolved identity may log out — the gate answers this route for a
758        // `can_access_panel=false` user too (GH #146), so demanding panel
759        // access here would strand their session row + cookie to expiry.
760        if current_user(cx).is_none() {
761            return Err(unauthenticated_error(cx));
762        }
763        // Login/logout carry no file parts: the values half is all they read.
764        let values = crate::panel::parse_form_body(cx, body).await?.values;
765        crate::csrf::verify(cx, &values)?;
766        if let Some(hash) = session::stop(cx).await? {
767            delete_session(cx, &hash).await?;
768        }
769        let target = login_url(cx);
770        topcoat::router::error::see_other(target).into_response(cx)
771    })
772}

774/// The standalone login document: brand and dark mode honored, CSRF hidden
775/// field, one error slot carrying a [`LoginError`]'s deliberate copy, no
776/// sidebar (ADR-0013).
777async fn render_login_page<'a>(
778    cx: &'a Cx,
779    error: Option<LoginError>,
780    next: String,
781) -> topcoat::Result<BoxView<'a>> {
782    let csrf = crate::csrf::ensure_token(cx);
783    let action = login_url(cx);
784    let brand = Panel::render_brand(cx).await?;
785    let hint = try_app_context::<LoginHint>(cx).map(|hint| hint.0.clone());
786    let body = topcoat::view::view! {
787        cx =>
788        <div class="flex min-h-svh items-center justify-center bg-muted p-6">
789            <div
790                class="flex w-full max-w-sm flex-col gap-6 rounded-xl border border-border bg-card p-6 text-card-foreground shadow-sm"
791            >
792                if let Some(error) = error {
793                    (error.status())
794                }
795                <div class="flex flex-col items-center gap-2">
796                    (brand)
797                    <h1 class="text-lg font-semibold text-foreground">"Sign in"</h1>
798                </div>
799                <form method="post" action=(action) class="flex flex-col gap-4">
800                    (crate::csrf::field(cx, &csrf))
801                    <input type="hidden" name=(NEXT_FIELD) value=(next)>
802                    if let Some(error) = error {
803                        argentum_ui::alert(
804                            variant: argentum_ui::AlertVariant::Destructive,
805                            attrs: topcoat::view::attributes! { role="alert" },
806                            argentum_ui::alert_title((error.message()))
807                        )
808                    }
809                    argentum_ui::field(
810                        argentum_ui::field_label(
811                            attrs: topcoat::view::attributes! { for="email" },
812                            "Email or username"
813                        )
814                        argentum_ui::input(
815                            attrs: topcoat::view::attributes! {
816                                id="email"
817                                name=(LOGIN_FIELD)
818                                type="text"
819                                required=""
820                                autocomplete="username"
821                                autofocus=""
822                            }
823                        )
824                    )
825                    argentum_ui::field(
826                        argentum_ui::field_label(
827                            attrs: topcoat::view::attributes! { for="password" },
828                            "Password"
829                        )
830                        argentum_ui::input(
831                            attrs: topcoat::view::attributes! {
832                                id="password"
833                                name=(PASSWORD_FIELD)
834                                type="password"
835                                required=""
836                                autocomplete="current-password"
837                            }
838                        )
839                    )
840                    argentum_ui::button(
841                        variant: argentum_ui::ButtonVariant::Primary,
842                        attrs: topcoat::view::attributes! { type="submit" class="w-full" },
843                        "Sign in"
844                    )
845                </form>
846                if let Some(hint) = hint {
847                    <p class="text-center text-xs text-muted-foreground">(hint)</p>
848                }
849            </div>
850        </div>
851    }
852    .boxed();
853    Panel::render_document(cx, "Sign in".to_string(), body).await
854}

864/// Fail loudly at startup when a required shipped model is missing from the
865/// app's `Db` (ADR-0013): the table is never pushed, and the first login
866/// would otherwise be a confusing runtime error.
867pub(crate) fn assert_models_registered(db: &Db, auth: &Auth) {
868    if auth.is_disabled() {
869        return;
870    }
871    let registered = |name: &str| {
872        db.schema()
873            .app
874            .models()
875            .any(|model| model.name().upper_camel_case() == name)
876    };
877    let missing: Vec<&str> = [
878        (!registered("AuthSession")).then_some("AuthSession"),
879        (matches!(auth, Auth::Password(_)) && !registered("AdminUser")).then_some("AdminUser"),
880    ]
881    .into_iter()
882    .flatten()
883    .collect();
884    assert!(
885        missing.is_empty(),
886        "argentum auth is on by default but its shipped models are not registered on the Db \
887         (missing {}). Register them with `toasty::models!(…, argentum_core::auth::AdminUser, \
888         argentum_core::auth::AuthSession)`, or opt out with `.auth(Auth::disabled())`.",
889        missing.join(", "),
890    );
891}
