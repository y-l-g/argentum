1//! Authentication — credentials, server-side sessions, and the panel gate
2//! (ADR-0013, spec #127).
3//!
4//! A [`Panel`] is gated by default. The shipped [`PasswordAuth`]
5//! verifies Argon2id PHC hashes against the [`AdminUser`] model, a session
6//! cookie issued by Topcoat's token transport identifies one server-side
7//! [`AuthSession`] row, and the resolved [`CurrentUser`] travels in request
8//! `Cx` for pages, shards, and app code. An app with its own user table
9//! implements [`Authenticator`] and swaps it in with
10//! [`Panel::auth`](crate::Panel::auth); [`Auth::disabled`] is the explicit,
11//! greppable opt-out for public demos.
12//!
13//! Sessions are always the framework's: the shipped [`AuthSession`] table maps
14//! a token hash to a user id, so an app registers it whatever authenticator it
15//! uses. A custom user model does not have to be an `AdminUser`.
16

39/// Max body the login route accepts: 64 KiB.
40///
41/// A credential form submits an email, a password, a `next` and a CSRF token,
42/// all short strings; the panel's 10 MiB form cap is for multipart uploads,
43/// which the login route never carries. 64 KiB leaves room for the fields plus
44/// percent-encoding growth while keeping the one unauthenticated POST route
45/// from buffering a megabyte-scale body.
46pub(crate) const MAX_LOGIN_BYTES: usize = 64 * 1024;

56/// The one error every failed login renders, so accounts cannot be
57/// enumerated and panel membership stays private (ADR-0013).
58const GENERIC_ERROR: &str = "Invalid email or password.";

60/// What a failed login attempt renders when the database behind it could not
61/// answer.
62///
63/// [`GENERIC_ERROR`] is deliberately non-specific about *why* credentials were
64/// refused, so reusing it for an outage would tell a user their password was
65/// wrong while the database was down — the one place they would retry
66/// uselessly. This copy names the machinery that is down instead, and like the
67/// generic one it never carries driver text: that goes to the log.
68const UNAVAILABLE_ERROR: &str = "Sign-in is unavailable right now. Try again shortly.";

70/// A dummy Argon2id PHC string verified against when the account does not

97/// The shipped server-side session record (ADR-0013): the SHA-256 hash of the
98/// client token, the user it authenticates (opaque id), and its expiry.
99///
100/// The raw token is never stored; a leaked session table contains nothing a
101/// client could present. Register this model alongside the app's user model.

103pub struct AuthSession {

105    #[key]
106    pub token_hash: String,
107    /// [`CurrentUser::id`] of the authenticated user.
108    #[index]
109    pub user_id: String,
110    pub expires_at: Timestamp,
111    pub created_at: Timestamp,
112}

116/// `id` is an opaque string so non-UUID keys fit; read it only through

123    pub login: String,

138/// The one authentication seam (ADR-0013).
139///
140/// The default [`PasswordAuth`] implements it against the shipped
141/// [`AdminUser`] model; an app with an existing user table implements it and
142/// passes the value to [`Panel::auth`](crate::Panel::auth) via
143/// [`Auth::custom`]. Sessions are the framework's, so a custom implementation
144/// only maps credentials to a [`CurrentUser`] and back.
145///
146/// Every *credential* failure must return `Ok(None)`, never a distinguishable
147/// error: the login response is one generic message for all of them. An
148/// infrastructure failure is not a credential verdict, so an implementation
149/// that cannot reach its store may return the driver's error instead — the
150/// login handler maps it to the opaque outage page, which is what
151/// keeps a database outage from rendering as a rejected password. An
152/// implementation's own error keeps its own mapping.
153pub trait Authenticator: Send + Sync + 'static {
154    /// Verify `login`/`password`, returning the user on success.
155    ///
156    /// Implementations must run comparable work for unknown accounts (the
157    /// shipped [`PasswordAuth`] verifies a dummy hash) so timing does not leak
158    /// account existence.
159    fn verify<'a>(
160        &'a self,
161        cx: &'a Cx,
162        login: &'a str,
163        password: &'a str,
164    ) -> AuthFuture<'a, Option<CurrentUser>>;
165
166    /// Resolve the live session user by [`CurrentUser::id`].
167    ///
168    /// Loading the row each request is what makes deactivation and revocation
169    /// take effect immediately; return `None` when the user does not
170    /// authenticate.
171    fn find_by_id<'a>(&'a self, cx: &'a Cx, id: &'a str) -> AuthFuture<'a, Option<CurrentUser>>;
172}

174/// The opaque error an infrastructure failure on an auth or session path
175/// carries.
176///
177/// Its `Display` is [`UNAVAILABLE_ERROR`], so nothing driver-shaped can travel
178/// inside it, and its concrete type is what lets the login handler recognise
179/// the one case it answers with the outage page rather than a 500.
180#[derive(Debug)]
181struct Unavailable;
182
183impl std::fmt::Display for Unavailable {
184    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
185        f.write_str(UNAVAILABLE_ERROR)
186    }
187}
188
189impl std::error::Error for Unavailable {}
190
191/// Map a failed auth or session operation to the error the response carries
192/// The counterpart of [`crate::db::hook_failure`] for this module.
193///
194/// An error that is the driver's is an infrastructure failure: it becomes
195/// [`Unavailable`], and the driver's own text goes to the log under this
196/// event, never to the page. Anything else is app-authored (a custom
197/// [`Authenticator`]'s own error) and keeps its mapping, so the seam does not
198/// swallow it. Every auth and session path maps through here, which is what
199/// keeps a database outage from reading as a rejected password — and what
200/// makes the two distinguishable in the logs: a rejection is `Ok(None)`, the
201/// generic 403 and no error line, while an outage logs the driver's text.
202fn infrastructure_failure(error: impl Into<topcoat::Error>) -> topcoat::Error {
203    let error = error.into();
204    if error.is::<toasty::Error>() {
205        tracing::error!(error = %error, "auth infrastructure failure");
206        Unavailable.into()
207    } else {
208        error
209    }
210}

212/// The shipped default authenticator: Argon2id verification against
213/// [`AdminUser`].
214#[derive(Debug, Default, Clone, Copy)]
215pub struct PasswordAuth;
216
217impl PasswordAuth {
218    /// Creates the shipped authenticator.
219    #[must_use]
220    pub fn new() -> Self {
221        Self
222    }
223}
224
225impl Authenticator for PasswordAuth {
226    fn verify<'a>(
227        &'a self,
228        cx: &'a Cx,
229        login: &'a str,
230        password: &'a str,
231    ) -> AuthFuture<'a, Option<CurrentUser>> {
232        Box::pin(async move {
233            let mut db = crate::db::db(cx);
234            let found = AdminUser::filter(AdminUser::fields().email().eq(login.trim().to_string()))
235                .first()
236                .exec(&mut db)
237                .await
238                .map_err(infrastructure_failure)?;
239            let Some(user) = found else {
240                // Unknown account: pay a verification anyway (ADR-0013).
241                let _ = verify_password(password, DUMMY_PASSWORD_HASH);
242                return Ok(None);
243            };
244            if !verify_password(password, &user.password_hash) {
245                return Ok(None);
246            }
247            Ok(Some(current_user_from(&user)))
248        })
249    }
250
251    fn find_by_id<'a>(&'a self, cx: &'a Cx, id: &'a str) -> AuthFuture<'a, Option<CurrentUser>> {
252        Box::pin(async move {
253            let Ok(id) = Uuid::parse_str(id) else {
254                return Ok(None);
255            };
256            let mut db = crate::db::db(cx);
257            let user = AdminUser::filter(AdminUser::fields().id().eq(id))
258                .first()
259                .exec(&mut db)
260                .await
261                .map_err(infrastructure_failure)?;
262            // A deactivated account stops resolving: its live sessions are
263            // purged and the next request redirects to login (spec #127 US11).
264            Ok(user
265                .filter(|user| user.active)
266                .map(|user| current_user_from(&user)))
267        })
268    }
269}

271/// Hash a password with Argon2id into a PHC string (the shipped storage
272/// format). Seeds and record fns call this; it never stores the plaintext.
273pub fn hash_password(password: &str) -> topcoat::Result<String> {
274    use argon2::password_hash::PasswordHasher;
275
276    argon2::Argon2::default()
277        .hash_password(password.as_bytes())
278        .map(|hash| hash.to_string())
279        .map_err(topcoat::Error::from)
280}
281
282/// Verify a password against a PHC hash; `false` on any malformed input.
283fn verify_password(password: &str, phc: &str) -> bool {
284    use argon2::password_hash::{PasswordVerifier, phc::PasswordHash};
285
286    let Ok(parsed) = PasswordHash::new(phc) else {
287        return false;
288    };
289    argon2::Argon2::default()
290        .verify_password(password.as_bytes(), &parsed)
291        .is_ok()
292}

305/// The panel's authentication configuration (ADR-0013).
306///
307/// The default is [`Auth::password`]; [`Auth::custom`] swaps in an app-owned
308/// [`Authenticator`]; [`Auth::disabled`] is the explicit fail-open opt-out.
309pub enum Auth {
310    /// The shipped Argon2id + [`AdminUser`] authenticator.
311    Password(PasswordAuth),
312    /// An app-owned authenticator over its own user table.
313    Custom(Box<dyn Authenticator>),
314    /// Explicit opt-out: no gate, no login routes, sessions unused.
315    Disabled,
316}
317
318impl Auth {
319    /// The shipped default.
320    #[must_use]
321    pub fn password() -> Self {
322        Self::Password(PasswordAuth::new())
323    }
324
325    /// An app-owned authenticator, erased to the trait object the panel keeps.
326    #[must_use]
327    pub fn custom(authenticator: impl Authenticator) -> Self {
328        Self::Custom(Box::new(authenticator))
329    }
330
331    /// Explicit fail-open opt-out for public demos (ADR-0013).
332    #[must_use]
333    pub fn disabled() -> Self {
334        Self::Disabled
335    }
336
337    /// The authenticator, or `None` when auth is explicitly disabled.
338    #[must_use]
339    pub fn authenticator(&self) -> Option<&dyn Authenticator> {
340        match self {
341            Self::Password(password) => Some(password),
342            Self::Custom(authenticator) => Some(&**authenticator),
343            Self::Disabled => None,
344        }
345    }
346
347    /// Whether the explicit opt-out is set.
348    #[must_use]
349    pub fn is_disabled(&self) -> bool {
350        matches!(self, Self::Disabled)
351    }
352}

373/// Require an authenticated, panel-permitted user.
374///
375/// Answers per request kind (ADR-0013): pages redirect to the login route
376/// with a same-origin-relative `next`, runtime endpoints, non-GET requests,
377/// and page re-runs (marked POSTs the runtime layer rewrites into GETs)
378/// answer 401, and an authenticated user without panel access answers 403.
379pub fn require_authenticated(cx: &Cx) -> topcoat::Result<CurrentUser> {
380    if let Some(user) = current_user(cx) {
381        if user.can_access_panel {
382            return Ok(user);
383        }
384        return Err(forbidden().into());
385    }
386    Err(unauthenticated_error(cx))
387}

389/// The error an unauthenticated request answers with, by request kind.
390fn unauthenticated_error(cx: &Cx) -> topcoat::Error {
391    let path = uri(cx).path();
392    let page_method = matches!(*method(cx), http::Method::GET | http::Method::HEAD);
393    // A page re-run reaches the gate as a rewritten GET: Topcoat's runtime
394    // layer rewrites the browser's marked POST into a GET for the page's own
395    // URL. The original request tells it apart from a plain page load, so a
396    // logged-out re-run answers 401 like any other non-page request instead
397    // of redirecting into the login page.
398    let rerun = !matches!(*original_method(cx), http::Method::GET | http::Method::HEAD)
399        && original_headers(cx).get(&topcoat::runtime::RUNTIME_HEADER) == Some(&RERUN_MARKER);
400    if path.starts_with(RUNTIME_PREFIX) || !page_method || rerun {
401        unauthorized().into()
402    } else {
403        redirect(login_url_with_next(cx)).into()
404    }
405}

417/// The login URL with a validated `next` back to the requested page.
418fn login_url_with_next(cx: &Cx) -> String {
419    let mut url = login_url(cx);
420    let request = uri(cx);
421    let requested = match request.query() {
422        Some(query) => format!("{}?{query}", request.path()),
423        None => request.path().to_string(),
424    };
425    if let Some(next) = safe_next(&requested) {
426        let mut serializer = form_urlencoded::Serializer::new(String::new());
427        serializer.append_pair(NEXT_FIELD, next);
428        url.push('?');
429        url.push_str(&serializer.finish());
430    }
431    url
432}

446/// Accept only same-origin relative paths as a post-login destination
447/// (ADR-0013): absolute URLs, scheme-relative `//host` targets, backslash
448/// tricks, and control characters are rejected.
449pub(crate) fn safe_next(next: &str) -> Option<&str> {
450    let next = next.trim();
451    if !next.starts_with('/') || next.starts_with("//") {
452        return None;
453    }
454    if next.contains('\\') || next.chars().any(|c| c.is_control()) {
455        return None;
456    }
457    Some(next)
458}

460/// The validated `?next=` the login page embeds as a hidden field.
461fn next_from_query(cx: &Cx) -> Option<String> {
462    let query = uri(cx).query()?;
463    form_urlencoded::parse(query.as_bytes())
464        .find(|(key, _)| key == NEXT_FIELD)
465        .map(|(_, value)| value.into_owned())
466        .filter(|value| safe_next(value).is_some())
467}

469/// Hex-encode a token hash into its storage key.
470fn token_key(hash: &TokenHash) -> String {
471    use std::fmt::Write as _;
472
473    let mut key = String::with_capacity(64);
474    for byte in hash.iter() {
475        write!(key, "{byte:02x}").expect("writing to a String cannot fail");
476    }
477    key
478}

496/// Revoke every live session of `user_id` (ADR-0013).
497///
498/// Call this whenever a credential changes out from under live sessions —
499/// password reset/change and deactivation alike. Nothing in-core calls it
500/// (there is no password-change flow in the framework); sessions otherwise
501/// stay valid for their full fixed lifetime, so a reset that skips this
502/// leaves a stolen session usable. A password-reset flow must call it,
503/// and custom `Authenticator` apps own the same obligation.
504pub async fn revoke_sessions_for_user(cx: &Cx, user_id: &str) -> topcoat::Result<()> {
505    let mut db = crate::db::db(cx);
506    AuthSession::filter(AuthSession::fields().user_id().eq(user_id.to_string()))
507        .delete()
508        .exec(&mut db)
509        .await
510        .map_err(infrastructure_failure)?;
511    Ok(())
512}

514/// Drop the expired session rows of `user_id`.
515///
516/// [`resolve`] purges a session row when its token is looked up expired, so
517/// without this a row whose token is never presented again would stay in the
518/// table; a sweep that does not wait for the owner is tracked in GH #302.
519/// Login is the bounded sweep: the user is present, the table is already open,
520/// and only their rows are touched. Revocation ([`revoke_sessions_for_user`]) is the
521/// unbounded counterpart that drops the live rows too.
522async fn purge_expired_sessions_for_user(cx: &Cx, user_id: &str) -> topcoat::Result<()> {
523    let now = Timestamp::now();
524    let mut db = crate::db::db(cx);
525    AuthSession::filter(
526        AuthSession::fields()
527            .user_id()
528            .eq(user_id.to_string())
529            .and(AuthSession::fields().expires_at().le(now)),
530    )
531    .delete()
532    .exec(&mut db)
533    .await
534    .map_err(infrastructure_failure)?;
535    Ok(())
536}

538/// Resolve the request's session into a user, lazily and without touching the
539/// database when no session cookie is present.
540pub(crate) async fn resolve(
541    cx: &Cx,
542    authenticator: &dyn Authenticator,
543) -> topcoat::Result<Option<CurrentUser>> {
544    let Some(hash) = session::token_hash(cx).await? else {
545        return Ok(None);
546    };
547    let key = token_key(&hash);
548    let mut db = crate::db::db(cx);
549    let row = AuthSession::filter(AuthSession::fields().token_hash().eq(key))
550        .first()
551        .exec(&mut db)
552        .await
553        .map_err(infrastructure_failure)?;
554    let Some(row) = row else {
555        return Ok(None);
556    };
557    if row.expires_at <= Timestamp::now() {
558        // Expired sessions do not resolve; purge the row on the way out.
559        delete_session_row(cx, &row.token_hash).await?;
560        return Ok(None);
561    }
562    match authenticator
563        .find_by_id(cx, &row.user_id)
564        .await
565        .map_err(infrastructure_failure)?
566    {
567        Some(user) => Ok(Some(user)),
568        None => {
569            // The session names a user who no longer authenticates (deleted
570            // or deactivated): purge it so removal is real (US11).
571            delete_session_row(cx, &row.token_hash).await?;
572            Ok(None)
573        }
574    }
575}

577/// The layer that gates the panel and runtime prefixes (ADR-0013): resolves

587        Self {
588            path: route_path(path),
589        }
590    }

592
593impl Layer for AuthGate {
594    fn path(&self) -> Option<&Path> {
595        Some(&self.path)
596    }
597
598    fn handle<'a>(&'a self, cx: &'a Cx, body: Body, next: Next<'a>) -> LayerFuture<'a> {
599        Box::pin(async move {
600            // The login page must answer while logged out.
601            if uri(cx).path() == login_url(cx) {
602                return next.run(cx, body).await;
603            }
604            let auth = app_context::<Auth>(cx);
605            let Some(authenticator) = auth.authenticator() else {
606                return next.run(cx, body).await;
607            };
608            // The logout route must answer for any resolved user, even one
609            // whose panel access was revoked after login — clearing
610            // the session row + cookie must not require panel permission, or
611            // the session lingers to expiry. The bypass is POST-only at the
612            // exact logout path: the route table registers nothing else
613            // there, and a non-POST method must not smuggle a resolved
614            // identity to any handler an app might mount at the same path.
615            let logout_route =
616                uri(cx).path() == logout_url(cx) && matches!(*method(cx), http::Method::POST);
617            match resolve(cx, authenticator).await? {
618                Some(user) if user.can_access_panel || logout_route => {
619                    // The logged-in user's optional tenant becomes the request
620                    // tenant; auth never requires one (ADR-0013).
621                    let tenant_id = user.tenant_id;
622                    let mut child = cx.with(user);
623                    if let Some(tenant_id) = tenant_id {
624                        child = child.with(crate::tenancy::Tenant(tenant_id));
625                    }
626                    next.run(&child, body).await
627                }
628                // Authenticated but not permitted: 403, indistinguishable
629                // from bad credentials at login (ADR-0013). The logout route
630                // is answered above.
631                Some(_) => Err(forbidden().into()),
632                // Pages redirect to the login route with a validated `next`;
633                // runtime endpoints, non-GET requests, and page re-runs
634                // answer 401.
635                None => Err(unauthenticated_error(cx)),
636            }
637        })
638    }
639}

652/// What a failed login attempt renders: the generic credential rejection, or
653/// the sign-in outage.
654///
655/// Two variants and no more, each with its own copy and status, so a failed
656/// attempt is always a deliberate answer: a driver failure is never rendered
657/// as a credential verdict, and a credential verdict never borrows the outage
658/// copy. Neither variant carries driver text.
659#[derive(Debug, Clone, Copy)]
660enum LoginError {
661    /// Wrong password, unknown account, empty fields, or valid credentials
662    /// without panel access: one 403 with one message (ADR-0013).
663    Credentials,
664    /// The database behind sign-in could not answer: a 503 that says so,
665    /// because the user must not read an outage as a rejected password.
666    Unavailable,
667}
668
669impl LoginError {
670    /// The message the login page's alert carries.
671    fn message(self) -> &'static str {
672        match self {
673            Self::Credentials => GENERIC_ERROR,
674            Self::Unavailable => UNAVAILABLE_ERROR,
675        }
676    }
677
678    /// The status a failed attempt answers with.
679    fn status(self) -> http::StatusCode {
680        match self {
681            Self::Credentials => http::StatusCode::FORBIDDEN,
682            Self::Unavailable => http::StatusCode::SERVICE_UNAVAILABLE,
683        }
684    }
685}
686
687/// The status and route a login attempt renders: a [`LoginError`] page or a
688/// redirect back to `next`.
689///
690/// The login page is a settled view, so [`ViewExt::single`] resolves it into
691/// an owned handle before the response is built — no borrowed view escapes.
692async fn login_response(
693    cx: &Cx,
694    error: Option<LoginError>,
695    next: String,
696) -> topcoat::Result<topcoat::router::response::Response> {
697    let page = render_login_page(cx, error, next).await?;
698    page.single().await?.into_response(cx)
699}

703    let next = next_from_query(cx).unwrap_or_default();

707/// `POST {prefix}/login` — verify, rotate the session, redirect to `next`.
708pub(crate) fn login_post(cx: &Cx, body: Body) -> RouteFuture<'_> {
709    Box::pin(async move {
710        // Login/logout carry no file parts: the values half is all they read.
711        let values = crate::panel::parse_form_body(cx, body).await?.values;
712        crate::csrf::verify(cx, &values)?;
713        // Keep a validated destination across a failed attempt so the retry
714        // form still returns where the visitor was headed (US6).
715        let next = values
716            .get(NEXT_FIELD)
717            .and_then(|value| safe_next(value))
718            .map(str::to_string)
719            .or_else(|| next_from_query(cx))
720            .unwrap_or_default();
721        let email = values.get(LOGIN_FIELD).map(|value| value.trim());
722        let password = values.get(PASSWORD_FIELD).map(String::as_str);
723        let auth = app_context::<Auth>(cx);
724        let verified = match (auth.authenticator(), email, password) {
725            (Some(authenticator), Some(email), Some(password))
726                if !email.is_empty() && !password.is_empty() =>
727            {
728                match authenticator.verify(cx, email, password).await {
729                    Ok(user) => user,
730                    // A driver failure is not a credential verdict: the page
731                    // says sign-in is unavailable instead of rendering a
732                    // rejection. The seam logs the driver's text and
733                    // hands back an app-authored error untouched.
734                    Err(error) => {
735                        let error = infrastructure_failure(error);
736                        if error.is::<Unavailable>() {
737                            return login_response(cx, Some(LoginError::Unavailable), next).await;
738                        }
739                        return Err(error);
740                    }
741                }
742            }
743            _ => None,
744        };
745        // One path for every failure: wrong password, unknown account, empty
746        // fields, or valid credentials without panel access (ADR-0013).
747        let Some(user) = verified.filter(|user| user.can_access_panel) else {
748            return login_response(cx, Some(LoginError::Credentials), next).await;
749        };
750        // Rotate on login: a token this request presented cannot be replayed.
751        if let Some(hash) = session::token_hash(cx).await? {
752            delete_session(cx, &hash).await?;
753        }
754        // Bounded housekeeping: the expired rows of the user signing
755        // in go with the rotation. A failure is logged rather than fatal — a
756        // credential that verified must not become a 503 because cleanup could
757        // not run.
758        if let Err(error) = purge_expired_sessions_for_user(cx, &user.id).await {
759            tracing::error!(error = %error, "expired-session purge failed");
760        }
761        let session = session::start(cx).await?;
762        let mut db = crate::db::db(cx);
763        let recorded = toasty::create!(AuthSession {
764            token_hash: token_key(&session.token_hash),
765            user_id: user.id.clone(),
766            expires_at: Timestamp::try_from(session.expires_at).map_err(topcoat::Error::from)?,
767            created_at: Timestamp::now(),
768        })
769        .exec(&mut db)
770        .await;
771        if let Err(error) = recorded {
772            // The credentials were right; the session row could not be
773            // recorded. Same outage page as a failed verification — never the
774            // driver's text.
775            let error = infrastructure_failure(error);
776            if error.is::<Unavailable>() {
777                return login_response(cx, Some(LoginError::Unavailable), next).await;
778            }
779            return Err(error);
780        }
781        let target = if next.is_empty() {
782            panel_root(cx)
783        } else {
784            next
785        };
786        // Success stays on the `Ok` path so `Set-Cookie` flushes
787        // (upstream topcoat#126).
788        topcoat::router::error::see_other(target).into_response(cx)
789    })
790}

792/// `POST {prefix}/logout` — delete the session row and clear the cookie.
793pub(crate) fn logout_post(cx: &Cx, body: Body) -> RouteFuture<'_> {
794    Box::pin(async move {
795        // Defense in depth: the route only exists on gated panels, but it
796        // re-checks so a missing layer cannot leave logout ungated. Any
797        // resolved identity may log out — the gate answers this route for a
798        // `can_access_panel=false` user too, so demanding panel
799        // access here would strand their session row + cookie to expiry.
800        if current_user(cx).is_none() {
801            return Err(unauthenticated_error(cx));
802        }
803        // Login/logout carry no file parts: the values half is all they read.
804        let values = crate::panel::parse_form_body(cx, body).await?.values;
805        crate::csrf::verify(cx, &values)?;
806        if let Some(hash) = session::stop(cx).await? {
807            delete_session(cx, &hash).await?;
808        }
809        let target = login_url(cx);
810        topcoat::router::error::see_other(target).into_response(cx)
811    })
812}
