1//! Tenancy via the `Cx` scoped value `Tenant(id)`.
2//!
3//! Since GH #223 the framework owns the tenant *filter*, not only the tenant
4//! gate. For a resource whose `requires_tenant()` is `true`, every loader runs
5//! [`scoped_query`](crate::resource::scoped_query) — the resource's own
6//! [`query`](crate::resource::Resource::query) with `tenant_id = <tenant>`
7//! ANDed onto it — and the column is discovered from the model's schema here
8//! (`derived_tenant_filter`, the default body of `Resource::tenant_scope`),
9//! not named by hand in each resource. A gated resource
10//! therefore cannot serve unscoped rows by forgetting an override, and
11//! `Resource::query` is left as the app's *non-tenant* scoping seam (soft
12//! deletes, includes, row-level visibility). ADR-0002's 2026-09-22 note records
13//! the change.
14//!
15//! Discovery is deliberately narrow and fails closed: the model must declare a
16//! field whose application name is `tenant_id` and whose type is a UUID.
17//! Anything else — no such field, or a same-named field of another type — reads
18//! as "no tenant column", and a gated resource that hits that is refused by
19//! `Panel::build` at boot (GH #231), because the declaration is checkable
20//! without a request. A predicate that is only `None` for some tenants still
21//! answers an error naming the resource rather than querying unscoped.
22//!
23//! The authenticated user's tenant is the production source: the auth layer
24//! (ADR-0013) injects `Tenant` into the request `Cx` when the logged-in user
25//! carries one. A `Tenant` request extension — server-set only, never a header
26//! — takes precedence, so app middleware and `Router::handle` tests can
27//! override it deliberately. The `x-tenant-id` header fallback was removed in
28//! GH #131: learning another tenant's UUID no longer makes anyone that tenant.
29
30use toasty::stmt::Expr;
31use topcoat::context::{Cx, try_request_context};
32
33/// Request-scoped tenant identifier.
34///
35/// The auth layer sets it via `cx.with(Tenant(id))` from the logged-in user;
36/// server code and tests may also carry it as a request extension.
37#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
38pub struct Tenant(pub uuid::Uuid);
39
40/// Returns the tenant id from `cx`, if present.
41///
42/// Checks a `Tenant` request extension first (a server-set override — the
43/// deliberate seam app middleware and `Router::handle` tests use), then the
44/// `Tenant` scoped value the auth layer injects from the authenticated user.
45/// No request header is consulted (GH #131).
46pub fn tenant_id(cx: &Cx) -> Option<uuid::Uuid> {
47    if let Some(parts) = try_request_context::<http::request::Parts>(cx)
48        && let Some(t) = parts.extensions.get::<Tenant>()
49    {
50        return Some(t.0);
51    }
52    try_request_context::<Tenant>(cx).map(|t| t.0)
53}
54
55/// Requires a tenant, returning an error if missing (for tenancy-gated resources).
56pub fn require_tenant(cx: &Cx) -> Result<uuid::Uuid, topcoat::Error> {
57    tenant_id(cx).ok_or_else(|| topcoat::router::error::forbidden().into())
58}

60/// The application name of the column the framework scopes a gated resource by
61/// (GH #223).
62///
63/// The convention is public — it is the contract a model signs up to when its
64/// resource declares
65/// [`requires_tenant`](crate::resource::Resource::requires_tenant) — and this
66/// is the one place the discovery reads it.
67const TENANT_FIELD: &str = "tenant_id";
68
69/// Position of `M`'s tenant column in its own schema, or `None` when `M`
70/// declares none the framework can recognize (GH #223).
71///
72/// Found by **name and type** over [`Model::schema`]'s field list: a primitive
73/// field whose application name is `tenant_id` and whose type is
74/// [`toasty::stmt::Type::Uuid`]. The index is taken from the field's own
75/// [`FieldId`](toasty::schema::app::FieldId) — the index
76/// [`Model::path_field`] addresses — rather than from the position in the
77/// vector, so the filter and the generated `tenant_id()` accessor cannot drift.
78///
79/// Name-based discovery is the fragile half, so a miss is `None` and every
80/// caller treats it as *fail closed*: nothing falls back to the unscoped query,
81/// and nothing guesses. A `tenant_id` declared as `String`, or a tenant column
82/// spelled any other way, is invisible here on purpose — comparing a UUID
83/// against it would be a driver-level type error at best and a cross-tenant
84/// match at worst.
85pub(crate) fn tenant_field_index<M: toasty::schema::Model>() -> Option<usize> {
86    M::schema()
87        .fields()
88        .iter()
89        .find(|field| {
90            field.name.app.as_deref() == Some(TENANT_FIELD)
91                && field
92                    .ty
93                    .as_primitive()
94                    .is_some_and(|primitive| primitive.ty.is_uuid())
95        })
96        .map(|field| field.id.index)
97}
98
99/// The **derived** `tenant_id = tenant` over `M`, or `None` when
100/// [`tenant_field_index`] finds no tenant column (GH #223).
101///
102/// This is the default body of
103/// [`Resource::tenant_scope`](crate::resource::Resource::tenant_scope) — the
104/// public hook a resource overrides when its rows inherit their tenant instead
105/// of carrying one — so the name says *derived*: it is the name-based
106/// convenience, not the only way to scope a gated resource.
107///
108/// Built generically on purpose: `Model::path_field` addresses the column by
109/// index and `Value::Uuid` types the comparison, so no generated accessor — and
110/// no per-resource copy of the filter — is needed.
111pub(crate) fn derived_tenant_filter<M: toasty::schema::Model>(
112    tenant: uuid::Uuid,
113) -> Option<Expr<bool>> {
114    let index = tenant_field_index::<M>()?;
115    Some(M::path_field::<uuid::Uuid>(index).eq(tenant))
116}

123    fn cx_with_header(value: &str) -> Cx {
124        let mut parts = http::Request::builder()
125            .uri("/")
126            .body(())
127            .unwrap()
128            .into_parts()
129            .0;
130        parts.headers.insert(
131            "x-tenant-id",
132            value.parse().expect("header value must parse"),
133        );
134        CxTestBuilder::new().request_context(parts).build()
135    }

137    #[test]
138    fn tenant_extension_wins_over_scoped_value_and_defaults_to_none() {
139        let id = uuid::Uuid::new_v4();
140        let other = uuid::Uuid::new_v4();
141
142        // The auth layer's scoped value is the production source.
143        let cx = CxTestBuilder::new().request_context(Tenant(id)).build();
144        assert_eq!(tenant_id(&cx), Some(id));
145
146        // A server-set request extension overrides it deliberately.
147        let cx = CxTestBuilder::new()
148            .request_context({
149                let mut parts = http::Request::builder()
150                    .uri("/")
151                    .body(())
152                    .unwrap()
153                    .into_parts()
154                    .0;
155                parts.extensions.insert(Tenant(id));
156                parts
157            })
158            .request_context(Tenant(other))
159            .build();
160        assert_eq!(tenant_id(&cx), Some(id));
161
162        // Nothing set → None (callers must reject tenantless access).
163        let cx = CxTestBuilder::new().build();
164        assert_eq!(tenant_id(&cx), None);
165    }

167    #[test]
168    fn tenant_header_is_never_trusted() {
169        // GH #131: the header fallback is gone; spoofing another tenant's
170        // UUID in `x-tenant-id` must not resolve a tenant.
171        let id = uuid::Uuid::new_v4();
172        assert_eq!(tenant_id(&cx_with_header(&id.to_string())), None);
173    }

202    #[test]
203    fn tenant_column_is_discovered_by_name_and_uuid_type() {
204        // `id` is index 0, `tenant_id` index 1, `name` index 2.
205        assert_eq!(tenant_field_index::<Scoped>(), Some(1));
206        // No column at all, and a `tenant_id` that is not a UUID: both are
207        // "cannot scope", never "scope by something else" (GH #223).
208        assert_eq!(tenant_field_index::<Unscoped>(), None);
209        assert_eq!(tenant_field_index::<WronglyTyped>(), None);
210    }

212    /// The derived filter is only real if it reaches SQL: the discovered index
213    /// and the UUID comparison must narrow a live query (GH #223).
214    #[tokio::test]
215    async fn derived_tenant_filter_scopes_a_live_query() {
216        let mut db = toasty::Db::builder()
217            .models(toasty::models!(Scoped))
218            .connect("sqlite::memory:")
219            .await
220            .unwrap();
221        db.push_schema().await.unwrap();
222        let mine = uuid::Uuid::new_v4();
223        let theirs = uuid::Uuid::new_v4();
224        toasty::create!(Scoped {
225            tenant_id: mine,
226            name: "Mine"
227        })
228        .exec(&mut db)
229        .await
230        .unwrap();
231        toasty::create!(Scoped {
232            tenant_id: theirs,
233            name: "Theirs"
234        })
235        .exec(&mut db)
236        .await
237        .unwrap();
238
239        let filter = derived_tenant_filter::<Scoped>(mine).expect("Scoped declares tenant_id");
240        let rows = toasty::stmt::Query::<toasty::stmt::List<Scoped>>::all()
241            .filter(filter)
242            .exec(&mut db)
243            .await
244            .unwrap();
245        assert_eq!(rows.len(), 1);
246        assert_eq!(rows[0].name, "Mine");
247
248        // No discoverable column → no filter → the caller must fail rather than
249        // run the query (GH #223).
250        assert!(derived_tenant_filter::<Unscoped>(mine).is_none());
251    }
