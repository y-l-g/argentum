21/// Retry link for a failed streamed table load (GH #110).
22///
23/// A malformed `?after=`/`?before=` cursor, a conflicting `after` + `before`
24/// pair (GH #155), or a cursor the query's ordering refuses (GH #294) is the
25/// failure itself: retrying the identical URL loops forever, so drop
26/// pagination from the link and keep the rest of the state
27/// (search/sort/filters/grouping). Every other failure keeps pagination too
28/// (GH #98) so a transient blip retries the same evidence.
29pub(crate) fn retry_url_for_error(
30    state: &TableState,
31    error: &topcoat::Error,
32    path: &str,
33) -> String {
34    if crate::cursor::is_cursor_error(error) {
35        state.without_cursor(path)
36    } else {
37        state.list_url(path)
38    }
39}

41/// The action chrome a resource declares (GH #207): the one derivation both
42/// [`wire_table_actions`] and the build-time declaration check
43/// ([`check_resource`](super::check_resource)) read, so the table the panel
44/// serves and the table it validated cannot disagree about whether a record
45/// key is required.
46pub(crate) fn declared_chrome<R: Resource>(cx: &Cx) -> TableChrome {
47    TableChrome {
48        delete: R::deletable(),
49        edit: R::editable(),
50        // GH #187: the View link follows the declaration, not a flag — a
51        // resource with no `view` schema has no page to link to.
52        view: R::viewed(cx),
53    }
54}

56/// Delete/bulk/edit action wiring shared by the streamed list and the
57/// live-search shard (GH #134): the delete form posts to `{list url}/{id}/delete`
58/// and the bulk bar to `{list url}/bulk-delete`, both derived from the panel
59/// declaration (not the request path) so the URLs are right wherever the table
60/// renders.
61///
62/// Chrome is opt-in (GH #226) and each affordance is gated by the flag that
63/// promises it — `deletable()` for row + bulk delete, `editable()` for the
64/// per-row Edit link — because the alternative ships Edit/Delete controls whose
65/// actions always answer 403. A resource that wants them declares the flag and
66/// the matching policy predicate together; see [`Resource::deletable`].
67///
68/// The flags are the coarse gate; the per-*record* gate rides the same call
69/// (GH #235). The table's row policy is wired from the resource's predicates,
70/// each action paired with exactly what its route checks: `can_view` for View,
71/// `can_view` + `can_update` for Edit (GH #86), `can_view` + `can_delete` for
72/// Delete and the bulk checkbox (GH #168). A row the predicate refuses renders
73/// no link and a disabled checkbox, while the handler keeps its all-or-nothing
74/// check as the safety net for a hand-crafted POST.
75///
76/// `live` selects the shard variant: the swapped region is everything except
77/// the toolbar the page owns eagerly (the live host owns those slots, so swaps
78/// must never nest invocations or duplicate inputs), hence the shard forces
79/// `.search(false).filter_bar(false)` while the streamed page keeps the declared
80/// table as-is. The filter bar joins the search toolbar there (GH #166): a
81/// control rebuilt by its own rerun loses focus.
82pub(crate) fn wire_table_actions<R: Resource>(cx: &Cx, live: bool) -> Table<R::Model> {
83    let mut table = R::table(cx);
84    if live {
85        table = table.search(false).filter_bar(false);
86    }
87    // `Cx` is Arc-backed and `Clone`, so the projection owns one: the policy
88    // outlives the request borrow without copying request state.
89    let policy_cx = cx.clone();
90    table = table.row_actions(move |record| {
91        // Read once: every route pairs its own predicate with `can_view`, so a
92        // record that cannot be viewed allows no action (GH #168).
93        let view = R::can_view(&policy_cx, record);
94        RowActions {
95            view,
96            edit: view && R::can_update(&policy_cx, record),
97            delete: view && R::can_delete(&policy_cx, record),
98        }
99    });
100    let chrome = declared_chrome::<R>(cx);
101    if chrome.delete {
102        table = table
103            .with_delete(list_url(cx, &R::slug()))
104            .with_bulk_delete(true);
105    }
106    if chrome.edit {
107        table = table.with_edit(list_url(cx, &R::slug()));
108    }
109    if chrome.view {
110        table = table.with_view(list_url(cx, &R::slug()));
111    }
112    table
113}

115/// Branded in-region table failure shared by the streamed list and the
116/// live-search shard (GH #134, GH #158): the trace line, the cursor-aware
117/// retry link ([`retry_url_for_error`]), and the `ErrorState` render are one
118/// copy so the three load sites cannot drift.
119///
120/// On a live table (`signals`) the retry stays in place (GH #166) instead of
121/// navigating, and re-runs the request that failed with the query it failed
122/// with: the click increments a retry token the shard reads (GH #294), so the
123/// rerun does not depend on the query signals changing — a write of an
124/// unchanged value re-runs nothing. A cursor failure resets only the cursor,
125/// the same reset its `href` spells out; every other failure keeps the whole
126/// query. `href` stays as the no-JS fallback, so it retries the URL as it
127/// stands.
128pub(crate) fn table_error_view<'a, R: Resource>(
129    cx: &'a Cx,
130    state: &TableState,
131    error: &topcoat::Error,
132    path: &str,
133    signals: Option<&TableSignals>,
134) -> BoxView<'a> {
135    tracing::error!(resource = R::slug(), error = %error, "table load failed");
136    let retry_url = retry_url_for_error(state, error, path);
137    let action: BoxView<'a> = match signals {
138        Some(signals) => {
139            // Retry re-runs the shard (GH #294). The token is the rerun's only
140            // cause: the shard reads it (declaring the dependency below), and
141            // every click increments it, so the write always changes even when
142            // the query signals already hold the values that failed.
143            let attempt = topcoat::runtime::signal(cx, || 0u64);
144            let cursor = signals.cursor.clone();
145            let none = crate::resource::cursor_none();
146            let cursor_error = crate::cursor::is_cursor_error(error);
147            let attrs = if cursor_error {
148                attributes! {
149                    cx =>
150                    href=(retry_url)
151                    data-retry-attempt=(attempt.get())
152                    @click=$(|e: Event| {
153                        e.prevent_default();
154                        cursor.set(none.clone());
155                        attempt.increment();
156                    })
157                }
158            } else {
159                attributes! {
160                    cx =>
161                    href=(retry_url)
162                    data-retry-attempt=(attempt.get())
163                    @click=$(|e: Event| {
164                        e.prevent_default();
165                        attempt.increment();
166                    })
167                }
168            };
169            view! { cx => <a (attrs)>"Retry"</a> }.boxed()
170        }
171        None => view! { cx => <a href=(retry_url)>"Retry"</a> }.boxed(),
172    };
173    view! {
174        cx =>
175        argentum_ui::error_state(
176            title: format!("Couldn't load {}", R::navigation_label()),
177            detail: "Something went wrong while loading the records.",
178            action: Some(action.into()),
179            attrs: attributes! { role="alert" }
180        )
181    }
182    .boxed()
183}

185/// The list page every declared [`Resource`] gets at `{prefix}/{slug}`.
186///
187/// One generic handler drives all resources: resolve the [`TableState`] from
188/// the URL, scope through the tenant-scoped query (GH #223),
189/// apply the table's search/sort/pagination declarations, render through
190/// `Resource::table`. The page title is the resource's navigation label.
191///
192/// The page streams: shell and header go out with the first content, while the
193/// table body (toolbar/filter/bulk/pager included) loads inside a `suspense`
194/// region that swaps in the skeleton → table without any client-side fetching
195/// (GH #98: the skeleton is thead + placeholders only, so chrome pops in with
196/// the swap by design).
197pub(crate) fn resource_list<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
198    Box::pin(HoistView::new(ThenView::new(async move {
199        enforce_auth(cx)?;
200        enforce_tenant::<R>(cx)?;
201        if !R::can_view_any(cx) {
202            return Err(forbidden().into());
203        }
204        // Ensure the CSRF cookie before streaming starts (GH #99): streamed
205        // children can only read it via current_token.
206        crate::csrf::ensure_token(cx);
207        let state = TableState::from_cx(cx);
208        let table = wire_table_actions::<R>(cx, false);
209        let title = R::navigation_label();
210        let list_path = list_url(cx, &R::slug());
211        // Create entry point (GH #162, Filament's List page `CreateAction` in
212        // the page header): a real link so no-JS keeps working. Gated on
213        // `can_create`; the POST handler enforces it again.
214        let create_url = R::can_create(cx).then(|| create_page_url(&list_path));
215        let create_label = format!("Create {}", R::navigation_label());
216        if table.is_live_search() {
217            return Ok(resource_list_live::<R>(cx, table, state, title, list_path));
218        }
219
220        // The skeleton table (`Table::render_skeleton_normalized`) streams
221        // while the rows load below.
222        // The load catches its own errors: post-stream
223        // the status line is fixed, so a failed load must render the branded
224        // ErrorState inside the region instead of truncating the body.
225        // Pre-stream failures (e.g. the skeleton itself) still propagate and
226        // map onto the response status. (For children that partially stream
227        // before failing, topcoat's `error_boundary` is the replace-in-place
228        // seam.)
229        //
230        // One normalization per request (GH #224): the skeleton, the load,
231        // the render and the retry link all read the state this page parsed,
232        // so it normalizes here and every seam below takes the proof (GH
233        // #153: the retry link must not echo an unknown `?group_by=`).
234        let state = table.normalize_state(&state);
235        let skeleton = table.render_skeleton_normalized(cx, &state).await?;
236        let lazy_rows = ThenView::new(async move {
237            let rendered = async {
238                let page = load_table_page::<R>(cx, &table, &state).await?;
239                table.render_normalized(cx, page, &state, &list_path).await
240            };
241            match rendered.await {
242                Ok(view) => Ok(view),
243                Err(error) => Ok(table_error_view::<R>(cx, &state, &error, &list_path, None)),
244            }
245        });
246
247        Ok(view! {
248            cx =>
249            argentum_ui::page(
250                argentum_ui::page_header(
251                    <div class="flex items-center justify-between gap-4">
252                        argentum_ui::page_title((title.clone()))
253                        if let Some(url) = create_url {
254                            <a
255                                href=(url)
256                                class=(argentum_ui::button_variants(
257                                    argentum_ui::ButtonVariant::Primary,
258                                    argentum_ui::ButtonSize::Md,
259                                ))
260                            >
261                                (create_label)
262                            </a>
263                        }
264                    </div>
265                )
266                argentum_ui::page_content(
267                    <div class="flex flex-col gap-4">
268                        suspense(fallback: skeleton, (lazy_rows.boxed()))
269                    </div>
270                )
271            )
272        }
273        .boxed())
274    })))
275}

277/// Live list page for `Table::live_search` tables (GH #104, GH #151): the
278/// page owns the interaction signals (`q`, `filters`, `sort`, `dir`,
279/// `cursor`, `group_by`, `bulk`) and renders the search toolbar eagerly above
280/// the streamed region while the `table_search` shard invocation fills the
281/// table below — one table per response, so rows can never duplicate. Every
282/// interaction writes a signal, so search, sort, filters, and pagination
283/// re-render only the invocation output, morphing in place with focus and
284/// scroll surviving.
285pub(crate) fn resource_list_live<R: Resource>(
286    cx: &Cx,
287    table: Table<R::Model>,
288    state: TableState,
289    title: String,
290    list_path: String,
291) -> BoxView<'_> {
292    Box::pin(HoistView::new(ThenView::new(async move {
293        // One state→signal conversion (GH #224), seeded from the state the
294        // page parsed — before normalizing, so an unknown `?group_by=` seeds
295        // the signal as written and is dropped on the way back in.
296        let signals = state.to_signals(cx);
297        // One normalization per request (GH #224): the toolbar, the hoisted
298        // filter bar, the skeleton, the dialog and the retry link all read
299        // the state this page parsed, so it normalizes here and every seam
300        // below takes the proof (GH #153: no unknown `?group_by=` in a link).
301        let state = table.normalize_state(&state);
302        let host = if table.search_enabled() {
303            Some(
304                table
305                    .render_live_search_bar_normalized(cx, &state, &list_path, &signals)
306                    .await?,
307            )
308        } else {
309            None
310        };
311        // The filter bar is hoisted next to the search host (GH #166): a
312        // `<select>` change re-renders the table, and a control inside the
313        // swapped region would lose focus and collapse its popup mid-change.
314        let filter_bar = if table.filter_bar_enabled() {
315            Some(
316                table
317                    .render_live_filter_bar_normalized(cx, &state, &list_path, &signals)
318                    .await?,
319            )
320        } else {
321            None
322        };
323        let skeleton = table.render_skeleton_normalized(cx, &state).await?;
324        // The delete confirmation dialog is not part of the swapped table
325        // region: a keystroke starts a new result set and must never carry
326        // (or re-open) a dialog, so the live page renders it eagerly once
327        // (GH #151).
328        let delete_dialog = table.render_delete_dialog_normalized(cx, &state).await?;
329        // Create entry point (GH #162): same header button as the streamed
330        // list — a real link above the swapped region, gated on `can_create`.
331        let create_url = R::can_create(cx).then(|| create_page_url(&list_path));
332        let create_label = format!("Create {}", R::navigation_label());
333        let lazy_rows = ThenView::new(async move {
334            // The retry link inside the table writes the same signals the
335            // toolbar does (GH #166), so a bad cursor recovers in place.
336            let retry_signals = signals.clone();
337            let rendered = table.render_live_invocation(cx, &list_path, signals).await;
338            match rendered {
339                Ok(view) => Ok(view),
340                Err(error) => Ok(table_error_view::<R>(
341                    cx,
342                    &state,
343                    &error,
344                    &list_path,
345                    Some(&retry_signals),
346                )),
347            }
348        });
349
350        Ok(view! {
351            cx =>
352            argentum_ui::page(
353                argentum_ui::page_header(
354                    <div class="flex items-center justify-between gap-4">
355                        argentum_ui::page_title((title.clone()))
356                        if let Some(url) = create_url {
357                            <a
358                                href=(url)
359                                class=(argentum_ui::button_variants(
360                                    argentum_ui::ButtonVariant::Primary,
361                                    argentum_ui::ButtonSize::Md,
362                                ))
363                            >
364                                (create_label)
365                            </a>
366                        }
367                    </div>
368                )
369                argentum_ui::page_content(
370                    <div class="flex flex-col gap-4">
371                        if let Some(host) = host {
372                            (host)
373                        }
374                        if let Some(bar) = filter_bar {
375                            (bar)
376                        }
377                        suspense(fallback: skeleton, (lazy_rows.boxed()))
378                        if let Some(dialog) = delete_dialog {
379                            (dialog)
380                        }
381                    </div>
382                )
383            )
384        })
385    })))
386}

388/// Resolve the declared table (search / filters / sort / pagination) against
389/// the tenant-scoped [`scoped_query_with`](crate::resource::scoped_query_with)
390/// — the data-loading half of
391/// [`resource_list`], kept separate so the page shell can stream before it.
392///
393/// The load asks for the includes the table's columns declared
394/// ([`Table::include_needs`], GH #298), so a resource that narrows its loaders
395/// loads exactly what the rendered cells read; the cursor-existence probes ask
396/// for none, because they only test whether a row exists. A resource that
397/// overrides nothing keeps its full
398/// [`query`](crate::resource::Resource::query) at both, the same safe default
399/// the export takes.
400///
401/// Resource lists must declare a page size (GH #172): without
402/// [`Table::paginate`] the load would be an unbounded `exec`, so the missing
403/// declaration fails loudly here — like a missing row key at render — instead
404/// of silently loading the whole table. Page-owned tables (the showcase
405/// demos, GH #154 §2) load through [`Table::load`] directly and keep the
406/// unbounded branch for previews.
407pub(crate) async fn load_table_page<R: Resource>(
408    cx: &Cx,
409    table: &Table<R::Model>,
410    state: &TableState,
411) -> Result<TablePage<R::Model>> {
412    if table.page_size().is_none() {
413        return Err(std::io::Error::other(
414            "resource list requires Table::paginate(..) — unbounded tables are previews only (GH #172)",
415        )
416        .into());
417    }
418    table
419        .load_with_probe(
420            cx,
421            crate::resource::scoped_query_with::<R>(cx, &table.include_needs())?,
422            crate::resource::scoped_query_with::<R>(cx, &crate::resource::IncludeNeeds::default())?,
423            state,
424        )
425        .await
426}

612        /// The signal id a rendered refresh control writes (GH #234): read
613        /// from the control's own `data-topcoat-on:change` handler, which is
614        /// the side that re-runs the shard. Locating it by offset from the
615        /// marker instead would read whatever payload happened to follow.
616        fn revision_signal_id(html: &str) -> &str {
617            const MARKER: &str = r#"id&quot;:&quot;"#;
618            let at = html
619                .find("data-table-revision")
620                .unwrap_or_else(|| panic!("the refresh control, got {html}"));
621            let tag_start = html[..at].rfind('<').expect("the control's opening tag");
622            // The tag runs to the next `<`: an attribute value escapes its own
623            // `>` (the handler's `=&gt;`), so the first `>` is not the tag's.
624            let next = html[tag_start + 1..]
625                .find('<')
626                .map(|i| tag_start + 1 + i)
627                .unwrap_or(html.len());
628            let tag = &html[tag_start..next];
629            let write = tag
630                .find("data-topcoat-on:change")
631                .unwrap_or_else(|| panic!("the control's write handler, got {tag}"));
632            let id_at = tag[write..]
633                .find(MARKER)
634                .unwrap_or_else(|| panic!("the handler's signal, got {tag}"))
635                + write
636                + MARKER.len();
637            let id = &tag[id_at..id_at + 32];
638            assert!(
639                id.chars().all(|c| c.is_ascii_hexdigit()),
640                "the handler must name a signal id, got {tag}"
641            );
642            id
643        }
644        // Args are positional shard inputs: q, filters, sort, dir, the single
645        // cursor wire (GH #166), group_by, and the bulk handle the table binds
646        // its selection transport to.
647        let shard_args =
648            |path: &str, q: &str, filters: &str, sort: &str, dir: &str, cursor: &str| {
649                format!(
650                    r#"["{path}",{}, {}, {}, {}, {}, {}, {}]"#,
651                    sig(1, q),
652                    sig(2, filters),
653                    sig(3, sort),
654                    sig(4, dir),
655                    sig(5, cursor),
656                    sig(6, ""),
657                    sig(7, "")
658                )
659            };
660        async fn call_shard(
661            router: &topcoat::router::Router,
662            args: String,
663        ) -> http::Response<Body> {
664            router
665                .handle(
666                    http::Request::builder()
667                        .method(http::Method::POST)
668                        .uri(TABLE_SEARCH_PATH)
669                        .header(http::header::CONTENT_TYPE, "application/json")
670                        .header(topcoat::router::request::IDENTITY_HEADER, "A".repeat(22))
671                        .body(Body::from(format!(r#"{{"args":{args},"signals":{{}}}}"#)))
672                        .unwrap(),
673                )
674                .await
675        }
676        let nope = call_shard(&router, shard_args("/admin/nope", "Ada", "", "", "", "")).await;
677        assert!(

679            "unknown shard path must fail, got {}",

1556    /// The `(record key, name cell)` pairs `html` renders, in document order.
1557    ///
1558    /// A test cannot assume the seeding order — a paginated table with no
1559    /// sortable column orders by the PK fallback, and the keys are random — so
1560    /// it reads each row's own cells.
1561    fn rendered_rows(html: &str) -> Vec<(String, String)> {
1562        let mut rows = Vec::new();
1563        let mut rest = html;
1564        while let Some(at) = rest.find("data-row-select") {
1565            let start = rest[..at]
1566                .rfind("<input")
1567                .expect("the marker's opening tag");
1568            let tag = &rest[start..];
1569            let value_at = tag.find("value=\"").expect("a checkbox value");
1570            let after = &tag[value_at + "value=\"".len()..];
1571            let end = after.find('"').expect("a closed value");
1572            let id = after[..end].to_string();
1573            // The name is the cell after the checkbox's own `<td>`.
1574            let cell = &rest[at..];
1575            let cell_at = cell.find("<td").expect("the name cell");
1576            let text = &cell[cell_at..];
1577            let text_at = text.find('>').expect("the cell's opening tag") + 1;
1578            let text_end = text[text_at..].find('<').expect("the cell's text end");
1579            rows.push((id, text[text_at..text_at + text_end].trim().to_string()));
1580            rest = &rest[at + 1..];
1581        }
1582        rows
1583    }
