22/// One live-table shard invocation (GH #224): the list path the page asks for
23/// and the interaction signals it owns.
24///
25/// The one argument the shard's *handler* takes. The `#[shard]` entry packs
26/// its wire parameters into this, and every seam below — the registry lookup,
27/// the state rebuild, the load, the render, and the retry link — reads the
28/// same value, so a new interaction dimension never changes a signature
29/// here.
30pub(crate) struct TableSearchArgs {
31    /// The list path to rerun: resolved through the registry (an allow-list,
32    /// never a raw route).
33    pub(crate) path: String,
34    /// The page's signals, untrusted by the time the shard reads them back.
35    pub(crate) signals: TableSignals,
36}
37
38/// A monomorphized live-search table loader, one per declared resource.
39///
40/// `#[shard]` inventory only discovers concrete fns (GH #104), so the single
41/// concrete [`table_search`] shard dispatches through this registry instead
42/// of going generic. Built by [`Panel::resource`], keyed by list path.
43pub(crate) type SearchFn = Arc<
44    dyn for<'a> Fn(
45            &'a Cx,
46            TableSearchArgs,
47        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
48        + Send
49        + Sync,
50>;
51
52/// Live-search handlers installed on the app context by [`Panel::build`].
53#[derive(Clone, Default)]
54pub(crate) struct SearchRegistry(pub(crate) HashMap<String, SearchFn>);
55
56/// Monomorphize `R`'s table loader into a [`SearchFn`]: tenancy + policy gate,
57/// then the same load + render the streamed list uses.
58///
59/// The table catches its own load errors (GH #158): a tampered `after=` /
60/// `before=` signal fails to decode inside the shard invocation, and the
61/// invocation must render the branded in-region `ErrorState` + retry link
62/// (via `super::list::table_error_view`, same as the streamed list) instead of
63/// erroring the shard. Auth/tenancy/policy failures still propagate — they
64/// are not table evidence.
65pub(crate) fn search_handler_for<R: Resource>() -> SearchFn {
66    Arc::new(
67        |cx: &Cx,
68         args: TableSearchArgs|
69         -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
70            Box::pin(async move {
71                enforce_auth(cx)?;
72                enforce_tenant::<R>(cx)?;
73                if !R::can_view_any(cx) {
74                    return Err(forbidden().into());
75                }
76                let table = wire_table_actions::<R>(cx, true);
77                let TableSearchArgs { path, signals } = args;
78                // One shared bound and one normalization per request (GH
79                // #148, GH #206, GH #224): `TableSignals::to_state` applies
80                // the same `q` clamp and `filters` bound the GET path applies
81                // (GH #205), and the shard `group_by` arg is client input — an
82                // unknown value must not echo through the retry link
83                // (GH #153). The render below takes the proof and does not
84                // normalize again.
85                let state = table.normalize_state(&signals.to_state());
86                // The retry link inside a failed table writes the same signals
87                // the toolbar does (GH #166), so keep a handle for it.
88                let retry_signals = signals.clone();
89                let rendered = async {
90                    let page = load_table_page::<R>(cx, &table, &state).await?;
91                    table
92                        .render_live_normalized(cx, page, &state, &path, signals)
93                        .await
94                };
95                match rendered.await {
96                    Ok(view) => Ok(view),
97                    Err(error) => Ok(table_error_view::<R>(
98                        cx,
99                        &state,
100                        &error,
101                        &path,
102                        Some(&retry_signals),
103                    )),
104                }
105            })
106        },
107    )
108}

110/// Resolve the registered live-search handler for `path`, answering the gate
111/// first (GH #146 defense in depth): the registry lookup runs only for an
112/// authenticated request, so an unknown `path` cannot be distinguished from a
113/// registered one by an unauthenticated probe (404-vs-401 oracle).
114fn search_entry(cx: &Cx, path: &str) -> Result<SearchFn> {
115    enforce_auth(cx)?;
116    topcoat::context::try_app_context::<SearchRegistry>(cx)
117        .and_then(|reg| reg.0.get(path).cloned())
118        .ok_or_else(|| topcoat::router::error::not_found().into())
119}

121/// Live table interactions (GH #104, GH #151): re-renders one resource's table
122/// as its signals change, morphing in place per Topcoat #392 (focus, scroll,
123/// and typing survive; rows carry stable `id`s from #104 prep).
124///
125/// The shard owns no state: the page creates the signals ([`TableSignals`]),
126/// renders the toolbar against them, and passes their handles here. Search,
127/// sort, filters, and pagination all write those signals, so one dependency
128/// graph re-renders the table — no navigation, no scroll jump. The swapped
129/// region is the table without the search toolbar (the live host owns that
130/// slot, so swaps never nest invocations or duplicate inputs).
131///
132/// Every arg is untrusted shard input: `path` must name a registered list
133/// (allow-list, never a raw route), and every signal value is clamped or
134/// re-parsed through [`TableSignals::to_state`] like the GET path.
135/// Authorization mirrors the list page (`requires_tenant` + `can_view_any`,
136/// row scoping via the tenant-scoped query, GH #223); shard POSTs carry no
137/// CSRF token, and
138/// none is needed for this read-only rerun. The GET toolbar stays as the
139/// no-JS fallback.
140///
141/// The module exists only to carry `allow(too_many_arguments)`: the shard's
142/// arity *is* the interaction list — one named parameter per dimension — and
143/// that exceeds the lint's default before the macro adds the ambient context.
144///
145/// The wire stays scalar by choice, not by necessity (GH #224). A struct
146/// cannot travel as a shard argument at all (topcoat requires the `expr!`
147/// vocabulary, and a struct has no `Surrogated` surrogate the browser's
148/// `cx.hydrate` can rebuild), but a *list* can: `Vec<Signal<String>>` and
149/// `[Signal<String>; N]` are both vocabulary types that round-trip. Packing
150/// the dimensions into one is still refused, because a list couples the
151/// browser to this server's ordering — adding or reordering a dimension would
152/// silently mismatch the two halves, where a named parameter cannot. So the
153/// handler packs its named parameters into [`TableSearchArgs`] for everything
154/// *below* the shard, and the wire keeps naming each dimension.
155#[allow(clippy::too_many_arguments)]
156mod shard_body {
157    use super::*;

159    #[shard]
160    pub(crate) async fn table_search(
161        cx: &Cx,
162        path: String,
163        q: topcoat::runtime::Signal<String>,
164        filters: topcoat::runtime::Signal<String>,
165        sort: topcoat::runtime::Signal<String>,
166        dir: topcoat::runtime::Signal<String>,
167        cursor: topcoat::runtime::Signal<String>,
168        group_by: topcoat::runtime::Signal<String>,
169        bulk: topcoat::runtime::Signal<String>,
170    ) -> Result<impl View> {
171        let entry = search_entry(cx, &path)?;
172        // One argument struct from here down (GH #224): the handler owns the
173        // wire arity, nothing below it does.
174        entry(
175            cx,
176            TableSearchArgs {
177                path,
178                signals: TableSignals {
179                    q,
180                    filters,
181                    sort,
182                    dir,
183                    cursor,
184                    group_by,
185                    bulk,
186                },
187            },
188        )
189        .await
190    }
191}

202    /// (GH #146): an unauthenticated probe cannot distinguish a registered

205    #[tokio::test]
206    async fn search_shard_answers_auth_before_the_registry_lookup() {
207        use topcoat::{context::CxTestBuilder, router::response::IntoResponse};
208
209        use crate::resource::Resource;
210
211        #[derive(Debug, toasty::Model, Clone)]
212        struct Dummy {
213            #[key]
214            #[auto]
215            id: uuid::Uuid,
216            name: String,
217        }
218        struct DummyResource;
219        impl Resource for DummyResource {
220            type Model = Dummy;
221        }
222
223        // A registry that really knows the `users` slug, so the known-path
224        // probe is a resolution the gate must preempt.
225        let (parts, ()) = http::Request::builder()
226            .method(http::Method::POST)
227            .uri(crate::auth::RUNTIME_PREFIX)
228            .body(())
229            .unwrap()
230            .into_parts();
231        let cx = CxTestBuilder::new()
232            .request_context(parts)
233            .app_context(crate::Auth::password())
234            .app_context(SearchRegistry(HashMap::from([(
235                "users".to_string(),
236                search_handler_for::<DummyResource>(),
237            )])))
238            .build();
239
240        // The gate's answer comes before the lookup: both a registered and an
241        // unregistered slug answer 401 identically (no 404 oracle).
242        let unknown = match search_entry(&cx, "not-a-slug") {
243            Ok(_) => panic!("unauthenticated probe must not resolve an entry"),
244            Err(err) => err,
245        };
246        let registered = match search_entry(&cx, "users") {
247            Ok(_) => panic!("an unauthenticated probe must never reach the registry"),
248            Err(err) => err,
249        };
250        let unknown_status = unknown
251            .into_response(&cx)
252            .expect("gate answer renders")
253            .status();
254        let registered_status = registered
255            .into_response(&cx)
256            .expect("gate answer renders")
257            .status();
258        assert_eq!(
259            unknown_status, registered_status,
260            "unauthenticated probes must not distinguish registered slugs"
261        );
262        assert_eq!(
263            registered_status,
264            http::StatusCode::UNAUTHORIZED,
265            "runtime probes answer 401 (ADR-0013), got {registered_status}"
266        );
267
268        // Auth disabled (the shard's own lookup is what remains): a
269        // registered path resolves and an unknown path is a plain 404 again.
270        let (parts, ()) = http::Request::builder()
271            .method(http::Method::POST)
272            .uri(crate::auth::RUNTIME_PREFIX)
273            .body(())
274            .unwrap()
275            .into_parts();
276        let cx = CxTestBuilder::new()
277            .request_context(parts)
278            .app_context(crate::Auth::disabled())
279            .app_context(SearchRegistry(HashMap::from([(
280                "users".to_string(),
281                search_handler_for::<DummyResource>(),
282            )])))
283            .build();
284        assert!(
285            search_entry(&cx, "users").is_ok(),
286            "with auth disabled a registered path resolves through the lookup"
287        );
288        let err = match search_entry(&cx, "not-a-slug") {
289            Ok(_) => panic!("an unregistered path must not resolve"),
290            Err(err) => err,
291        };
292        assert!(
293            err.downcast_ref::<topcoat::router::error::NotFoundError>()
294                .is_some(),
295            "with auth disabled the unknown path is a plain 404, got {err}"
296        );
297    }

611    ) -> http::Response<Body> {

613        let args = format!(
614            r#"["{path}",{}, {}, {}, {}, {}, {}, {}]"#,
615            sig(1, ""),
616            sig(2, ""),
617            sig(3, ""),
618            sig(4, ""),
619            sig(5, ""),
620            sig(6, ""),
621            sig(7, ""),
622        );
623        let request = http::Request::builder()
624            .method(http::Method::POST)
625            .uri(format!(
626                "/_topcoat/runtime/shards/{}",
627                topcoat::runtime::Shard::id(&table_search).as_str()
628            ))
629            .header(http::header::CONTENT_TYPE, "application/json")

631            .body(Body::from(format!(r#"{{"args":{args},"signals":{{}}}}"#)))
632            .unwrap();
633        let (mut parts, body) = request.into_parts();
634        if let Some(tenant) = tenant {
635            parts.extensions.insert(crate::tenancy::Tenant(tenant));
636        }
637        router.handle(http::Request::from_parts(parts, body)).await
638    }

640    /// The live-search shard re-checks the tenant and policy gates itself,
641    /// because page guards do not run on shard requests (GH #281): a gated
642    /// resource with no tenant must be refused instead of running an unscoped
643    /// query, a tenanted rerun must serve only that tenant's rows, and a
644    /// `can_view_any` denial is refused even with a tenant present.
645    #[tokio::test]
646    async fn live_shard_enforces_tenant_and_policy_gates() {
647        use std::collections::HashMap;
648
649        use http_body_util::BodyExt;
650
651        use crate::resource::Resource;
652
653        #[derive(Debug, Clone, toasty::Model)]
654        struct TenantDummy {
655            #[key]
656            #[auto]
657            id: uuid::Uuid,
658            #[index]
659            tenant_id: uuid::Uuid,
660            name: String,
661        }
662
663        struct TenantLive;
664        impl Resource for TenantLive {
665            type Model = TenantDummy;
666            fn slug() -> String {
667                "tenant-dummies".to_string()
668            }
669            fn requires_tenant() -> bool {
670                true
671            }
672            fn can_view_any(_cx: &Cx) -> bool {
673                true
674            }
675            fn can_view(_cx: &Cx, _record: &TenantDummy) -> bool {
676                true
677            }
678            fn table(cx: &Cx) -> crate::resource::Table<TenantDummy> {
679                crate::resource::Table::r#for(cx)
680                    .id(|d: &TenantDummy| d.id.to_string())
681                    .pk(|d: &TenantDummy| d.id.to_string())
682                    .columns(
683                        crate::resource::TextColumn::r#for(
684                            TenantDummy::fields().name(),
685                            |d: &TenantDummy| d.name.clone(),
686                        )
687                        .searchable()
688                        .sortable(),
689                    )
690                    .paginate(10)
691                    .live_search(true)
692            }
693            fn hydrate_form_values(_cx: &Cx, _record: &TenantDummy) -> HashMap<String, String> {
694                HashMap::new()
695            }
696        }
697
698        struct DeniedLive;
699        impl Resource for DeniedLive {
700            type Model = TenantDummy;
701            fn slug() -> String {
702                "denied-dummies".to_string()
703            }
704            fn requires_tenant() -> bool {
705                true
706            }
707            fn can_view_any(_cx: &Cx) -> bool {
708                false
709            }
710            fn table(cx: &Cx) -> crate::resource::Table<TenantDummy> {
711                crate::resource::Table::r#for(cx)
712                    .id(|d: &TenantDummy| d.id.to_string())
713                    .pk(|d: &TenantDummy| d.id.to_string())
714                    .columns(
715                        crate::resource::TextColumn::r#for(
716                            TenantDummy::fields().name(),
717                            |d: &TenantDummy| d.name.clone(),
718                        )
719                        .searchable()
720                        .sortable(),
721                    )
722                    .paginate(10)
723                    .live_search(true)
724            }
725            fn hydrate_form_values(_cx: &Cx, _record: &TenantDummy) -> HashMap<String, String> {
726                HashMap::new()
727            }
728        }
729
730        let tenant_a = uuid::Uuid::from_u128(1);
731        let tenant_b = uuid::Uuid::from_u128(2);
732        let mut db = Db::builder()
733            .models(toasty::models!(TenantDummy))
734            .connect("sqlite::memory:")
735            .await
736            .unwrap();
737        db.push_schema().await.unwrap();
738        for (tenant, name) in [(tenant_a, "Alpha"), (tenant_b, "Bravo")] {
739            toasty::create!(TenantDummy {
740                tenant_id: tenant,
741                name: name.to_string(),
742            })
743            .exec(&mut db)
744            .await
745            .unwrap();
746        }
747        let router = Panel::new("admin")
748            .app_context(db)
749            .resource::<TenantLive>()
750            .resource::<DeniedLive>()
751            .auth(crate::Auth::disabled())
752            .build()
753            .expect("panel builds");
754
755        let response = post_table_shard(&router, "/admin/tenant-dummies", None).await;
756        assert_eq!(
757            response.status(),
758            http::StatusCode::FORBIDDEN,
759            "a tenantless shard rerun must be refused"
760        );
761
762        let response = post_table_shard(&router, "/admin/tenant-dummies", Some(tenant_a)).await;
763        assert_eq!(
764            response.status(),
765            http::StatusCode::OK,
766            "a tenanted shard rerun must succeed"
767        );
768        let bytes = response.into_body().collect().await.unwrap().to_bytes();
769        let table_html = String::from_utf8_lossy(&bytes).to_string();
770        assert!(
771            table_html.contains("Alpha"),
772            "the shard must render the tenant's own row: {table_html}"
773        );
774        assert!(
775            !table_html.contains("Bravo"),
776            "the shard must not render another tenant's row: {table_html}"
777        );
778
779        let response = post_table_shard(&router, "/admin/denied-dummies", Some(tenant_a)).await;
780        assert_eq!(
781            response.status(),
782            http::StatusCode::FORBIDDEN,
783            "a can_view_any denial must refuse the shard rerun"
784        );
785    }
