22/// One live-table shard invocation (GH #224): the list path the page asks for
23/// and the interaction signals it owns.
24///
25/// The one argument the shard's *handler* takes. The `#[shard]` entry packs
26/// its wire parameters into this, and every seam below — the registry lookup,
27/// the state rebuild, the load, the render, and the retry link — reads the
28/// same value, so a new interaction dimension never changes a signature
29/// here.
30pub(crate) struct TableSearchArgs {
31    /// The list path to rerun: resolved through the registry (an allow-list,
32    /// never a raw route).
33    pub(crate) path: String,
34    /// The page's signals, untrusted by the time the shard reads them back.
35    pub(crate) signals: TableSignals,
36}

38/// A monomorphized live-search table loader, one per declared resource.
39///
40/// `#[shard]` inventory only discovers concrete fns (GH #104), so the single
41/// concrete [`table_search`] shard dispatches through this registry instead
42/// of going generic. Built by [`Panel::resource`], keyed by list path.
43pub(crate) type SearchFn = Arc<
44    dyn for<'a> Fn(
45            &'a Cx,
46            TableSearchArgs,
47        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
48        + Send
49        + Sync,
50>;

52/// Live-search handlers installed on the app context by [`Panel::build`].
53#[derive(Clone, Default)]
54pub(crate) struct SearchRegistry(pub(crate) HashMap<String, SearchFn>);

56/// Monomorphize `R`'s table loader into a [`SearchFn`]: tenancy + policy gate,
57/// then the same load + render the streamed list uses.
58///
59/// The table catches its own load errors (GH #158): a tampered `after=` /
60/// `before=` signal fails to decode inside the shard invocation, and the
61/// invocation must render the branded in-region `ErrorState` + retry link
62/// (via `super::list::table_error_view`, same as the streamed list) instead of
63/// erroring the shard. Auth/tenancy/policy failures still propagate — they
64/// are not table evidence.
65pub(crate) fn search_handler_for<R: Resource>() -> SearchFn {
66    Arc::new(
67        |cx: &Cx,
68         args: TableSearchArgs|
69         -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
70            Box::pin(async move {
71                gate::<R>(cx)?;
72                if !R::can_view_any(cx) {
73                    return Err(forbidden().into());
74                }
75                let table = wire_table_actions::<R>(cx, true);
76                let TableSearchArgs { path, signals } = args;
77                // One shared bound and one normalization per request (GH
78                // #148, GH #206, GH #224): `TableSignals::to_state` applies
79                // the same `q` clamp and `filters` bound the GET path applies
80                // (GH #205), and the shard `group_by` arg is client input — an
81                // unknown value must not echo through the retry link
82                // (GH #153). The render below takes the proof and does not
83                // normalize again.
84                let state = table.normalize_state(&signals.to_state());
85                // The retry link inside a failed table writes the same signals
86                // the toolbar does (GH #166), so keep a handle for it.
87                let retry_signals = signals.clone();
88                let rendered = async {
89                    let page = load_table_page::<R>(cx, &table, &state).await?;
90                    table
91                        .render_live_normalized(cx, page, &state, &path, signals)
92                        .await
93                };
94                match rendered.await {
95                    Ok(view) => Ok(view),
96                    Err(error) => Ok(table_error_view::<R>(
97                        cx,
98                        &state,
99                        &error,
100                        &path,
101                        Some(&retry_signals),
102                    )),
103                }
104            })
105        },
106    )
107}

109/// Resolve the registered live-search handler for `path`, answering the gate
110/// first (GH #146 defense in depth): the registry lookup runs only for an
111/// authenticated request, so an unknown `path` cannot be distinguished from a
112/// registered one by an unauthenticated probe (404-vs-401 oracle).
113fn search_entry(cx: &Cx, path: &str) -> Result<SearchFn> {
114    enforce_auth(cx)?;
115    topcoat::context::try_app_context::<SearchRegistry>(cx)
116        .and_then(|reg| reg.0.get(path).cloned())
117        .ok_or_else(|| topcoat::router::error::not_found().into())
118}

120/// Live table interactions (GH #104, GH #151): re-renders one resource's table
121/// as its signals change, morphing in place per Topcoat #392 (focus, scroll,
122/// and typing survive; rows carry stable `id`s from #104 prep).
123///
124/// The shard owns no state: the page creates the signals ([`TableSignals`]),
125/// renders the toolbar against them, and passes their handles here. Search,
126/// sort, filters, and pagination all write those signals, so one dependency
127/// graph re-renders the table — no navigation, no scroll jump. The swapped
128/// region is the table without the search toolbar (the live host owns that
129/// slot, so swaps never nest invocations or duplicate inputs).
130///
131/// Every arg is untrusted shard input: `path` must name a registered list
132/// (allow-list, never a raw route), and every signal value is clamped or
133/// re-parsed through [`TableSignals::to_state`] like the GET path.
134/// Authorization mirrors the list page (`requires_tenant` + `can_view_any`,
135/// row scoping via the tenant-scoped query, GH #223); shard POSTs carry no
136/// CSRF token, and
137/// none is needed for this read-only rerun. The GET toolbar stays as the
138/// no-JS fallback.
139///
140/// The module exists only to carry `allow(too_many_arguments)`: the shard's
141/// arity *is* the interaction list — one named parameter per dimension — and
142/// that exceeds the lint's default before the macro adds the ambient context.
143///
144/// The wire stays scalar by choice, not by necessity (GH #224). A struct
145/// cannot travel as a shard argument at all (topcoat requires the `expr!`
146/// vocabulary, and a struct has no `Surrogated` surrogate the browser's
147/// `cx.hydrate` can rebuild), but a *list* can: `Vec<Signal<String>>` and
148/// `[Signal<String>; N]` are both vocabulary types that round-trip. Packing
149/// the dimensions into one is still refused, because a list couples the
150/// browser to this server's ordering — adding or reordering a dimension would
151/// silently mismatch the two halves, where a named parameter cannot. So the
152/// handler packs its named parameters into [`TableSearchArgs`] for everything
153/// *below* the shard, and the wire keeps naming each dimension.
154#[allow(clippy::too_many_arguments)]
155mod shard_body {
156    use super::*;

158    #[shard("/_topcoat/runtime/shards/argentum-table-search")]
159    pub(crate) async fn table_search(
160        cx: &Cx,
161        path: String,
162        q: topcoat::runtime::Signal<String>,
163        filters: topcoat::runtime::Signal<String>,
164        sort: topcoat::runtime::Signal<String>,
165        dir: topcoat::runtime::Signal<String>,
166        cursor: topcoat::runtime::Signal<String>,
167        group_by: topcoat::runtime::Signal<String>,
168        bulk: topcoat::runtime::Signal<String>,
169    ) -> Result<impl View> {
170        let entry = search_entry(cx, &path)?;
171        // One argument struct from here down (GH #224): the handler owns the
172        // wire arity, nothing below it does.
173        entry(
174            cx,
175            TableSearchArgs {
176                path,
177                signals: TableSignals {
178                    q,
179                    filters,
180                    sort,
181                    dir,
182                    cursor,
183                    group_by,
184                    bulk,
185                },
186            },
187        )
188        .await
189    }

214    /// The signal id the live retry link writes (GH #294): read from the
215    /// control's own `increment()` handler, which is the side that re-runs the
216    /// shard. Locating it by offset from the marker instead would read whatever
217    /// payload happened to follow.
218    fn retry_signal_id(html: &str) -> &str {
219        const MARKER: &str = r#"id&quot;:&quot;"#;
220        let at = html
221            .find("data-retry-attempt")
222            .unwrap_or_else(|| panic!("the retry control, got {html}"));
223        let tag_start = html[..at].rfind('<').expect("the control's opening tag");
224        // The tag runs to the next `<`: an attribute value escapes its own
225        // `>` (the handler's `=&gt;`), so the first `>` is not the tag's.
226        let next = html[tag_start + 1..]
227            .find('<')
228            .map(|i| tag_start + 1 + i)
229            .unwrap_or(html.len());
230        let tag = &html[tag_start..next];
231        let write = tag
232            .find("increment()")
233            .unwrap_or_else(|| panic!("the control's write handler, got {tag}"));
234        // The id sits in the handler's `cx.hydrate({"t":"Signal","id":"…"})`,
235        // the last one before the increment.
236        let id_at = tag[..write]
237            .rfind(MARKER)
238            .unwrap_or_else(|| panic!("the handler's signal, got {tag}"))
239            + MARKER.len();
240        let id = &tag[id_at..id_at + 32];
241        assert!(
242            id.chars().all(|c| c.is_ascii_hexdigit()),
243            "the handler must name a signal id, got {tag}"
244        );
245        id
246    }

624    #[tokio::test]
625    async fn live_shard_retry_preserves_the_query() {
626        // GH #294: a failure the cursor did not cause must be retried with the
627        // query that failed — search, filters, and sort included. The retry
628        // re-runs through the token it increments, so it is not inert when the
629        // query signals already hold the values the failed request used.
630        //
631        // The unpaginated table is the deterministic non-cursor failure: the
632        // list loader refuses it before any cursor is decoded.
633
634        use http_body_util::BodyExt;
635
636        use crate::resource::Resource;
637
638        #[derive(Debug, toasty::Model, Clone)]
639        struct Dummy {
640            #[key]
641            #[auto]
642            id: uuid::Uuid,
643            name: String,
644            featured: bool,
645        }
646        struct UnpaginatedLive;
647        impl Resource for UnpaginatedLive {
648            type Model = Dummy;
649            fn slug() -> String {
650                "dummies".to_string()
651            }
652            fn can_view_any(_cx: &Cx) -> bool {
653                true
654            }
655            fn can_view(_cx: &Cx, _record: &Dummy) -> bool {
656                true
657            }
658            fn table(cx: &Cx) -> crate::resource::Table<Dummy> {
659                crate::resource::Table::r#for(cx)
660                    .id(|d: &Dummy| d.id.to_string())
661                    .pk(|d: &Dummy| d.id.to_string())
662                    .columns(
663                        crate::resource::TextColumn::r#for(Dummy::fields().name(), |d: &Dummy| {
664                            d.name.clone()
665                        })
666                        .searchable()
667                        .sortable(),
668                    )
669                    .filters(crate::resource::TernaryFilter::r#for(
670                        Dummy::fields().featured(),
671                    ))
672                    .live_search(true)
673            }
674        }
675
676        let mut db = Db::builder()
677            .models(toasty::models!(Dummy))
678            .connect("sqlite::memory:")
679            .await
680            .unwrap();
681        db.push_schema().await.unwrap();
682        toasty::create!(Dummy {
683            name: "Ada".to_string(),
684            featured: false,
685        })
686        .exec(&mut db)
687        .await
688        .unwrap();
689        let router = panel_for::<UnpaginatedLive>(db)
690            .build()
691            .expect("panel builds");
692
693        let sig = |n: u8, v: &str| format!(r#"{{"t":"Signal","id":"{:032x}","v":"{v}"}}"#, n);
694        let args = format!(
695            r#"["/admin/dummies",{}, {}, {}, {}, {}, {}, {}]"#,
696            sig(1, "Ada"),
697            sig(2, "featured:true"),
698            sig(3, "name"),
699            sig(4, "desc"),
700            sig(5, ""),
701            sig(6, ""),
702            sig(7, ""),
703        );
704        let response = router
705            .handle(
706                http::Request::builder()
707                    .method(http::Method::POST)
708                    .uri(TABLE_SEARCH_PATH)
709                    .header(http::header::CONTENT_TYPE, "application/json")
710                    .header(topcoat::router::request::IDENTITY_HEADER, "A".repeat(22))
711                    .body(Body::from(format!(r#"{{"args":{args},"signals":{{}}}}"#)))
712                    .unwrap(),
713            )
714            .await;
715        assert_eq!(response.status(), http::StatusCode::OK);
716        let bytes = response.into_body().collect().await.unwrap().to_bytes();
717        let table_html = String::from_utf8_lossy(&bytes).to_string();
718        assert!(
719            table_html.contains("Couldn't load Dummies"),
720            "the unpaginated live load must render the error state: {table_html}"
721        );
722        // The href keeps the failed query; the no-JS fallback retries it.
723        let href = table_html
724            .split("href=\"")
725            .nth(1)
726            .and_then(|rest| rest.split('"').next())
727            .unwrap_or_default()
728            .to_string();
729        assert!(
730            href.contains("q=Ada") && href.contains("filters=") && href.contains("sort=name"),
731            "the retry href must keep the failed query, got {href}"
732        );
733        assert!(
734            table_html.contains("data-retry-attempt=") && table_html.contains("increment()"),
735            "the retry must re-run through its token: {table_html}"
736        );
737        // The shard re-runs only if it read the token: assert the token's own
738        // dep marker, not the shard arguments' dep markers.
739        let retry = retry_signal_id(&table_html);
740        assert!(
741            table_html.contains(&format!(r#"::topcoat::dep("{retry}")"#)),
742            "the retry token must be a shard dependency, got {table_html}"
743        );
744        assert!(
745            !table_html.contains("set((cx.hydrate"),
746            "a non-cursor failure must not clear the query signals: {table_html}"
747        );
748    }

860    /// Post one live-table shard rerun with an optional `Tenant` request
861    /// extension (GH #281).
862    ///
863    /// The first positional arg is the list path the registry is keyed by, so
864    /// the caller chooses the resource; the identity header is the one the
865    /// browser runtime sends.
866    async fn post_table_shard(
867        router: &topcoat::router::Router,
868        path: &str,
869        tenant: Option<uuid::Uuid>,
870    ) -> http::Response<Body> {
871        let sig = |n: u8, v: &str| format!(r#"{{"t":"Signal","id":"{:032x}","v":"{v}"}}"#, n);
872        let args = format!(
873            r#"["{path}",{}, {}, {}, {}, {}, {}, {}]"#,
874            sig(1, ""),
875            sig(2, ""),
876            sig(3, ""),
877            sig(4, ""),
878            sig(5, ""),
879            sig(6, ""),
880            sig(7, ""),
881        );
882        let request = http::Request::builder()
883            .method(http::Method::POST)
884            .uri(TABLE_SEARCH_PATH)
885            .header(http::header::CONTENT_TYPE, "application/json")
886            .header(topcoat::router::request::IDENTITY_HEADER, "A".repeat(22))
887            .body(Body::from(format!(r#"{{"args":{args},"signals":{{}}}}"#)))
888            .unwrap();
889        let (mut parts, body) = request.into_parts();
890        if let Some(tenant) = tenant {
891            parts.extensions.insert(crate::tenancy::Tenant(tenant));
892        }
893        router.handle(http::Request::from_parts(parts, body)).await
894    }
