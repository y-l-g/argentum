1//! The detail page (GH #187): `GET {prefix}/{slug}/{id}`, read-only.
2//!
3//! One record, rendered through [`Resource::view`] — the same `Schema` a form
4//! uses, read the other way round. It lives beside the form handlers rather
5//! than in `list.rs` because it is a record page: it loads through the same
6//! tenant-scoped query (`find_by_key`, GH #223), checks the same `can_view`
7//! policy,
8//! and answers the same 404 for an unknown or out-of-scope id.
9
10use topcoat::view::internal::ThenView;
11use topcoat::{
12    context::Cx,
13    router::{Body, error::not_found, path_param_segment},
14    view::{BoxView, HoistView, ViewExt, view},
15};
16
17use super::actions::load_viewable;
18use super::{enforce_auth, enforce_tenant, list_url};
19use crate::db::db;
20use crate::resource::Resource;
21
22/// Detail page GET (GH #187).
23///
24/// A resource that declares no [`view`](Resource::view) has no detail page:
25/// the handler 404s rather than rendering an empty shell, which keeps
26/// "declares nothing" and "no such page" the same answer for a hand-typed URL
27/// and makes the row link's absence honest.
28///
29/// The record loads through the tenant-scoped query (`find_by_key` — the
30/// tenancy half derived by the framework, GH #223 — and the resource's own
31/// soft-delete scope, ADR-0002), so an unknown id and an id outside the
32/// request's scope get one answer, as everywhere else in the panel. `can_view`
33/// on the loaded record is a 403 rather than a 404: the record exists and this
34/// caller may not see it.
35pub(crate) fn resource_view<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
36    Box::pin(HoistView::new(ThenView::new(async move {
37        enforce_auth(cx)?;
38        enforce_tenant::<R>(cx)?;
39        if !R::viewed(cx) {
40            return Err(not_found().into());
41        }
42        let id = path_param_segment(cx, "id").to_string();
43        let mut db = db(cx);
44        let record = load_viewable::<R>(cx, &mut db).await?;
45        // Values come from the same hydration the edit form uses, so the page
46        // and the form cannot disagree about what a field holds.
47        let values = R::hydrate_form_values(cx, &record);
48        let body = R::view(cx).render_readonly(cx, &values).await?;
49        // Relations render from the record itself (GH #187): the `Schema`
50        // above carries only its string projection, and the related rows are
51        // already loaded by `query`'s `include`, so this adds no query.
52        let relations = R::view_relations(cx, &record);
53        // The title names the page and the record's key, which is what the
54        // list and the URL call it (`Table::id` is the display key, `pk` the
55        // record key — the URL carries the latter, GH #168).
56        let title = format!("{} {id}", R::navigation_label());
57        let back = list_url(cx, &R::slug());
58        Ok(view! {
59            cx =>
60            argentum_ui::page(
61                argentum_ui::page_header(
62                    argentum_ui::page_title((title))
63                    <a href=(back) class="text-sm text-muted-foreground underline">
64                        "Back to list"
65                    </a>
66                )
67                argentum_ui::page_content(
68                    <div class="flex flex-col gap-4">
69                        (body)
70                        if let Some(relations) = relations {
71                            (relations)
72                        }
73                    </div>
74                )
75            )
76        }
77        .boxed())
78    })))
79}
