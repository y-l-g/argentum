1//! The detail page (GH #187): `GET {prefix}/{slug}/{id}`, read-only.
2//!
3//! One record, rendered through [`Resource::view`] — the same `Schema` a form
4//! uses, read the other way round. It lives beside the form handlers rather
5//! than in `list.rs` because it is a record page: it loads through the same
6//! tenant-scoped query (`find_by_key`, GH #223), checks the same `can_view`
7//! policy,
8//! and answers the same 404 for an unknown or out-of-scope id.
9
10use topcoat::{
11    context::Cx,
12    router::{Body, error::not_found, path_param_segment},
13    view::{BoxView, HoistView, ViewExt, internal::ThenView, view},
14};
15
16use super::{actions::load_viewable, gate, list_url};
17use crate::{db::db, resource::Resource};

19/// Detail page GET (GH #187).
20///
21/// A resource that declares no [`view`](Resource::view) has no detail page:
22/// the handler 404s rather than rendering an empty shell, which keeps
23/// "declares nothing" and "no such page" the same answer for a hand-typed URL
24/// and makes the row link's absence honest.
25///
26/// The record loads through the tenant-scoped query (`find_by_key` — the
27/// tenancy half derived by the framework, GH #223 — and the resource's own
28/// soft-delete scope, ADR-0002), so an unknown id and an id outside the
29/// request's scope get one answer, as everywhere else in the panel. `can_view`
30/// on the loaded record is a 403 rather than a 404: the record exists and this
31/// caller may not see it.
32pub(crate) fn resource_view<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
33    Box::pin(HoistView::new(ThenView::new(async move {
34        gate::<R>(cx)?;
35        if !R::viewed(cx) {
36            return Err(not_found().into());
37        }
38        let id = path_param_segment(cx, "id").to_string();
39        let mut db = db(cx);
40        let record = load_viewable::<R>(cx, &mut db).await?;
41        // Values come from the same hydration the edit form uses, so the page
42        // and the form cannot disagree about what a field holds.
43        let values = R::hydrate_form_values(cx, &record);
44        let body = R::view(cx).render_readonly(cx, &values).await?;
45        // Relations render from the record itself (GH #187): the `Schema`
46        // above carries only its string projection, and the related rows are
47        // already loaded by `query`'s `include`, so this adds no query.
48        let relations = R::view_relations(cx, &record);
49        // The record's own label titles the page when the resource declares
50        // one (GH #241). The fallback is the page's name plus the URL's record
51        // key, which is what the route carries (`Table::id` is the list's
52        // display key and `pk` its record key, GH #168).
53        let title = detail_title::<R>(cx, &record, &id);
54        let back = list_url(cx, &R::slug());
55        Ok(view! {
56            cx =>
57            argentum_ui::page(
58                argentum_ui::page_header(
59                    argentum_ui::page_title((title))
60                    <a href=(back) class="text-sm text-muted-foreground underline">
61                        "Back to list"
62                    </a>
63                )
64                argentum_ui::page_content(
65                    <div class="flex flex-col gap-4">
66                        (body)
67                        if let Some(relations) = relations {
68                            (relations)
69                        }
70                    </div>
71                )
72            )
73        }
74        .boxed())
75    })))
76}

78/// The detail page's title (GH #241): the record's label when the resource
79/// declares one ([`Resource::record_label`]), else the page's name and the
80/// URL's record key.
81fn detail_title<R: Resource>(cx: &Cx, record: &R::Model, id: &str) -> String {
82    R::record_label(cx, record).unwrap_or_else(|| format!("{} {id}", R::navigation_label()))
83}

85#[cfg(test)]
86mod tests {
87    use topcoat::context::CxTestBuilder;
88
89    use super::*;
90
91    #[derive(Debug, Clone, toasty::Model)]
92    struct Note {
93        #[key]
94        #[auto]
95        id: uuid::Uuid,
96        title: String,
97    }
98
99    /// A resource with no label: the title keeps the page name and the record
100    /// key.
101    struct Unlabelled;
102
103    impl Resource for Unlabelled {
104        type Model = Note;
105    }
106
107    /// A resource that labels its records with the note's title.
108    struct Labelled;
109
110    impl Resource for Labelled {
111        type Model = Note;
112
113        fn record_label(_cx: &Cx, record: &Note) -> Option<String> {
114            Some(record.title.clone())
115        }
116    }
117
118    fn note() -> Note {
119        Note {
120            id: uuid::Uuid::nil(),
121            title: "A Title".to_string(),
122        }
123    }
124
125    #[test]
126    fn a_resource_without_a_label_titles_the_page_with_the_record_key() {
127        let cx = CxTestBuilder::new().build();
128        assert_eq!(
129            detail_title::<Unlabelled>(&cx, &note(), "8f14e45f"),
130            "Notes 8f14e45f"
131        );
132    }
133
134    #[test]
135    fn a_declared_label_titles_the_page() {
136        let cx = CxTestBuilder::new().build();
137        assert_eq!(
138            detail_title::<Labelled>(&cx, &note(), "8f14e45f"),
139            "A Title"
140        );
141    }
