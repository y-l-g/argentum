25/// Branding for the admin shell (panel header + sidebar header).
26#[derive(Debug, Clone)]
27pub struct Brand {
28    /// Display name (e.g. `"Acme"`).
29    pub name: String,
30    /// Optional logo URL (e.g. `"/logo.svg"`). Rendered as an `<img>` when present.
31    pub logo: Option<String>,
32}

34impl Brand {
35    /// Create a brand with the given name (GH #102: surrounding whitespace is
36    /// trimmed so `" Acme "` cannot break the `flex h-16` header).
37    pub fn new(name: impl Into<String>) -> Self {
38        Self {
39            name: name.into().trim().to_string(),
40            logo: None,
41        }
42    }
43
44    /// Attach a logo URL (GH #102: blank values are ignored so an empty
45    /// `logo("")` falls back to the name-only render instead of a
46    /// broken-image icon).
47    pub fn logo(mut self, logo: impl Into<String>) -> Self {
48        let logo = logo.into().trim().to_string();
49        if !logo.is_empty() {
50            self.logo = Some(logo);
51        }
52        self
53    }
54}

56/// Whether the shell starts in dark mode for a visitor with no stored choice.
57/// Persisted via `theme.js` (`localStorage` + `theme` cookie).
58///
59/// Precedence (GH #102, corrected in GH #184): this build-time default only
60/// sets the initial `<html class>` and is handed to the blocking
61/// `theme_init_script` as its fallback. The stored preference — `localStorage`
62/// first, then the `theme` cookie — wins in **both** directions: a stored
63/// `light` removes the class this default added. The server never reads the
64/// cookie per request, so a toggle is client-side until the next navigation,
65/// at which point this default would otherwise re-darken the page — which is
66/// exactly the bug GH #184 fixed.
67#[derive(Debug, Clone, Copy)]
68pub struct DarkMode(pub bool);

71///
72/// Tailwind's generated stylesheet is necessarily a call-site asset because

77pub(crate) struct ShellAssets {
78    pub(crate) stylesheet: Asset,

99        use topcoat::context::try_app_context;
100        let (name, logo) = if let Some(brand) = try_app_context::<Brand>(cx) {
101            (brand.name.clone(), brand.logo.clone())
102        } else {
103            ("Argentum".to_string(), None)

106            let alt = name.clone();
107            Ok(view! {

109                <div class="flex items-center gap-2 font-semibold text-foreground">
110                    <img
111                        src=(logo_url)
112                        alt=(alt)
113                        width="24"
114                        height="24"
115                        class="h-6 w-6 rounded"
116                    >
117                    (name)
118                </div>
119            }
120            .boxed())
121        } else {
122            Ok(view! {
123                cx =>
124                <div class="flex items-center gap-2 font-semibold text-foreground">
125                    (name)
126                </div>
127            }
128            .boxed())
129        }

132    async fn sidebar_navigation<'a>(
133        cx: &'a Cx,
134        nav_items: &[NavigationItem],
135        current_path: &str,
136        mobile_open: Signal<bool>,
137    ) -> Result<BoxView<'a>> {
138        use argentum_ui::{
139            sidebar_group, sidebar_group_content, sidebar_group_label, sidebar_menu,
140            sidebar_menu_button, sidebar_menu_item,
141        };
142        let mut nav_items = nav_items.to_vec();
143        // Stable order (GH #102): explicit `order` first, declaration order
144        // breaking ties — a resource's `navigation()` override interleaves by
145        // setting it.
146        nav_items.sort_by_key(|item| item.order);
147        // A Panel resolves every item it owns (`Panel::resource`); one that
148        // reaches the sidebar unresolved has no URL to render, which is a
149        // framework bug rather than user error (GH #165).
150        debug_assert!(
151            nav_items.iter().all(|item| item.url().is_some()),
152            "navigation items are resolved by the Panel that owns them"
153        );
154        let current_path = current_path.to_string();
155
156        Ok(view! {
157            cx =>
158            sidebar_group(
159                sidebar_group_label("Navigation")
160                sidebar_group_content(
161                    sidebar_menu(
162                        for item in &nav_items {
163                            let is_active = item.is_current_path(&current_path);
164                            sidebar_menu_item(
165                                sidebar_menu_button(
166                                    active: is_active,
167                                    href: item.url(),
168                                    tooltip: Some(item.label.as_str()),
169                                    attrs: attributes! {
170                                        // Tapping a link in the mobile sheet closes
171                                        // it; on desktop the navigation is the effect.
172                                        @click=$(|_e: Event| mobile_open.set(false))
173                                    },
174                                    <span>(item.label.clone())</span>
175                                )
176                            )
177                        }
178                    )
179                )
180            )
181        }
182        .boxed())
183    }

185    /// Whether the persisted `sidebar_state` cookie asks for an expanded
186    /// desktop sidebar (default: expanded).
187    ///
188    /// Parsed from the raw `Cookie` header on purpose: `topcoat::cookie::cookies`
189    /// panics when the cookie router layer is absent (tests, minimal routers),
190    /// and the shell must render everywhere. The value only seeds the runtime
191    /// signal's initial `data-state`; after hydration the browser owns the
192    /// state, and `assets/sidebar.js` mirrors changes back to the cookie.
193    fn sidebar_starts_open(cx: &Cx) -> bool {
194        !try_request_context::<http::request::Parts>(cx)
195            .and_then(|parts| parts.headers.get(COOKIE))
196            .and_then(|value| value.to_str().ok())
197            .is_some_and(|cookie| {
198                cookie
199                    .split(';')
200                    .any(|part| part.trim().strip_prefix("sidebar_state=") == Some("collapsed"))
201            })
202    }

204    /// Render the Filament-grade Shell that frames every admin page.
205    ///
206    /// Composes Topcoat's upstream `sidebar` primitives (ADR-0007): the
207    /// desktop panel and the mobile sheet share one navigation rendering, and
208    /// the open state lives in runtime signals — `open` seeds from the
209    /// `sidebar_state` cookie for the first paint, the triggers carry
210    /// `@click` handlers, and `assets/sidebar.js` persists changes back to
211    /// the cookie. Includes dark-mode toggle (Ghost button, persisted by
212    /// `theme.js` to `localStorage` + the `theme` cookie) and the toast stack
213    /// (shadcn/Sonner surface, fixed bottom-right). Additive `class` is
214    /// allowed on the outer container only (narrow seam).
215    ///
216    /// Asset note: desktop persistence needs `assets/sidebar.js`
217    /// (`argentum_ui::SIDEBAR_JS`), which [`Self::layout_shell`]'s document —
218    /// not this function — emits (scripts are owned by the document). The
219    /// mobile sheet (`#mobile-sidebar-sheet`) instead dismisses through its
220    /// own runtime `@keydown`/`@click` handlers, so it needs no asset; the
221    /// vendored `sheet`/`sidebar` primitives carry no note themselves
222    /// (ADR-0007 sync guard). See ADR-0014.
223    pub async fn render_shell<'a>(
224        cx: &'a Cx,
225        nav_items: &[NavigationItem],
226        current_path: &str,
227        slot: Child<'a>,
228        extra_class: Option<String>,
229    ) -> Result<BoxView<'a>> {
230        // A hoisting body: the sidebar signals are declared while this view
231        // resolves, and a page re-run resumes them from the client. The owned
232        // copies pin the caller's navigation to the lazy body's lifetime.
233        let nav_items = nav_items.to_vec();
234        let current_path = current_path.to_string();
235        Ok(Box::pin(HoistView::new(ThenView::new(async move {
236            Self::render_shell_body(cx, &nav_items, &current_path, slot, extra_class).await
237        }))))
238    }

240    async fn render_shell_body<'a>(
241        cx: &'a Cx,
242        nav_items: &[NavigationItem],
243        current_path: &str,
244        slot: Child<'a>,
245        extra_class: Option<String>,
246    ) -> Result<BoxView<'a>> {
247        use argentum_ui::{
248            SeparatorOrientation, SidebarCollapsible, separator, sidebar, sidebar_content,
249            sidebar_footer, sidebar_header, sidebar_inset, sidebar_provider, sidebar_trigger,
250        };
251
252        let sidebar_open = signal(cx, || Self::sidebar_starts_open(cx));
253        let mobile_open = signal(cx, || false);
254        let outer_class = extra_class.clone().unwrap_or_default();
255        let header_title = topcoat::context::try_app_context::<Brand>(cx)
256            .map(|b| b.name.clone())
257            .unwrap_or_else(|| "Argentum".to_string());
258        // One navigation tree: the upstream sidebar renders its children once
259        // and shares them between the desktop panel and the mobile sheet.
260        let navigation =
261            Self::sidebar_navigation(cx, nav_items, current_path, mobile_open.clone()).await?;
262        let sidebar_brand = Self::render_brand(cx).await?;
263        let sidebar_theme_toggle = Self::theme_toggle(cx).await?;
264        let header_theme_toggle = Self::theme_toggle(cx).await?;
265        // Signed-in identity + logout control, present only with a session
266        // (ADR-0013). `ensure_token` runs before any streaming starts so the
267        // logout form always carries a matching CSRF pair.
268        #[cfg(feature = "auth")]
269        let account_view: BoxView<'_> = match crate::auth::current_user(cx) {
270            Some(user) => {
271                let csrf = crate::csrf::ensure_token(cx);
272                let logout = crate::auth::logout_url(cx);
273                view! {
274                    cx =>
275                    <div class="flex items-center gap-2">
276                        <span class="text-sm text-muted-foreground">
277                            (user.display_name)
278                        </span>
279                        <form method="post" action=(logout)>
280                            (crate::csrf::field(cx, &csrf))
281                            <button
282                                type="submit"
283                                class="text-sm text-muted-foreground underline"
284                            >
285                                "Sign out"
286                            </button>
287                        </form>
288                    </div>
289                }
290                .boxed()
291            }
292            None => view! { cx => <span></span> }.boxed(),
293        };
294        #[cfg(not(feature = "auth"))]
295        let account_view: BoxView<'_> = view! { cx => <span></span> }.boxed();
296        let notification_view: BoxView<'_> = match take_notification(cx) {
297            Some(notification) => {
298                crate::notification::render_notification(cx, notification, Default::default())
299                    .await?
300            }
301            // No `<span>` placeholder (GH #160): the toaster renders an `<ol>`,
302            // which permits only `li`/`script`/`template` children — the empty
303            // view renders nothing.
304            None => ().boxed(),
305        };
306        // The page owns the live-toast signals; resolve the same handles here
307        // (same helper, same request identity) and hand them to the shard
308        // (GH #154 §3).
309        let LiveToast {
310            status: toast_status,
311            title: toast_title,
312            description: toast_description,
313            serial: toast_serial,
314        } = live_toast(cx);
315
316        Ok(view! {
317            cx =>
318            sidebar_provider(
319                attrs: attributes! { class=(outer_class) },
320                // `sidebar_rail` is intentionally not rendered: the header
321                // `sidebar_trigger` is the explicit toggle, and the rail's
322                // edge hit-area reads as stray chrome as a primary toggle.
323                // Keep the component available (`argentum_ui::sidebar_rail`)
324                // for opt-in `variant=inset`/`floating` layouts.
325                sidebar(
326                    open: $(sidebar_open.get()),
327                    mobile_open: $(mobile_open.get()),
328                    collapsible: SidebarCollapsible::Offcanvas,
329                    sheet_attrs: attributes! {
330                        id="mobile-sidebar-sheet"
331                        aria-label="Navigation"
332                        @keydown=$(|e: Event| {
333                            if e.key == "Escape" {
334                                mobile_open.set(false);
335                            }
336                        })
337                        @click=$(|e: Event| {
338                            if e.target.id == "mobile-sidebar-sheet" {
339                                mobile_open.set(false);
340                            }
341                        })
342                    },
343                    sidebar_header(
344                        <div class="flex items-center gap-2">
345                            (sidebar_brand)
346                            // Below `md` the sheet covers the inset header,
347                            // so the sheet's own header carries the close
348                            // control (upstream `examples/ui`); the mobile
349                            // trigger in the inset header is behind the veil.
350                            argentum_ui::button(
351                                variant: argentum_ui::ButtonVariant::Ghost,
352                                size: argentum_ui::ButtonSize::Icon,
353                                attrs: attributes! {
354                                    type="button"
355                                    class="md:hidden ml-auto"
356                                    aria-label="Close sidebar"
357                                    @click=$(|_e: Event| mobile_open.set(false))
358                                },
359                                icon(data: argentum_ui::icons::X)
360                            )
361                        </div>
362                    )
363                    sidebar_content((navigation))
364                    sidebar_footer((sidebar_theme_toggle))
365                )
366                sidebar_inset(
367                    sidebar_header(
368                        // The desktop trigger collapses the rail; below md the
369                        // mobile trigger opens the sheet instead (shadcn
370                        // SidebarTrigger pair, upstream `examples/ui`).
371                        sidebar_trigger(
372                            open: $(sidebar_open.get()),
373                            attrs: attributes! {
374                                class="max-md:hidden"
375                                aria-controls="mobile-sidebar-sheet"
376                                @click=$(|_e: Event| sidebar_open.toggle())
377                            }
378                        )
379                        sidebar_trigger(
380                            open: $(mobile_open.get()),
381                            attrs: attributes! {
382                                class="md:hidden"
383                                aria-controls="mobile-sidebar-sheet"
384                                @click=$(|_e: Event| mobile_open.toggle())
385                            }
386                        )
387                        separator(orientation: SeparatorOrientation::Vertical)
388                        <div class="font-semibold text-foreground">(header_title)</div>
389                        <div class="ml-auto flex items-center gap-2">
390                            (account_view)
391                            (header_theme_toggle)
392                        </div>
393                    )
394                    // `sidebar_inset` is the document's one `<main>`; a second
395                    // nested landmark is invalid and confuses landmark
396                    // navigation (upstream `examples/ui` uses a plain div).
397                    <div class="flex-1 mx-auto max-w-7xl w-full p-6">(slot)</div>
398                )
399                // Toast stack — the shadcn/Sonner surface, fixed bottom-right
400                // and a polite live region so streamed swaps are announced
401                // (GH #98, GH #151). `live_toaster` is the page-owned
402                // in-place transport (GH #154 §3); the flash cookie's toast
403                // rides beside it.
404                argentum_ui::toaster(
405                    (notification_view)
406                    live_toaster(
407                        status: $(toast_status),
408                        title: $(toast_title),
409                        description: $(toast_description),
410                        serial: $(toast_serial)
411                    )
412                )
413            ) // Scripts are owned by the document (layout_shell).
414        }
415        .boxed())
416    }

418    /// Convenience wrapper for `#[layout]` handlers: takes the layout's
419    /// `slot: Slot<'_>` and renders the complete HTML document around the
420    /// shell.
421    ///
422    /// The stylesheet and font are supplied to the Panel builder with
423    /// [`Self::shell_assets`]. A Panel without those values remains renderable
424    /// for tests and custom document owners, but does not pretend that a CSS
425    /// bundle exists. Errors from the page slot propagate unchanged when the
426    /// document view is resolved.
427    pub async fn layout_shell<'a>(cx: &'a Cx, slot: Slot<'a>) -> Result<impl View + 'a> {
428        use topcoat::{context::try_app_context, router::request::uri};
429        let current = uri(cx).path().to_string();
430        // Prefer declarative nav_items from Panel::resource, fallback to Home.
431        let nav_items = try_app_context::<Vec<NavigationItem>>(cx)
432            .cloned()
433            .unwrap_or_else(|| vec![NavigationItem::at("Home", super::panel_prefix(cx))]);
434        let shell = Self::render_shell(cx, &nav_items, &current, slot, None).await?;
435        let brand_title = try_app_context::<Brand>(cx)
436            .map(|b| b.name.clone())
437            .unwrap_or_else(|| "Argentum".to_string());
438        Self::render_document(cx, brand_title, shell).await
439    }

441    /// The complete HTML document around a rendered body: assets, dark-mode
442    /// class, and title. [`Self::layout_shell`] frames the panel shell with
443    /// it; the standalone login page (ADR-0013) uses the same document so
444    /// brand and dark mode carry over.
445    ///
446    /// The ten shell scripts ship `defer`red (deliberate all-load policy,
447    /// ADR-0014): parsing never waits for them, and every one is safe
448    /// deferred — document-level listeners install after parse, and the
449    /// `DOMContentLoaded` handlers still run, since deferred scripts execute
450    /// first. The blocking `theme_init_script` stays inline so the `dark`
451    /// class lands pre-paint.
452    pub(crate) async fn render_document<'a>(
453        cx: &'a Cx,
454        title: String,
455        body: BoxView<'a>,
456    ) -> Result<BoxView<'a>> {
457        use topcoat::context::try_app_context;
458        // The build-time default: the `<html class>` a first-time visitor gets,
459        // and the fallback the blocking script uses when nothing is stored
460        // (GH #102, GH #184).
461        let default_dark = try_app_context::<DarkMode>(cx).is_some_and(|dm| dm.0);
462        let head: BoxView<'_> = match try_app_context::<ShellAssets>(cx).copied() {
463            Some(ShellAssets { stylesheet, font }) => view! {
464                cx =>
465                topcoat::dev::script()
466                argentum_ui::theme_init_script(default_dark: default_dark)
467                topcoat::runtime::script()
468                topcoat::font::link(font: font)
469                <link rel="stylesheet" href=(stylesheet)>
470                <script src=(argentum_ui::SIDEBAR_JS) defer=""></script>
471                <script src=(argentum_ui::THEME_JS) defer=""></script>
472                <script src=(argentum_ui::DIALOG_JS) defer=""></script>
473                <script src=(argentum_ui::BULK_JS) defer=""></script>
474                <script src=(argentum_ui::FILTERS_JS) defer=""></script>
475                <script src=(argentum_ui::LIVE_SEARCH_JS) defer=""></script>
476                <script src=(argentum_ui::SELECTS_JS) defer=""></script>
477                <script src=(argentum_ui::VARIANT_JS) defer=""></script>
478                <script src=(argentum_ui::NOTIFICATION_JS) defer=""></script>
479                <script src=(argentum_ui::MUTATION_SUBMIT_JS) defer=""></script>
480            }
481            .boxed(),
482            None => view! {
483                cx =>
484                topcoat::dev::script()
485                argentum_ui::theme_init_script(default_dark: default_dark)
486            }
487            .boxed(),
488        };
489        let html_class = default_dark.then_some("dark");
490        Ok(view! {
491            cx =>
492            <!DOCTYPE html>
493            <html class=(html_class)>
494                <head>
495                    <title>(title)</title>
496                    (head)
497                </head>
498                <body>(body)</body>
499            </html>
500        }
501        .boxed())
502    }

711    /// The opening tag that starts at `start`, sliced up to the `>` closing it.
712    ///
713    /// `Attributes` renders in no guaranteed order (topcoat#122), so a test
714    /// locates a tag by whichever attribute it can and asserts on the whole
715    /// tag. Quoting is honoured, so a `>` inside an attribute value (Tailwind
716    /// selectors and arrow-function handlers both carry them) does not end
717    /// the slice.
718    fn opening_tag_at(html: &str, start: usize) -> &str {
719        let mut quoted = false;
720        for (offset, byte) in html.as_bytes()[start..].iter().enumerate() {
721            match byte {
722                b'"' => quoted = !quoted,
723                b'>' if !quoted => return &html[start..start + offset],
724                _ => {}
725            }
726        }
727        panic!("unterminated tag at byte {start} in {html}");
728    }
