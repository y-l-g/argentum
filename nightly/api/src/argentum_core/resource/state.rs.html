16/// The live table's browser state (GH #151).
17///
18/// The page owns these signals and hands their handles to the `table_search`
19/// shard through `Table::render_live_with_state`; each tracked read inside
20/// the shard becomes a `dep` marker the browser watches, so writing any signal
21/// re-renders the table in place — no navigation, no scroll jump. Sort links,
22/// the pager, the filter transport, the bulk selection, and the clear links
23/// rendered by the table write them.
24///
25/// The two carriers meet in exactly two methods (GH #224):
26/// [`TableState::to_signals`] seeds these handles from a parsed state, and
27/// [`TableSignals::to_state`] rebuilds the state from their current values.
28/// A new interaction dimension is a field here plus one arm in each, instead
29/// of a hand-written conversion at every seam.
30///
31/// `q`/`filters`/`sort`/`dir`/`group_by` reset the cursor when they change;
32/// [`Self::cursor`] pages within the current result set. All values are
33/// untrusted by the time the shard reads them back (the client owns the
34/// signal).
35#[derive(Clone)]
36pub struct TableSignals {
37    /// `?q=` — the search term (escaped substring match, GH #116).
38    pub q: Signal<String>,
39    /// `?filters=` — the composed `key:value,key2:value2` transport.
40    pub filters: Signal<String>,
41    /// `?sort=` — the active sort column name (`""` = the table default).
42    pub sort: Signal<String>,
43    /// `?dir=` — `asc`/`desc` for [`Self::sort`].
44    pub dir: Signal<String>,
45    /// The live cursor (GH #166), as one signal: `""` (no cursor),
46    /// `after:<token>` or `before:<token>`. One signal makes the
47    /// `after`+`before` pair Toasty rejects unrepresentable in the browser — no
48    /// cross-write interleaving can produce it — and lets every result-set
49    /// transition clear pagination with a single write. Written through
50    /// `cursor_after` / `cursor_before` / `cursor_none`, read through
51    /// `split_cursor`.
52    pub cursor: Signal<String>,
53    /// `?group_by=` — the active grouping (`""` = ungrouped, GH #157).
54    /// Seeded from the page-load state and changed via navigation
55    /// (`?group_by=` links); no live control writes it yet, so it persists
56    /// across in-place reruns. A future control writing it must clear the
57    /// cursor like the other result-set dimensions.
58    pub group_by: Signal<String>,
59    /// The bulk selection: comma-separated record keys, `""` when nothing is
60    /// selected (GH #166). Row checkboxes carry no `checked` attribute —
61    /// `bulk.js` sets `checked` from the transport after every swap and change
62    /// — and the script writes the transport, whose bound `change` handler
63    /// writes this signal, so a live rerun re-renders the boxes from the
64    /// selection instead of dropping it. The shard carries the handle without
65    /// reading it: the table needs it to bind the boxes, but a checkbox click
66    /// must not reload rows.
67    pub bulk: Signal<String>,
68}

70/// The live cursor wire format (GH #166): `after:<token>` / `before:<token>`,
71/// with the empty string meaning "no cursor". One signal carries it, so the
72/// browser can never hold both cursors at once.
73const CURSOR_AFTER: &str = "after:";
74const CURSOR_BEFORE: &str = "before:";
75
76/// Wire value for a forward cursor (`?after=<token>`).
77pub(crate) fn cursor_after(token: &str) -> String {
78    format!("{CURSOR_AFTER}{token}")
79}
80
81/// Wire value for a backward cursor (`?before=<token>`).
82pub(crate) fn cursor_before(token: &str) -> String {
83    format!("{CURSOR_BEFORE}{token}")
84}
85
86/// Wire value for no cursor — what every result-set transition writes, and
87/// what a fresh page seeds.
88pub(crate) fn cursor_none() -> String {
89    String::new()
90}

92/// Split the live cursor wire value into the `(after, before)` pair the loader
93/// consumes. At most one side is ever `Some`: a value naming neither direction
94/// (a tampered signal, or one the client never sent) degrades to "no cursor" —
95/// the drop-pagination retry contract of GH #110 — rather than the pair error
96/// of GH #155 that a URL carrying both cursors reaches at load time.
97pub(crate) fn split_cursor(wire: &str) -> (Option<String>, Option<String>) {
98    let wire = wire.trim();
99    for (prefix, forward) in [(CURSOR_AFTER, true), (CURSOR_BEFORE, false)] {
100        if let Some(token) = wire.strip_prefix(prefix) {
101            let token = token.trim();
102            if token.is_empty() {
103                break;
104            }
105            return if forward {
106                (Some(token.to_string()), None)
107            } else {
108                (None, Some(token.to_string()))
109            };
110        }
111    }
112    (None, None)
113}

115/// Whether `key` is selected in the live bulk wire (GH #166).
116///
117/// The wire is comma-delimited on both ends — `,a,b,`, empty when nothing is
118/// selected — so the client-side `checked` binding tests membership with a
119/// plain `contains(",<key>,")` instead of a substring test that would confuse
120/// `b` with `ab`. [`parse_bulk_ids`](crate::panel) already ignores the empty
121/// segments the delimiters produce, so the same wire is the form transport.
122#[cfg(test)]
123pub(crate) fn bulk_wire_contains(wire: &str, key: &str) -> bool {
124    wire.split(',').any(|segment| segment == key)
125}

127/// One executed page of rows for `Table::render`.
128///
129/// For paginated tables build it from toasty's `Page` via
130/// [`Self::from_toasty_page`] (which URL-encodes the engine cursors); for
131/// unpaginated tables `Vec<M>` converts directly. An absent cursor simply
132/// means no Previous/Next link is rendered — the chrome never invents pages.
133#[derive(Debug, Clone)]
134pub struct TablePage<M> {
135    /// The rows of this page.
136    pub rows: Vec<M>,
137    /// Encoded cursor for the next page (`?after=`), when one exists.
138    pub next_cursor: Option<String>,
139    /// Encoded cursor for the previous page (`?before=`), when one exists.
140    pub prev_cursor: Option<String>,
141}
142
143impl<M> From<Vec<M>> for TablePage<M> {
144    fn from(rows: Vec<M>) -> Self {
145        Self {
146            rows,
147            next_cursor: None,
148            prev_cursor: None,
149        }
150    }
151}

153impl<M: toasty::schema::Model> TablePage<M> {
154    /// Wrap a toasty cursor-pagination result, encoding its cursors for URLs.
155    ///
156    /// # Errors
157    ///
158    /// Errors when a cursor contains a value the URL codec cannot represent
159    /// (see `crate::cursor`).
160    pub fn from_toasty_page(page: toasty::stmt::Page<M>) -> Result<Self> {
161        Ok(Self {
162            rows: page.items,
163            next_cursor: page
164                .next_cursor
165                .as_ref()
166                .map(crate::cursor::encode)
167                .transpose()?,
168            prev_cursor: page
169                .prev_cursor
170                .as_ref()
171                .map(crate::cursor::encode)
172                .transpose()?,
173        })
174    }
175}

187/// Request-scoped table state, parsed from the current URL query.
188///
189/// The single parse point shared by loaders (the search term, ordering via
190/// `Table::order_bys_for`) and render (active sort, toolbar values,
191/// pagination links), so the URL is the one truth for list state. The fixed parameter
192/// names assume one table per page — per-table prefixes are deferred until a
193/// real page needs two tables.
194#[derive(Debug, Clone, Default, PartialEq, Eq)]
195pub struct TableState {
196    /// `?q=` — trimmed and clamped to `MAX_QUERY_TERM` chars; `None` when
197    /// absent or blank.
198    pub search: Option<String>,
199    /// `?sort=` + `?dir=` — `None` when absent or blank.
200    pub sort: Option<Sort>,
201    /// `?after=` — encoded forward cursor.
202    pub after: Option<String>,
203    /// `?before=` — encoded backward cursor.
204    pub before: Option<String>,
205    /// `?filters=` — `key:value,key2:value2` (comma-separated, colon-delimited),
206    /// bounded at parse time by `MAX_FILTERS_PARAM`/`MAX_FILTER_SEGMENTS`
207    /// (GH #205).
208    pub filters: HashMap<String, String>,
209    /// `?filters=` segments that carry no `key:value` pair (GH #148): kept so
210    /// `Table::unapplied_filters` can flag them (list banner, export 400)
211    /// instead of silently dropping them, and so [`Self::filters_param`]
212    /// round-trips them — a link built from this state keeps the warning
213    /// until a valid `?filters=` replaces it.
214    pub malformed_filters: Vec<String>,
215    /// `?group_by=` — field name to group by (in-memory, `count` summarizer).
216    pub group_by: Option<String>,
217    /// `?delete=` — the row key whose delete confirmation dialog opens on the
218    /// list page (GH #151). The dialog's confirmed POST re-enters the delete
219    /// route; the parameter itself is never a write.
220    pub delete: Option<String>,
221    /// `?open=false` — set by `dialog.js` when Escape/backdrop dismisses the
222    /// delete dialog, so the next render stays closed. Absent (or `true`)
223    /// renders it open.
224    pub open: Option<bool>,
225}

227/// Longest `?filters=` transport parsed (GH #205): the live shard hands this
228/// the client-owned `filters` signal, and the router buffers shard bodies up
229/// to megabytes — so the same bounded-echoed-state posture as
230/// [`MAX_QUERY_TERM`](crate::query_term::MAX_QUERY_TERM) has to hold here,
231/// where the transport is parsed, rather than at the shard that happens to read
232/// it.
233pub(crate) const MAX_FILTERS_PARAM: usize = 1024;
234
235/// Most segments one `?filters=` transport may carry (GH #205): the byte cap
236/// alone still admits a thousand one-character segments, and each surviving
237/// segment becomes a map entry every rebuilt URL echoes.
238pub(crate) const MAX_FILTER_SEGMENTS: usize = 32;
239
240/// The one segment an over-long or over-full `?filters=` collapses to
241/// (GH #205).
242///
243/// It carries no `key:value` pair, so it rides the GH #148 malformed channel:
244/// [`Table::unapplied_filters`](crate::resource::Table::unapplied_filters)
245/// flags it (the list warns, the export refuses with 400 instead of exporting
246/// an over-broad CSV) and [`TableState::filters_param`] re-emits it, so the
247/// warning survives pagination and sort links.
248pub(crate) const FILTERS_OVERFLOW_SEGMENT: &str = "filters=overflow";

250impl TableState {
251    /// Parse the state from the request in `cx`.
252    ///
253    /// A blank or unknown query parses as neutral state rather than failing
254    /// the request. Duplicate keys (`?filters=a&filters=b`) resolve to the
255    /// first occurrence, so a repeated filter never vanishes: rejecting the
256    /// duplicate would fail the whole decode, and answering empty state instead
257    /// would drop every filter — including export's fail-closed guard (GH #93).
258    /// Cursor errors still surface later, at decode time, where they are
259    /// precise — including the conflicting `after` + `before` pair, which fails
260    /// at load time (GH #155). Renders without a request context (e.g. unit
261    /// tests) get neutral state.
262    pub fn from_cx(cx: &Cx) -> Self {
263        let Some(parts) = topcoat::context::try_request_context::<http::request::Parts>(cx) else {
264            return Self::default();
265        };
266        let params = first_wins_query_params(parts.uri.query().unwrap_or(""));
267        Self::from_parts(|key| params.get(key).map(String::as_str))
268    }
269
270    /// One shared constructor behind [`Self::from_cx`] and
271    /// [`Self::from_live_args`] (GH #133, GH #206): every query-state parse
272    /// funnels through one contract, so the public live-args entry point — the
273    /// documented seam for a page owning its own signals — cannot be the
274    /// looser one. `q` is trimmed and clamped to
275    /// [`MAX_QUERY_TERM`](crate::query_term::MAX_QUERY_TERM), `dir` is
276    /// trimmed before comparing, and the `filters` transport is bounded at
277    /// [`MAX_FILTERS_PARAM`]/[`MAX_FILTER_SEGMENTS`].
278    ///
279    /// The live args arrive named at the single `match` below, where a
280    /// transposed positional pair would not compile silently.
281    fn from_parts<'a>(get: impl Fn(&str) -> Option<&'a str>) -> Self {
282        let non_empty = |v: Option<&str>| {
283            v.map(str::trim)
284                .filter(|t| !t.is_empty())
285                .map(str::to_string)
286        };
287        let (filters, malformed_filters) = parse_filters_param(get("filters").unwrap_or_default());
288        Self {
289            search: get("q").map(clamp_query_term).filter(|t| !t.is_empty()),
290            sort: non_empty(get("sort")).map(|column| Sort {
291                column,
292                descending: get("dir").map(str::trim) == Some("desc"),
293            }),
294            after: non_empty(get("after")),
295            before: non_empty(get("before")),
296            filters,
297            malformed_filters,
298            group_by: non_empty(get("group_by")),
299            // The delete dialog is opt-in through `?delete=`; `?open=false`
300            // is the dismissal mirror `dialog.js` writes (GH #151). Any other
301            // `open` value stays neutral (open).
302            delete: non_empty(get("delete")),
303            open: match get("open") {
304                Some("false") => Some(false),
305                Some("true") => Some(true),
306                _ => None,
307            },
308        }
309    }

311    /// Serialized `filters` for URL (`key:value,key2:value2`), or `None` when empty.
312    ///
313    /// Keys/values escape `%`, `:`, `,` (`%25`/`%3A`/`%2C`, GH #93) so a
314    /// free-text value like `a,b` round-trips instead of splitting.
315    ///
316    /// This is the expensive half of a URL projection (a `format!` per pair,
317    /// a sort, a join, and a percent-encode per byte), so a table render must
318    /// encode it a bounded number of times — never once per row (GH #205).
319    /// `TableState::row_url_base` exists to make that structural.
320    pub fn filters_param(&self) -> Option<String> {
321        #[cfg(test)]
322        FILTERS_PARAM_ENCODES.with(|count| count.set(count.get() + 1));
323        if self.filters.is_empty() && self.malformed_filters.is_empty() {
324            return None;
325        }
326        let mut pairs: Vec<String> = self
327            .filters
328            .iter()
329            .map(|(k, v)| {
330                format!(
331                    "{}:{}",
332                    encode_filter_component(k),
333                    encode_filter_component(v)
334                )
335            })
336            .collect();
337        pairs.sort();
338        // Malformed segments ride along verbatim (GH #148): they have no
339        // colon to protect and re-enter `parse_filters_param` as malformed on
340        // the next request, keeping the banner (and export's fail-closed 400)
341        // alive across pagination.
342        pairs.extend(self.malformed_filters.iter().cloned());
343        Some(pairs.join(","))
344    }

346    /// URL projection: `TableState` owns the table's URL vocabulary (GH #153).
347    /// Callers ask for a user intent, never a parameter list, so adding a
348    /// parameter cannot silently drop it from half the links (GH #93).
349    ///
350    /// One private encoder ([`Self::project_url`]) holds the vocabulary in
351    /// canonical order `q, sort, dir, filters, group_by, after, before`
352    /// (`delete` appended by its intent). The parser is first-wins with unique
353    /// keys, so order is semantically irrelevant.
354    ///
355    /// Expects `group_by` pre-normalized: render seams normalize through
356    /// [`Table::normalize_state`], so the projection echoes `state.group_by`
357    /// as-is. `open` is never emitted by any link; `delete` only by
358    /// [`Self::row_url_base`]'s dialog intent.
359    ///
360    /// Full state, including cursors; never `delete`/`open`. The streamed
361    /// retry link for failures that keep their evidence (GH #98).
362    pub(crate) fn list_url(&self, path: &str) -> String {
363        let filters = self.filters_param();
364        self.project_url(
365            path,
366            self.search.as_deref(),
367            self.sort_pair(),
368            filters.as_deref(),
369            self.group_by.as_deref(),
370            self.after.as_deref(),
371            self.before.as_deref(),
372            None,
373        )
374    }

531    /// Rebuild list state from live-search shard args (GH #74).
532    ///
533    /// Shard requests hit the `table_search` shard's own endpoint, so
534    /// [`Self::from_cx`] would see the endpoint URI — not the list page's
535    /// query. The page hands over the current values of its interaction
536    /// signals instead: the search term, the filter transport, the sort column
537    /// and its direction, and the page's grouping. Live search resets
538    /// pagination (`after`/`before` are always `None` — a new search is a new
539    /// result set, same as the GET toolbar) and keeps the page's `group_by`.
540    ///
541    /// Every argument is client-owned by the time the shard reads it back, so
542    /// this applies [`Self::from_cx`]'s bounds through the shared
543    /// `Self::from_parts` (GH #206) — the public constructor is not the
544    /// looser one.
545    ///
546    /// [`TableSignals::to_state`] is the shard's call site for this (GH #224):
547    /// it reads the signals and passes their values here, so the shard never
548    /// rebuilds state by hand.
549    pub fn from_live_args(
550        q: &str,
551        filters_param: &str,
552        sort: &str,
553        dir: &str,
554        group_by: &str,
555    ) -> Self {
556        // `after`/`before`/`delete`/`open` stay `None`: live search resets
557        // pagination and the delete dialog with every keystroke (a new search
558        // is a new result set, same as the GET toolbar), and the panel
559        // renders the dialog outside the shard region for live tables
560        // (GH #151). Missing keys read as absent through `from_parts`.
561        Self::from_parts(|key| match key {
562            "q" => Some(q),
563            "filters" => Some(filters_param),
564            "sort" => Some(sort),
565            "dir" => Some(dir),
566            "group_by" => Some(group_by),
567            _ => None,
568        })
569    }

571    /// Seed the page's interaction signals from this state (GH #224).
572    ///
573    /// The one state→signal conversion: the page owns the handles
574    /// ([`TableSignals`]) and every control it renders writes them, so the
575    /// values the page loaded with are what the signals start from. The
576    /// cursor is one signal (GH #166) seeded from the `after`/`before` pair
577    /// in this state; the bulk selection starts empty (the page never loads a
578    /// selection); `group_by` seeds from the page-load value and persists
579    /// across in-place reruns (GH #157).
580    ///
581    /// Call it with the *parsed* state, before normalizing: an unknown
582    /// `?group_by=` seeds the signal as written, and the shard's
583    /// [`TableSignals::to_state`] + `Table::normalize_state` drop it on the
584    /// way back in, exactly as the GET path does (GH #153).
585    ///
586    /// Creates the signals, so it carries [`topcoat::runtime::signal`]'s
587    /// contract: call it while a view is collecting signal declarations — the
588    /// panel calls it from the live page's render, and the declarations ride
589    /// that page's hoisted parts.
590    pub fn to_signals(&self, cx: &Cx) -> TableSignals {
591        TableSignals {
592            q: signal(cx, || self.search.clone().unwrap_or_default()),
593            filters: signal(cx, || self.filters_param().unwrap_or_default()),
594            // The projection's own spelling of `?sort=`/`?dir=` (GH #153), so
595            // the signal and every link agree on the direction word.
596            sort: signal(cx, || {
597                self.sort_pair()
598                    .map(|(column, _)| column.to_string())
599                    .unwrap_or_default()
600            }),
601            dir: signal(cx, || {
602                self.sort_pair()
603                    .map(|(_, dir)| dir.to_string())
604                    .unwrap_or_else(|| "asc".to_string())
605            }),
606            cursor: signal(cx, || match (&self.after, &self.before) {
607                (Some(token), _) => cursor_after(token),
608                (None, Some(token)) => cursor_before(token),
609                (None, None) => cursor_none(),
610            }),
611            group_by: signal(cx, || self.group_by.clone().unwrap_or_default()),
612            bulk: signal(cx, String::new),
613        }
614    }
615}

617/// The live seam's other direction (GH #224): the signals the page owns,
618/// read back into the request state the shard loads and renders with.
619impl TableSignals {
620    /// Rebuild request state from the live signals (GH #224).
621    ///
622    /// The one signal→state conversion: every value is client-owned by the time
623    /// the shard reads it back, so it goes through
624    /// [`TableState::from_live_args`] — the same `q` clamp and `filters` bound
625    /// the GET path applies (GH #148, GH #205, GH #206) — and the one cursor
626    /// wire is split into the `(after, before)` pair the loader consumes
627    /// (GH #166). Pagination and the delete dialog always reset (GH #151); a
628    /// token that does not decode fails loudly at load time (GH #158).
629    pub fn to_state(&self) -> TableState {
630        let mut state = TableState::from_live_args(
631            &self.q.get(),
632            &self.filters.get(),
633            &self.sort.get(),
634            &self.dir.get(),
635            &self.group_by.get(),
636        );
637        (state.after, state.before) = split_cursor(&self.cursor.get());
638        state
639    }
640}

717/// Query-string pairs with the first occurrence winning.
718///
719/// A duplicate key keeps its first value rather than failing the decode:
720/// rejecting it would fail the whole parse, and answering empty state instead
721/// would silently drop filters and export's fail-closed guard (GH #93). Unknown
722/// keys are ignored, like the typed decode.
723fn first_wins_query_params(query: &str) -> HashMap<String, String> {
724    let mut out = HashMap::new();
725    for (key, value) in form_urlencoded::parse(query.as_bytes()) {
726        if !out.contains_key(key.as_ref()) {
727            out.insert(key.into_owned(), value.into_owned());
728        }
729    }
730    out
731}

733/// Parse `filters` query param: `key:value,key2:value2` (trimmed, blank ignored).
734///
735/// `,`/`:`/`%` inside keys/values are `%`-escaped by [`TableState::filters_param`]
736/// (GH #93); decoding restores them. Duplicate keys keep the first occurrence
737/// instead of silent last-wins.
738///
739/// An over-long or over-full transport is refused *whole* — never partially
740/// applied, which would silently drop filters the caller did send — and the
741/// refusal rides the GH #148 malformed channel as `FILTERS_OVERFLOW_SEGMENT`
742/// (GH #205), so the list warns and the export 400s instead of running
743/// unfiltered. The bound lives here, where the value is parsed: the live shard
744/// hands this the client-owned `filters` signal, which the router buffers up to
745/// megabytes of.
746fn parse_filters_param(raw: &str) -> (HashMap<String, String>, Vec<String>) {
747    // The length test first: it is O(1) and short-circuits the segment scan
748    // for the oversized input this bound exists for.
749    if raw.len() > MAX_FILTERS_PARAM || raw.split(',').count() > MAX_FILTER_SEGMENTS {
750        return (HashMap::new(), vec![FILTERS_OVERFLOW_SEGMENT.to_string()]);
751    }
752    let mut map = HashMap::new();
753    let mut malformed = Vec::new();
754    for part in raw.split(',') {
755        let part = part.trim();
756        if part.is_empty() {
757            continue;
758        }
759        // Split on the first *unescaped* colon: `%3A` stays inside the key/value,
760        // so a plain `split_once(':')` is correct on the encoded form.
761        let Some((k_enc, v_enc)) = part.split_once(':') else {
762            malformed.push(part.to_string());
763            continue;
764        };
765        let k = decode_filter_component(k_enc.trim());
766        let v = decode_filter_component(v_enc.trim());
767        if k.is_empty() || v.is_empty() {
768            malformed.push(part.to_string());
769        } else {
770            map.entry(k).or_insert(v);
771        }
772    }
773    (map, malformed)
774}

776/// Escape `%`, `:`, `,` inside a filter key/value (GH #93).
777fn encode_filter_component(s: &str) -> String {
778    s.replace('%', "%25")
779        .replace(':', "%3A")
780        .replace(',', "%2C")
781}
782
783/// Decode [`encode_filter_component`] (case-insensitive hex, single pass).
784fn decode_filter_component(s: &str) -> String {
785    s.replace("%2C", ",")
786        .replace("%2c", ",")
787        .replace("%3A", ":")
788        .replace("%3a", ":")
789        .replace("%25", "%")
790}

792/// Percent-encode a query parameter value (`unreserved` RFC 3986 set passes).
793fn encode_query_value(value: &str) -> String {
794    let mut out = String::with_capacity(value.len());
795    for b in value.bytes() {
796        match b {
797            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'_' | b'.' | b'~' => {
798                out.push(b as char);
799            }
800            _ => out.push_str(&format!("%{b:02X}")),
801        }
802    }
803    out
804}

806/// Percent-encode a single path segment (GH #96).
807///
808/// Row keys are `String` by contract, so `/`, `?`, `#`, `%`, `+` inside a key
809/// must not rewrite the action URL. Topcoat's `path_param_segment` returns
810/// the percent-decoded segment, so this round-trips; UUID keys pass through
811/// unchanged.
812pub(crate) fn encode_path_segment(value: &str) -> String {
813    encode_query_value(value)
814}

816/// FNV-1a (32-bit): stable across runs and Rust versions, no dependency.
817/// Used only to disambiguate DOM ids, never for anything security-relevant.
818fn fnv1a_32(s: &str) -> u32 {
819    let mut hash: u32 = 0x811c_9dc5;
820    for byte in s.bytes() {
821        hash ^= u32::from(byte);
822        hash = hash.wrapping_mul(0x0100_0193);
823    }
824    hash
825}

827/// Stable DOM id for a table row (GH #104): Topcoat's morph (#392) follows
828/// elements by `id` across reruns, so reorderable row content needs one in
829/// addition to the keyed-diff `key:`. Derived from the row key (stable for
830/// the record, unlike a loop index), sanitized to an HTML-safe token plus a
831/// short hash: distinct keys (`Ada Lovelace`, `Ada-Lovelace`) can sanitize to
832/// the same token, and duplicate DOM ids would make the morph follow one row.
833pub(crate) fn row_dom_id(key: &str) -> String {
834    dom_id("row", key)
835}
836
837/// Stable DOM id for a page-local group header (GH #219).
838///
839/// Same contract as [`row_dom_id`]: the header is injected, removed and moved
840/// as the page is re-sorted, so the in-place morph needs an id derived from the
841/// group label it belongs to rather than from its position in the page.
842pub(crate) fn group_header_dom_id(label: &str) -> String {
843    dom_id("group", label)
844}
845
846/// The one sanitizer behind both ids: `{prefix}-{token}-{hash}`.
847fn dom_id(prefix: &str, key: &str) -> String {
848    let mut out = String::with_capacity(prefix.len() + key.len() + 14);
849    out.push_str(prefix);
850    out.push('-');
851    for c in key.chars() {
852        if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | ':' | '.') {
853            out.push(c);
854        } else {
855            out.push('-');
856        }
857    }
858    out.push_str(&format!("-{:08x}", fnv1a_32(key)));
859    out
860}

862/// Build `path?k=v&…` from ordered optional parameters, skipping `None`.
863pub(crate) fn build_url(path: &str, params: &[(&str, Option<&str>)]) -> String {
864    let query = params
865        .iter()
866        .filter_map(|(k, v)| v.map(|v| format!("{k}={}", encode_query_value(v))))
867        .collect::<Vec<_>>()
868        .join("&");
869    if query.is_empty() {
870        path.to_string()
871    } else {
872        format!("{path}?{query}")
873    }
874}

1375    /// GH #166: the live cursor travels as one wire value, so the browser can
1376    /// never hold `after` and `before` at once — the pair Toasty rejects
1377    /// (GH #155) is unreachable from the live path.
1378    #[test]
1379    fn cursor_wire_carries_at_most_one_direction() {
1380        assert_eq!(cursor_after("tok"), "after:tok");
1381        assert_eq!(cursor_before("tok"), "before:tok");
1382        assert_eq!(cursor_none(), "");
1383        assert_eq!(
1384            split_cursor(&cursor_after("tok")),
1385            (Some("tok".into()), None)
1386        );
1387        assert_eq!(
1388            split_cursor(&cursor_before("tok")),
1389            (None, Some("tok".into()))
1390        );
1391        assert_eq!(split_cursor(&cursor_none()), (None, None));
1392        // Whitespace from the signal is tolerated, like the GET path's trims.
1393        assert_eq!(
1394            split_cursor("  after: tok  "),
1395            (Some("tok".to_string()), None)
1396        );
1397        // A tampered or half-written value degrades to no cursor (GH #110's
1398        // drop-pagination retry contract) instead of erroring the table.
1399        assert_eq!(split_cursor("tok"), (None, None));
1400        assert_eq!(split_cursor("after:"), (None, None));
1401        assert_eq!(split_cursor("before:"), (None, None));
1402        // Only the prefix is a direction; a token may contain colons.
1403        assert_eq!(split_cursor("after:a:b"), (Some("a:b".to_string()), None));
1404    }
