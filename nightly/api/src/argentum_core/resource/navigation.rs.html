1//! Sidebar navigation: [`NavigationItem`] and the [`NavTarget`] it points with.
2
3use super::Resource;
4
5/// Where a sidebar entry points (GH #165).
6///
7/// A [`Resource`] cannot name its own URL: [`Resource::navigation`] takes no
8/// `Cx` and no prefix, so the entry it declares by default carries no URL at
9/// all — [`NavTarget::Derived`] — and the [`Panel`](crate::panel::Panel) that
10/// owns the item resolves it from its own mount prefix plus the resource's
11/// [`slug`](Resource::slug). [`NavTarget::Url`] is a URL its author wrote out,
12/// and a Panel passes it through untouched.
13#[derive(Clone, Default, PartialEq, Eq)]
14pub enum NavTarget {
15    /// No URL yet: the owning Panel resolves it to `{prefix}/{slug}` of the
16    /// resource whose `navigation()` declared this item. What
17    /// [`NavigationItem::for_resource`] — and so the default
18    /// [`Resource::navigation`] — returns.
19    #[default]
20    Derived,
21    /// An explicit URL: a custom path, a query view, another panel's mount.
22    /// Active state is string matching (exact, or a slash-boundary prefix).
23    Url(String),
24}
25
26impl NavTarget {
27    /// The URL this target names, or `None` while it is still
28    /// [`Self::Derived`] — i.e. before the owning Panel has resolved it.
29    pub fn url(&self) -> Option<&str> {
30        match self {
31            Self::Derived => None,
32            Self::Url(url) => Some(url),
33        }
34    }
35}
36
37impl std::fmt::Debug for NavTarget {
38    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
39        match self {
40            Self::Derived => f.write_str("Derived"),
41            Self::Url(url) => f.debug_tuple("Url").field(url).finish(),
42        }
43    }
44}

46/// Sidebar entry derived from a `Resource` (see `CONTEXT.md`).
47#[derive(Clone, Debug, Default, PartialEq, Eq)]
48pub struct NavigationItem {
49    pub label: String,
50    /// Where this entry points. [`NavTarget::Derived`] until the owning Panel
51    /// resolves it — see [`NavTarget`]. Build items with
52    /// [`NavigationItem::for_resource`] or [`NavigationItem::at`] rather than
53    /// spelling the variant out.
54    pub target: NavTarget,
55    /// Sort key for the sidebar (GH #102): items render in stable `order`
56    /// order, so declaration order breaks ties. Resources declare in
57    /// `Panel::resource` order (all default `0`); a
58    /// [`Resource::navigation`] override interleaves by setting a lower value,
59    /// e.g. `NavigationItem { order: -1, ..NavigationItem::for_resource::<Self>() }`
60    /// pins above the resources.
61    pub order: i32,
62}
63
64impl NavigationItem {
65    /// The default sidebar entry for `R`: its
66    /// [`navigation_label`](Resource::navigation_label), and no URL yet.
67    ///
68    /// This is what [`Resource::navigation`] returns unless overridden — and
69    /// what an override decorates, e.g.
70    /// `NavigationItem { order: -1, ..NavigationItem::for_resource::<Self>() }`.
71    /// The resource cannot know where it is mounted, so the URL stays
72    /// [`NavTarget::Derived`] until the owning [`Panel`](crate::panel::Panel)
73    /// resolves it; an entry declared here can therefore never link at a mount
74    /// the resource guessed (GH #165).
75    pub fn for_resource<R: Resource>() -> Self {
76        Self {
77            label: R::navigation_label(),
78            target: NavTarget::Derived,
79            order: 0,
80        }
81    }
82
83    /// A sidebar entry at an explicit `url`.
84    ///
85    /// Active state is string matching: exact path, or a slash-boundary prefix,
86    /// so `/admin/users` is current on `/admin/users/create`.
87    pub fn at(label: impl Into<String>, url: impl Into<String>) -> Self {
88        Self {
89            label: label.into(),
90            target: NavTarget::Url(url.into()),
91            order: 0,
92        }
93    }
94
95    /// Resolve a [`NavTarget::Derived`] entry against the Panel that owns it
96    /// (GH #165), leaving an explicit target untouched.
97    ///
98    /// [`Resource::navigation`] cannot know its panel — it takes no `Cx` and no
99    /// prefix — so the entry it declares carries no URL. The Panel consumes it
100    /// through `Panel::resource`, which passes its own prefix and the
101    /// resource's mount slug; there is no panel-root case, since nothing
102    /// generates an entry without a resource behind it (GH #228).
103    ///
104    /// There is no guessing here: a URL an author wrote out — including one
105    /// that happens to look like `/admin/{slug}` — is a different
106    /// [`NavTarget`] variant and is never rewritten.
107    pub(crate) fn resolved(mut self, prefix: &str, slug: &str) -> Self {
108        if matches!(self.target, NavTarget::Derived) {
109            self.target = NavTarget::Url(format!("{}/{slug}", mount(prefix)));
110        }
111        self
112    }
113
114    /// The URL this entry points at, or `None` while it is unresolved — i.e.
115    /// still [`NavTarget::Derived`], not yet handed to a Panel. Panel-owned
116    /// items are always resolved (`Panel::resource`).
117    pub fn url(&self) -> Option<&str> {
118        self.target.url()
119    }
120
121    /// Whether this item is current for the given request path (without query):
122    /// an exact match, or a prefix match on a slash boundary (so
123    /// `/admin/users` is active on `/admin/users/create` but not on
124    /// `/admin/userships`). Uniform for every item — since resources mount at
125    /// `{prefix}/{slug}` (GH #39), no generated item points at the bare panel
126    /// prefix that needed the old root-exact special case.
127    ///
128    /// This is the whole active-state contract: `Panel::render_shell` takes the
129    /// request path as a parameter, so it can judge an item without a `Cx` (and
130    /// the shell stays testable without a full `http::request::Parts` in `Cx`).
131    /// The `Cx`-taking wrapper went with its last caller (GH #228).
132    pub fn is_current_path(&self, current_path: &str) -> bool {
133        let Some(url) = self.url() else {
134            // Unresolved: no URL to be current for.
135            return false;
136        };
137        if current_path == url {
138            return true;
139        }
140        current_path
141            .strip_prefix(url)
142            .is_some_and(|rest| rest.starts_with('/'))
143    }
144}
145
146/// The mount a panel prefix normalises to: `/admin` when it is empty — the same
147/// rule [`Panel::new`](crate::panel::Panel::new) applies.
148fn mount(prefix: &str) -> String {
149    let trimmed = prefix.trim_matches('/').trim();
150    if trimmed.is_empty() {
151        "/admin".to_string()
152    } else {
153        format!("/{trimmed}")
154    }
155}

157#[cfg(test)]
158mod tests {
159    use super::*;
160    use toasty::stmt::List;
161    use topcoat::context::Cx;
162
163    #[derive(Debug, Clone, toasty::Model)]
164    struct User {
165        #[key]
166        #[auto]
167        id: uuid::Uuid,
168        name: String,
169    }
170
171    struct UserResource;
172
173    impl Resource for UserResource {
174        type Model = User;
175
176        fn query(_cx: &Cx) -> toasty::stmt::Query<List<User>> {
177            // Custom scoping example: only users named Ada
178            toasty::stmt::Query::<List<User>>::all().filter(User::fields().name().eq("Ada"))
179        }
180    }
181
182    #[test]
183    fn for_resource_derives_label_only() {
184        // The default `Resource::navigation` entry: label from the pluralized
185        // model name, and no URL at all — the owning Panel supplies it, so a
186        // resource can never link at a mount it guessed (GH #165).
187        let item = NavigationItem::for_resource::<UserResource>();
188        assert_eq!(item.label, "Users");
189        assert!(matches!(item.target, NavTarget::Derived));
190        assert_eq!(item.url(), None);
191        assert_eq!(item.order, 0);
192        // Unresolved, it is current nowhere.
193        assert!(!item.is_current_path("/admin/users"));
194    }
195
196    /// GH #165: `Derived` is resolved by the owning Panel from its own prefix
197    /// plus the resource's slug — exactly once — while an explicit URL is the
198    /// author's, even one shaped like another panel's mount.
199    #[test]
200    fn derived_targets_resolve_against_the_owning_panel() {
201        let derived = NavigationItem::for_resource::<UserResource>();
202
203        // Non-`/admin` panel → this panel's mount, never `/admin/users`.
204        let resolved = derived.clone().resolved("/backoffice", "users");
205        assert_eq!(resolved.url(), Some("/backoffice/users"));
206        assert_eq!(resolved.label, "Users");
207        assert_eq!(
208            NavigationItem {
209                order: -1,
210                ..derived.clone()
211            }
212            .resolved("/backoffice", "users")
213            .order,
214            -1
215        );
216        // Already resolved: resolving again is a no-op, however the second
217        // panel is mounted.
218        assert_eq!(
219            resolved.clone().resolved("elsewhere", "users").url(),
220            Some("/backoffice/users")
221        );
222
223        // Mount normalisation follows `Panel::new`.
224        assert_eq!(
225            derived.clone().resolved("/admin", "users").url(),
226            Some("/admin/users")
227        );
228        assert_eq!(
229            derived.clone().resolved("/backoffice/", "users").url(),
230            Some("/backoffice/users")
231        );
232        assert_eq!(
233            derived.clone().resolved("", "users").url(),
234            Some("/admin/users")
235        );
236
237        // Explicit URLs survive verbatim — including `/admin/users`, which the
238        // old origin-mount heuristic could not tell from a derived default.
239        for url in [
240            "/admin/users",
241            "/admin/posts?filters=status:draft",
242            "/backoffice/users/drafts",
243            "/reports/users",
244        ] {
245            let spelled_out = NavigationItem::at("Users", url);
246            assert_eq!(
247                spelled_out.resolved("backoffice", "users").url(),
248                Some(url),
249                "explicit URL must survive resolution"
250            );
251        }
252    }

254    #[test]
255    fn slugs_follow_the_filament_convention() {
256        // UserResource → strip "Resource" → pluralize → kebab-case
257        assert_eq!(<UserResource as Resource>::slug(), "users");
258        assert_eq!(UserResource::navigation_label(), "Users");
259    }

261    #[test]
262    fn navigation_item_is_current_path() {
263        let users = NavigationItem::at("Users", "/admin/users");
264        let showcase = NavigationItem::at("Showcase", "/admin/showcase");
265        // exact
266        assert!(users.is_current_path("/admin/users"));
267        assert!(showcase.is_current_path("/admin/showcase"));
268        // slash-boundary — sub-pages active
269        assert!(users.is_current_path("/admin/users/create"));
270        assert!(showcase.is_current_path("/admin/showcase/table"));
271        // slash-boundary — near-misses inactive
272        assert!(!users.is_current_path("/admin/userships"));
273        assert!(!showcase.is_current_path("/admin/showcases"));
274        assert!(!showcase.is_current_path("/admin/showcase-table"));
275        // unrelated
276        assert!(!users.is_current_path("/other"));
277        assert!(!showcase.is_current_path("/admin/users"));
278    }
