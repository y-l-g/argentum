1//! Read-only relation rendering for detail pages (GH #187).
2//!
3//! A detail page shows a record's related rows — a post's comments — from the
4//! rows `Resource::query`'s `include` already loaded. This renders them, and
5//! deliberately renders *only* them: no pager, no search, no row actions, no
6//! bulk column. A relation on a record page is a fixed, already-loaded set,
7//! and the list's chrome exists to narrow a query this page never runs.
8//!
9//! Everything here is owned: a column's projection returns a `String`, so the
10//! rendered view borrows the request context and nothing else. That is what
11//! lets `Resource::view_relations(cx, record)` return a view that outlives the
12//! record it read — the page is rendered before the handler's `record` binding
13//! drops, and the borrow checker says so if it is not.
14
15use argentum_ui::{table, table_body, table_cell, table_head, table_header, table_row};
16use topcoat::{
17    context::Cx,
18    view::{BoxView, ViewExt, view},
19};
20
21/// One column of a relation's read-only table (GH #187).
22///
23/// The projection is the same shape a list column uses — a typed closure over
24/// the related record — minus everything that only makes sense against a
25/// query: no `sortable`, no `searchable`, no key.
26pub struct RelationColumn<R> {
27    label: String,
28    display: Box<dyn Fn(&R) -> String + Send + Sync>,
29}
30
31impl<R> RelationColumn<R> {
32    /// Declare a column by label and projection.
33    ///
34    /// Named for the shape it is, as
35    /// [`TextColumn::computed`](crate::resource::TextColumn::computed) is on the list's side: a
36    /// relation column has no lens to bind, because the related rows arrive as values rather
37    /// than as a query. What a column *is* is its projection.
38    pub fn computed(
39        label: impl Into<String>,
40        display: impl Fn(&R) -> String + Send + Sync + 'static,
41    ) -> Self {
42        Self {
43            label: label.into(),
44            display: Box::new(display),
45        }
46    }
47
48    /// The column's heading.
49    pub fn label(&self) -> &str {
50        &self.label
51    }
52
53    /// This column's cell for `row`.
54    pub fn render_cell(&self, row: &R) -> String {
55        (self.display)(row)
56    }
57}
58
59impl<R> std::fmt::Debug for RelationColumn<R> {
60    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
61        f.debug_struct("RelationColumn")
62            .field("label", &self.label)
63            .finish_non_exhaustive()
64    }
65}

67/// The columns of a relation's table, declared as a tuple or a single column.
68///
69/// A separate collection from the list's `IntoColumns` because the two tables
70/// answer different questions: one is queryable, this one is not.
71pub struct RelationColumns<R> {
72    pub(crate) columns: Vec<RelationColumn<R>>,
73}
74
75impl<R> RelationColumns<R> {
76    /// Collect one or more columns.
77    pub fn columns(columns: impl IntoRelationColumns<R>) -> Self {
78        columns.into_relation_columns()
79    }
80}
81
82/// What can be collected into [`RelationColumns`].
83pub trait IntoRelationColumns<R> {
84    fn into_relation_columns(self) -> RelationColumns<R>;
85}
86
87impl<R> IntoRelationColumns<R> for RelationColumn<R> {
88    fn into_relation_columns(self) -> RelationColumns<R> {
89        RelationColumns {
90            columns: vec![self],
91        }
92    }
93}
94
95impl<R, A, B> IntoRelationColumns<R> for (A, B)
96where
97    A: IntoRelationColumns<R>,
98    B: IntoRelationColumns<R>,
99{
100    fn into_relation_columns(self) -> RelationColumns<R> {
101        let mut columns = self.0.into_relation_columns().columns;
102        columns.extend(self.1.into_relation_columns().columns);
103        RelationColumns { columns }
104    }
105}

107/// Render `rows` as a titled, read-only table (GH #187).
108///
109/// `rows` is the related records the record already carries — the caller passes
110/// `record.comments.get().iter().cloned().collect()` or the equivalent — so this
111/// runs no query. An empty set renders the title with an honest "none" line
112/// rather than an empty table, which would read as a failure to load.
113///
114/// The rows are rendered in the order given; a detail page shows what the query
115/// loaded, and `Resource::query` owns that order.
116pub fn render_relation<'a, R>(
117    cx: &'a Cx,
118    title: &str,
119    columns: RelationColumns<R>,
120    rows: &[R],
121) -> BoxView<'a> {
122    let title = title.to_string();
123    // Own everything before the `view!` block: the emitted view must borrow
124    // the request context and nothing else, or the caller's `record` (which
125    // owns these rows) would have to outlive the page.
126    let heads: Vec<String> = columns
127        .columns
128        .iter()
129        .map(|c| c.label().to_string())
130        .collect();
131    let cells: Vec<Vec<String>> = rows
132        .iter()
133        .map(|row| {
134            columns
135                .columns
136                .iter()
137                .map(|column| column.render_cell(row))
138                .collect()
139        })
140        .collect();
141    let has_rows = !cells.is_empty();
142    // Row ids are positional: this table does not reorder or swap, so it needs
143    // no record key — the list's `Table::id` contract exists for keyed diffs
144    // and action URLs, and neither exists here.
145    view! {
146        cx =>
147        <section class="flex flex-col gap-3">
148            <h2 class="text-base font-semibold">(title)</h2>
149            if !has_rows {
150                <p class="text-sm text-muted-foreground">"None."</p>
151            } else {
152                table(
153                    table_header(
154                        table_row(
155                            for head in heads {
156                                table_head((head))
157                            }
158                        )
159                    )
160                    table_body(
161                        for row in cells {
162                            table_row(
163                                for cell in row {
164                                    table_cell((cell))
165                                }
166                            )
167                        }
168                    )
169                )
170            }
171        </section>
172    }
173    .boxed()
174}
