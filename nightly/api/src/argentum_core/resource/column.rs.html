9/// The relations a query must `include`, named in the resource's vocabulary.
10///
11/// A column's projection closure reads relations off the row (`|p| p.author
12/// .get().name.clone()`), and Toasty has no instance→field reflection
13/// (upstream #119), so the framework cannot see *which* relation a closure
14/// touches. Each column therefore **declares** the includes its closure reads
15/// ([`TextColumn::needs`]), a [`Table`](super::Table) gathers the declarations
16/// of the columns it renders into one `IncludeNeeds`, and
17/// [`Resource::export_query`](super::Resource::export_query) answers `wants`
18/// per `include(..)` call.
19///
20/// The names are an opaque vocabulary shared between the declaring column and
21/// the resource that maps them onto `include(..)` calls, because includes are
22/// typed (`Include<Post, Author>`) and a type-erased column cannot name one.
23/// Nothing else reads them: an unknown name is not an error, it just never
24/// matches a branch.
25#[derive(Clone, Debug, Default)]
26pub struct IncludeNeeds {
27    names: BTreeSet<&'static str>,
28}
29
30impl IncludeNeeds {
31    /// Whether `name` was declared by a rendered column.
32    ///
33    /// This is the one question a resource's
34    /// [`export_query`](super::Resource::export_query) asks, once per
35    /// `include(..)` it could add.
36    pub fn wants(&self, name: &str) -> bool {
37        self.names.contains(name)
38    }
39
40    /// Whether no column declared anything — the narrowed query needs no
41    /// relation at all.
42    pub fn is_empty(&self) -> bool {
43        self.names.is_empty()
44    }
45}
46
47impl FromIterator<&'static str> for IncludeNeeds {
48    fn from_iter<T: IntoIterator<Item = &'static str>>(iter: T) -> Self {
49        Self {
50            names: iter.into_iter().collect(),
51        }
52    }
53}

55/// `IncludeNeeds::from(["author", "comments"])` — the whole set up front, which
56/// is what a resource's `query` needs (its `export_query` gets the set handed
57/// to it instead).
58impl<const N: usize> From<[&'static str; N]> for IncludeNeeds {
59    fn from(names: [&'static str; N]) -> Self {
60        names.into_iter().collect()
61    }
62}

64/// The share of the table a [`ColumnWidth::Narrow`] column claims, in whole
65/// percent.
66pub(crate) const NARROW_DEFAULT_PERCENT: u8 = 10;
67
68/// The width a [`TextColumn`] claims in the table's fixed layout.
69///
70/// Widths are **shares of the table**, so what a table declares is a fraction
71/// of its container rather than a length that can outgrow it: the columns that
72/// declare none take what the declared ones leave. A length
73/// ([`Rem`](Self::Rem)) is the exception — lengths do not shrink with the
74/// table, and a table whose lengths exceed its width gives the columns that
75/// declare none no space at all, header text included.
76///
77/// The renderer writes the width into the column's `th` and every row's `td`
78/// as an inline `style` attribute — data, never a generated Tailwind class.
79/// Tailwind generates only the class literals it finds in source, so a width
80/// assembled at render (`w-[{n}%]`) would emit no CSS at all (ADR-0006); a
81/// declared width is read by the layout directly.
82///
83/// A column's **kind** picks the default: `TextColumn::r#for` binds a `String`
84/// field, so its cells hold the row's own text — a title, a name, a body — and
85/// it defaults to [`Wide`](Self::Wide), taking a share of what the declared
86/// columns leave; [`TextColumn::computed`] derives its cell (a status, a
87/// boolean, a date, a count) and defaults to [`Narrow`](Self::Narrow), a share
88/// of the table. [`TextColumn::width`] overrides either, which is the seam for
89/// a column whose content disagrees with its kind.
90#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
91pub enum ColumnWidth {
92    /// Take a share of whatever the declared columns leave: the column
93    /// declares no width, and `table-fixed` splits the remainder between the
94    /// wide columns instead of measuring the rows currently rendered.
95    #[default]
96    Wide,
97    /// A share of the table for a status, boolean, date or count cell: the
98    /// default for a [`TextColumn::computed`] column. The renderer resolves
99    /// the share (10% nominally) against the table's other kind defaults.
100    Narrow,
101    /// An explicit length in whole rem: `Rem(14)` declares `14rem`. A length
102    /// does not shrink with the table, so a table narrower than the lengths it
103    /// declares gives the columns that declare none no space at all.
104    Rem(u8),
105    /// An explicit share of the table in whole percent: `Percent(30)`
106    /// declares `30%`.
107    Percent(u8),
108}
109
110impl ColumnWidth {
111    /// The share of the table this column claims as a **kind default**, in
112    /// whole percent, or `None` for a column that declares an explicit width
113    /// or none at all.
114    ///
115    /// A nominal: the renderer scales the kind defaults down together when
116    /// their total would leave the wide columns less than their share of the
117    /// table.
118    pub(crate) fn default_percent(self) -> Option<u8> {
119        match self {
120            Self::Narrow => Some(NARROW_DEFAULT_PERCENT),
121            Self::Wide | Self::Rem(_) | Self::Percent(_) => None,
122        }
123    }
124
125    /// The `style` attribute value an **explicit** declaration emits, or
126    /// `None` for [`Wide`](Self::Wide), which declares nothing, and for
127    /// [`Narrow`](Self::Narrow), whose share the renderer resolves against the
128    /// rest of the table.
129    pub(crate) fn explicit_css(self) -> Option<Cow<'static, str>> {
130        match self {
131            Self::Rem(rem) => Some(Cow::Owned(format!("width: {rem}rem"))),
132            Self::Percent(percent) => Some(Cow::Owned(format!("width: {percent}%"))),
133            Self::Wide | Self::Narrow => None,
134        }
135    }
136}

138/// Text column bound to a typed lens **and** a typed projection.
139///
140/// The lens (`FieldLens<M, String>`) is the query side: it names the column
141/// and produces search/sort predicates — `TextColumn::for(User::fields().name(), ..)`
142/// fails to compile if the field does not exist (ADR-0001).
143///
144/// The projection closure is the render side: it reads the value off a model
145/// instance for the cell (`|u| u.name.clone()`). Toasty models are plain
146/// structs and expose no instance→field reflection, so the closure is the
147/// only way to read a field generically (upstream gap #119: instance →
148/// field-value extraction). A typo in the closure body fails at compile
149/// time — there is no string dispatch and no panic at render.
150///
151/// A projection that reads a **relation** declares it with [`Self::needs`],
152/// because the closure is opaque to the framework and the export builds its
153/// query from those declarations. The `(unloaded)` guard in the
154/// closure is what catches a missed declaration, in test builds, at render.
155#[derive(Clone)]
156pub struct TextColumn<M> {
157    /// The query-side lens; `None` for [`Self::computed`] columns, which
158    /// render a value but declare no predicates.
159    path: Option<FieldLens<M, String>>,
160    name: String,
161    label: String,
162    project: Arc<dyn Fn(&M) -> String + Send + Sync>,
163    searchable: bool,
164    sortable: bool,
165    /// The width this column claims in the table's fixed layout.
166    width: ColumnWidth,
167    /// Relations this column's projection reads, in the resource's vocabulary.
168    needs: Vec<&'static str>,
169}
170
171/// The escape character the search pattern declares to `LIKE`:
172/// backslash, escaped in the pattern by [`escape_like_pattern`].
173pub(crate) const LIKE_ESCAPE: char = '\\';
174
175/// Wrap `term` as a `LIKE` pattern matching it anywhere in the column, with
176/// `%`, `_` and the escape character itself escaped so the term stays literal.
177///
178/// Toasty ships the SQL half (`like_with_escape`) but not this one: escaping is
179/// app-side because only the app knows whether it is building a literal or a
180/// pattern.
181pub(crate) fn escape_like_pattern(term: &str) -> String {
182    let mut pattern = String::with_capacity(term.len() + 2);
183    pattern.push('%');
184    for c in term.chars() {
185        if c == LIKE_ESCAPE || c == '%' || c == '_' {
186            pattern.push(LIKE_ESCAPE);
187        }
188        pattern.push(c);
189    }
190    pattern.push('%');
191    pattern
192}

194impl<M> TextColumn<M>
195where
196    M: toasty::schema::Model,
197{
198    /// Bind a column to a `String` field lens plus a projection closure:
199    /// `TextColumn::for(User::fields().name(), |u| u.name.clone())`.
200    ///
201    /// The closure receives each rendered row and returns the cell text, so
202    /// computed cells (`|u| u.active.then(|| "Active".into()).unwrap_or_default()`)
203    /// are as natural as plain field reads.
204    pub fn r#for(
205        path: FieldLens<M, String>,
206        project: impl Fn(&M) -> String + Send + Sync + 'static,
207    ) -> Self {
208        let field = lens_field(path.clone(), &M::schema());
209        Self {
210            path: Some(path),
211            name: field.name.app_unwrap().to_string(),
212            label: lens_label(&field),
213            project: Arc::new(project),
214            searchable: false,
215            sortable: false,
216            width: ColumnWidth::Wide,
217            needs: Vec::new(),
218        }
219    }
220
221    /// A computed, display-only column (CONTEXT.md Column: "a computed value").
222    ///
223    /// No field lens — so it cannot be searchable or sortable (it maps to no
224    /// query predicate) — but any cell projection compiles: booleans,
225    /// timestamps, joined values. Calling `.searchable()` / `.sortable()` on
226    /// a computed column panics: a lying sort link / search promise
227    /// is worse than a loud build error.
228    pub fn computed(
229        label: impl Into<String>,
230        project: impl Fn(&M) -> String + Send + Sync + 'static,
231    ) -> Self {
232        let label = label.into();
233        let name = label.to_lowercase();
234        Self {
235            path: None,
236            name,
237            label,
238            project: Arc::new(project),
239            searchable: false,
240            sortable: false,
241            width: ColumnWidth::Narrow,
242            needs: Vec::new(),
243        }
244    }
245
246    /// Declare the relations this column's projection reads, under
247    /// the names the resource's
248    /// [`export_query`](super::Resource::export_query) matches on:
249    /// `.needs(["author"])` for `|p| p.author.get().name.clone()`.
250    ///
251    /// **Declare every relation the closure reads.** A missing name is not a
252    /// compile error: the export's query arrives without that relation, and
253    /// the closure's `is_unloaded` guard — the unloaded-relation contract of
254    /// ADR-0011, `"(unloaded)"` plus a `debug_assert!` — is what turns it
255    /// into a loud failure instead of a silent `"-"`. Declaring a name nothing
256    /// reads is harmless.
257    ///
258    /// Repeat calls accumulate: `.needs(["author"]).needs(["comments"])`.
259    pub fn needs(mut self, names: impl IntoIterator<Item = &'static str>) -> Self {
260        self.needs.extend(names);
261        self
262    }
263
264    /// The relations this column declared, in declaration order. Internal: the
265    /// public read is [`Table::include_needs`](super::Table::include_needs),
266    /// the union the export hands its resource.
267    pub(crate) fn include_names(&self) -> &[&'static str] {
268        &self.needs
269    }
270
271    pub fn searchable(mut self) -> Self {
272        assert!(
273            self.path.is_some(),
274            "searchable() on computed column '{}': computed columns map to no query predicate",
275            self.label
276        );
277        self.searchable = true;
278        self
279    }

281    pub fn sortable(mut self) -> Self {
282        assert!(
283            self.path.is_some(),
284            "sortable() on computed column '{}': computed columns map to no query predicate",
285            self.label
286        );
287        self.sortable = true;
288        self
289    }

299    /// Declare this column's width in the table's fixed layout:
300    /// `.width(ColumnWidth::Percent(20))` for a column that knows its own
301    /// measure.
302    ///
303    /// The default follows the column's kind — see [`ColumnWidth`]. Override
304    /// it when the content disagrees with the kind: a `computed` column that
305    /// holds a name or a title is [`Wide`](ColumnWidth::Wide), a `String` field
306    /// that holds a status is [`Narrow`](ColumnWidth::Narrow).

308        self.width = width;
309        self
310    }

329    pub fn render_cell(&self, row: &M) -> String {
330        (self.project)(row)
331    }

333    /// The search predicate for this column: a portable, escaped
334    /// **substring** match.
335    ///
336    /// `like_with_escape` keeps the pattern parameterised and lowers to the
337    /// same `LIKE … ESCAPE '\\'` on every driver, and
338    /// `escape_like_pattern` makes the term literal — a `%` or `_` the user
339    /// typed matches that character, it does not act as a wildcard. Note the
340    /// driver difference `LIKE` brings: SQLite compares ASCII
341    /// case-insensitively, PostgreSQL case-sensitively.
342    pub fn to_search_expr(&self, term: &str) -> Option<Expr<bool>> {
343        let t = term.trim();
344        if !self.searchable || t.is_empty() {
345            return None;
346        }
347        Some(
348            self.path
349                .clone()?
350                .like_with_escape(escape_like_pattern(t), LIKE_ESCAPE),
351        )
352    }
353
354    pub fn to_order_by(&self, descending: bool) -> Option<OrderByExpr> {
355        if self.sortable {
356            // Cursor determinism is the engine's job: toasty's
357            // `normalize_cursor_order` appends the physical PK columns to
358            // ambiguous cursor orderings internally.
359            let path = self.path.clone()?;
360            Some(if descending { path.desc() } else { path.asc() })
361        } else {
362            None
363        }
364    }
365}

380/// Convert a single column or tuple of columns into `Vec<TextColumn<M>>`.
381///
382/// Tuple members are `TextColumn<M>` themselves, so nothing sits between the
383/// column types. A single column converts on its own, with no
384/// one-element tuple.
385///
386/// Tuple arities stop at eight, the ceiling every tuple-collection trait
387/// shares: `IntoFilters` in `resource/filter.rs`, `IntoSchema` in
388/// `schema/tree.rs`, and `IntoRelationColumns` in `resource/relation.rs`.
389/// Without variadic generics the idiom is one `macro_rules!` invocation per
390/// arity, and eight covers the widest tuple a Resource declares. Extend every
391/// list together when a real Resource needs more.
392pub trait IntoColumns<M> {
393    fn into_columns(self) -> Vec<TextColumn<M>>;
394}
395
396impl<M> IntoColumns<M> for TextColumn<M> {
397    fn into_columns(self) -> Vec<TextColumn<M>> {
398        vec![self]
399    }
400}
401
402/// Generate the tuple impls of [`IntoColumns`] from one list per arity.
403///
404/// Each list names the binding a tuple element moves through; the `@element`
405/// rule supplies the single element type every position shares.
406macro_rules! into_columns_tuples {
407    ($($v:ident),+ $(,)?) => {
408        impl<M> IntoColumns<M> for ($(into_columns_tuples!(@element $v)),+) {
409            fn into_columns(self) -> Vec<TextColumn<M>> {
410                let ($($v,)+) = self;
411                vec![$($v,)+]
412            }
413        }
414    };
415    (@element $v:ident) => { TextColumn<M> };
416}
417
418into_columns_tuples!(a, b);
419into_columns_tuples!(a, b, c);
420into_columns_tuples!(a, b, c, d);
421into_columns_tuples!(a, b, c, d, e);
422into_columns_tuples!(a, b, c, d, e, f);
423into_columns_tuples!(a, b, c, d, e, f, g);
424into_columns_tuples!(a, b, c, d, e, f, g, h);
