10/// The relations a query must `include`, named in the resource's vocabulary
11/// (GH #177).
12///
13/// A column's projection closure reads relations off the row (`|p| p.author
14/// .get().name.clone()`), and Toasty has no instance→field reflection
15/// (upstream #119), so the framework cannot see *which* relation a closure
16/// touches. Each column therefore **declares** the includes its closure reads
17/// ([`TextColumn::needs`]), a [`Table`](super::Table) gathers the declarations
18/// of the columns it renders into one `IncludeNeeds`, and
19/// [`Resource::export_query`](super::Resource::export_query) answers `wants`
20/// per `include(..)` call.
21///
22/// The names are an opaque vocabulary shared between the declaring column and
23/// the resource that maps them onto `include(..)` calls, because includes are
24/// typed (`Include<Post, Author>`) and a type-erased column cannot name one.
25/// Nothing else reads them: an unknown name is not an error, it just never
26/// matches a branch.
27#[derive(Clone, Debug, Default)]
28pub struct IncludeNeeds {
29    names: BTreeSet<&'static str>,
30}
31
32impl IncludeNeeds {
33    /// Whether `name` was declared by a rendered column.
34    ///
35    /// This is the one question a resource's
36    /// [`export_query`](super::Resource::export_query) asks, once per
37    /// `include(..)` it could add.
38    pub fn wants(&self, name: &str) -> bool {
39        self.names.contains(name)
40    }
41
42    /// Whether no column declared anything — the narrowed query needs no
43    /// relation at all.
44    pub fn is_empty(&self) -> bool {
45        self.names.is_empty()
46    }
47}
48
49impl FromIterator<&'static str> for IncludeNeeds {
50    fn from_iter<T: IntoIterator<Item = &'static str>>(iter: T) -> Self {
51        Self {
52            names: iter.into_iter().collect(),
53        }
54    }
55}
56
57/// `IncludeNeeds::from(["author", "comments"])` — the whole set up front, which
58/// is what a resource's `query` needs (its `export_query` gets the set handed
59/// to it instead).
60impl<const N: usize> From<[&'static str; N]> for IncludeNeeds {
61    fn from(names: [&'static str; N]) -> Self {
62        names.into_iter().collect()
63    }
64}

66/// Text column bound to a typed lens **and** a typed projection.
67///
68/// The lens (`FieldLens<M, String>`) is the query side: it names the column
69/// and produces search/sort predicates — `TextColumn::for(User::fields().name(), ..)`
70/// fails to compile if the field does not exist (ADR-0001).
71///
72/// The projection closure is the render side: it reads the value off a model
73/// instance for the cell (`|u| u.name.clone()`). Toasty models are plain
74/// structs and expose no instance→field reflection, so the closure is the
75/// only way to read a field generically (upstream gap #119: instance →
76/// field-value extraction). A typo in the closure body fails at compile
77/// time — there is no string dispatch and no panic at render.
78///
79/// A projection that reads a **relation** declares it with [`Self::needs`],
80/// because the closure is opaque to the framework and the export builds its
81/// query from those declarations (GH #177). The `(unloaded)` guard in the
82/// closure is what catches a missed declaration, in test builds, at render.
83#[derive(Clone)]
84pub struct TextColumn<M> {
85    /// The query-side lens; `None` for [`Self::computed`] columns, which
86    /// render a value but declare no predicates.
87    path: Option<FieldLens<M, String>>,
88    name: String,
89    label: String,
90    project: Arc<dyn Fn(&M) -> String + Send + Sync>,
91    searchable: bool,
92    sortable: bool,
93    /// Relations this column's projection reads, in the resource's vocabulary
94    /// (GH #177).
95    needs: Vec<&'static str>,
96}
97
98/// The escape character the search pattern declares to `LIKE` (GH #116):
99/// backslash, escaped in the pattern by [`escape_like_pattern`].
100pub(crate) const LIKE_ESCAPE: char = '\\';
101
102/// Wrap `term` as a `LIKE` pattern matching it anywhere in the column, with
103/// `%`, `_` and the escape character itself escaped so the term stays literal
104/// (GH #116).
105///
106/// Toasty ships the SQL half (`like_with_escape`) but not this one: escaping is
107/// app-side because only the app knows whether it is building a literal or a
108/// pattern.
109pub(crate) fn escape_like_pattern(term: &str) -> String {
110    let mut pattern = String::with_capacity(term.len() + 2);
111    pattern.push('%');
112    for c in term.chars() {
113        if c == LIKE_ESCAPE || c == '%' || c == '_' {
114            pattern.push(LIKE_ESCAPE);
115        }
116        pattern.push(c);
117    }
118    pattern.push('%');
119    pattern
120}

122impl<M> TextColumn<M>
123where
124    M: toasty::schema::Model,
125{
126    /// Bind a column to a `String` field lens plus a projection closure:
127    /// `TextColumn::for(User::fields().name(), |u| u.name.clone())`.
128    ///
129    /// The closure receives each rendered row and returns the cell text, so
130    /// computed cells (`|u| u.active.then(|| "Active".into()).unwrap_or_default()`)
131    /// are as natural as plain field reads.
132    pub fn r#for(
133        path: FieldLens<M, String>,
134        project: impl Fn(&M) -> String + Send + Sync + 'static,
135    ) -> Self {
136        let field = lens_field(path.clone(), &M::schema());
137        Self {
138            path: Some(path),
139            name: field.name.app_unwrap().to_string(),
140            label: lens_label(&field),
141            project: Arc::new(project),
142            searchable: false,
143            sortable: false,
144            needs: Vec::new(),
145        }
146    }
147
148    /// A computed, display-only column (CONTEXT.md Column: "a computed value").
149    ///
150    /// No field lens — so it cannot be searchable or sortable (it maps to no
151    /// query predicate) — but any cell projection compiles: booleans,
152    /// timestamps, joined values. Calling `.searchable()` / `.sortable()` on
153    /// a computed column panics (GH #101): a lying sort link / search promise
154    /// is worse than a loud build error.
155    pub fn computed(
156        label: impl Into<String>,
157        project: impl Fn(&M) -> String + Send + Sync + 'static,
158    ) -> Self {
159        let label = label.into();
160        let name = label.to_lowercase();
161        Self {
162            path: None,
163            name,
164            label,
165            project: Arc::new(project),
166            searchable: false,
167            sortable: false,
168            needs: Vec::new(),
169        }
170    }
171
172    /// Declare the relations this column's projection reads (GH #177), under
173    /// the names the resource's
174    /// [`export_query`](super::Resource::export_query) matches on:
175    /// `.needs(["author"])` for `|p| p.author.get().name.clone()`.
176    ///
177    /// **Declare every relation the closure reads.** A missing name is not a
178    /// compile error: the export's query arrives without that relation, and
179    /// the closure's `is_unloaded` guard — the unloaded-relation contract of
180    /// ADR-0011, `"(unloaded)"` plus a `debug_assert!` — is what turns it
181    /// into a loud failure instead of a silent `"-"`. Declaring a name nothing
182    /// reads is harmless.
183    ///
184    /// Repeat calls accumulate: `.needs(["author"]).needs(["comments"])`.
185    pub fn needs(mut self, names: impl IntoIterator<Item = &'static str>) -> Self {
186        self.needs.extend(names);
187        self
188    }
189
190    /// The relations this column declared, in declaration order. Internal: the
191    /// public read is [`Table::include_needs`](super::Table::include_needs),
192    /// the union the export hands its resource.
193    pub(crate) fn include_names(&self) -> &[&'static str] {
194        &self.needs
195    }
196
197    pub fn searchable(mut self) -> Self {
198        assert!(
199            self.path.is_some(),
200            "searchable() on computed column '{}': computed columns map to no query predicate",
201            self.label
202        );
203        self.searchable = true;
204        self
205    }
206
207    pub fn sortable(mut self) -> Self {
208        assert!(
209            self.path.is_some(),
210            "sortable() on computed column '{}': computed columns map to no query predicate",
211            self.label
212        );
213        self.sortable = true;
214        self
215    }
216
217    pub fn is_searchable(&self) -> bool {
218        self.searchable
219    }
220
221    pub fn is_sortable(&self) -> bool {
222        self.sortable
223    }
224
225    pub fn label(&self) -> &str {
226        &self.label
227    }
228
229    /// App-level field name (from the lens). Identifies the column in the
230    /// `?sort=` URL parameter.
231    pub fn name(&self) -> &str {
232        &self.name
233    }
234
235    /// Render the cell for one row via the typed projection.
236    pub fn render_cell(&self, row: &M) -> String {
237        (self.project)(row)
238    }
239
240    /// The search predicate for this column (GH #116): a portable, escaped
241    /// **substring** match.
242    ///
243    /// `like_with_escape` keeps the pattern parameterised and lowers to the
244    /// same `LIKE … ESCAPE '\\'` on every driver, and
245    /// `escape_like_pattern` makes the term literal — a `%` or `_` the user
246    /// typed matches that character, it does not act as a wildcard. Note the
247    /// driver difference `LIKE` brings: SQLite compares ASCII
248    /// case-insensitively, PostgreSQL case-sensitively.
249    pub fn to_search_expr(&self, term: &str) -> Option<Expr<bool>> {
250        let t = term.trim();
251        if !self.searchable || t.is_empty() {
252            return None;
253        }
254        Some(
255            self.path
256                .clone()?
257                .like_with_escape(escape_like_pattern(t), LIKE_ESCAPE),
258        )
259    }
260
261    pub fn to_order_by(&self, descending: bool) -> Option<OrderByExpr> {
262        if self.sortable {
263            // Cursor determinism is the engine's job: toasty's
264            // `normalize_cursor_order` appends the physical PK columns to
265            // ambiguous cursor orderings internally (GH #76).
266            let path = self.path.clone()?;
267            Some(if descending { path.desc() } else { path.asc() })
268        } else {
269            None
270        }
271    }
272}

286/// Convert a single column or tuple of columns into `Vec<TextColumn<M>>`.
287///
288/// Tuple members are `TextColumn<M>` themselves: the `Into` bounds these impls
289/// once carried existed for the removed `Column<M>` enum, and nothing else ever
290/// implemented `From<_> for TextColumn<M>` but the reflexive impl (GH #228).
291///
292/// 5-tuple limit: without variadic generics this is idiomatic Rust — one arity
293/// past `IntoSchema` in `schema/tree.rs`, which stops at four. Tables wider
294/// than five columns are rare in admin UIs; extend (or macro-ify) when a real
295/// Resource needs it.
296pub trait IntoColumns<M> {
297    fn into_columns(self) -> Vec<TextColumn<M>>;
298}
299
300impl<M> IntoColumns<M> for TextColumn<M> {
301    fn into_columns(self) -> Vec<TextColumn<M>> {
302        vec![self]
303    }
304}

309    }
310}

350    /// GH #116: `%` and `_` in a search term are literal characters, not
351    /// wildcards, and the term is wrapped for a substring match.
352    #[test]
353    fn search_pattern_escapes_like_metacharacters() {
354        assert_eq!(escape_like_pattern("Ada"), "%Ada%");
355        assert_eq!(escape_like_pattern("100%"), "%100\\%%");
356        assert_eq!(escape_like_pattern("a_b"), "%a\\_b%");
357        assert_eq!(escape_like_pattern("back\\slash"), "%back\\\\slash%");
358    }

424    /// GH #177: a column that reads no relation declares nothing, and repeat
425    /// `.needs(..)` calls accumulate in declaration order.
426    #[test]
427    fn text_column_include_declarations_accumulate() {
428        let plain = TextColumn::r#for(User::fields().name(), |u| u.name.clone());
429        assert!(plain.include_names().is_empty());
430
431        let declared = TextColumn::computed("Audit", |u: &User| u.name.clone())
432            .needs(["author"])
433            .needs(["comments", "post"]);
434        assert_eq!(declared.include_names(), ["author", "comments", "post"]);

437    /// GH #177: the gathered set is what a resource's `export_query` asks, so
438    /// membership and the empty case are the whole contract.
439    #[test]
440    fn include_needs_gathers_declarations() {
441        let needs: IncludeNeeds = ["author", "comments"].into_iter().collect();
442        assert!(needs.wants("author") && needs.wants("comments"));
443        assert!(!needs.wants("post"));
444        assert!(!needs.is_empty());
445
446        // The whole set up front, as a resource's `query` declares it.
447        let declared = IncludeNeeds::from(["author", "comments"]);
448        assert!(declared.wants("author") && declared.wants("comments"));
449
450        assert!(IncludeNeeds::default().is_empty());
451        assert!(!IncludeNeeds::default().wants("author"));
452    }
