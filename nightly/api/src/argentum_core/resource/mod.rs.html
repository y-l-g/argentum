48/// Maps one Toasty `Model` to its admin UI.
49///
50/// # Contract
51///
52/// **Every method has a default**, so a resource compiles the moment it
53/// declares a [`Model`](Self::Model) — and an omission must fail loudly
54/// rather than silently:
55///
56/// - **Checked at [`Panel::build`](crate::panel::Panel::build)**: the table must declare columns
57///   and a row key, and with [`can_create`](Self::can_create) the [`form`](Self::form) must declare
58///   fields. `table`, `form` and `can_create` are declarations and take no request context, because
59///   build checks them with a Db-only context.
60/// - **Loud at request time**: the record fns ([`create_record`](Self::create_record),
61///   [`update_record`](Self::update_record), [`delete_record`](Self::delete_record)) default to an
62///   error naming the type, so a resource that never implemented delete says so instead of writing
63///   nothing quietly. [`bulk_delete_records`](Self::bulk_delete_records) loops `delete_record` by
64///   default, so it stays loud through the same stub.
65/// - **Opt-in chrome, gated per record**: [`deletable`](Self::deletable) and
66///   [`editable`](Self::editable) are whole-resource flags, false by default; see
67///   [`deletable`](Self::deletable) for how the row predicates narrow them.
68/// - **Default-deny is untouched**: every `can_*` defaults to `false`, so an unconfigured resource
69///   exposes no data and no mutation.
70pub trait Resource: Sized + Send + Sync + 'static {
71    /// The persisted model this resource administers.
72    ///
73    /// `Send + Sync` holds for every data-only model struct and is required
74    /// for concurrent rendering of the resource's pages.
75    ///
76    /// `Clone` is part of the contract because a committed mutation names its
77    /// rows: a handler keeps a copy of the rows it loaded while the
78    /// record fn consumes them, so the hook can be handed what was written.
79    type Model: toasty::schema::Model + Send + Sync + Clone + 'static;
80
81    /// Whether the current user may view the list page.
82    ///
83    /// Also gates relationship option loads: a related resource
84    /// that denies this cannot offer its records as options at all.
85    fn can_view_any(_cx: &Cx) -> bool {
86        false
87    }
88
89    /// Whether the current user may view the given record.
90    ///
91    /// Checked on the edit page (GET), the edit POST (which requires both
92    /// `can_view` and `can_update`), per row in CSV export, on each
93    /// record behind a relationship `Select`'s options, on each row
94    /// of a detail page's relation table, and on the list page as
95    /// the per-row gate of every action link (GH #235: the View link, and the
96    /// `can_view` half of Edit and Delete). Note both hooks default-deny: a
97    /// resource used as a relationship target for option loads must allow
98    /// `can_view_any` **and** `can_view` (overriding one does not imply the
99    /// other), while a relation table consults `can_view` alone. The
100    /// list page deliberately checks only
101    /// `can_view_any` for *membership*: `can_view` is an in-memory Rust
102    /// predicate that cannot run in SQL, and filtering rows after cursor
103    /// pagination would mislabel pages. Row-level visibility that must hold on
104    /// the list belongs in [`Self::query`].
105    fn can_view(_cx: &Cx, _record: &Self::Model) -> bool {
106        false
107    }
108
109    /// Whether the current user may create a new record.
110    fn can_create(_cx: &Cx) -> bool {
111        false
112    }
113
114    /// Whether the current user may update the given record.
115    fn can_update(_cx: &Cx, _record: &Self::Model) -> bool {
116        false
117    }
118
119    /// Whether the current user may delete the given record.
120    ///
121    /// Checked on the single-delete and bulk-delete POSTs together with
122    /// `can_view` — the edit contract: a record that cannot be
123    /// viewed cannot be deleted by UUID-guessing the route.
124    fn can_delete(_cx: &Cx, _record: &Self::Model) -> bool {
125        false
126    }
127
128    /// Whether this resource exposes row and bulk delete chrome.
129    ///
130    /// Chrome is opt-in: the default renders no Delete button, no bulk bar and
131    /// no confirmation dialog, because the server policy
132    /// ([`can_view`](Self::can_view) and [`can_delete`](Self::can_delete), both
133    /// default-deny) answers 403 to every one of them. Override to `true`
134    /// alongside those predicates.
135    ///
136    /// This flag is the whole-resource gate; the panel wires the predicates into
137    /// the table's row policy ([`Table::row_actions`]), so a row they refuse
138    /// renders no Delete link and a disabled bulk checkbox. The handler keeps
139    /// its all-or-nothing check as the safety net for a hand-crafted POST.
140    fn deletable() -> bool {
141        false
142    }
143
144    /// Whether this resource exposes row edit chrome.
145    ///
146    /// The same opt-in flag and per-row rule as [`Self::deletable`], gated by
147    /// [`can_view`](Self::can_view) + [`can_update`](Self::can_update).
148    fn editable() -> bool {
149        false
150    }

152    /// How one record is displayed on the detail page, read-only.
153    ///
154    /// The same [`Schema`] a form uses, rendered for reading: a `TextInput`
155    /// shows its stored value instead of an `<input>`, a `Select` shows the
156    /// option label the form offered, and a layout block keeps the structure it
157    /// declares. Declaring a view is what turns the detail page on — the default
158    /// declares nothing, so the route 404s and no `View` row action renders.
159    ///
160    /// Values come from [`hydrate_form_values`](Self::hydrate_form_values). A
161    /// relation is not one of these fields — it is a list of records, not a
162    /// string — and renders through [`view_relations`](Self::view_relations).
163    ///
164    /// Read-only is a promise, not a disabled form: nothing here validates or
165    /// submits, and no field renders a required marker or an error slot.
166    fn view(_cx: &Cx) -> crate::schema::Schema {
167        crate::schema::Schema::empty()
168    }
169
170    /// The related records on this resource's detail page.
171    ///
172    /// [`view`](Self::view) renders from the record's *string projection* — one
173    /// `HashMap<String, String>` — because that is what every field binds. A
174    /// relation is not a string and may be a list of records, so it cannot ride
175    /// that map; this hook renders it instead, with the loaded record in hand.
176    ///
177    /// This is the half that makes [`Self::query`]'s `include` pay: the related
178    /// rows are already loaded on the record, so a hook that reads
179    /// `record.comments.get()` issues no query at all. Touching an un-included
180    /// relation panics ([`Deferred::get`](toasty::Deferred)), so check
181    /// [`Deferred::is_unloaded`](toasty::Deferred::is_unloaded) — the same marker
182    /// the list columns check.
183    ///
184    /// Returns `None` (the default) for a resource with no related records to
185    /// show, which renders nothing. The two lifetimes are deliberately separate:
186    /// the returned view may borrow the request context, never the record — a
187    /// view holding the record would pin the handler's local binding for as long
188    /// as the page, which does not compile, and the projections return owned
189    /// strings ([`render_relation`]), so nothing needs to.
190    fn view_relations<'a>(
191        _cx: &'a Cx,
192        _record: &Self::Model,
193    ) -> Option<topcoat::view::BoxView<'a>> {
194        None
195    }
196
197    /// Whether this resource declares a detail page.
198    ///
199    /// Derived from [`view`](Self::view) rather than declared twice, so the
200    /// route and the row link cannot disagree with the schema that renders
201    /// them. The detail handler uses it to 404 a resource that declares
202    /// nothing, and the row chrome uses it to leave the link off.
203    fn viewed(cx: &Cx) -> bool {
204        !Self::view(cx).is_empty()
205    }

207    /// The record's label in the detail page's title, or `None` when
208    /// the record has no label to show.
209    ///
210    /// The detail page titles itself with this label when a resource returns
211    /// `Some`, and with [`navigation_label`](Self::navigation_label) plus the
212    /// URL's record key when it returns `None` — the default, so a resource
213    /// that declares nothing keeps the title it has. The showcase's
214    /// `PostResource` is the worked example: it returns the post's title, so
215    /// its heading reads the title instead of `Blog Posts <record key>`.
216    ///
217    /// `cx` is the request's context — the same one [`view`](Self::view) and
218    /// [`hydrate_form_values`](Self::hydrate_form_values) receive — so a label
219    /// can read request state (a locale, a tenant). The default ignores both
220    /// arguments and returns `None`.
221    ///
222    /// A label is display text, not a key. Two records can share one (two
223    /// users named Ada), so it cannot replace [`Table::id`], whose projection
224    /// must stay injective within a page for keyed diffs and bulk selection
225    /// or [`Table::pk`], which the action routes resolve as the
226    /// model's typed PK.
227    fn record_label(_cx: &Cx, _record: &Self::Model) -> Option<String> {
228        None
229    }

231    /// The URL slug for this resource's pages, e.g. `"users"` mounts the list
232    /// at `{panel prefix}/users`.
233    ///
234    /// Defaults to the Filament convention (`HasRoutes::resolveDefaultSlug`):
235    /// take the resource type's name, strip a trailing `Resource`, pluralize
236    /// (`UserResource` → `Users`, `CategoryResource` → `Categories`), then
237    /// kebab-case (`BlogPostResource` → `blog-posts`). Override for irregular
238    /// naming the rules cannot guess (`UsersResource` pluralizes to
239    /// `userses` — name resources singular, or override).
240    fn slug() -> String {
241        let name = type_short_name::<Self>();
242        let singular = name.strip_suffix("Resource").unwrap_or(name);
243        kebab_case(&pluralize(singular))
244    }

246    /// The sidebar label, e.g. `"Users"`.
247    ///
248    /// Defaults to the pluralized `Model` type name (Filament's plural model
249    /// label): `User` → `Users`, `Category` → `Categories`, `Person` →
250    /// `People`. Override for custom wording.
251    fn navigation_label() -> String {
252        pluralize(type_short_name::<Self::Model>())
253    }

255    /// Base query — the seam for a resource's **own** row scoping (ADR-0002):
256    /// soft deletes, row-level visibility, and the relations a page loads.
257    ///
258    /// **Tenancy is not this method's job.** When
259    /// [`requires_tenant`](Self::requires_tenant) is `true` the framework ANDs
260    /// the tenant filter, derived from the model's `tenant_id` column, onto
261    /// whatever this returns, at every loader through [`scoped_query`]. Do not
262    /// re-state `tenant_id().eq(tenant_id(cx))`: the copy is redundant, and one
263    /// that disagreed with the derived column would hide rows rather than widen
264    /// access. Code outside the framework's loaders starts from
265    /// [`scoped_query`].
266    ///
267    /// Returns the raw typed statement query, which composes generically —
268    /// `filter`, `order_by` and `Paginate::new` work on the raw form for any
269    /// `M: Model` — so one panel handler drives every resource's list page.
270    ///
271    /// # Keep unique constraints in step with this scope
272    ///
273    /// The app-side unique pre-check probes through the same tenant-scoped
274    /// query, so a `#[unique]` index *broader* than the scope is invisible: the
275    /// probe misses the colliding row and the user gets a 500 instead of the
276    /// inline "has already been taken". Scope the constraint to match —
277    /// `#[unique(tenant_id, email)]`. Not checkable at declaration time, since
278    /// a query's filters are not introspectable; upstream #117 is the fix.
279    fn query(_cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
280        toasty::stmt::Query::<List<Self::Model>>::all()
281    }
282
283    /// [`Self::query`] narrowed to the relations `needs` asks for.
284    ///
285    /// A loader that reads only part of what [`Self::query`] loads states the
286    /// includes it reads here, and the resource answers with the matching branch
287    /// of its base query. The names are the opaque vocabulary [`IncludeNeeds`]
288    /// documents; the resource maps them onto its typed `include(..)` calls.
289    ///
290    /// **The default ignores `needs` and returns [`Self::query`] unchanged**, so
291    /// narrowing is opt-in per resource. An override must keep the non-tenant
292    /// scope [`Self::query`] carries and any relation its
293    /// [`can_view`](Self::can_view) reads, because the loaders run that
294    /// predicate over the loaded rows; the tenant half is the framework's to AND
295    /// on. The list and the export ask for their table's
296    /// [`include_needs`](Table::include_needs). The detail page reads
297    /// [`view_relations`](Self::view_relations), an opaque hook with no
298    /// declaration, so it loads [`Self::query`] unchanged; edit, delete, bulk
299    /// delete and the option and pagination probes read no relation, so they ask
300    /// for an empty set.
301    fn query_with(cx: &Cx, _needs: &IncludeNeeds) -> toasty::stmt::Query<List<Self::Model>> {
302        Self::query(cx)
303    }
304
305    /// The CSV export's base query: [`Self::query`], narrowed to the relations
306    /// the rendered columns declared.
307    ///
308    /// The export writes a cell per column, so it asks the table which relations
309    /// those columns' projections read ([`Table::include_needs`] — each column
310    /// declares them with [`TextColumn::needs`]) and hands the answer here.
311    /// `docs/guide/src/resources.md` states the override contract: one branch per
312    /// declared name, so an include `query` carries for the detail page or the
313    /// live list rides along on an export only when a rendered column declared
314    /// it.
315    ///
316    /// **The default delegates to [`Self::query_with`]**, which returns
317    /// [`Self::query`] unchanged unless the resource overrides it. Over-fetching
318    /// costs a join; dropping a relation a column reads breaks the render, so the
319    /// default over-fetches.
320    ///
321    /// # What an override must keep
322    ///
323    /// - **The non-tenant scope of [`Self::query`]** (soft deletes, row-level visibility): the
324    ///   export is a reader like any other, and an override that drops that half exports other
325    ///   rows. The tenant half is the framework's.
326    /// - **Whatever the policy path reads.** The visibility scan calls
327    ///   [`Self::can_view`](Self::can_view) on every row of both passes before any cell is written,
328    ///   so a `can_view` that reads a relation needs it included even though no column declared it.
329    /// - **Every name a column declared.** A declared name with no matching include renders an
330    ///   unloaded relation, which the column's `is_unloaded` guard (ADR-0011) reports in test
331    ///   builds instead of a silent `"-"`.
332    fn export_query(cx: &Cx, needs: &IncludeNeeds) -> toasty::stmt::Query<List<Self::Model>> {
333        Self::query_with(cx, needs)
334    }

336    /// Whether this resource requires a tenant in every handler.
337    ///
338    /// Opt-in and default-open: `false` preserves [`Self::query`] exactly as
339    /// written. `true` means two things:
340    ///
341    /// 1. **The gate.** Every handler 403s when the request carries no tenant, instead of leaking
342    ///    unscoped rows or minting nil-tenant orphans.
343    /// 2. **The scope.** Every loader ANDs `tenant_id = <request tenant>` onto the base query,
344    ///    deriving the column from the model's own schema or the resource's [`Self::tenant_scope`].
345    ///    A gated resource that declares neither a `tenant_id` UUID column nor an override is
346    ///    refused by [`Panel::build`](crate::Panel::build) at boot rather than served unscoped or
347    ///    failing per request.
348    ///
349    /// A resource that must genuinely serve more than the request tenant
350    /// declares `false` and scopes in [`Self::query`] by hand, giving up the
351    /// gate above along with the derived filter.
352    fn requires_tenant() -> bool {
353        false
354    }
355
356    /// The predicate the framework ANDs onto this resource's base query to
357    /// scope it to `tenant`, or `None` when there is nothing to AND.
358    ///
359    /// The default derives it from the model: `tenant_id = tenant`, on the
360    /// field named `tenant_id` whose type is a UUID (see [`crate::tenancy`]).
361    /// Override it when the resource's tenancy is not a column on its own
362    /// model — a row that inherits its parent's tenant states the relation
363    /// path here instead, and the framework applies it exactly as it applies
364    /// the derived one. The showcase's comments are the worked example.
365    ///
366    /// Only consulted when [`requires_tenant`](Self::requires_tenant) is
367    /// `true`. `None` from a gated resource is a **misdeclaration**, not a way
368    /// to be unscoped: [`Panel::build`](crate::Panel::build) refuses it at boot
369    /// and every loader keeps answering an error naming the resource
370    /// rather than running its query without a tenant predicate — the backstop
371    /// for a predicate that is only `None` for some tenants. There is
372    /// deliberately no override that *removes* the scope — a resource that
373    /// must serve more than one tenant declares `requires_tenant() = false`
374    /// and owns the scope in [`Self::query`], visibly, with the gate given up.
375    fn tenant_scope(tenant: uuid::Uuid) -> Option<toasty::stmt::Expr<bool>> {
376        crate::tenancy::derived_tenant_filter::<Self::Model>(tenant)
377    }

415    /// Create a new record from form values, returning the row it wrote.
416    ///
417    /// The `Panel` create handler validates `required`/`email` inline and checks
418    /// `Resource::can_create` before calling this, inside a framework-owned
419    /// transaction: `ex` is the open tx — run every statement
420    /// through it (`exec(&mut *ex)`) and never open a second handle, so the
421    /// write commits atomically with the handler's checks. The default
422    /// implementation returns an error; resources should override to perform
423    /// the actual `toasty::create!` (or `Insert`).
424    ///
425    /// Return the created row — `toasty::create!` already hands it back, and
426    /// the framework cannot otherwise name what a create wrote: the primary key
427    /// is the database's (or the app's) to generate, so it is only knowable
428    /// from the row. That is what [`Self::after_commit`] receives for a create.
429    fn create_record(
430        _cx: &Cx,
431        _values: HashMap<String, String>,
432        _ex: &mut dyn toasty::Executor,
433    ) -> impl std::future::Future<Output = Result<Self::Model>> + Send
434    where
435        Self: Sized,
436    {
437        async move {
438            Err(std::io::Error::other(format!(
439                "create not implemented for {}",
440                std::any::type_name::<Self>()
441            ))
442            .into())
443        }
444    }

446    /// Update the already-authorized `record` from form values,
447    /// returning the row as it now stands.
448    ///
449    /// The handler loads `record` through the tenancy-scoped query **inside
450    /// the framework transaction** and checks `can_view` + `can_update` on
451    /// that snapshot before calling this — use the passed record directly,
452    /// never re-query by id (re-loading outside the checked snapshot was the
453    /// TOCTOU hole). Run writes through `ex`; commit/rollback is the
454    /// handler's job. Residual (documented, not fixed): a concurrent
455    /// cross-transaction policy flip landing between this tx's snapshot and
456    /// its commit is backend-isolation territory, out of scope here.
457    ///
458    /// Return the updated row, the way [`Self::create_record`] returns the
459    /// created one: `toasty::update!` already resolves to it, and
460    /// [`Self::after_commit`] needs what was written rather than what was
461    /// loaded — the pre-write snapshot would have a watcher notify its
462    /// subscribers with stale values.

464        _cx: &Cx,
465        _record: Self::Model,
466        _values: HashMap<String, String>,
467        _ex: &mut dyn toasty::Executor,
468    ) -> impl std::future::Future<Output = Result<Self::Model>> + Send

470        Self: Sized,
471    {

473            Err(std::io::Error::other(format!(
474                "update not implemented for {}",
475                std::any::type_name::<Self>()
476            ))
477            .into())
478        }
479    }

501    /// Bulk-delete the already-authorized `records`: the handler
502    /// fetches through the tenancy-scoped `IN` query inside the framework
503    /// transaction and checks `can_delete` on every row before calling this.
504    /// The default deletes each record through [`Self::delete_record`] in
505    /// order, through the same `ex` — any error rolls the whole batch back,
506    /// so mid-loop failures delete zero rows. Override for a single-statement
507    /// batch.
508    fn bulk_delete_records(
509        cx: &Cx,
510        records: Vec<Self::Model>,
511        ex: &mut dyn toasty::Executor,
512    ) -> impl std::future::Future<Output = Result<()>> + Send
513    where
514        Self: Sized,
515    {
516        async move {
517            for record in records {
518                Self::delete_record(cx, record, &mut *ex).await?;
519            }
520            Ok(())
521        }
522    }

524    /// Post-commit work for a mutation this resource committed.
525    ///
526    /// The place for a side effect that must not survive a rollback: an email, a
527    /// webhook, an audit row, cache invalidation. Called once per successful
528    /// write, after `tx.commit()` and before the response — running it inside a
529    /// record fn would leak the effect on rollback, and the write handlers' pool
530    /// discipline forbids a second handle while the transaction is open.
531    ///
532    /// [`Committed`] names the mutation and the rows it wrote: the committed row
533    /// a create or update returned, the rows a delete or bulk delete removed
534    /// (one call with every row). It is never called when nothing committed: a
535    /// validation error, a policy denial, a failed record fn, or a failed commit
536    /// all leave the hook untouched, so a rollback cannot produce the effect. A
537    /// hook that returns `Err` is logged and ignored — the write is committed,
538    /// and retries are the app's to build; a panic surfaces as Topcoat's
539    /// panic-isolated 500.
540    ///
541    /// ```ignore
542    /// async fn after_commit(cx: &Cx, committed: Committed<Post>) -> Result<()> {
543    ///     let mut db = db(cx); // a fresh handle is allowed here
544    ///     for post in committed.records() { notify(post).await?; }
545    ///     Ok(())
546    /// }
547    /// ```
548    fn after_commit(
549        _cx: &Cx,
550        _committed: Committed<Self::Model>,
551    ) -> impl std::future::Future<Output = Result<()>> + Send
552    where
553        Self: Sized,
554    {
555        async move { Ok(()) }
556    }

558    /// Hydrate form values from a record for the Edit and View pages.
559    ///
560    /// The record's **string projection**: the flat map every field binds, keyed
561    /// by the name the control posts. Default returns empty; a resource
562    /// overrides it to map its record to those keys (`name -> record.name`).
563    ///
564    /// `cx` carries the app schema. A scalar projection needs no
565    /// request context, but an embedded **value** does: its keys are the
566    /// columns the compiled mapping resolves
567    /// ([`write_embedded`](crate::schema::write_embedded)), so this override
568    /// does not re-derive them. The context is the
569    /// request's, the same one `form(cx)` and the record fns receive.
570    fn hydrate_form_values(_cx: &Cx, _record: &Self::Model) -> HashMap<String, String> {
571        HashMap::new()
572    }
573}

575/// The tenant-scoped full base query.
576///
577/// [`Resource::query`] with the predicate from [`Resource::tenant_scope`]
578/// ANDed onto it, so a resource that overrides `query` for includes or soft
579/// deletes cannot drop the tenant scope by forgetting to re-state it. This is
580/// the entry point for a reader that wants the full base query — the detail
581/// page and app code. A framework loader that reads only part of it uses
582/// `scoped_query_with`.
583///
584/// # Errors
585///
586/// - A gated resource and no tenant in `cx` → 403, the same fail-closed answer the handler gate
587///   gives.
588/// - A gated resource that supplies no tenant predicate — no discoverable `tenant_id` UUID column,
589///   no [`Resource::tenant_scope`] override → an error naming the resource and the model.
590///   [`Panel::build`](crate::Panel::build) refuses that declaration at boot, so this is the
591///   backstop for a predicate that is `None` for the request's tenant, and for app code outside a
592///   panel. It is deliberately **not** a fallback to the unscoped query: a silent miss would be the
593///   leak [`Resource::requires_tenant`] exists to prevent.
594///
595/// App code that loads rows itself must call this — on a gated resource
596/// [`Resource::query`] is the *tenant-unscoped* base by design, so calling it
597/// directly is safe only for rows whose tenant membership is already settled (a
598/// write by id against a record the framework loaded and authorized).
599pub fn scoped_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
600    apply_tenant_scope::<R>(cx, R::query(cx))
601}
602
603/// [`scoped_query`] over a loader's declared includes: the same tenant gate
604/// and derived predicate, seeded from [`Resource::query_with`] instead of
605/// [`Resource::query`]. A loader that reads no relation passes
606/// `IncludeNeeds::default()`.
607pub(crate) fn scoped_query_with<R: Resource>(
608    cx: &Cx,
609    needs: &IncludeNeeds,
610) -> Result<Query<List<R::Model>>> {
611    apply_tenant_scope::<R>(cx, R::query_with(cx, needs))
612}
613
614/// AND the framework's tenant predicate onto `query`.
615///
616/// The body of [`scoped_query`] and [`scoped_query_with`], split out so every
617/// seed — the base query, a loader's narrowed branch, the export's
618/// [`Resource::export_query`](Resource::export_query) — shares one gate, one
619/// predicate, and one fail-closed error, and so the seeds cannot drift apart.
620/// It is crate-internal because a caller outside the crate always has a
621/// `Resource`, and so always wants one of the `scoped_query*` entry points.
622pub(crate) fn apply_tenant_scope<R: Resource>(
623    cx: &Cx,
624    query: Query<List<R::Model>>,
625) -> Result<Query<List<R::Model>>> {
626    if !R::requires_tenant() {
627        return Ok(query);
628    }
629    let tenant = crate::tenancy::require_tenant(cx)?;
630    let Some(filter) = R::tenant_scope(tenant) else {
631        // Fail closed and loudly: the resource declared a gate whose scope the
632        // framework cannot derive and the resource did not state, and running
633        // the query unscoped is the one outcome that declaration exists to
634        // prevent. `Panel::build` already refused the resource if *no* tenant
635        // could scope it; this is the backstop for a `tenant_scope`
636        // that answers `None` only for this tenant, and for callers outside a
637        // panel.
638        tracing::error!(
639            resource = R::slug(),
640            model = std::any::type_name::<R::Model>(),
641            "requires_tenant is true but the resource supplies no tenant predicate: no `tenant_id` \
642             UUID column on the model and no `tenant_scope` override (GH #223)"
643        );
644        return Err(std::io::Error::other(format!(
645            "resource '{}' requires a tenant, but the framework cannot scope it: {} declares no \
646             `tenant_id` UUID column to derive the filter from, and the resource does not override \
647             `tenant_scope` (GH #223); declare the column, override `tenant_scope`, or drop \
648             `requires_tenant` and scope in `query`",
649            R::slug(),
650            std::any::type_name::<R::Model>(),
651        ))
652        .into());
653    };
654    Ok(query.filter(filter))
655}

657/// Every `Resource` is an [`OptionSource`](crate::schema::OptionSource) — the
658/// bridge that lets the relationship option loaders be generic over the source
659/// surface instead of over `Resource`, so `schema` does not depend on
660/// `resource`.
661///
662/// A resource answers the loaders here, so the loaders never name `Resource`.
663/// [`scoped_query`](crate::schema::OptionSource::scoped_query), the one required
664/// method, forwards to [`scoped_query`], so an option load inherits the tenant
665/// gate and the derived tenant predicate exactly as every other loader does; it
666/// is deliberately not [`Resource::query`], which on a gated resource is the
667/// *tenant-unscoped* base.
668///
669/// [`options_query`](crate::schema::OptionSource::options_query) forwards to
670/// `scoped_query_with` with an empty [`IncludeNeeds`]: an option load projects a
671/// value and a label off each row and reads no relation of its own, so a
672/// resource that narrows its loaders narrows option loads too. A source that
673/// overrides nothing keeps the full [`scoped_query`]. The policy predicates and
674/// the tenant declaration forward unchanged, and the search expression and
675/// default ordering come from the resource's declared [`table`](Resource::table),
676/// which is where "the option search searches the related resource's searchable
677/// columns" lives.
678impl<R: Resource> crate::schema::OptionSource for R {
679    type Model = R::Model;
680
681    fn scoped_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
682        scoped_query::<R>(cx)
683    }
684
685    fn options_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
686        scoped_query_with::<R>(cx, &IncludeNeeds::default())
687    }
688
689    fn can_view_any(cx: &Cx) -> bool {
690        <R as Resource>::can_view_any(cx)
691    }
692
693    fn can_view(cx: &Cx, record: &R::Model) -> bool {
694        <R as Resource>::can_view(cx, record)
695    }
696
697    fn requires_tenant() -> bool {
698        <R as Resource>::requires_tenant()
699    }
700
701    fn slug() -> String {
702        <R as Resource>::slug()
703    }
704
705    fn search_expr(cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
706        <R as Resource>::table(cx).search_expr(term)
707    }
708
709    fn order_by(cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
710        <R as Resource>::table(cx).order_by(false)
711    }
712}
