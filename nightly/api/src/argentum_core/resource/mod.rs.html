48/// Maps one Toasty `Model` to its admin UI.
49///
50/// # Contract (GH #138)
51///
52/// **Every method has a default**, so a resource compiles the moment it
53/// declares a [`Model`](Self::Model) — and an omission must therefore fail
54/// loudly rather than silently:
55///
56/// - **Checked at [`Panel::build`](crate::panel::Panel::build)**, which returns `Err` naming the
57///   type: the table must be renderable ([`table`](Self::table) declares columns and a row key)
58///   and, where [`can_create`](Self::can_create) allows it, the [`form`](Self::form) must declare
59///   fields. `table`, `form` and `can_create` are declarations: they must not need request-scoped
60///   context, because build checks them with a Db-only context, and each list and form request
61///   calls `table` / `form` again.
62/// - **Loud at request time**: the record fns ([`create_record`](Self::create_record),
63///   [`update_record`](Self::update_record), [`delete_record`](Self::delete_record),
64///   [`bulk_delete_records`](Self::bulk_delete_records)) default to an error naming the type, so a
65///   resource that never implemented delete answers "delete not implemented for …" instead of
66///   writing nothing quietly.
67/// - **Opt-in chrome, gated per record**: [`deletable`](Self::deletable) and
68///   [`editable`](Self::editable) default to `false`, so a resource that never mentions them
69///   renders no Edit or Delete affordance and cannot advertise an action its default-deny predicate
70///   refuses. A resource that wants the chrome declares the flag *and* the matching policy
71///   predicates (`can_view` + `can_delete` for `deletable`, `can_view` + `can_update` for
72///   `editable`). The flag is the whole-resource gate; the predicates are applied per row, because
73///   the panel wires them into the table's row policy ([`Table::row_actions`], GH #235): a row
74///   `can_update` refuses renders no Edit link, and a row `can_delete` refuses renders no Delete
75///   link and a disabled bulk checkbox. A row-level rule therefore narrows the chrome instead of
76///   leaving a control the route answers 403. [`viewed`](Self::viewed) is per-record exact the same
77///   way, derived from the declared [`view`](Self::view) schema rather than declared beside it.
78/// - **Default-deny is untouched**: every `can_*` still defaults to `false`, so an unconfigured
79///   resource exposes no data and no mutation.
80pub trait Resource: Sized + Send + Sync + 'static {
81    /// The persisted model this resource administers.
82    ///
83    /// `Send + Sync` holds for every data-only model struct and is required
84    /// for concurrent rendering of the resource's pages.
85    ///
86    /// `Clone` is part of the contract because a committed mutation names its
87    /// rows (GH #112): a handler keeps a copy of the rows it loaded while the
88    /// record fn consumes them, so the hook can be handed what was written.
89    type Model: toasty::schema::Model + Send + Sync + Clone + 'static;

91    /// Whether the current user may view the list page.
92    ///
93    /// Also gates relationship option loads (GH #108): a related resource
94    /// that denies this cannot offer its records as options at all.
95    fn can_view_any(_cx: &Cx) -> bool {
96        false
97    }
98
99    /// Whether the current user may view the given record.
100    ///
101    /// Checked on the edit page (GET), the edit POST (which requires both
102    /// `can_view` and `can_update`, GH #86), per row in CSV export, on each
103    /// record behind a relationship `Select`'s options (GH #108), and on the
104    /// list page as the per-row gate of every action link (GH #235: the View
105    /// link, and the `can_view` half of Edit and Delete). Note
106    /// both hooks default-deny: a resource used as a relationship target
107    /// must allow `can_view_any` **and** `can_view` (overriding one does not
108    /// imply the other). The list page deliberately checks only
109    /// `can_view_any` for *membership* (GH #86): `can_view` is an in-memory
110    /// Rust predicate that cannot run in SQL, and filtering rows after cursor
111    /// pagination would mislabel pages (holes, wrong Next/Prev). Row-level
112    /// visibility that must hold on the list belongs in [`Self::query`]
113    /// (ADR-0002), which every loader — list, edit, delete, bulk, export —
114    /// funnels through *inside* [`scoped_query`], so the tenant half of the
115    /// scope is applied after the override rather than by it (GH #223).
116    fn can_view(_cx: &Cx, _record: &Self::Model) -> bool {
117        false
118    }
119
120    /// Whether the current user may create a new record.
121    fn can_create(_cx: &Cx) -> bool {
122        false
123    }

125    /// Whether the current user may update the given record.
126    fn can_update(_cx: &Cx, _record: &Self::Model) -> bool {
127        false
128    }

130    /// Whether the current user may delete the given record.
131    ///
132    /// Checked on the single-delete and bulk-delete POSTs together with
133    /// `can_view` (GH #168) — the edit contract: a record that cannot be
134    /// viewed cannot be deleted by UUID-guessing the route.
135    fn can_delete(_cx: &Cx, _record: &Self::Model) -> bool {
136        false
137    }

139    /// Whether this resource exposes row and bulk delete chrome (GH #96).
140    ///
141    /// Chrome is opt-in (GH #226): the default renders no Delete button, no
142    /// bulk bar and no confirmation dialog, because server policy
143    /// ([`can_view`](Self::can_view) +
144    /// [`can_delete`](Self::can_delete), both default-deny) would answer 403 to
145    /// every one of them. Override to `true` alongside those predicates.
146    ///
147    /// This flag is the whole-resource gate; the predicates are applied per row
148    /// (GH #235). The panel wires them into the table's row policy, so a row
149    /// they refuse renders no Delete link and a disabled bulk checkbox — the
150    /// affordance narrows with the rule instead of leaving a control the POST
151    /// answers 403 to. The handler keeps its all-or-nothing check as the safety
152    /// net for a hand-crafted POST.
153    fn deletable() -> bool {
154        false
155    }

157    /// Whether this resource exposes row edit chrome (GH #162).
158    ///
159    /// Chrome is opt-in (GH #226): the default renders no `Edit` link per row,
160    /// because server policy ([`can_view`](Self::can_view) +
161    /// [`can_update`](Self::can_update), both default-deny) would answer 403 to
162    /// the edit GET. Override to `true` alongside those predicates.
163    ///
164    /// The same per-row rule as [`Self::deletable`] applies (GH #235): the
165    /// panel wires `can_view` + `can_update` into the table's row policy, so a
166    /// row the caller may not edit renders no link, matching the edit GET.
167    fn editable() -> bool {
168        false
169    }

171    /// How one record is displayed on the detail page (GH #187), read-only.
172    ///
173    /// The same [`Schema`] a form uses, rendered for
174    /// reading: a `TextInput` shows its stored value instead of an `<input>`,
175    /// a `Select` shows the option label the form offered, and a layout block
176    /// keeps the structure it declares (`Grid` stays a grid). Declaring a view
177    /// is what turns the detail page on — the default declares nothing, so a
178    /// resource that does not override this renders no page content, links no
179    /// `View` row action, and 404s the route.
180    ///
181    /// Values come from [`hydrate_form_values`](Self::hydrate_form_values), so
182    /// a field bound here is one the resource already knows how to read. A
183    /// relation is not one of these fields — it is a list of records, not a
184    /// string — and renders through
185    /// [`view_relations`](Self::view_relations), which the detail page draws
186    /// under this Schema.
187    ///
188    /// Read-only is a promise, not a disabled form: nothing here validates or
189    /// submits, and no field renders a required marker or an error slot —
190    /// including a `Repeater`, which renders its label over its children's
191    /// values.
192    fn view(_cx: &Cx) -> crate::schema::Schema {
193        crate::schema::Schema::empty()
194    }
195
196    /// The related records on this resource's detail page (GH #187).
197    ///
198    /// [`view`](Self::view) is a `Schema`, and a `Schema` is rendered from the
199    /// record's *string projection* — one `HashMap<String, String>` — because
200    /// that is what every field binds. A relation is not a string and may be a
201    /// list of records, so it cannot ride that map; this hook is where it
202    /// renders instead, with the loaded record in hand.
203    ///
204    /// The reason is structural, not stylistic: the render tree is monomorphic
205    /// (one walk renders every resource) while a record's type is the
206    /// resource's, so a `Schema` node cannot be handed `&Self::Model` without
207    /// type erasure across a borrow — the workspace denies `unsafe`, and the
208    /// safe erasures either require a `'static` record or capture the caller's
209    /// locals in the render's lifetime. A generic method has none of those
210    /// problems and keeps the typing.
211    ///
212    /// This is the half that makes `Resource::query`'s `include` pay: the
213    /// related rows are already loaded on the record, so a hook that reads
214    /// `record.comments.get()` issues no query at all. Touching an
215    /// un-included relation panics (`Deferred::get`), and the way to notice is
216    /// [`Deferred::is_unloaded`](toasty::Deferred::is_unloaded) — the same
217    /// marker the list columns check.
218    ///
219    /// Returns `None` (the default) for a resource with no related records to
220    /// show, which renders nothing.
221    /// The two lifetimes are deliberately separate: the returned view may
222    /// borrow the request context, never the record. A view that held the
223    /// record would pin the handler's local binding for as long as the page,
224    /// which does not compile — the renderer's projections return owned
225    /// strings ([`render_relation`]), so nothing needs to.
226    fn view_relations<'a>(
227        _cx: &'a Cx,
228        _record: &Self::Model,
229    ) -> Option<topcoat::view::BoxView<'a>> {
230        None
231    }

233    /// Whether this resource declares a detail page (GH #187).
234    ///
235    /// Derived from [`view`](Self::view) rather than declared twice, so the
236    /// route and the row link cannot disagree with the schema that renders
237    /// them. The detail handler uses it to 404 a resource that declares
238    /// nothing, and the row chrome uses it to leave the link off.
239    fn viewed(cx: &Cx) -> bool {
240        !Self::view(cx).is_empty()
241    }

243    /// The record's label in the detail page's title (GH #241), or `None` when
244    /// the record has no label to show.
245    ///
246    /// The detail page titles itself with this label when a resource returns
247    /// `Some`, and with [`navigation_label`](Self::navigation_label) plus the
248    /// URL's record key when it returns `None` — the default, so a resource
249    /// that declares nothing keeps the title it has. The showcase's
250    /// `PostResource` is the worked example: it returns the post's title, so
251    /// its heading reads the title instead of `Blog Posts <record key>`.
252    ///
253    /// `cx` is the request's context — the same one [`view`](Self::view) and
254    /// [`hydrate_form_values`](Self::hydrate_form_values) receive — so a label
255    /// can read request state (a locale, a tenant). The default ignores both
256    /// arguments and returns `None`.
257    ///
258    /// A label is display text, not a key. Two records can share one (two
259    /// users named Ada), so it cannot replace [`Table::id`], whose projection
260    /// must stay injective within a page for keyed diffs and bulk selection
261    /// (GH #96), or [`Table::pk`], which the action routes resolve as the
262    /// model's typed PK (GH #168).
263    fn record_label(_cx: &Cx, _record: &Self::Model) -> Option<String> {
264        None
265    }

267    /// The URL slug for this resource's pages, e.g. `"users"` mounts the list
268    /// at `{panel prefix}/users`.
269    ///
270    /// Defaults to the Filament convention (`HasRoutes::resolveDefaultSlug`):
271    /// take the resource type's name, strip a trailing `Resource`, pluralize
272    /// (`UserResource` → `Users`, `CategoryResource` → `Categories`), then
273    /// kebab-case (`BlogPostResource` → `blog-posts`). Override for irregular
274    /// naming the rules cannot guess (`UsersResource` pluralizes to
275    /// `userses` — name resources singular, or override).
276    fn slug() -> String {
277        let name = type_short_name::<Self>();
278        let singular = name.strip_suffix("Resource").unwrap_or(name);
279        kebab_case(&pluralize(singular))
280    }

283    ///
284    /// Defaults to the pluralized `Model` type name (Filament's plural model
285    /// label): `User` → `Users`, `Category` → `Categories`, `Person` →
286    /// `People`. Override for custom wording.

288        pluralize(type_short_name::<Self::Model>())
289    }

291    /// Base query — the seam for a resource's **own** row scoping (ADR-0002 and
292    /// its 2026-09-22 status note): soft deletes, row-level visibility, and the
293    /// relations a page loads.
294    ///
295    /// **Tenancy is not this method's job (GH #223).** When
296    /// [`requires_tenant`](Self::requires_tenant) is `true` the framework ANDs
297    /// the tenant filter — derived from the model's `tenant_id` column — onto
298    /// whatever this returns, at every loader, through [`scoped_query`]. Do not
299    /// re-state `tenant_id().eq(tenant_id(cx))` here: the copy is redundant,
300    /// and one that disagreed with the derived column would hide rows rather
301    /// than widen access. The gate that makes a missing tenant a 403 is
302    /// unchanged (GH #87).
303    ///
304    /// An override of this method is **not** the tenant seam: code outside the
305    /// framework's loaders must start from [`scoped_query`], because on a gated
306    /// resource this is the *tenant-unscoped* base. See [`scoped_query`] for
307    /// why.
308    ///
309    /// Returns the raw typed statement query:
310    /// raw queries compose generically — `filter`, `order_by`, and
311    /// `Paginate::new` are available on the raw form for any `M: Model` —
312    /// which is what lets [`crate::panel::Panel`] drive every resource's list
313    /// page through one handler. Scoping it via `Model::filter(..)` in an
314    /// override is enough; the wrapper's extra methods are only needed by
315    /// hand-written loaders.
316    ///
317    /// # Keep unique constraints in step with this scope (GH #88)
318    ///
319    /// The app-side unique pre-check probes submitted values **through
320    /// [`scoped_query`]**, so it only sees the rows that query returns. A
321    /// `#[unique]` index *broader* than the scope is therefore invisible to it:
322    /// the probe misses the colliding row, the database refuses the write, and
323    /// the user gets a 500 instead of the inline "has already been taken".
324    ///
325    /// The tenant case is the one that bites — the framework scopes to
326    /// `tenant_id` while the column carries a plain `#[unique]` (global), which
327    /// makes two tenants sharing a value a legitimate pair to the probe and a
328    /// constraint violation to the database. Scope the constraint to match:
329    /// `#[unique(tenant_id, email)]`, which also makes it say what it means.
330    /// `Author.email` in the showcase is the worked example.
331    ///
332    /// The invariant cannot be *checked* at declaration time — a query's filters
333    /// are not introspectable, so nothing can compare the two automatically.
334    /// Upstream #117 (a driver-level unique-violation predicate) is what would
335    /// make a mismatch safe rather than merely documented.
336    fn query(_cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
337        toasty::stmt::Query::<List<Self::Model>>::all()
338    }

340    /// The CSV export's base query: [`Self::query`], narrowed to the relations
341    /// the rendered columns declared (GH #177).
342    ///
343    /// The export writes a cell per column, so it asks the table which
344    /// relations those columns' projections read ([`Table::include_needs`] —
345    /// each column declares them with [`TextColumn::needs`]) and hands the
346    /// answer here. `docs/guide/src/resources.md` states the override contract:
347    /// a resource splits its base query in two, one branch per declared name,
348    /// so an include `query` carries for the detail page or the live list rides
349    /// along on an export only when a rendered column declared it.
350    ///
351    /// **The default ignores `needs` and returns [`Self::query`] unchanged.**
352    /// Over-fetching a relation nothing reads costs a join; dropping one a
353    /// column does read breaks the render — the default takes the safe side.
354    /// The **list page does not use this**: it keeps inheriting [`Self::query`],
355    /// so only the export's constant-factor over-fetch is addressed here.
356    ///
357    /// # What an override must keep
358    ///
359    /// - **The non-tenant scope of [`Self::query`].** This is the same soft-delete/row-level seam
360    ///   (ADR-0002), and the export is a reader like any other: an override that drops that half
361    ///   exports other rows. The *tenant* half is not the override's to keep — the framework ANDs
362    ///   it onto what this returns, exactly as it does for [`Self::query`] (GH #223), so a gated
363    ///   export is scoped whether or not the override re-states the filter.
364    /// - **Whatever the policy path reads.** The export's visibility scan calls [`Self::can_view`]
365    ///   on every row of both passes, before any cell is written, so a `can_view` that reads a
366    ///   relation needs that relation included even though no column declared it — include it
367    ///   unconditionally in the narrowed branch. Reading an un-included relation panics in
368    ///   `Deferred::get`.
369    /// - **Every name a column declared.** A declared name with no matching include renders an
370    ///   unloaded relation, which the column's `is_unloaded` guard (ADR-0011) reports in test
371    ///   builds instead of a silent `"-"`.
372    fn export_query(cx: &Cx, _needs: &IncludeNeeds) -> toasty::stmt::Query<List<Self::Model>> {
373        Self::query(cx)
374    }

376    /// Whether this resource requires a tenant in every handler (GH #87).
377    ///
378    /// Opt-in and default-open: `false` preserves the default behavior — no
379    /// gate, and [`Self::query`] exactly as written. Override to `true` on a
380    /// resource whose model carries rows per tenant.
381    ///
382    /// `true` means two things, and the second is GH #223:
383    ///
384    /// 1. **The gate.** Every handler 403s when the request carries no tenant, instead of leaking
385    ///    unscoped rows or minting nil-tenant orphans (#87's tenantless-create rejection,
386    ///    unchanged).
387    /// 2. **The scope.** Every loader ANDs `tenant_id = <request tenant>` onto the resource's base
388    ///    query, deriving the column from the model's own schema — see [`scoped_query`] and
389    ///    [`Self::tenant_scope`]. A gated resource is therefore never unscoped because an override
390    ///    forgot to re-state the filter, and it is never unscoped because the derivation failed
391    ///    either: the model must declare a `tenant_id` UUID column, or the resource must declare
392    ///    its own predicate in [`Self::tenant_scope`], and a gated resource that does neither is
393    ///    refused by [`Panel::build`](crate::Panel::build) at boot (GH #231) — the declaration is
394    ///    checkable without a request — rather than serving rows unscoped or failing per request.
395    ///
396    /// A resource that must genuinely serve more than the request tenant — a
397    /// deliberate cross-tenant view — declares `false` and scopes in
398    /// [`Self::query`] by hand. That is the explicit, visible way out, and it
399    /// gives up the gate above along with the derived filter.
400    fn requires_tenant() -> bool {
401        false
402    }
403
404    /// The predicate the framework ANDs onto this resource's base query to
405    /// scope it to `tenant` (GH #223), or `None` when there is nothing to AND.
406    ///
407    /// The default derives it from the model: `tenant_id = tenant`, on the
408    /// field named `tenant_id` whose type is a UUID (see [`crate::tenancy`]).
409    /// Override it when the resource's tenancy is not a column on its own
410    /// model — a row that inherits its parent's tenant states the relation
411    /// path here instead, and the framework applies it exactly as it applies
412    /// the derived one. The showcase's comments are the worked example.
413    ///
414    /// Only consulted when [`requires_tenant`](Self::requires_tenant) is
415    /// `true`. `None` from a gated resource is a **misdeclaration**, not a way
416    /// to be unscoped: [`Panel::build`](crate::Panel::build) refuses it at boot
417    /// (GH #231), and every loader keeps answering an error naming the resource
418    /// rather than running its query without a tenant predicate — the backstop
419    /// for a predicate that is only `None` for some tenants. There is
420    /// deliberately no override that *removes* the scope — a resource that
421    /// must serve more than one tenant declares `requires_tenant() = false`
422    /// and owns the scope in [`Self::query`], visibly, with the gate given up.
423    fn tenant_scope(tenant: uuid::Uuid) -> Option<toasty::stmt::Expr<bool>> {
424        crate::tenancy::derived_tenant_filter::<Self::Model>(tenant)
425    }

463    /// Create a new record from form values, returning the row it wrote.
464    ///
465    /// The `Panel` create handler validates `required`/`email` inline and checks
466    /// `Resource::can_create` before calling this, inside a framework-owned
467    /// transaction (GH #84): `ex` is the open tx — run every statement
468    /// through it (`exec(&mut *ex)`) and never open a second handle, so the
469    /// write commits atomically with the handler's checks. The default
470    /// implementation returns an error; resources should override to perform
471    /// the actual `toasty::create!` (or `Insert`).
472    ///
473    /// Return the created row — `toasty::create!` already hands it back, and
474    /// the framework cannot otherwise name what a create wrote: the primary key
475    /// is the database's (or the app's) to generate, so it is only knowable
476    /// from the row. That is what [`Self::after_commit`] receives for a create
477    /// (GH #112).
478    fn create_record(
479        _cx: &Cx,
480        _values: HashMap<String, String>,
481        _ex: &mut dyn toasty::Executor,
482    ) -> impl std::future::Future<Output = Result<Self::Model>> + Send
483    where
484        Self: Sized,
485    {
486        async move {
487            Err(std::io::Error::other(format!(
488                "create not implemented for {}",
489                std::any::type_name::<Self>()
490            ))
491            .into())
492        }
493    }

495    /// Update the already-authorized `record` from form values (GH #86),
496    /// returning the row as it now stands.
497    ///

572    /// Post-commit work for a mutation this resource committed (GH #112).
573    ///
574    /// The one place a side effect that must not survive a rollback belongs —
575    /// an email, a webhook, an audit row, cache invalidation. Called by the
576    /// framework **once per successful write**, after `tx.commit()` and before
577    /// the response: running it inside a record fn would leak the effect when
578    /// the transaction rolls back, and the write handlers' pool discipline
579    /// (GH #84) forbids a second handle while the transaction is open. By the
580    /// time this runs the transaction is gone, so it may open its own `Db`
581    /// handle — `db(cx)` — or none at all.
582    ///
583    /// [`Committed`] names the mutation and the rows it wrote: the row a create
584    /// returned, the row an update returned (the committed state, not the
585    /// snapshot the handler loaded), the rows a delete or bulk delete removed —
586    /// deletes are the one case where the row is gone by the time you see it.
587    /// A bulk delete is **one call** with every row, not one call per row.
588    ///
589    /// It is never called when nothing committed: a validation error, a policy
590    /// denial, a failed record fn, or a failed commit all leave the hook
591    /// untouched, so a rollback can never produce the effect.
592    ///
593    /// A hook that returns `Err` is logged and ignored — the write is
594    /// committed, and an error page would misreport it. Retries and delivery
595    /// guarantees are the app's to build (an outbox written here is the usual
596    /// shape); the framework promises neither. A hook that panics is a bug in
597    /// the hook and surfaces as Topcoat's panic-isolated 500.
598    ///
599    /// Defaults to a no-op, so a resource that declares nothing is unaffected.
600    ///
601    /// ```ignore
602    /// impl Resource for PostResource {
603    ///     type Model = Post;
604    ///
605    ///     async fn after_commit(cx: &Cx, committed: Committed<Post>) -> Result<()> {
606    ///         // The transaction is committed and gone, so this opens its own handle.
607    ///         let mut db = db(cx);
608    ///         for post in committed.records() {
609    ///             notify_watchers(post, committed.mutation()).await?;
610    ///         }
611    ///         Ok(())
612    ///     }
613    /// }
614    /// ```
615    fn after_commit(
616        _cx: &Cx,
617        _committed: Committed<Self::Model>,
618    ) -> impl std::future::Future<Output = Result<()>> + Send
619    where
620        Self: Sized,
621    {
622        async move { Ok(()) }
623    }

642/// The tenant-scoped query every loader executes (GH #223).
643///
644/// [`Resource::query`] with the tenant predicate from
645/// [`Resource::tenant_scope`] ANDed onto it: for a resource whose
646/// [`requires_tenant`](Resource::requires_tenant) is `true`, the rows are
647/// narrowed to the request tenant — derived from `tenant_id` by default,
648/// declared by the resource when its tenancy is inherited. The filter is
649/// applied *here*, outside the resource's `query`, so a resource that
650/// overrides `query` for includes or soft deletes cannot drop the tenant scope
651/// by forgetting to re-state it.
652///
653/// This is the entry point; `apply_tenant_scope` is the same composition for
654/// a seed that is not [`Resource::query`].
655///
656/// # Errors
657///
658/// - A gated resource and no tenant in `cx` → 403, the same fail-closed answer the handler gate
659///   gives (GH #87).
660/// - A gated resource that supplies no tenant predicate — no discoverable `tenant_id` UUID column,
661///   no [`Resource::tenant_scope`] override → an error naming the resource and the model.
662///   [`Panel::build`](crate::Panel::build) refuses that declaration at boot (GH #231), so this is
663///   the backstop for a predicate that exists but is `None` for the request's tenant, and for app
664///   code that calls this outside a panel. It is deliberately **not** a fallback to the unscoped
665///   query: discovery is by name, and a silent miss would be exactly the leak
666///   [`Resource::requires_tenant`] exists to prevent.
667///
668/// # When to call this
669///
670/// Every framework loader does, and app code that loads rows itself must too —
671/// a record fn double-checking a foreign key, a custom page, a test. On a gated
672/// resource [`Resource::query`] is the *tenant-unscoped* base by design, so
673/// calling it directly is safe only for rows whose tenant membership is already
674/// settled (a write by id against a record the framework loaded and
675/// authorized).
676pub fn scoped_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
677    apply_tenant_scope::<R>(cx, R::query(cx))
678}
679
680/// AND the framework's tenant predicate onto `query` (GH #223).
681///
682/// [`scoped_query`]'s body, split out for the one loader whose base is not
683/// [`Resource::query`]: the export seeds from
684/// [`Resource::export_query`](Resource::export_query), which carries the
685/// includes the rendered columns declared (GH #177). Everything else — the
686/// gate, the predicate, the fail-closed error — is shared, so the two seeds can
687/// not drift apart. It is crate-internal because a caller outside the crate
688/// always has a `Resource`, and so always wants [`scoped_query`].
689pub(crate) fn apply_tenant_scope<R: Resource>(
690    cx: &Cx,
691    query: Query<List<R::Model>>,
692) -> Result<Query<List<R::Model>>> {
693    if !R::requires_tenant() {
694        return Ok(query);
695    }
696    let tenant = crate::tenancy::require_tenant(cx)?;
697    let Some(filter) = R::tenant_scope(tenant) else {
698        // Fail closed and loudly: the resource declared a gate whose scope the
699        // framework cannot derive and the resource did not state, and running
700        // the query unscoped is the one outcome that declaration exists to
701        // prevent. `Panel::build` already refused the resource if *no* tenant
702        // could scope it (GH #231); this is the backstop for a `tenant_scope`
703        // that answers `None` only for this tenant, and for callers outside a
704        // panel.
705        tracing::error!(
706            resource = R::slug(),
707            model = std::any::type_name::<R::Model>(),
708            "requires_tenant is true but the resource supplies no tenant predicate: no `tenant_id` \
709             UUID column on the model and no `tenant_scope` override (GH #223)"
710        );
711        return Err(std::io::Error::other(format!(
712            "resource '{}' requires a tenant, but the framework cannot scope it: {} declares no \
713             `tenant_id` UUID column to derive the filter from, and the resource does not override \
714             `tenant_scope` (GH #223); declare the column, override `tenant_scope`, or drop \
715             `requires_tenant` and scope in `query`",
716            R::slug(),
717            std::any::type_name::<R::Model>(),
718        ))
719        .into());
720    };
721    Ok(query.filter(filter))
722}

724/// Every `Resource` is an [`OptionSource`](crate::schema::OptionSource) — the
725/// bridge that lets the relationship option loaders be generic over the source
726/// surface instead of over `Resource`, so `schema` does not depend on
727/// `resource` (GH #208).
728///
729/// A resource answers the loaders here, so the loaders never name `Resource`:
730///
731/// - [`scoped_query`](crate::schema::OptionSource::scoped_query) — the one required method —
732///   forwards to [`scoped_query`], so an option load inherits the tenant gate and the framework's
733///   derived tenant predicate exactly as every other loader does (GH #223). It is deliberately not
734///   [`Resource::query`], which on a gated resource is the *tenant-unscoped* base.
735/// - the policy predicates and the tenant declaration forward unchanged.
736/// - the search expression and the default ordering come from the resource's declared
737///   [`table`](Resource::table), which is where "the option search searches the related resource's
738///   searchable columns" lives (GH #150).
739impl<R: Resource> crate::schema::OptionSource for R {
740    type Model = R::Model;
741
742    fn scoped_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
743        scoped_query::<R>(cx)
744    }
745
746    fn can_view_any(cx: &Cx) -> bool {
747        <R as Resource>::can_view_any(cx)
748    }
749
750    fn can_view(cx: &Cx, record: &R::Model) -> bool {
751        <R as Resource>::can_view(cx, record)
752    }
753
754    fn requires_tenant() -> bool {
755        <R as Resource>::requires_tenant()
756    }
757
758    fn slug() -> String {
759        <R as Resource>::slug()
760    }
761
762    fn search_expr(cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
763        <R as Resource>::table(cx).search_expr(term)
764    }
765
766    fn order_by(cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
767        <R as Resource>::table(cx).order_by(false)
768    }
769}
