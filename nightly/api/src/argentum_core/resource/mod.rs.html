49/// Maps one Toasty `Model` to its admin UI.
50///
51/// # Contract (GH #138)
52///
53/// **Every method has a default**, so a resource compiles the moment it
54/// declares a [`Model`](Self::Model) — and an omission must therefore fail
55/// loudly rather than silently:
56///
57/// - **Checked at [`Panel::build`](crate::panel::Panel::build)**, which returns
58///   `Err` naming the type: the table must be renderable
59///   ([`table`](Self::table) declares columns and a row key) and, where
60///   [`can_create`](Self::can_create) allows it, the
61///   [`form`](Self::form) must declare fields. `table`, `form` and `can_create`
62///   are declarations: they must not need request-scoped context, because the
63///   panel calls them once at boot with the app's values and no request.
64/// - **Loud at request time**: the record fns
65///   ([`create_record`](Self::create_record),
66///   [`update_record`](Self::update_record),
67///   [`delete_record`](Self::delete_record),
68///   [`bulk_delete_records`](Self::bulk_delete_records)) default to an error
69///   naming the type, so a resource that never implemented delete answers
70///   "delete not implemented for …" instead of writing nothing quietly.
71/// - **Opt-in chrome, gated per record**: [`deletable`](Self::deletable) and
72///   [`editable`](Self::editable) default to `false`, so a resource that never
73///   mentions them renders no Edit or Delete affordance and cannot advertise an
74///   action its default-deny predicate refuses. A resource that wants the chrome
75///   declares the flag *and* the matching policy predicates (`can_view` +
76///   `can_delete` for `deletable`, `can_view` + `can_update` for `editable`).
77///   The flag is the
78///   whole-resource gate; the predicates are applied per row, because the panel
79///   wires them into the table's row policy ([`Table::row_actions`], GH #235):
80///   a row `can_update` refuses renders no Edit link, and a row `can_delete`
81///   refuses renders no Delete link and a disabled bulk checkbox. A row-level
82///   rule therefore narrows the chrome instead of leaving a control the route
83///   answers 403. [`viewed`](Self::viewed) is per-record exact the same way,
84///   derived from the declared [`view`](Self::view) schema rather than declared
85///   beside it.
86/// - **Default-deny is untouched**: every `can_*` still defaults to `false`, so
87///   an unconfigured resource exposes no data and no mutation.
88pub trait Resource: Sized + Send + Sync + 'static {
89    /// The persisted model this resource administers.
90    ///
91    /// `Send + Sync` holds for every data-only model struct and is required
92    /// for concurrent rendering of the resource's pages.
93    ///
94    /// `Clone` is part of the contract because a committed mutation names its
95    /// rows (GH #112): a handler keeps a copy of the rows it loaded while the
96    /// record fn consumes them, so the hook can be handed what was written.
97    type Model: toasty::schema::Model + Send + Sync + Clone + 'static;

99    /// Whether the current user may view the list page.
100    ///
101    /// Also gates relationship option loads (GH #108): a related resource
102    /// that denies this cannot offer its records as options at all.
103    fn can_view_any(_cx: &Cx) -> bool {
104        false
105    }

107    /// Whether the current user may view the given record.
108    ///
109    /// Checked on the edit page (GET), the edit POST (which requires both
110    /// `can_view` and `can_update`, GH #86), per row in CSV export, on each
111    /// record behind a relationship `Select`'s options (GH #108), and on the
112    /// list page as the per-row gate of every action link (GH #235: the View
113    /// link, and the `can_view` half of Edit and Delete). Note
114    /// both hooks default-deny: a resource used as a relationship target
115    /// must allow `can_view_any` **and** `can_view` (overriding one does not
116    /// imply the other). The list page deliberately checks only
117    /// `can_view_any` for *membership* (GH #86): `can_view` is an in-memory
118    /// Rust predicate that cannot run in SQL, and filtering rows after cursor
119    /// pagination would mislabel pages (holes, wrong Next/Prev). Row-level
120    /// visibility that must hold on the list belongs in [`Self::query`]
121    /// (ADR-0002), which every loader — list, edit, delete, bulk, export —
122    /// funnels through *inside* [`scoped_query`], so the tenant half of the
123    /// scope is applied after the override rather than by it (GH #223).
124    fn can_view(_cx: &Cx, _record: &Self::Model) -> bool {
125        false
126    }

129    fn can_create(_cx: &Cx) -> bool {
130        false
131    }

134    fn can_update(_cx: &Cx, _record: &Self::Model) -> bool {
135        false
136    }

139    ///
140    /// Checked on the single-delete and bulk-delete POSTs together with
141    /// `can_view` (GH #168) — the edit contract: a record that cannot be
142    /// viewed cannot be deleted by UUID-guessing the route.
143    fn can_delete(_cx: &Cx, _record: &Self::Model) -> bool {
144        false
145    }

147    /// Whether this resource exposes row and bulk delete chrome (GH #96).
148    ///
149    /// Chrome is opt-in (GH #226): the default renders no Delete button, no
150    /// bulk bar and no confirmation dialog, because server policy
151    /// ([`can_view`](Self::can_view) +
152    /// [`can_delete`](Self::can_delete), both default-deny) would answer 403 to
153    /// every one of them. Override to `true` alongside those predicates.
154    ///
155    /// This flag is the whole-resource gate; the predicates are applied per row
156    /// (GH #235). The panel wires them into the table's row policy, so a row
157    /// they refuse renders no Delete link and a disabled bulk checkbox — the
158    /// affordance narrows with the rule instead of leaving a control the POST
159    /// answers 403 to. The handler keeps its all-or-nothing check as the safety
160    /// net for a hand-crafted POST.
161    fn deletable() -> bool {
162        false
163    }

165    /// Whether this resource exposes row edit chrome (GH #162).
166    ///
167    /// Chrome is opt-in (GH #226): the default renders no `Edit` link per row,
168    /// because server policy ([`can_view`](Self::can_view) +
169    /// [`can_update`](Self::can_update), both default-deny) would answer 403 to
170    /// the edit GET. Override to `true` alongside those predicates.
171    ///
172    /// The same per-row rule as [`Self::deletable`] applies (GH #235): the
173    /// panel wires `can_view` + `can_update` into the table's row policy, so a
174    /// row the caller may not edit renders no link, matching the edit GET.
175    fn editable() -> bool {
176        false
177    }

179    /// How one record is displayed on the detail page (GH #187), read-only.
180    ///
181    /// The same [`Schema`] a form uses, rendered for
182    /// reading: a `TextInput` shows its stored value instead of an `<input>`,
183    /// a `Select` shows the option label the form offered, and a layout block
184    /// keeps the structure it declares (`Grid` stays a grid). Declaring a view
185    /// is what turns the detail page on — the default declares nothing, so a
186    /// resource that does not override this renders no page content, links no
187    /// `View` row action, and 404s the route.
188    ///
189    /// Values come from [`hydrate_form_values`](Self::hydrate_form_values), so
190    /// a field bound here is one the resource already knows how to read. A
191    /// relation is not one of these fields — it is a list of records, not a
192    /// string — and renders through
193    /// [`view_relations`](Self::view_relations), which the detail page draws
194    /// under this Schema.
195    ///
196    /// Read-only is a promise, not a disabled form: nothing here validates or
197    /// submits, and no field renders a required marker or an error slot —
198    /// including a `Repeater`, which renders its label over its children's
199    /// values.
200    fn view(_cx: &Cx) -> crate::schema::Schema {
201        crate::schema::Schema::empty()
202    }
203
204    /// The related records on this resource's detail page (GH #187).
205    ///
206    /// [`view`](Self::view) is a `Schema`, and a `Schema` is rendered from the
207    /// record's *string projection* — one `HashMap<String, String>` — because
208    /// that is what every field binds. A relation is not a string and may be a
209    /// list of records, so it cannot ride that map; this hook is where it
210    /// renders instead, with the loaded record in hand.
211    ///
212    /// The reason is structural, not stylistic: the render tree is monomorphic
213    /// (one walk renders every resource) while a record's type is the
214    /// resource's, so a `Schema` node cannot be handed `&Self::Model` without
215    /// type erasure across a borrow — the workspace denies `unsafe`, and the
216    /// safe erasures either require a `'static` record or capture the caller's
217    /// locals in the render's lifetime. A generic method has none of those
218    /// problems and keeps the typing.
219    ///
220    /// This is the half that makes `Resource::query`'s `include` pay: the
221    /// related rows are already loaded on the record, so a hook that reads
222    /// `record.comments.get()` issues no query at all. Touching an
223    /// un-included relation panics (`Deferred::get`), and the way to notice is
224    /// [`Deferred::is_unloaded`](toasty::Deferred::is_unloaded) — the same
225    /// marker the list columns check.
226    ///
227    /// Returns `None` (the default) for a resource with no related records to
228    /// show, which renders nothing.
229    /// The two lifetimes are deliberately separate: the returned view may
230    /// borrow the request context, never the record. A view that held the
231    /// record would pin the handler's local binding for as long as the page,
232    /// which does not compile — the renderer's projections return owned
233    /// strings ([`render_relation`]), so nothing needs to.
234    fn view_relations<'a>(
235        _cx: &'a Cx,
236        _record: &Self::Model,
237    ) -> Option<topcoat::view::BoxView<'a>> {
238        None
239    }

241    /// Whether this resource declares a detail page (GH #187).
242    ///
243    /// Derived from [`view`](Self::view) rather than declared twice, so the
244    /// route and the row link cannot disagree with the schema that renders
245    /// them. The detail handler uses it to 404 a resource that declares
246    /// nothing, and the row chrome uses it to leave the link off.
247    fn viewed(cx: &Cx) -> bool {
248        !Self::view(cx).is_empty()
249    }

251    /// The record's label in the detail page's title (GH #241), or `None` when
252    /// the record has no label to show.
253    ///
254    /// The detail page titles itself with this label when a resource returns
255    /// `Some`, and with [`navigation_label`](Self::navigation_label) plus the
256    /// URL's record key when it returns `None` — the default, so a resource
257    /// that declares nothing keeps the title it has. The showcase's
258    /// `PostResource` is the worked example: it returns the post's title, so
259    /// its heading reads the title instead of `Blog Posts <record key>`.
260    ///
261    /// `cx` is the request's context — the same one [`view`](Self::view) and
262    /// [`hydrate_form_values`](Self::hydrate_form_values) receive — so a label
263    /// can read request state (a locale, a tenant). The default ignores both
264    /// arguments and returns `None`.
265    ///
266    /// A label is display text, not a key. Two records can share one (two
267    /// users named Ada), so it cannot replace [`Table::id`], whose projection
268    /// must stay injective within a page for keyed diffs and bulk selection
269    /// (GH #96), or [`Table::pk`], which the action routes resolve as the
270    /// model's typed PK (GH #168).
271    fn record_label(_cx: &Cx, _record: &Self::Model) -> Option<String> {
272        None
273    }

275    /// The URL slug for this resource's pages, e.g. `"users"` mounts the list
276    /// at `{panel prefix}/users`.
277    ///
278    /// Defaults to the Filament convention (`HasRoutes::resolveDefaultSlug`):
279    /// take the resource type's name, strip a trailing `Resource`, pluralize
280    /// (`UserResource` → `Users`, `CategoryResource` → `Categories`), then
281    /// kebab-case (`BlogPostResource` → `blog-posts`). Override for irregular
282    /// naming the rules cannot guess (`UsersResource` pluralizes to
283    /// `userses` — name resources singular, or override).
284    fn slug() -> String {
285        let name = type_short_name::<Self>();
286        let singular = name.strip_suffix("Resource").unwrap_or(name);
287        kebab_case(&pluralize(singular))
288    }

291    ///
292    /// Defaults to the pluralized `Model` type name (Filament's plural model
293    /// label): `User` → `Users`, `Category` → `Categories`, `Person` →
294    /// `People`. Override for custom wording.

296        pluralize(type_short_name::<Self::Model>())
297    }

299    /// Base query — the seam for a resource's **own** row scoping (ADR-0002 and
300    /// its 2026-09-22 status note): soft deletes, row-level visibility, and the
301    /// relations a page loads.
302    ///
303    /// **Tenancy is not this method's job any more (GH #223).** When
304    /// [`requires_tenant`](Self::requires_tenant) is `true` the framework ANDs
305    /// the tenant filter — derived from the model's `tenant_id` column — onto
306    /// whatever this returns, at every loader, through [`scoped_query`]. Do not
307    /// re-state `tenant_id().eq(tenant_id(cx))` here: the copy is redundant,
308    /// and one that disagreed with the derived column would hide rows rather
309    /// than widen access. The gate that makes a missing tenant a 403 is
310    /// unchanged (GH #87).
311    ///
312    /// That also means an override of this method is **not** the tenant seam it
313    /// once was: code outside the framework's loaders must start from
314    /// [`scoped_query`], because on a gated resource this is the
315    /// *tenant-unscoped* base. See [`scoped_query`] for why.
316    ///
317    /// Returns the raw typed statement query (the spec's original signature):
318    /// raw queries compose generically — `filter`, `order_by`, and
319    /// `Paginate::new` are available on the raw form for any `M: Model` —
320    /// which is what lets [`crate::panel::Panel`] drive every resource's list
321    /// page through one handler. Scoping it via `Model::filter(..)` in an
322    /// override stays as ergonomic as before; the wrapper's extra methods are
323    /// only needed by hand-written loaders.
324    ///
325    /// # Keep unique constraints in step with this scope (GH #88)
326    ///
327    /// The app-side unique pre-check probes submitted values **through
328    /// [`scoped_query`]**, so it only sees the rows that query returns. A
329    /// `#[unique]` index *broader* than the scope is therefore invisible to it:
330    /// the probe misses the colliding row, the database refuses the write, and
331    /// the user gets a 500 instead of the inline "has already been taken".
332    ///
333    /// The tenant case is the one that bites — the framework scopes to
334    /// `tenant_id` while the column carries a plain `#[unique]` (global), which
335    /// makes two tenants sharing a value a legitimate pair to the probe and a
336    /// constraint violation to the database. Scope the constraint to match:
337    /// `#[unique(tenant_id, email)]`, which also makes it say what it means.
338    /// `Author.email` in the showcase is the worked example.
339    ///
340    /// The invariant cannot be *checked* at declaration time — a query's filters
341    /// are not introspectable, so nothing can compare the two automatically.
342    /// Upstream #117 (a driver-level unique-violation predicate) is what would
343    /// make a mismatch safe rather than merely documented.
344    fn query(_cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
345        toasty::stmt::Query::<List<Self::Model>>::all()
346    }

348    /// The CSV export's base query: [`Self::query`], narrowed to the relations
349    /// the rendered columns declared (GH #177).
350    ///
351    /// The export writes a cell per column, so it asks the table which
352    /// relations those columns' projections read ([`Table::include_needs`] —
353    /// each column declares them with [`TextColumn::needs`]) and hands the
354    /// answer here. `docs/guide/src/resources.md` states the override contract:
355    /// a resource splits its base query in two, one branch per declared name,
356    /// so an include `query` carries for the detail page or the live list rides
357    /// along on an export only when a rendered column declared it.
358    ///
359    /// **The default ignores `needs` and returns [`Self::query`] unchanged.**
360    /// Over-fetching a relation nothing reads costs a join; dropping one a
361    /// column does read breaks the render — the default takes the safe side.
362    /// The **list page does not use this**: it keeps inheriting [`Self::query`],
363    /// so only the export's constant-factor over-fetch is addressed here.
364    ///
365    /// # What an override must keep
366    ///
367    /// - **The non-tenant scope of [`Self::query`].** This is the same
368    ///   soft-delete/row-level seam (ADR-0002), and the export is a reader like
369    ///   any other: an override that drops that half exports other rows. The
370    ///   *tenant* half is not the override's to keep — the framework ANDs it
371    ///   onto what this returns, exactly as it does for [`Self::query`]
372    ///   (GH #223), so a gated export is scoped whether or not the override
373    ///   re-states the filter.
374    /// - **Whatever the policy path reads.** The export's visibility scan calls
375    ///   [`Self::can_view`] on every row of both passes, before any cell is
376    ///   written, so a `can_view` that reads a relation needs that relation
377    ///   included even though no column declared it — include it
378    ///   unconditionally in the narrowed branch. Reading an un-included
379    ///   relation panics in `Deferred::get`.
380    /// - **Every name a column declared.** A declared name with no matching
381    ///   include renders an unloaded relation, which the column's `is_unloaded`
382    ///   guard (ADR-0011) reports in test builds instead of a silent `"-"`.
383    fn export_query(cx: &Cx, _needs: &IncludeNeeds) -> toasty::stmt::Query<List<Self::Model>> {
384        Self::query(cx)
385    }

387    /// Whether this resource requires a tenant in every handler (GH #87).
388    ///
389    /// Opt-in and default-open: `false` preserves the default behavior — no
390    /// gate, and [`Self::query`] exactly as written. Override to `true` on a
391    /// resource whose model carries rows per tenant.
392    ///
393    /// `true` means two things, and the second is GH #223:
394    ///
395    /// 1. **The gate.** Every handler 403s when the request carries no tenant,
396    ///    instead of leaking unscoped rows or minting nil-tenant orphans
397    ///    (#87's tenantless-create rejection, unchanged).
398    /// 2. **The scope.** Every loader ANDs `tenant_id = <request tenant>` onto
399    ///    the resource's base query, deriving the column from the model's own
400    ///    schema — see [`scoped_query`] and [`Self::tenant_scope`]. A gated
401    ///    resource is therefore never unscoped because an override forgot to
402    ///    re-state the filter, and it is never unscoped because the derivation
403    ///    failed either: the model must declare a `tenant_id` UUID column, or
404    ///    the resource must declare its own predicate in
405    ///    [`Self::tenant_scope`], and a gated resource that does neither is
406    ///    refused by [`Panel::build`](crate::Panel::build) at boot (GH #231) —
407    ///    the declaration is checkable without a request — rather than serving
408    ///    rows unscoped or failing per request.
409    ///
410    /// A resource that must genuinely serve more than the request tenant — a
411    /// deliberate cross-tenant view — declares `false` and scopes in
412    /// [`Self::query`] by hand. That is the explicit, visible way out, and it
413    /// gives up the gate above along with the derived filter.
414    fn requires_tenant() -> bool {
415        false
416    }

418    /// The predicate the framework ANDs onto this resource's base query to
419    /// scope it to `tenant` (GH #223), or `None` when there is nothing to AND.
420    ///
421    /// The default derives it from the model: `tenant_id = tenant`, on the
422    /// field named `tenant_id` whose type is a UUID (see [`crate::tenancy`]).
423    /// Override it when the resource's tenancy is not a column on its own
424    /// model — a row that inherits its parent's tenant states the relation
425    /// path here instead, and the framework applies it exactly as it applies
426    /// the derived one. The showcase's comments are the worked example.
427    ///
428    /// Only consulted when [`requires_tenant`](Self::requires_tenant) is
429    /// `true`. `None` from a gated resource is a **misdeclaration**, not a way
430    /// to be unscoped: [`Panel::build`](crate::Panel::build) refuses it at boot
431    /// (GH #231), and every loader keeps answering an error naming the resource
432    /// rather than running its query without a tenant predicate — the backstop
433    /// for a predicate that is only `None` for some tenants. There is
434    /// deliberately no override that *removes* the scope — a resource that
435    /// must serve more than one tenant declares `requires_tenant() = false`
436    /// and owns the scope in [`Self::query`], visibly, with the gate given up.
437    fn tenant_scope(tenant: uuid::Uuid) -> Option<toasty::stmt::Expr<bool>> {
438        crate::tenancy::derived_tenant_filter::<Self::Model>(tenant)
439    }

477    /// Create a new record from form values, returning the row it wrote.
478    ///
479    /// The `Panel` create handler validates `required`/`email` inline and checks
480    /// `Resource::can_create` before calling this, inside a framework-owned
481    /// transaction (GH #84): `ex` is the open tx — run every statement
482    /// through it (`exec(&mut *ex)`) and never open a second handle, so the
483    /// write commits atomically with the handler's checks. The default
484    /// implementation returns an error; resources should override to perform
485    /// the actual `toasty::create!` (or `Insert`).
486    ///
487    /// Return the created row — `toasty::create!` already hands it back, and
488    /// the framework cannot otherwise name what a create wrote: the primary key
489    /// is the database's (or the app's) to generate, so it is only knowable
490    /// from the row. That is what [`Self::after_commit`] receives for a create
491    /// (GH #112).
492    fn create_record(
493        _cx: &Cx,
494        _values: HashMap<String, String>,
495        _ex: &mut dyn toasty::Executor,
496    ) -> impl std::future::Future<Output = Result<Self::Model>> + Send
497    where
498        Self: Sized,
499    {
500        async move {
501            Err(std::io::Error::other(format!(
502                "create not implemented for {}",

586    /// Post-commit work for a mutation this resource committed (GH #112).
587    ///
588    /// The one place a side effect that must not survive a rollback belongs —
589    /// an email, a webhook, an audit row, cache invalidation. Called by the
590    /// framework **once per successful write**, after `tx.commit()` and before
591    /// the response: running it inside a record fn would leak the effect when
592    /// the transaction rolls back, and the write handlers' pool discipline
593    /// (GH #84) forbids a second handle while the transaction is open. By the
594    /// time this runs the transaction is gone, so it may open its own `Db`
595    /// handle — `db(cx)` — or none at all.
596    ///
597    /// [`Committed`] names the mutation and the rows it wrote: the row a create
598    /// returned, the row an update returned (the committed state, not the
599    /// snapshot the handler loaded), the rows a delete or bulk delete removed —
600    /// deletes are the one case where the row is gone by the time you see it.
601    /// A bulk delete is **one call** with every row, not one call per row.
602    ///
603    /// It is never called when nothing committed: a validation error, a policy
604    /// denial, a failed record fn, or a failed commit all leave the hook
605    /// untouched, so a rollback can never produce the effect.
606    ///
607    /// A hook that returns `Err` is logged and ignored — the write is
608    /// committed, and an error page would misreport it. Retries and delivery
609    /// guarantees are the app's to build (an outbox written here is the usual
610    /// shape); the framework promises neither. A hook that panics is a bug in
611    /// the hook and surfaces as Topcoat's panic-isolated 500.
612    ///
613    /// Defaults to a no-op, so a resource that declares nothing is unaffected.
614    ///
615    /// ```ignore
616    /// impl Resource for PostResource {
617    ///     type Model = Post;
618    ///
619    ///     async fn after_commit(cx: &Cx, committed: Committed<Post>) -> Result<()> {
620    ///         // The transaction is committed and gone, so this opens its own handle.
621    ///         let mut db = db(cx);
622    ///         for post in committed.records() {
623    ///             notify_watchers(post, committed.mutation()).await?;
624    ///         }
625    ///         Ok(())
626    ///     }
627    /// }
628    /// ```
629    fn after_commit(
630        _cx: &Cx,
631        _committed: Committed<Self::Model>,
632    ) -> impl std::future::Future<Output = Result<()>> + Send
633    where
634        Self: Sized,
635    {
636        async move { Ok(()) }
637    }

656/// The tenant-scoped query every loader executes (GH #223).
657///
658/// [`Resource::query`] with the tenant predicate from
659/// [`Resource::tenant_scope`] ANDed onto it: for a resource whose
660/// [`requires_tenant`](Resource::requires_tenant) is `true`, the rows are
661/// narrowed to the request tenant — derived from `tenant_id` by default,
662/// declared by the resource when its tenancy is inherited. The filter is
663/// applied *here*, outside the resource's `query`, so a resource that
664/// overrides `query` for includes or soft deletes cannot drop the tenant scope
665/// by forgetting to re-state it.
666///
667/// This is the entry point; `apply_tenant_scope` is the same composition for
668/// a seed that is not [`Resource::query`].
669///
670/// # Errors
671///
672/// - A gated resource and no tenant in `cx` → 403, the same fail-closed answer
673///   the handler gate gives (GH #87).
674/// - A gated resource that supplies no tenant predicate — no discoverable
675///   `tenant_id` UUID column, no [`Resource::tenant_scope`] override → an
676///   error naming the resource and the model. [`Panel::build`](crate::Panel::build)
677///   refuses that declaration at boot (GH #231), so this is the backstop for a
678///   predicate that exists but is `None` for the request's tenant, and for app
679///   code that calls this outside a panel. It is deliberately **not** a
680///   fallback to the unscoped query: discovery is by name, and a silent miss
681///   would be exactly the leak [`Resource::requires_tenant`] exists to
682///   prevent.
683///
684/// # When to call this
685///
686/// Every framework loader does, and app code that loads rows itself must too —
687/// a record fn double-checking a foreign key, a custom page, a test. On a gated
688/// resource [`Resource::query`] is the *tenant-unscoped* base by design, so
689/// calling it directly is safe only for rows whose tenant membership is already
690/// settled (a write by id against a record the framework loaded and
691/// authorized).
692pub fn scoped_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
693    apply_tenant_scope::<R>(cx, R::query(cx))
694}
695
696/// AND the framework's tenant predicate onto `query` (GH #223).
697///
698/// [`scoped_query`]'s body, split out for the one loader whose base is not
699/// [`Resource::query`]: the export seeds from
700/// [`Resource::export_query`](Resource::export_query), which carries the
701/// includes the rendered columns declared (GH #177). Everything else — the
702/// gate, the predicate, the fail-closed error — is shared, so the two seeds can
703/// not drift apart. It is crate-internal because a caller outside the crate
704/// always has a `Resource`, and so always wants [`scoped_query`].
705pub(crate) fn apply_tenant_scope<R: Resource>(
706    cx: &Cx,
707    query: Query<List<R::Model>>,
708) -> Result<Query<List<R::Model>>> {
709    if !R::requires_tenant() {
710        return Ok(query);
711    }
712    let tenant = crate::tenancy::require_tenant(cx)?;
713    let Some(filter) = R::tenant_scope(tenant) else {
714        // Fail closed and loudly: the resource declared a gate whose scope the
715        // framework cannot derive and the resource did not state, and running
716        // the query unscoped is the one outcome that declaration exists to
717        // prevent. `Panel::build` already refused the resource if *no* tenant
718        // could scope it (GH #231); this is the backstop for a `tenant_scope`
719        // that answers `None` only for this tenant, and for callers outside a
720        // panel.
721        tracing::error!(
722            resource = R::slug(),
723            model = std::any::type_name::<R::Model>(),
724            "requires_tenant is true but the resource supplies no tenant predicate: no `tenant_id` \
725             UUID column on the model and no `tenant_scope` override (GH #223)"
726        );
727        return Err(std::io::Error::other(format!(
728            "resource '{}' requires a tenant, but the framework cannot scope it: {} declares no \
729             `tenant_id` UUID column to derive the filter from, and the resource does not override \
730             `tenant_scope` (GH #223); declare the column, override `tenant_scope`, or drop \
731             `requires_tenant` and scope in `query`",
732            R::slug(),
733            std::any::type_name::<R::Model>(),
734        ))
735        .into());
736    };
737    Ok(query.filter(filter))
738}

740/// Every `Resource` is an [`OptionSource`](crate::schema::OptionSource) — the
741/// bridge that lets the relationship option loaders be generic over the source
742/// surface instead of over `Resource`, so `schema` no longer depends on
743/// `resource` (GH #208).
744///
745/// A resource answers the loaders here, so the loaders never name `Resource`:
746///
747/// - [`scoped_query`](crate::schema::OptionSource::scoped_query) — the one
748///   required method — forwards to [`scoped_query`], so an option load inherits
749///   the tenant gate and the framework's derived tenant predicate exactly as
750///   every other loader does (GH #223). It is deliberately not
751///   [`Resource::query`], which on a gated resource is the *tenant-unscoped*
752///   base.
753/// - the policy predicates and the tenant declaration forward unchanged.
754/// - the search expression and the default ordering come from the resource's
755///   declared [`table`](Resource::table), which is where "the option search
756///   searches the related resource's searchable columns" lives (GH #150).
757impl<R: Resource> crate::schema::OptionSource for R {
758    type Model = R::Model;
759
760    fn scoped_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
761        scoped_query::<R>(cx)
762    }
763
764    fn can_view_any(cx: &Cx) -> bool {
765        <R as Resource>::can_view_any(cx)
766    }
767
768    fn can_view(cx: &Cx, record: &R::Model) -> bool {
769        <R as Resource>::can_view(cx, record)
770    }
771
772    fn requires_tenant() -> bool {
773        <R as Resource>::requires_tenant()
774    }
775
776    fn slug() -> String {
777        <R as Resource>::slug()
778    }
779
780    fn search_expr(cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
781        <R as Resource>::table(cx).search_expr(term)
782    }
783
784    fn order_by(cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
785        <R as Resource>::table(cx).order_by(false)
786    }
787}
