1//! Where uploaded bytes go (GH #188): the `Uploader` seam, and the one place
2//! the framework hands bytes to it.
3//!
4//! `FileUpload` binds a `String` column, the panel renders a file input, and
5//! the form parser decodes the multipart body — but *where* the bytes live and
6//! what path the record stores is the app's decision: an object store, a
7//! directory on disk, a CDN. The framework owns everything up to the bytes and
8//! nothing after them, so this module is deliberately small: a trait, the app
9//! context value that carries it, and the call that runs it.
10
11use std::collections::HashMap;
12use std::future::Future;
13use std::pin::Pin;

15use topcoat::context::{Cx, try_app_context};

19/// Store one uploaded file and name the value a record stores (GH #188).
20///
21/// Installed once per panel with [`Panel::uploads`](crate::Panel::uploads) —
22/// the way `Db` is — and found on the app context wherever a `FileUpload`
23/// stores, because an object store is an app-level dependency: threading it
24/// through every `.for(..)` call site would put it in the schema declaration.
25///
26/// `filename` arrives already sanitized to a basename (GH #90): no directory
27/// components, no control characters, capped at 255 bytes, and never empty (an
28/// empty filename is "no file chosen", which never reaches an uploader).
29/// `bytes` are the part's content, bounded by the form-body cap (10 MiB,
30/// `MAX_FORM_BYTES`). Everything else is the
31/// app's business: generating a collision-free name, choosing a directory or
32/// bucket, and deciding what the returned path means — the framework stores it
33/// verbatim and renders it as the stored value.
34pub trait Uploader: Send + Sync + 'static {
35    /// Store `bytes` and return the value to store for this field.
36    ///
37    /// The `Err` string is rendered to the user inside the field's inline
38    /// error ("`<Label>` could not be uploaded: `<reason>`"), so it must be
39    /// something they can act on — never a filesystem path, a driver message,
40    /// or anything else the deployment would rather not print. A rejection is
41    /// user input going wrong, not infrastructure: it re-renders the form with
42    /// the submitted values instead of failing the request.
43    fn store(
44        &self,
45        filename: &str,
46        bytes: &[u8],
47    ) -> impl Future<Output = Result<String, String>> + Send;
48}

50/// A file part a form submitted: the sanitized basename and its bytes.
51///
52/// Staged by the multipart parser (`panel::forms`) only when an [`Uploader`] is
53/// installed. Without one the bytes are still drained and dropped, exactly as
54/// they were before GH #188, so an app that installs no uploader keeps the
55/// constant-memory path it had.
56#[derive(Debug, Clone)]
57pub(crate) struct StagedUpload {
58    pub(crate) filename: String,
59    pub(crate) bytes: Vec<u8>,
60}
61
62/// The one uploader a panel was built with, on the app context the way `Db` is.
63pub(crate) struct InstalledUploader(Box<dyn DynUploader + Send + Sync>);
64
65impl InstalledUploader {
66    pub(crate) fn new(uploader: impl Uploader) -> Self {
67        Self(Box::new(uploader))
68    }
69}

71/// Boxed future of [`DynUploader::store`].
72type StoreFuture<'a> = Pin<Box<dyn Future<Output = Result<String, String>> + Send + 'a>>;
73
74/// Dyn-compatible view of [`Uploader`].
75///
76/// [`Uploader`] returns `impl Future` (the house style: no `async_trait`
77/// dependency, no hand-boxed signatures for implementors), which is not
78/// dyn-compatible. The panel holds whichever uploader the app installed
79/// without becoming generic over it, so the single call it makes goes through
80/// this shim — the public trait stays the shape an app implements, and the
81/// box stays here.
82pub(crate) trait DynUploader: Send + Sync {
83    fn store<'a>(&'a self, filename: &'a str, bytes: &'a [u8]) -> StoreFuture<'a>;
84}
85
86impl<U: Uploader> DynUploader for U {
87    fn store<'a>(&'a self, filename: &'a str, bytes: &'a [u8]) -> StoreFuture<'a> {
88        Box::pin(Uploader::store(self, filename, bytes))
89    }
90}

92/// Inline errors from a failed store, keyed by field name: the shape
93/// [`Schema::validate`](crate::schema::Schema::validate) answers with, so the
94/// handler merges the two without translating.
95pub(crate) type UploadErrors = HashMap<String, Vec<String>>;
96
97/// Whether this panel has an uploader installed (GH #188).
98///
99/// The multipart parser asks before staging bytes: with no uploader they would
100/// be buffered only to be dropped, and today's drain-and-discard is what keeps
101/// a large upload off the heap for every app that never installs one.
102pub(crate) fn installed(cx: &Cx) -> bool {
103    try_app_context::<InstalledUploader>(cx).is_some()
104}
105
106/// Run the installed uploader over the file parts this form submitted
107/// (GH #188), returning `field_name -> inline errors`.
108///
109/// For each declared [`FileUpload`](crate::schema::FileUpload) that carried
110/// bytes, the returned path replaces the sanitized basename the parser put in
111/// `values` — so the record fn sees the stored path and nothing else changes
112/// about its contract. Without an installed uploader this is a no-op: the
113/// basename stays, which is exactly the pre-#188 behaviour an app that never
114/// installs one must keep.
115///
116/// A failed store becomes an inline field error and **drops the submitted
117/// value**, because there is no path to store: nothing was written, and the
118/// client's filename is not a stored file. Dropping it is what lets the caller
119/// re-render honestly — run this *before* the edit handler's untouched-value
120/// backfill, which then restores the path that is actually stored, while a
121/// create renders the field empty beside the reason. Files stored earlier in
122/// the same call are *not* rolled back — their paths never reach the record, so
123/// they are unreferenced rather than wrong; a store with a real write cost
124/// wants its own janitor, which is the app's call, not the framework's.
125///
126/// Call this outside the write transaction: an upload is a side effect in
127/// another system, and a rolled-back transaction must not have to undo it.
128pub(crate) async fn store_uploads(
129    cx: &Cx,
130    schema: &Schema,
131    files: &HashMap<String, StagedUpload>,
132    values: &mut HashMap<String, String>,
133) -> UploadErrors {
134    let Some(uploader) = try_app_context::<InstalledUploader>(cx).map(|installed| &*installed.0)
135    else {
136        return UploadErrors::new();
137    };
138    let mut errors = UploadErrors::new();
139    // Declared uploads only: a file part the schema does not declare is not a
140    // field this form may write (the unknown-key allow-list answers for it).
141    for (name, upload) in schema.file_uploads() {
142        let Some(staged) = files.get(&name) else {
143            continue;
144        };
145        match uploader.store(&staged.filename, &staged.bytes).await {
146            Ok(path) => {
147                values.insert(name, path);
148            }
149            Err(reason) => {
150                values.remove(&name);
151                errors.insert(
152                    name,
153                    vec![format!(
154                        "{} could not be uploaded: {reason}",
155                        upload.label_str()
156                    )],
157                );
158            }
159        }
160    }
161    errors
162}
