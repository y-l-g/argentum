1//! Where uploaded bytes go (GH #188): the `Uploader` seam, and the one place
2//! the framework hands bytes to it.
3//!
4//! `FileUpload` binds a `String` column, the panel renders a file input, and
5//! the form parser decodes the multipart body — but *where* the bytes live and
6//! what path the record stores is the app's decision: an object store, a
7//! directory on disk, a CDN. The framework owns everything up to the bytes and
8//! nothing after them, so this module is deliberately small: a trait, the app
9//! context value that carries it, and the call that runs it.
10

21/// Store one uploaded file and name the value a record stores (GH #188).
22///
23/// Installed once per panel with [`Panel::uploads`](crate::Panel::uploads) —
24/// the way `Db` is — and found on the app context wherever a `FileUpload`
25/// stores, because an object store is an app-level dependency: threading it
26/// through every `.for(..)` call site would put it in the schema declaration.
27///
28/// `filename` arrives already sanitized to a basename (GH #90): no directory
29/// components, no control characters, capped at 255 bytes, and never empty (an
30/// empty filename is "no file chosen", which never reaches an uploader).
31/// `bytes` are the part's content, bounded by the form-body cap (10 MiB,
32/// `MAX_FORM_BYTES`). Everything else is the
33/// app's business: generating a collision-free name, choosing a directory or
34/// bucket, and deciding what the returned path means — the framework stores it
35/// verbatim and renders it as the stored value.
36pub trait Uploader: Send + Sync + 'static {
37    /// Store `bytes` and return the value to store for this field.
38    ///
39    /// The `Err` string is rendered to the user inside the field's inline
40    /// error ("`<Label>` could not be uploaded: `<reason>`"), so it must be
41    /// something they can act on — never a filesystem path, a driver message,
42    /// or anything else the deployment would rather not print. A rejection is
43    /// user input going wrong, not infrastructure: it re-renders the form with
44    /// the submitted values instead of failing the request.
45    fn store(
46        &self,
47        filename: &str,
48        bytes: &[u8],
49    ) -> impl Future<Output = Result<String, String>> + Send;
50
51    /// Whether `path` is a value this store produced and still holds (GH #297).
52    ///
53    /// A form that re-renders with errors carries the path a just-finished
54    /// store returned, so the file survives the next submit; the framework asks
55    /// this before it re-uses that path, which keeps the value's origin in the
56    /// store rather than in whatever the client sent: a client-typed path is
57    /// stored only when the store itself vouches for it (GH #277).
58    ///
59    /// The contract is **ownership, not bare existence**: answer `true` only
60    /// for a path this store returned from [`store`](Self::store) and still
61    /// resolves **inside its own root**. An existence check on a
62    /// client-supplied path — `Path::exists`, a `HEAD` on any URL — would turn
63    /// this into a path-traversal gate.
64    ///
65    /// The default answers `false`, so a store that does not implement it keeps
66    /// the pre-#297 behaviour: a re-rendered form does not carry an upload, and
67    /// the next submit fails the field's `required` rule (create) or keeps the
68    /// record's stored file (edit).
69    fn holds(&self, path: &str) -> impl Future<Output = bool> + Send {
70        async move {
71            let _ = path;
72            false
73        }
74    }
75}

77/// A file part a form submitted: the sanitized basename and its bytes.
78///
79/// Staged by the multipart parser (`panel::forms`) only when an [`Uploader`] is
80/// installed. Without one the bytes are still drained and dropped, exactly as
81/// they were before GH #188, so an app that installs no uploader keeps the
82/// constant-memory path it had.
83#[derive(Debug, Clone)]
84pub(crate) struct StagedUpload {
85    pub(crate) filename: String,
86    pub(crate) bytes: Vec<u8>,
87}

98/// Boxed future of [`DynUploader::store`].
99type StoreFuture<'a> = Pin<Box<dyn Future<Output = Result<String, String>> + Send + 'a>>;
100
101/// Boxed future of [`DynUploader::holds`].
102type HoldFuture<'a> = Pin<Box<dyn Future<Output = bool> + Send + 'a>>;
103
104/// Dyn-compatible view of [`Uploader`].
105///
106/// [`Uploader`] returns `impl Future` (the house style: no `async_trait`
107/// dependency, no hand-boxed signatures for implementors), which is not
108/// dyn-compatible. The panel holds whichever uploader the app installed
109/// without becoming generic over it, so the calls it makes go through this
110/// shim — the public trait stays the shape an app implements, and the box
111/// stays here.
112pub(crate) trait DynUploader: Send + Sync {
113    fn store<'a>(&'a self, filename: &'a str, bytes: &'a [u8]) -> StoreFuture<'a>;
114    fn holds<'a>(&'a self, path: &'a str) -> HoldFuture<'a>;
115}
116
117impl<U: Uploader> DynUploader for U {
118    fn store<'a>(&'a self, filename: &'a str, bytes: &'a [u8]) -> StoreFuture<'a> {
119        Box::pin(Uploader::store(self, filename, bytes))
120    }
121
122    fn holds<'a>(&'a self, path: &'a str) -> HoldFuture<'a> {
123        Box::pin(Uploader::holds(self, path))
124    }
125}

127/// Inline errors from a failed store, keyed by field name: the shape
128/// [`Schema::validate`](crate::schema::Schema::validate) answers with, so the
129/// handler merges the two without translating.
130pub(crate) type UploadErrors = HashMap<String, Vec<String>>;

132/// Whether this panel has an uploader installed (GH #188).
133///
134/// The multipart parser asks before staging bytes: with no uploader they would
135/// be buffered only to be dropped, and today's drain-and-discard is what keeps
136/// a large upload off the heap for every app that never installs one.
137pub(crate) fn installed(cx: &Cx) -> bool {
138    try_app_context::<InstalledUploader>(cx).is_some()
139}
140
141/// The installed uploader, if this panel has one.
142///
143/// The concrete boxed trait object keeps its auto traits: dropping `Send +
144/// Sync` from the reference is not a coercion the compiler performs here.
145type DynUploaderRef<'a> = &'a (dyn DynUploader + Send + Sync);
146
147fn installed_uploader<'a>(cx: &'a Cx) -> Option<DynUploaderRef<'a>> {
148    try_app_context::<InstalledUploader>(cx).map(|installed| &*installed.0)
149}

151/// Whether the installed uploader still holds `path` (GH #297).
152///
153/// `false` without an installed uploader: nothing stored the bytes, so nothing
154/// can vouch for a path a re-rendered form carried.
155pub(crate) async fn holds(cx: &Cx, path: &str) -> bool {
156    match installed_uploader(cx) {
157        Some(uploader) => uploader.holds(path).await,
158        None => false,
159    }
160}

162/// Run the installed uploader over the file parts this form submitted
163/// (GH #188), returning `field_name -> inline errors` and the fields whose
164/// value is now the uploader's answer.
165///
166/// For each declared [`FileUpload`](crate::schema::FileUpload) that carried
167/// bytes, the returned path replaces the sanitized basename the parser put in
168/// `values` — so the record fn sees the stored path and nothing else changes
169/// about its contract. The second half of the answer is those field names: a
170/// form that re-renders carries them so the next submit can keep the upload
171/// (GH #297). Without an installed uploader this is a no-op: the basename
172/// stays, which is exactly the pre-#188 behaviour an app that never installs
173/// one must keep, and nothing is carried, because a client's filename is not a
174/// stored file.
175///
176/// A failed store becomes an inline field error and **drops the submitted
177/// value**, because there is no path to store: nothing was written, and the
178/// client's filename is not a stored file. Dropping it is what lets the caller
179/// re-render honestly — run this *before* the edit handler's untouched-value
180/// backfill, which then restores the path that is actually stored, while a
181/// create renders the field empty beside the reason. Files stored earlier in
182/// the same call are *not* rolled back — their paths never reach the record, so
183/// they are unreferenced rather than wrong; a store with a real write cost
184/// wants its own janitor, which is the app's call, not the framework's.
185///
186/// Call this outside the write transaction: an upload is a side effect in
187/// another system, and a rolled-back transaction must not have to undo it.
188pub(crate) async fn store_uploads(
189    cx: &Cx,
190    schema: &Schema,
191    files: &HashMap<String, StagedUpload>,
192    values: &mut HashMap<String, String>,
193) -> (UploadErrors, HashSet<String>) {
194    let Some(uploader) = installed_uploader(cx) else {
195        return (UploadErrors::new(), HashSet::new());
196    };
197    let mut errors = UploadErrors::new();
198    let mut stored = HashSet::new();
199    // Declared uploads only: a file part the schema does not declare is not a
200    // field this form may write (the unknown-key allow-list answers for it).
201    for (name, upload) in schema.file_uploads() {
202        let Some(staged) = files.get(&name) else {
203            continue;
204        };
205        match uploader.store(&staged.filename, &staged.bytes).await {
206            Ok(path) => {
207                values.insert(name.clone(), path);
208                stored.insert(name);
209            }
210            Err(reason) => {
211                values.remove(&name);
212                errors.insert(
213                    name,
214                    vec![format!(
215                        "{} could not be uploaded: {reason}",
216                        upload.label_str()
217                    )],
218                );
219            }
220        }
221    }
222    (errors, stored)
223}
