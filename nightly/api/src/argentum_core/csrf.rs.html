1//! CSRF protection via double-submit cookie (GH #99).
2//!
3//! Every state-changing form embeds `csrf_token`, and every POST handler
4//! verifies the form value matches the cookie. No server-side session is
5//! needed: the token is a random UUID the server sets (and reads) via the
6//! cookie layer, and the browser's same-origin policy keeps an attacker
7//! from reading the token to forge the form field. `confirm=1` stays
8//! a UX step, never a security boundary.
9//!
10//! The cookie is `__Host-`-prefixed and `Secure` (GH #149, matching the
11//! session cookie's hardened contract): a cookie-writable position
12//! (subdomain, cleartext HTTP) cannot pin a known token to the jar. The
13//! compare is constant-time so a mismatch cannot be probed byte-by-byte.
14//! The token deliberately stays a bare random UUID — binding it to the
15//! server (HMAC via Topcoat's signed jar) would need a `Key` app context
16//! `Panel::build` does not register; that stays an upstream-gap decision
17//! (#139), not a hand-rolled one.
18
19use subtle::ConstantTimeEq;
20use topcoat::{
21    context::{Cx, try_request_context},
22    cookie::{Cookie, CookieJarCell, Cookies, cookies},
23};

25/// Cookie carrying the CSRF token (`__Host-` prefix: `Secure` + `Path=/` +
26/// no `Domain` are required by the prefix contract, GH #149).
27pub const COOKIE_NAME: &str = "__Host-argentum_csrf";
28/// Hidden form field carrying the CSRF token.
29pub const FIELD_NAME: &str = "csrf_token";
30
31/// Ensure a token exists for this request, setting the cookie when needed.
32///
33/// Must be called before response headers are sent (i.e. in the page handler,
34/// not inside a streamed `suspense` child — setting cookies after headers
35/// panics). Returns the token to embed in forms. When no cookie layer is
36/// present (bare unit renders), returns an empty string so renders never panic;
37/// POST handlers behind the real router always have the layer and enforce.
38pub fn ensure_token(cx: &Cx) -> String {
39    if try_request_context::<CookieJarCell>(cx).is_none() {
40        return String::new();
41    }
42    let jar = cookies(cx);
43    if let Some(cookie) = jar.get(COOKIE_NAME) {
44        let value = cookie.value().to_string();
45        if is_valid_token(&value) {
46            return value;
47        }
48    }
49    let token = uuid::Uuid::new_v4().to_string();
50    let cookie = Cookie::build((COOKIE_NAME, token.clone()))
51        .path("/")
52        .http_only(true)
53        .secure(true)
54        .same_site(topcoat::cookie::SameSite::Lax)
55        .build();
56    jar.add(cookie);
57    token
58}

60/// Read the current token without setting one (GH #99).
61///
62/// Safe inside streamed `suspense` children that outlive header send: renders
63/// embed the already-ensured token, or `""` when none was ensured.
64pub fn current_token(cx: &Cx) -> String {
65    if try_request_context::<CookieJarCell>(cx).is_none() {
66        return String::new();
67    }
68    cookies(cx)
69        .get(COOKIE_NAME)
70        .map(|c| c.value().to_string())
71        .filter(|v| is_valid_token(v))
72        .unwrap_or_default()
73}

75/// The hidden field every state-changing form embeds.
76///
77/// The token is passed in, never resolved here: whether a site calls
78/// [`ensure_token`] (which sets the cookie and must run before response headers
79/// are sent) or [`current_token`] (the only one safe inside a streamed
80/// `suspense` child) is the site's decision, and a helper that guessed would
81/// either panic after header send or silently embed nothing. What the helper
82/// owns is the spelling — [`FIELD_NAME`] is what [`verify`] reads, so a rename
83/// that missed a form would be a silent 403 on every POST (GH #212).
84pub fn field<'a>(cx: &'a Cx, token: &str) -> topcoat::view::BoxView<'a> {
85    use topcoat::view::ViewExt;
86
87    // Own the token before the `view!` block: the emitted view must borrow the
88    // request context and nothing else, or a caller's local `String` would have
89    // to outlive the page.
90    let token = token.to_string();
91    topcoat::view::view! { cx => <input type="hidden" name=(FIELD_NAME) value=(token)> }.boxed()
92}

94/// Verify the submitted form token matches the cookie (GH #99).
95///
96/// Fails closed: missing cookie, missing field, or mismatch all yield 403.
97/// The mismatch compare is constant-time (GH #149) so a failed double-submit
98/// cannot be probed byte-by-byte; the token itself stays a random UUID, so
99/// a length difference is not a secret.
100pub fn verify(
101    cx: &Cx,
102    values: &std::collections::HashMap<String, String>,
103) -> Result<(), topcoat::Error> {
104    let cookie_ok = try_request_context::<CookieJarCell>(cx)
105        .map(|_| cookies(cx).get(COOKIE_NAME).map(|c| c.value().to_string()))
106        .unwrap_or(None);
107    let Some(expected) = cookie_ok else {
108        return Err(topcoat::router::error::forbidden().into());
109    };
110    let Some(submitted) = values.get(FIELD_NAME) else {
111        return Err(topcoat::router::error::forbidden().into());
112    };
113    if !is_valid_token(&expected) || submitted.as_bytes().ct_ne(expected.as_bytes()).into() {
114        return Err(topcoat::router::error::forbidden().into());
115    }
116    Ok(())
117}

119fn is_valid_token(value: &str) -> bool {
120    value.parse::<uuid::Uuid>().is_ok()
121}

123#[cfg(test)]
124mod tests {
125    use topcoat::context::CxTestBuilder;
126
127    use super::*;
128
129    fn cx_with_cookie(value: Option<&str>) -> Cx {
130        let mut parts = http::Request::builder()
131            .uri("/")
132            .body(())
133            .unwrap()
134            .into_parts()
135            .0;
136        if let Some(v) = value {
137            parts.headers.insert(
138                http::header::COOKIE,
139                format!("{COOKIE_NAME}={v}").parse().unwrap(),
140            );
141        }
142        CxTestBuilder::new()
143            .request_context(parts)
144            .request_context(CookieJarCell::new())
145            .build()
146    }
147
148    #[test]
149    fn valid_token_format() {
150        assert!(is_valid_token(&uuid::Uuid::new_v4().to_string()));
151        assert!(!is_valid_token(""));
152        assert!(!is_valid_token("not-a-uuid"));
153    }
154
155    #[test]
156    fn verify_matches_cookie_and_rejects_mismatch() {
157        let token = uuid::Uuid::new_v4().to_string();
158        let cx = cx_with_cookie(Some(&token));
159        let mut values = std::collections::HashMap::new();
160        values.insert(FIELD_NAME.to_string(), token.clone());
161        assert!(verify(&cx, &values).is_ok());
162        values.insert(FIELD_NAME.to_string(), uuid::Uuid::new_v4().to_string());
163        assert!(verify(&cx, &values).is_err());
164        assert!(verify(&cx, &std::collections::HashMap::new()).is_err());
165        assert!(
166            verify(
167                &cx_with_cookie(None),
168                &std::collections::HashMap::from([(FIELD_NAME.to_string(), token)])
169            )
170            .is_err()
171        );
172    }
173
174    /// The issued cookie carries the hardened `__Host-` contract (GH #149),
175    /// matching the session cookie: `Secure`, `HttpOnly`, `SameSite=Lax`,
176    /// `Path=/`, no `Domain`.
177    #[test]
178    fn ensured_cookie_is_host_prefixed_and_secure() {
179        let cx = cx_with_cookie(None);
180        let token = ensure_token(&cx);
181        assert!(is_valid_token(&token));
182        let cookie = cookies(&cx)
183            .get(COOKIE_NAME)
184            .expect("ensure_token set the cookie");
185        assert_eq!(cookie.value(), token);
186        assert_eq!(cookie.name(), COOKIE_NAME);
187        assert!(cookie.secure().unwrap_or(false), "{cookie:?}");
188        assert!(cookie.http_only().unwrap_or(false), "{cookie:?}");
189        assert_eq!(cookie.path(), Some("/"));
190        assert!(cookie.domain().is_none(), "{cookie:?}");
191        assert!(
192            matches!(
193                cookie.same_site(),
194                Some(topcoat::cookie::SameSite::Lax) | None
195            ),
196            "{cookie:?}"
197        );
198    }
