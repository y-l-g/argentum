1//! Primary-key bridge — URL ids to typed Toasty predicates.
2//!
3//! Sole home of the `pk_*` helpers: parsing path-segment ids into the
4//! model's key type and building equality/`IN` predicates through
5//! `toasty_core` (upstream #114). Generic callers (panel loaders) cannot
6//! build the typed expressions without this dynamic-value bridge.
7
8/// Parse a URL path segment into `M`'s primary-key value.
9///
10/// The PK's application type decides the [`stmt::Value`] variant (`Uuid` PK
11/// → `Value::Uuid`, `i64` PK → `Value::I64`, …). Returns `None` when `id`
12/// does not parse (an unparseable id cannot exist — callers render
13/// not-found) or the PK is not a single primitive field (composite keys
14/// have no URL representation in Argentum; row keys are plain `String`s).
15///
16/// Bridge helper — reaches into `toasty_core` for the schema walk and the
17/// untyped equality construction (upstream issue #114): the facade's
18/// `find_by_primary_key` takes a typed `Expr<M::PrimaryKey>`, which generic
19/// code cannot build from a `String` without a dynamic-value bridge.
20fn pk_field_value<M>(
21    id: &str,
22) -> Option<(toasty_core::schema::app::FieldId, toasty_core::stmt::Value)>
23where
24    M: toasty::schema::Model,
25{
26    let app_model = M::schema();
27    let root = app_model.as_root()?;
28    if root.primary_key.fields.len() != 1 {
29        return None;
30    }
31    let fid = root.primary_key.fields.first().copied()?;
32    let model_field = app_model.fields().get(fid.index)?;
33    let toasty_core::schema::app::FieldTy::Primitive(prim) = &model_field.ty else {
34        return None;
35    };
36    let value = match prim.ty {
37        toasty_core::stmt::Type::Uuid => toasty_core::stmt::Value::Uuid(id.parse().ok()?),
38        toasty_core::stmt::Type::String => toasty_core::stmt::Value::String(id.to_string()),
39        toasty_core::stmt::Type::Bool => toasty_core::stmt::Value::Bool(id.parse().ok()?),
40        toasty_core::stmt::Type::I8 => toasty_core::stmt::Value::I8(id.parse().ok()?),
41        toasty_core::stmt::Type::I16 => toasty_core::stmt::Value::I16(id.parse().ok()?),
42        toasty_core::stmt::Type::I32 => toasty_core::stmt::Value::I32(id.parse().ok()?),
43        toasty_core::stmt::Type::I64 => toasty_core::stmt::Value::I64(id.parse().ok()?),
44        toasty_core::stmt::Type::U8 => toasty_core::stmt::Value::U8(id.parse().ok()?),
45        toasty_core::stmt::Type::U16 => toasty_core::stmt::Value::U16(id.parse().ok()?),
46        toasty_core::stmt::Type::U32 => toasty_core::stmt::Value::U32(id.parse().ok()?),
47        toasty_core::stmt::Type::U64 => toasty_core::stmt::Value::U64(id.parse().ok()?),
48        toasty_core::stmt::Type::F32 => toasty_core::stmt::Value::F32(id.parse().ok()?),
49        toasty_core::stmt::Type::F64 => toasty_core::stmt::Value::F64(id.parse().ok()?),
50        // Bytes PKs have no canonical URL text form; accept the UTF-8 bytes so
51        // list/edit round-trip instead of 404ing (GH #95).
52        toasty_core::stmt::Type::Bytes => toasty_core::stmt::Value::Bytes(id.as_bytes().to_vec()),
53        // Temporal PKs parse from their canonical string forms, in lockstep
54        // with the cursor codec (GH #95). Decimal/net PKs need Toasty
55        // features this build doesn't enable (`rust_decimal`, `bigdecimal`,
56        // `net`) and composite keys have no URL representation — both stay
57        // documented limits.
58        toasty_core::stmt::Type::Timestamp => toasty_core::stmt::Value::Timestamp(id.parse().ok()?),
59        toasty_core::stmt::Type::Date => toasty_core::stmt::Value::Date(id.parse().ok()?),
60        toasty_core::stmt::Type::Time => toasty_core::stmt::Value::Time(id.parse().ok()?),
61        toasty_core::stmt::Type::DateTime => toasty_core::stmt::Value::DateTime(id.parse().ok()?),
62        toasty_core::stmt::Type::Zoned => toasty_core::stmt::Value::Zoned(id.parse().ok()?),
63        _ => return None,
64    };
65    Some((fid, value))
66}

68/// Whether `M`'s primary key is composite (GH #95): more than one field.
69/// Composite keys have no URL representation in Argentum (row keys are plain
70/// `String`s), so handlers fail loudly (500, programmer error) instead of
71/// 404ing every id and hiding the misconfiguration.
72pub(crate) fn pk_is_composite<M>() -> bool
73where
74    M: toasty::schema::Model,
75{
76    M::schema()
77        .as_root()
78        .is_some_and(|root| root.primary_key.fields.len() > 1)
79}

81/// Equality predicate on `M`'s primary key for a URL path-segment `id` —
82/// `pk == value`. `None` when the id does not parse as the PK's type or the
83/// PK is not a single primitive field.
84///
85/// Consumers: the panel's edit/delete loaders, which filter the
86/// tenant-scoped [`scoped_query`](crate::resource::scoped_query) instead of
87/// fetching
88/// every row and matching row keys in memory (GH #75 item 1).
89pub(crate) fn pk_eq_expr<M>(id: &str) -> Option<toasty::stmt::Expr<bool>>
90where
91    M: toasty::schema::Model,
92{
93    let (fid, value) = pk_field_value::<M>(id)?;
94    let cond = toasty_core::stmt::Expr::eq(
95        toasty_core::stmt::Expr::ref_self_field(fid),
96        toasty_core::stmt::Expr::from(value),
97    );
98    Some(toasty::stmt::Expr::from_untyped(cond))
99}

101/// `IN` predicate over primary keys for a bulk id list — `pk IN (a, b, …)`.
102/// `None` when any id fails to parse as the PK's type (an unparseable id
103/// cannot exist, so the batch fails closed) or the PK is not a single
104/// primitive field.
105///
106/// A single `IN` predicate, not an N-way `OR` chain (GH #85): the batch is
107/// still bounded by `MAX_BULK_IDS` in the bulk-delete handler.
108pub(crate) fn pk_in_expr<M>(ids: &[&str]) -> Option<toasty::stmt::Expr<bool>>
109where
110    M: toasty::schema::Model,
111{
112    let mut parsed = ids.iter().map(|id| pk_field_value::<M>(id));
113    let (fid, first) = parsed.next()??;
114    let mut values = vec![first];
115    for item in parsed {
116        let (f, v) = item?;
117        debug_assert_eq!(
118            f.index, fid.index,
119            "pk_in_expr: one model, one PK field — mixed fields are a bug"
120        );
121        values.push(v);
122    }
123    let cond = toasty_core::stmt::Expr::in_list(
124        toasty_core::stmt::Expr::ref_self_field(fid),
125        toasty_core::stmt::Expr::list(values),
126    );
127    Some(toasty::stmt::Expr::from_untyped(cond))
128}

133    use super::*;
134    #[derive(Debug, toasty::Model)]
135    struct DummyUser {

137        #[auto]

144    #[test]
145    fn composite_pk_has_no_url_representation() {
146        // GH #95: composite keys fail loudly (programmer error), never a
147        // per-id 404 that hides the misconfiguration.
148        #[derive(Debug, Clone, toasty::Model)]
149        struct Pair {
150            #[key]
151            a: String,
152            #[key]
153            b: String,
154            name: String,
155        }
156        assert!(pk_is_composite::<Pair>());
157        assert!(!pk_is_composite::<DummyUser>());
158        assert!(pk_eq_expr::<Pair>("anything").is_none());
159    }
160
161    #[test]
162    fn pk_in_expr_builds_one_in_predicate_and_fails_closed() {
163        // GH #85: single IN predicate; empty lists and unparseable ids yield
164        // None (empty posts redirect with a toast before reaching here, GH
165        // #151; an unparseable id cannot exist, so the batch must not
166        // silently drop it — the handler maps None to 404).
167        assert!(pk_in_expr::<DummyUser>(&[]).is_none());
168        assert!(pk_in_expr::<DummyUser>(&["not-a-uuid"]).is_none());
169        let a = uuid::Uuid::new_v4().to_string();
170        let b = uuid::Uuid::new_v4().to_string();
171        assert!(pk_in_expr::<DummyUser>(&[a.as_str(), b.as_str()]).is_some());
172        assert!(pk_in_expr::<DummyUser>(&[a.as_str(), "not-a-uuid"]).is_none());
173    }

186        name: String,

189    #[tokio::test]
190    async fn zoned_pks_parse_from_url_ids() {
191        // Zoned PKs parse from canonical forms, in lockstep with the cursor
192        // codec (GH #95); garbage stays a 404.
193        let z = jiff::civil::date(2024, 1, 15)
194            .at(9, 30, 0, 0)
195            .to_zoned(jiff::tz::TimeZone::UTC)
196            .unwrap();
197        assert!(pk_eq_expr::<ZonedPk>(&z.to_string()).is_some());
198        assert!(pk_eq_expr::<ZonedPk>("not-a-time").is_none());
199        // Round-trip through sqlite: the parsed value filters the row.
200        let mut db = toasty::Db::builder()
201            .models(toasty::models!(ZonedPk))
202            .connect("sqlite::memory:")
203            .await
204            .unwrap();
205        db.push_schema().await.unwrap();
206        toasty::create!(ZonedPk {
207            at: z.clone(),
208            name: "Ada".to_string(),
209        })
210        .exec(&mut db)
211        .await
212        .unwrap();
213        let mut db2 = db.clone();
214        let expr = pk_eq_expr::<ZonedPk>(&z.to_string()).unwrap();
215        let rows = ZonedPk::filter(expr).exec(&mut db2).await.unwrap();
216        assert_eq!(rows.len(), 1);
217        assert_eq!(rows[0].name, "Ada");
218    }
