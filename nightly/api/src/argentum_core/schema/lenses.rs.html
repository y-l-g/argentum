1//! Field lenses — typed Toasty paths and their app-level metadata.
2//!
3//! Base bridge layer (with `pk`): these two modules are the only schema modules
4//! that name `toasty_core` (upstream #114/#183), so upstream churn has one
5//! blast radius.
6//!
7//! [`lens_field`] hands callers the built `app::Field`, so name, label,
8//! nullability, storage name, `FieldTy`, `auto` and `constraints` all come from
9//! one walk; [`lens_field_unique`] covers uniqueness, which `Field` does not
10//! carry because Toasty keeps it on the model's index list.
11//!
12//! [`FieldResolver`] is the schema-aware walk: with the app schema in hand an
13//! embedded path resolves to its **flattened storage column**, so
14//! `TextInput::r#for_post(...)`-style bindings reach a field inside an embedded
15//! struct or a `#[document]`. Without a schema the single-segment rule applies,
16//! because the owned `app::Model` cannot see embedded models.
17
18use toasty_core::stmt::PathRoot;
19use topcoat::context::Cx;
20
21/// Spec alias — ADR-0001 typed lens. Currently uses `toasty::stmt::Path` directly;
22/// a richer `FieldLens` trait will replace this alias if Toasty exposes the
23/// metadata walk directly (see upstream issue #183).
24pub type FieldLens<M, T> = toasty::stmt::Path<M, T>;
25
26/// The storage column an embedded path flattens to, plus the metadata field
27/// constructors need.
28///
29/// Owned because the schema is borrowed from the request context for the length
30/// of one call and no borrow of it can be held by a `Schema`.
31#[derive(Debug, Clone)]
32pub(crate) struct LeafField {
33    /// The flattened storage column (`application_seo_title`), or the field's
34    /// own name for a single-segment path.
35    pub(crate) name: String,
36    pub(crate) label: String,
37    /// Whether the leaf is nullable. Every leaf the walk resolves reports
38    /// `true`: an embedded leaf is never required by default, because only the
39    /// matching enum variant writes a variant payload's column. That is this
40    /// walk's policy rather than the compiled column's own nullability — the
41    /// flattened column of a required embedded struct is `NOT NULL` — so it is
42    /// the *binding* default, not a storage fact.
43    pub(crate) nullable: bool,
44}
45
46/// The compiled schema for this request, or `None` when no `Db` is in context.
47///
48/// `Db::schema()` is public and gives all three halves the walk needs: `.app`
49/// carries the embedded models the owned `Model::schema()` cannot see,
50/// `.mapping` records which physical column each field resolves to, and `.db`
51/// holds the physical table and column names. So the request path can bind an
52/// embedded lens without opening a connection.
53///
54/// Borrowed, never cloned, and optional: a bare `CxTestBuilder` has no `Db`, and
55/// then the single-segment rule applies, so a schema-less test fails
56/// loudly on a traversal lens rather than silently binding the first segment.
57fn request_schema(cx: &Cx) -> Option<&toasty_core::Schema> {
58    topcoat::context::try_app_context::<toasty::Db>(cx).map(|db| &**db.schema())
59}

61/// Walks a lens path against the app schema, resolving embedded steps.
62///
63/// Kept as a struct rather than free functions so the schema borrow has one
64/// owner and the constructors can share one resolution between the name, the
65/// label, and the uniqueness check.
66pub(crate) struct FieldResolver<'a> {
67    /// The **compiled** schema: its `.app` half resolves the path, its
68    /// `.mapping` half names the column, its `.db` half holds the name.
69    schema: Option<&'a toasty_core::Schema>,
70}
71
72impl<'a> FieldResolver<'a> {
73    /// Build the resolver from the app schema this request carries, if any.
74    pub(crate) fn from_cx(cx: &'a Cx) -> Self {
75        Self {
76            schema: request_schema(cx),
77        }
78    }
79
80    /// Whether this request carries an app schema at all (a `Db` in context).
81    ///
82    /// The leaf constructors fall back to the single-segment rule without one;
83    /// a *value* binding has no fallback — every key comes from the schema — so
84    /// its entry points say so rather than reporting a traversal-lens error.
85    pub(crate) fn has_schema(&self) -> bool {
86        self.schema.is_some()
87    }
88
89    /// Resolve a lens to its leaf field.
90    ///
91    /// With a schema this walks the whole projection, so an embedded step lands
92    /// on the flattened column. Without one it falls back to the single-segment
93    /// rule and panics on a traversal lens — loudly, because silently binding
94    /// the first segment misbinds in release.
95    pub(crate) fn resolve<M, T>(&self, path: FieldLens<M, T>) -> LeafField
96    where
97        M: toasty::schema::Model,
98    {
99        let model = M::schema();
100        let core_path: toasty_core::stmt::Path = path.into();
101        let segments = core_path.projection.as_slice().len();
102        // A variant root is an embedded-enum payload accessor *whatever* its
103        // projection length: the generated accessor rebases onto the variant, so
104        // the steps are variant-local and a single-step projection is the
105        // payload field. Only a model root with one step is a plain field.
106        let is_embedded_path = segments > 1 || matches!(core_path.root, PathRoot::Variant { .. });
107        if is_embedded_path && let Some(schema) = self.schema {
108            return self
109                .walk_embedded(schema, &core_path, &model)
110                .unwrap_or_else(|| {
111                    panic!(
112                        "lens path {projection:?} does not resolve to a single column for {}: \
113                         only embedded steps (embedded structs, enum variant fields, and \
114                         `#[document]` fields) can be bound, not relation hops, and only when \
115                         the model is in the app schema this request carries (GH #185)",
116                        std::any::type_name::<M>(),
117                        projection = core_path.projection.as_slice(),
118                    )
119                });
120        }
121        // No schema: the single-segment rule is all an owned `app::Model` can
122        // answer, so resolve the first step against `ModelRoot.fields` and let
123        // `require_single_segment` reject a traversal lens.
124        require_single_segment(&core_path, "lens");
125        let idx = core_path
126            .projection
127            .as_slice()
128            .first()
129            .copied()
130            .expect("field lens must have a projection");
131        let field = model
132            .as_root_unwrap()
133            .fields
134            .get(idx)
135            .cloned()
136            .unwrap_or_else(|| {
137                panic!(
138                    "field index {idx} out of bounds for {}",
139                    std::any::type_name::<M>()
140                )
141            });
142        LeafField {
143            name: field.name.app_unwrap().to_string(),
144            label: lens_label(&field),
145            nullable: field.nullable(),
146        }
147    }
148
149    /// Walk a lens path to the physical column it names.
150    ///
151    /// Two sources, each authoritative for one thing:
152    ///
153    /// - the **app schema** drives the traversal, because it is the only side that knows a field is
154    ///   a `#[document]` — a *primitive* whose storage is a model, so its inner fields collapse
155    ///   into the one column named after it;
156    /// - the **compiled mapping** names the column. `Db::schema().mapping` records, per model,
157    ///   which column every field resolves to — flattened embedded structs, enum discriminant and
158    ///   payload columns, shared columns, and a document's single column alike — so this reads the
159    ///   name off `db::Table` rather than re-deriving Toasty's naming rules.
160    ///
161    /// Two roots reach here: a **model** root for a plain path (`seo.title`), and
162    /// a **variant** root for an enum payload accessor
163    /// (`media.video().poster().url()`), where the generated accessor rebases
164    /// onto the variant and the projection's steps are variant-local.
165    ///
166    /// Only embedded steps are followed. A relation hop yields `None`: this
167    /// exists for embedded binding, and binding anything else would reintroduce
168    /// the misbind the single-segment rule guards against.
169    fn walk_embedded(
170        &self,
171        schema: &'a toasty_core::Schema,
172        path: &toasty_core::stmt::Path,
173        owner: &toasty::schema::app::Model,
174    ) -> Option<LeafField> {
175        match &path.root {
176            PathRoot::Model(id) => {
177                let root = schema.app.get_model(*id)?.as_root()?;
178                // The accessor's `ModelId` and the schema's may come from
179                // different `models!(..)` expansions, so an id can name a
180                // *different* model here. Verify identity by root model name
181                // before trusting any index: binding another model's column is
182                // the misbind GH #100 exists to prevent.
183                if root.name.upper_camel_case() != owner.as_root()?.name.upper_camel_case() {
184                    return None;
185                }
186                let model = schema.mapping.models.get(&root.id)?;
187                descend(
188                    schema,
189                    &root.fields,
190                    &model.fields,
191                    path.projection.as_slice(),
192                )
193            }
194            // An enum payload: resolve the parent path (it ends at the enum
195            // field), then descend into the variant the accessor selected.
196            PathRoot::Variant { parent, variant_id } => {
197                let root = schema
198                    .app
199                    .get_model(parent.root.as_model_unwrap())?
200                    .as_root()?;
201                let model = schema.mapping.models.get(&root.id)?;
202                let parent_steps = parent.projection.as_slice();
203                // The app side supplies the enum's payload field list, the
204                // mapping side its per-variant column mappings.
205                let app_field = app_field_at(schema, &root.fields, parent_steps)?;
206                let Some(toasty::schema::app::Model::EmbeddedEnum(e)) =
207                    app_embedded(schema, app_field)
208                else {
209                    return None;
210                };
211                let MappingField::Enum(me) = mapping_field_at(&model.fields, parent_steps)? else {
212                    return None;
213                };
214                let variant = me.variants.get(variant_id.index)?;
215                let payloads: Vec<_> = e.variant_fields(variant_id.index).iter().collect();
216                descend(
217                    schema,
218                    &payloads,
219                    &variant.fields,
220                    path.projection.as_slice(),
221                )
222            }
223        }
224    }
225
226    /// Enumerate the bindable surface of the embedded **value** at `path`.
227    /// [`Self::resolve`] answers "which column does this one leaf occupy"; this
228    /// answers "what does this whole value consist of" — every leaf column,
229    /// plus the discriminant for an enum — which is what a value codec and a
230    /// generated form need.
231    ///
232    /// `path` addresses the embedded field itself (`Post::fields().seo()`,
233    /// `Post::fields().publication()`), not one of its leaves: a variant-rooted
234    /// path names a *variant* of a value, not the value, and yields `None`.
235    pub(crate) fn resolve_embedded_value<M, T>(
236        &self,
237        path: FieldLens<M, T>,
238    ) -> Option<EmbeddedValueSpec>
239    where
240        M: toasty::schema::Model,
241    {
242        let schema = self.schema?;
243        let core_path: toasty_core::stmt::Path = path.into();
244        let PathRoot::Model(id) = core_path.root else {
245            return None;
246        };
247        let owner = M::schema();
248        let root = schema.app.get_model(id)?.as_root()?;
249        // The same identity check `walk_embedded` makes: an accessor's
250        // `ModelId` and the schema's can come from different `models!(..)`
251        // expansions, so verify the root by name before trusting any index.
252        if root.name.upper_camel_case() != owner.as_root()?.name.upper_camel_case() {
253            return None;
254        }
255        let mapping = schema.mapping.models.get(&root.id)?;
256        let steps = core_path.projection.as_slice();
257        let app_field = app_field_at(schema, &root.fields, steps)?;
258        let mapping_field = mapping_field_at(&mapping.fields, steps)?;
259        let mut columns = Vec::new();
260        let enum_spec = match (app_embedded(schema, app_field)?, mapping_field) {
261            (toasty::schema::app::Model::EmbeddedStruct(e), MappingField::Struct(ms)) => {
262                collect_columns(schema, &e.fields, &ms.fields, &mut columns)?;
263                None
264            }
265            (toasty::schema::app::Model::EmbeddedEnum(e), MappingField::Enum(me)) => {
266                collect_enum_columns(schema, e, me, &mut columns)?;
267                let discriminant =
268                    column_name(schema, &MappingField::Primitive(me.discriminant.clone()))?;
269                let variants = e
270                    .variants
271                    .iter()
272                    .map(|v| {
273                        discriminant_text(&v.discriminant)
274                            .map(|value| (value, capitalize(&v.name.snake_case())))
275                    })
276                    .collect::<Option<Vec<_>>>()?;
277                Some(crate::schema::embedded::EnumSpec::new(
278                    discriminant,
279                    variants,
280                ))
281            }
282            _ => return None,
283        };
284        Some(EmbeddedValueSpec { columns, enum_spec })
285    }
286}

288/// The column one mapping field occupies, by name.
289fn column_name(schema: &toasty_core::Schema, field: &MappingField) -> Option<String> {
290    column_of(schema, field).map(|leaf| leaf.name)
291}

293/// Collect every leaf column under one embedded level.
294///
295/// A relation inside an embedded value is not bindable and yields `None`: this
296/// walk exists for value binding, the same boundary GH #100 draws for
297/// single lenses.
298fn collect_columns(
299    schema: &toasty_core::Schema,
300    app_fields: &[toasty::schema::app::Field],
301    mapping_fields: &[MappingField],
302    out: &mut Vec<String>,
303) -> Option<()> {
304    for (app_field, mapping_field) in app_fields.iter().zip(mapping_fields) {
305        if is_document(app_field) {
306            // A `#[document]`'s inner fields share its one column, so a *value*
307            // codec has nothing per-field to bind: refuse rather than hand back
308            // one column for several fields (the misbind GH #100 guards
309            // against). Leaf binding still reaches a document's column.
310            return None;
311        }
312        match &app_field.ty {
313            toasty::schema::app::FieldTy::Primitive(_) => {
314                push_column(schema, mapping_field, out)?;
315            }
316            toasty::schema::app::FieldTy::Embedded(embedded) => {
317                match schema.app.get_model(embedded.target)? {
318                    toasty::schema::app::Model::EmbeddedStruct(e) => {
319                        let MappingField::Struct(ms) = mapping_field else {
320                            return None;
321                        };
322                        collect_columns(schema, &e.fields, &ms.fields, out)?;
323                    }
324                    toasty::schema::app::Model::EmbeddedEnum(e) => {
325                        let MappingField::Enum(me) = mapping_field else {
326                            return None;
327                        };
328                        collect_enum_columns(schema, e, me, out)?;
329                    }
330                    toasty::schema::app::Model::Root(_) => return None,
331                }
332            }
333            _ => return None,
334        }
335    }
336    Some(())
337}

339/// Collect every variant's payload columns under one embedded enum.
340fn collect_enum_columns(
341    schema: &toasty_core::Schema,
342    e: &toasty::schema::app::EmbeddedEnum,
343    me: &toasty_core::schema::mapping::FieldEnum,
344    out: &mut Vec<String>,
345) -> Option<()> {
346    // The discriminant column first: it is a form key of this value (the
347    // control that carries the variant), whether the enum is the value itself
348    // or nested inside one.
349    push_column(
350        schema,
351        &MappingField::Primitive(me.discriminant.clone()),
352        out,
353    )?;
354    for (index, variant) in me.variants.iter().enumerate() {
355        collect_columns(schema, e.variant_fields(index), &variant.fields, out)?;
356    }
357    Some(())
358}

360/// Push a column name once: a `#[shared(..)]` column is declared by several
361/// variants and is still one column.
362fn push_column(
363    schema: &toasty_core::Schema,
364    field: &MappingField,
365    out: &mut Vec<String>,
366) -> Option<()> {
367    let name = column_name(schema, field)?;
368    if !out.contains(&name) {
369        out.push(name);
370    }
371    Some(())
372}

374/// The text an embedded enum's discriminant stores.
375///
376/// Toasty's discriminants are integers or strings, and the form carries that
377/// same text — so what a submission posts is what a row stores. Anything else
378/// is reported rather than guessed.
379fn discriminant_text(value: &toasty_core::stmt::Value) -> Option<String> {
380    use toasty_core::stmt::Value;
381    match value {
382        Value::Bool(v) => Some(v.to_string()),
383        Value::I8(v) => Some(v.to_string()),
384        Value::I16(v) => Some(v.to_string()),
385        Value::I32(v) => Some(v.to_string()),
386        Value::I64(v) => Some(v.to_string()),
387        Value::U8(v) => Some(v.to_string()),
388        Value::U16(v) => Some(v.to_string()),
389        Value::U32(v) => Some(v.to_string()),
390        Value::U64(v) => Some(v.to_string()),
391        Value::String(v) => Some(v.clone()),
392        _ => None,
393    }
394}

396/// A `mapping::Field`, aliased so the traversal signatures stay readable.
397type MappingField = toasty_core::schema::mapping::Field;
398
399/// Walk the path down to the column that stores it.
400///
401/// `app_fields` and `mapping_fields` are indexed by the same field index, so
402/// every step reads both: the app side decides *whether to descend*, the mapping
403/// side names *the column*. Neither alone is enough — the mapping cannot say a
404/// field is a `#[document]`, and the app schema cannot say which column a leaf
405/// occupies without re-deriving Toasty's naming.
406fn descend<F>(
407    schema: &toasty_core::Schema,
408    app_fields: &[F],
409    mapping_fields: &[MappingField],
410    steps: &[usize],
411) -> Option<LeafField>
412where
413    F: std::borrow::Borrow<toasty::schema::app::Field>,
414{
415    let (first, rest) = steps.split_first()?;
416    let app_field: &toasty::schema::app::Field = app_fields.get(*first)?.borrow();
417    let mapping_field = mapping_fields.get(*first)?;
418    // A `#[document]`'s inner fields share its one column, so reaching the
419    // document is reaching the leaf, however many steps remain.
420    if rest.is_empty() || is_document(app_field) {
421        return column_of(schema, mapping_field);
422    }
423    let toasty::schema::app::FieldTy::Embedded(embedded) = &app_field.ty else {
424        // A relation hop, or a primitive with steps left over: not an embedded
425        // leaf either way.
426        return None;
427    };
428    match schema.app.get_model(embedded.target)? {
429        toasty::schema::app::Model::EmbeddedStruct(e) => {
430            let MappingField::Struct(ms) = mapping_field else {
431                return None;
432            };
433            descend(schema, &e.fields, &ms.fields, rest)
434        }
435        toasty::schema::app::Model::EmbeddedEnum(e) => {
436            let MappingField::Enum(me) = mapping_field else {
437                return None;
438            };
439            // A variant consumes two steps: the variant index, then the field.
440            let (variant_index, tail) = rest.split_first()?;
441            let variant = me.variants.get(*variant_index)?;
442            let payloads: Vec<_> = e.variant_fields(*variant_index).iter().collect();
443            descend(schema, &payloads, &variant.fields, tail)
444        }
445        _ => None,
446    }
447}

449/// The column a single mapping field occupies, if it is a leaf.
450fn column_of(schema: &toasty_core::Schema, field: &MappingField) -> Option<LeafField> {
451    let MappingField::Primitive(p) = field else {
452        // A path stopping on a struct or enum names no single column, and a
453        // relation stores none.
454        return None;
455    };
456    let column = &schema.db.tables[p.column.table.0].columns[p.column.index];
457    Some(LeafField {
458        name: column.name.clone(),
459        label: capitalize(&column.name.replace('_', " ")),
460        // Reported nullable by policy, not read off the column: an embedded
461        // leaf is never required by default, because only the matching enum
462        // variant writes a variant payload's column. The compiled column can be
463        // `NOT NULL` — an embedded struct's flattened column is.
464        nullable: true,
465    })
466}

468/// Whether `field` is a `#[document]`: a *primitive* whose storage is a model,
469/// so its inner fields collapse into the one column named after it.
470fn is_document(field: &toasty::schema::app::Field) -> bool {
471    matches!(
472        &field.ty,
473        toasty::schema::app::FieldTy::Primitive(p)
474            if matches!(p.ty, toasty_core::stmt::Type::Model(_))
475    )
476}

478/// The bindable surface of one embedded **value**: every column its
479/// fields occupy, plus — for an enum — the discriminant column and each
480/// variant's stored value.
481///
482/// [`FieldResolver::resolve`] answers "which column does this one leaf occupy";
483/// this answers "what does this whole value consist of", which is what a value
484/// codec and a generated form need. Both read the same two authorities: the app
485/// schema decides the structure, the compiled mapping names every column.
486#[derive(Debug, Clone)]
487pub(crate) struct EmbeddedValueSpec {
488    /// Every form key the value occupies, in declaration order and
489    /// deduplicated: each leaf column, and the discriminant column of every
490    /// enum the value contains — the top-level one and any nested inside a
491    /// struct or a variant. A `#[shared(..)]` column declared by several
492    /// variants appears once, because it *is* one column.
493    pub(crate) columns: Vec<String>,
494    /// `Some` for an embedded enum: its discriminant column and its variants
495    /// (each one's stored value and the name it is labelled with). The one
496    /// `EnumSpec` type is shared with the public seam.
497    pub(crate) enum_spec: Option<crate::schema::embedded::EnumSpec>,
498}

500/// The embedded model an app field targets, if it is embedded.
501fn app_embedded<'a>(
502    schema: &'a toasty_core::Schema,
503    field: &toasty::schema::app::Field,
504) -> Option<&'a toasty::schema::app::Model> {
505    let toasty::schema::app::FieldTy::Embedded(embedded) = &field.ty else {
506        return None;
507    };
508    schema.app.get_model(embedded.target)
509}

511/// The app-level field `steps` reaches, used by the variant root to find the
512/// enum's payload list before descending.
513///
514/// Recurses through embedded structs: the variant root's parent path can walk
515/// through them (an embedded struct holding the enum), not just one step.
516fn app_field_at<'a>(
517    schema: &'a toasty_core::Schema,
518    fields: &'a [toasty::schema::app::Field],
519    steps: &[usize],
520) -> Option<&'a toasty::schema::app::Field> {
521    let (first, rest) = steps.split_first()?;
522    let field = fields.get(*first)?;
523    if rest.is_empty() {
524        return Some(field);
525    }
526    match app_embedded(schema, field)? {
527        toasty::schema::app::Model::EmbeddedStruct(e) => app_field_at(schema, &e.fields, rest),
528        // An enum nested in the parent path consumes two steps per level: the
529        // variant index, then a variant-local field index. `variant_fields`
530        // returns the variant's own slice of the enum's global field list
531        // (upstream 7ff180db), so the second step indexes it directly — the
532        // generated accessor's index is variant-local, which is the same field
533        // either way.
534        //
535        // The variant step is bounds-checked here first: `variant_fields`
536        // indexes `variants[i]` and panics out of range, and this walk answers
537        // `None` for a path it cannot resolve (a wrong lens must not abort a
538        // request).
539        toasty::schema::app::Model::EmbeddedEnum(e) => {
540            let (variant, tail) = rest.split_first()?;
541            e.variants.get(*variant)?;
542            let field = e.variant_fields(*variant).get(*tail.first()?)?;
543            if tail.len() == 1 {
544                Some(field)
545            } else {
546                app_field_at(schema, std::slice::from_ref(field), &tail[1..])
547            }
548        }
549        toasty::schema::app::Model::Root(_) => None,
550    }
551}

553/// The mapping field `steps` reaches, used by the variant root to find the
554/// enum's per-variant mappings before descending.
555fn mapping_field_at<'a>(fields: &'a [MappingField], steps: &[usize]) -> Option<&'a MappingField> {
556    let (first, rest) = steps.split_first()?;
557    let field = fields.get(*first)?;
558    if rest.is_empty() {
559        return Some(field);
560    }
561    match field {
562        MappingField::Struct(s) => mapping_field_at(&s.fields, rest),
563        MappingField::Enum(e) => {
564            let (variant, tail) = rest.split_first()?;
565            mapping_field_at(&e.variants.get(*variant)?.fields, tail)
566        }
567        _ => None,
568    }
569}

571/// Resolve a typed lens to the app-level [`Field`] behind it.
572///
573/// The single walk over `Path → toasty_core::stmt::Path → projection`, reading
574/// off the model's built schema (upstream issues #114/#183). Callers read
575/// name, label, nullability, storage name, `FieldTy`, `auto`, and `constraints`
576/// off the result; uniqueness is the one property a `Field` cannot answer, so
577/// it has [`lens_field_unique`] of its own.
578///
579/// `model` is the owning `Model::schema()`, passed in so one form-input
580/// construction resolves the schema once (never per row) and shares it with
581/// [`lens_field_unique`]. The returned `Field` is owned because
582/// `Model::schema()` builds the model by value with no cache, so no borrow of
583/// it can escape — but only the one field is cloned.
584///
585/// Traversal lenses are rejected: a multi-step path has no single
586/// field name, and silently binding its first segment misbinds in release.
587/// Use [`FieldResolver`] to bind an embedded path instead.
588pub(crate) fn lens_field<M, T>(
589    path: FieldLens<M, T>,
590    model: &toasty::schema::app::Model,
591) -> toasty::schema::app::Field
592where
593    M: toasty::schema::Model,
594{
595    let core_path: toasty_core::stmt::Path = path.into();
596    require_single_segment(&core_path, "lens");
597    let idx = core_path
598        .projection
599        .as_slice()
600        .first()
601        .copied()
602        .expect("field lens must have a projection");
603    model
604        .as_root_unwrap()
605        .fields
606        .get(idx)
607        .cloned()
608        .unwrap_or_else(|| {
609            panic!(
610                "field index {idx} out of bounds for {}",
611                std::any::type_name::<M>()
612            )
613        })
614}
615
616/// The capitalized label for a field's app-level name.
617pub(crate) fn lens_label(field: &toasty::schema::app::Field) -> String {
618    capitalize(field.name.app_unwrap())
619}
620
621/// Whether `field` is backed by a unique index it participates in.
622///
623/// Uniqueness is not a property of a `Field`: Toasty stores it on the model's
624/// index list, so this needs the owning `ModelRoot` too. The primary key is
625/// excluded — a key column is unique by construction, not by a declared
626/// constraint.
627///
628/// A **composite** unique index counts too: `#[unique(tenant_id, email)]` is how
629/// a tenant-scoped resource expresses "unique within the tenant", and the
630/// app-side pre-check has to recognise it or the field's `unique()` declaration
631/// is silently dead. Recognizing the index is not checking it exactly: the
632/// pre-check probes inside the tenant-scoped query's scope, so it enforces the
633/// constraint only when that scope matches the index's remaining components.
634/// An index with components outside the scope stays a gap, and it is not
635/// checkable here because a query's filters are not introspectable; see
636/// `Resource::query`'s note and upstream #117.
637///
638/// This reports declared schema uniqueness, not a global guarantee: SQL permits
639/// multiple `NULL`s in a unique index, and enum-variant columns are
640/// storage-nullable, so a nullable unique field can still repeat.
641pub(crate) fn lens_field_unique(
642    field: &toasty::schema::app::Field,
643    model: &toasty::schema::app::ModelRoot,
644) -> bool {
645    model.indices.iter().any(|index| {
646        index.unique && !index.primary_key && index.fields.iter().any(|f| f.field == field.id)
647    })
648}
649
650/// Panic unless a lens path addresses exactly one field.
651///
652/// A traversal lens (relation hops, embedded steps) has no single field name,
653/// nullability, or uniqueness — silently binding its first segment misbinds in
654/// release, so every lens helper rejects multi-segment paths loudly instead.
655pub(crate) fn require_single_segment(path: &toasty_core::stmt::Path, what: &str) {
656    assert_eq!(
657        path.projection.as_slice().len(),
658        1,
659        "{what} requires a single-field lens, got a {}-segment traversal path (GH #100)",
660        path.projection.as_slice().len()
661    );
662}
663
664/// A field's human label from its storage name.
665///
666/// Sentence case, with the underscores a Rust column name carries read as
667/// spaces: `word_count` is "Word count", not "Word_count". A label is the one
668/// place a column name becomes prose, so it should not leak the identifier —
669/// which is what `Media_poster_url` and `Seo_title` did everywhere an embedded
670/// or document leaf rendered its own name (#192). An explicit
671/// [`.label(..)`](crate::schema::TextInput::label) still wins.
672pub(crate) fn capitalize(s: &str) -> String {
673    let spaced = s.replace('_', " ");
674    let mut c = spaced.chars();
675    match c.next() {
676        None => String::new(),
677        Some(f) => f.to_uppercase().collect::<String>() + c.as_str(),
678    }
679}

866    ///
867    /// Two Rust model types never share a `ModelId` in one process, so the one
868    /// way an id can name a foreign model is an assembled set — which is what
869    /// the guard exists for: it trusts the root's **name**, never the id.
870    fn models_with_a_foreign_root() -> toasty::schema::ModelSet {
871        let mut set = toasty::schema::ModelSet::new();
872        let mut model = Impostor::schema();
873        let toasty_core::schema::app::Model::Root(root) = &mut model else {
874            panic!("a #[derive(Model)] type builds a root model");
875        };
876        let id = <LensPost as toasty::schema::Model>::id();
877        root.id = id;
878        // A field's own `FieldId` names its model too, so the forgery has to
879        // carry through or the root would be inconsistent with its fields.
880        for field in &mut root.fields {
881            field.id.model = id;
882        }
883        set.add(model);
884        // The forged root embeds both, so their models must be in the set too.
885        <Seo as toasty::schema::Field>::register(&mut set);
886        <Wrapper as toasty::schema::Field>::register(&mut set);
887        set
888    }
