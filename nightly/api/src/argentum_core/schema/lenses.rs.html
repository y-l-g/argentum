1//! Field lenses — typed Toasty paths and their app-level metadata.
2//!
3//! Base bridge layer (with `pk`): these two modules are the only schema
4//! modules that name `toasty_core` (upstream #114/#183), so upstream
5//! churn has one blast radius. Everything above reads lenses through
6//! `FieldLens` and the helpers here.
7//!
8//! [`lens_field`] hands callers the built `app::Field`, so name, label,
9//! nullability, storage name, `FieldTy`, `auto`, and `constraints` all come
10//! from one walk. [`lens_field_unique`] covers the one property `Field` does
11//! not carry, because Toasty keeps uniqueness on the model's index list.
12//!
13//! [`FieldResolver`] is the schema-aware walk (GH #185): with the app schema
14//! in hand an embedded path resolves to its **flattened storage column**, so
15//! `TextInput::r#for_post(...)`-style bindings can reach a field inside an
16//! embedded struct or a `#[document]`. Without a schema the single-segment
17//! rule (GH #100) still applies — the owned `app::Model` cannot see embedded
18//! models, which is exactly what the schema adds.
19//!
20//! The walk is covered directly by the test module at the foot of this file,
21//! over a fixture built for the shapes it resolves: a leaf inside an embedded
22//! struct, a variant-rooted path through nested structs and through a struct
23//! that holds the enum, a `#[shared]` column (one column for every variant that
24//! declares it), and a `#[document]`. A plain leaf is covered too, but it never
25//! enters the walk — a single segment on a model root reads the owned app field.
26//! The identity guard is covered on both entries, and `resolve_embedded_value`
27//! for both the value shapes (struct, enum, struct holding an enum) and the
28//! paths it refuses (variant root, leaf, document, foreign root, no schema).
29
30use toasty_core::stmt::PathRoot;
31use topcoat::context::Cx;

33/// Spec alias — ADR-0001 typed lens. Currently uses `toasty::stmt::Path` directly;
34/// a richer `FieldLens` trait will replace this alias if Toasty exposes the
35/// metadata walk directly (see GH #11, upstream issue #183).
36pub type FieldLens<M, T> = toasty::stmt::Path<M, T>;
37
38/// The storage column an embedded path flattens to, plus the metadata field
39/// constructors need.
40///
41/// Owned because the schema is borrowed from the request context for the length
42/// of one call and no borrow of it can be held by a `Schema`.
43#[derive(Debug, Clone)]
44pub(crate) struct LeafField {
45    /// The flattened storage column (`application_seo_title`), or the field's
46    /// own name for a single-segment path.
47    pub(crate) name: String,
48    pub(crate) label: String,
49    /// Whether the leaf is nullable. Every leaf the walk resolves reports
50    /// `true`: an embedded leaf is never required by default, because only the
51    /// matching enum variant writes a variant payload's column. That is this
52    /// walk's policy rather than the compiled column's own nullability — the
53    /// flattened column of a required embedded struct is `NOT NULL` — so it is
54    /// the *binding* default, not a storage fact.
55    pub(crate) nullable: bool,
56}
57
58/// The compiled schema for this request, or `None` when no `Db` is in context.
59///
60/// `Db::schema()` is public and gives all three halves the walk needs: `.app`
61/// carries the embedded models the owned `Model::schema()` cannot see,
62/// `.mapping` records which physical column each field resolves to, and `.db`
63/// holds the physical table and column names. So the request path can bind an
64/// embedded lens without opening a connection.
65///
66/// Borrowed, never cloned, and optional: a bare `CxTestBuilder` has no `Db`, and
67/// then the single-segment rule (GH #100) applies, so a schema-less test fails
68/// loudly on a traversal lens rather than silently binding the first segment.
69fn request_schema(cx: &Cx) -> Option<&toasty_core::Schema> {
70    topcoat::context::try_app_context::<toasty::Db>(cx).map(|db| &**db.schema())
71}

73/// Walks a lens path against the app schema, resolving embedded steps.
74///
75/// Kept as a struct rather than free functions so the schema borrow has one
76/// owner and the constructors can share one resolution between the name, the
77/// label, and the uniqueness check.
78pub(crate) struct FieldResolver<'a> {
79    /// The **compiled** schema: its `.app` half resolves the path, its
80    /// `.mapping` half names the column, its `.db` half holds the name.
81    schema: Option<&'a toasty_core::Schema>,
82}
83
84impl<'a> FieldResolver<'a> {
85    /// Build the resolver from the app schema this request carries, if any.
86    pub(crate) fn from_cx(cx: &'a Cx) -> Self {
87        Self {
88            schema: request_schema(cx),
89        }
90    }
91
92    /// Whether this request carries an app schema at all (a `Db` in context).
93    ///
94    /// The leaf constructors fall back to the single-segment rule without one;
95    /// a *value* binding has no fallback — every key comes from the schema — so
96    /// its entry points say so rather than reporting a traversal-lens error
97    /// (GH #191).
98    pub(crate) fn has_schema(&self) -> bool {
99        self.schema.is_some()
100    }
101
102    /// Resolve a lens to its leaf field.
103    ///
104    /// With a schema this walks the whole projection, so an embedded step lands
105    /// on the flattened column. Without one it falls back to the single-segment
106    /// rule and panics on a traversal lens — loudly, because silently binding
107    /// the first segment misbinds in release (GH #100).
108    pub(crate) fn resolve<M, T>(&self, path: FieldLens<M, T>) -> LeafField
109    where
110        M: toasty::schema::Model,
111    {
112        let model = M::schema();
113        let core_path: toasty_core::stmt::Path = path.into();
114        let segments = core_path.projection.as_slice().len();
115        // A variant root is an embedded-enum payload accessor *whatever* its
116        // projection length: the generated accessor rebases onto the variant, so
117        // the steps are variant-local and a single-step projection is the
118        // payload field. Only a model root with one step is a plain field.
119        let is_embedded_path = segments > 1 || matches!(core_path.root, PathRoot::Variant { .. });
120        if is_embedded_path && let Some(schema) = self.schema {
121            return self
122                .walk_embedded(schema, &core_path, &model)
123                .unwrap_or_else(|| {
124                    panic!(
125                        "lens path {projection:?} does not resolve to a single column for {}: \
126                         only embedded steps (embedded structs, enum variant fields, and \
127                         `#[document]` fields) can be bound, not relation hops, and only when \
128                         the model is in the app schema this request carries (GH #185)",
129                        std::any::type_name::<M>(),
130                        projection = core_path.projection.as_slice(),
131                    )
132                });
133        }
134        // No schema: the single-segment rule is all an owned `app::Model` can
135        // answer, so resolve the first step against `ModelRoot.fields` and let
136        // `require_single_segment` reject a traversal lens (GH #100).
137        require_single_segment(&core_path, "lens");
138        let idx = core_path
139            .projection
140            .as_slice()
141            .first()
142            .copied()
143            .expect("field lens must have a projection");
144        let field = model
145            .as_root_unwrap()
146            .fields
147            .get(idx)
148            .cloned()
149            .unwrap_or_else(|| {
150                panic!(
151                    "field index {idx} out of bounds for {}",
152                    std::any::type_name::<M>()
153                )
154            });
155        LeafField {
156            name: field.name.app_unwrap().to_string(),
157            label: lens_label(&field),
158            nullable: field.nullable(),
159        }
160    }
161
162    /// Walk a lens path to the physical column it names.
163    ///
164    /// Two sources, each authoritative for one thing:
165    ///
166    /// - the **app schema** drives the traversal, because it is the only side that knows a field is
167    ///   a `#[document]` — a *primitive* whose storage is a model, so its inner fields collapse
168    ///   into the one column named after it, and a path through one has steps left over on arrival;
169    /// - the **compiled mapping** names the column. `Db::schema().mapping` records, per model,
170    ///   which column every field resolves to — flattened embedded structs, enum discriminant and
171    ///   payload columns, shared columns, and a document's single column alike — so this reads the
172    ///   name off `db::Table` rather than re-deriving Toasty's naming rules. An earlier revision
173    ///   accumulated names from `app::Field.name` and had to encode those rules itself.
174    ///
175    /// Probed against the pinned rev the two agree for every reachable shape,
176    /// including the cases that are easy to get wrong: `#[shared(..)]` payloads
177    /// collapse onto one column (`publication_timestamp`), and a `#[document]`
178    /// is a single `Primitive` column named after the field (`stats`), not one
179    /// column per inner field.
180    ///
181    /// Two roots reach here:
182    ///
183    /// - a **model** root, for a plain path (`seo.title`);
184    /// - a **variant** root, for an enum payload accessor (`media.video().poster().url()`): the
185    ///   generated accessor rebases onto the variant, so the root carries the parent path *to the
186    ///   enum field* while the projection's steps are **variant-local**.
187    ///
188    /// Only embedded steps are followed. A relation hop yields `None`: this
189    /// exists for embedded binding (GH #185), and binding anything else here
190    /// would reintroduce the misbind GH #100 guards against.
191    fn walk_embedded(
192        &self,
193        schema: &'a toasty_core::Schema,
194        path: &toasty_core::stmt::Path,
195        owner: &toasty::schema::app::Model,
196    ) -> Option<LeafField> {
197        match &path.root {
198            PathRoot::Model(id) => {
199                let root = schema.app.get_model(*id)?.as_root()?;
200                // The accessor's `ModelId` and the schema's may come from
201                // different `models!(..)` expansions, so an id can name a
202                // *different* model here. Verify identity by root model name
203                // before trusting any index: binding another model's column is
204                // the misbind GH #100 exists to prevent.
205                if root.name.upper_camel_case() != owner.as_root()?.name.upper_camel_case() {
206                    return None;
207                }
208                let model = schema.mapping.models.get(&root.id)?;
209                descend(
210                    schema,
211                    &root.fields,
212                    &model.fields,
213                    path.projection.as_slice(),
214                )
215            }
216            // An enum payload: resolve the parent path (it ends at the enum
217            // field), then descend into the variant the accessor selected.
218            PathRoot::Variant { parent, variant_id } => {
219                let root = schema
220                    .app
221                    .get_model(parent.root.as_model_unwrap())?
222                    .as_root()?;
223                let model = schema.mapping.models.get(&root.id)?;
224                let parent_steps = parent.projection.as_slice();
225                // The app side supplies the enum's payload field list, the
226                // mapping side its per-variant column mappings.
227                let app_field = app_field_at(schema, &root.fields, parent_steps)?;
228                let Some(toasty::schema::app::Model::EmbeddedEnum(e)) =
229                    app_embedded(schema, app_field)
230                else {
231                    return None;
232                };
233                let MappingField::Enum(me) = mapping_field_at(&model.fields, parent_steps)? else {
234                    return None;
235                };
236                let variant = me.variants.get(variant_id.index)?;
237                let payloads: Vec<_> = e.variant_fields(variant_id.index).iter().collect();
238                descend(
239                    schema,
240                    &payloads,
241                    &variant.fields,
242                    path.projection.as_slice(),
243                )
244            }
245        }
246    }
247
248    /// Enumerate the bindable surface of the embedded **value** at `path`
249    /// (GH #191).
250    ///
251    /// [`Self::resolve`] answers "which column does this one leaf occupy"; this
252    /// answers "what does this whole value consist of" — every leaf column,
253    /// plus the discriminant for an enum — which is what a value codec and a
254    /// generated form need.
255    ///
256    /// `path` addresses the embedded field itself (`Post::fields().seo()`,
257    /// `Post::fields().publication()`), not one of its leaves: a variant-rooted
258    /// path names a *variant* of a value, not the value, and yields `None`.
259    pub(crate) fn resolve_embedded_value<M, T>(
260        &self,
261        path: FieldLens<M, T>,
262    ) -> Option<EmbeddedValueSpec>
263    where
264        M: toasty::schema::Model,
265    {
266        let schema = self.schema?;
267        let core_path: toasty_core::stmt::Path = path.into();
268        let PathRoot::Model(id) = core_path.root else {
269            return None;
270        };
271        let owner = M::schema();
272        let root = schema.app.get_model(id)?.as_root()?;
273        // The same identity check `walk_embedded` makes: an accessor's
274        // `ModelId` and the schema's can come from different `models!(..)`
275        // expansions, so verify the root by name before trusting any index.
276        if root.name.upper_camel_case() != owner.as_root()?.name.upper_camel_case() {
277            return None;
278        }
279        let mapping = schema.mapping.models.get(&root.id)?;
280        let steps = core_path.projection.as_slice();
281        let app_field = app_field_at(schema, &root.fields, steps)?;
282        let mapping_field = mapping_field_at(&mapping.fields, steps)?;
283        let mut columns = Vec::new();
284        let enum_spec = match (app_embedded(schema, app_field)?, mapping_field) {
285            (toasty::schema::app::Model::EmbeddedStruct(e), MappingField::Struct(ms)) => {
286                collect_columns(schema, &e.fields, &ms.fields, &mut columns)?;
287                None
288            }
289            (toasty::schema::app::Model::EmbeddedEnum(e), MappingField::Enum(me)) => {
290                collect_enum_columns(schema, e, me, &mut columns)?;
291                let discriminant =
292                    column_name(schema, &MappingField::Primitive(me.discriminant.clone()))?;
293                let variants = e
294                    .variants
295                    .iter()
296                    .map(|v| {
297                        discriminant_text(&v.discriminant)
298                            .map(|value| (value, capitalize(&v.name.snake_case())))
299                    })
300                    .collect::<Option<Vec<_>>>()?;
301                Some(crate::schema::embedded::EnumSpec::new(
302                    discriminant,
303                    variants,
304                ))
305            }
306            _ => return None,
307        };
308        Some(EmbeddedValueSpec { columns, enum_spec })
309    }
310}

312/// The column one mapping field occupies, by name.
313fn column_name(schema: &toasty_core::Schema, field: &MappingField) -> Option<String> {
314    column_of(schema, field).map(|leaf| leaf.name)
315}

317/// Collect every leaf column under one embedded level.
318///
319/// A relation inside an embedded value is not bindable and yields `None`: this
320/// walk exists for value binding (GH #191), the same boundary GH #100 draws for
321/// single lenses.
322fn collect_columns(
323    schema: &toasty_core::Schema,
324    app_fields: &[toasty::schema::app::Field],
325    mapping_fields: &[MappingField],
326    out: &mut Vec<String>,
327) -> Option<()> {
328    for (app_field, mapping_field) in app_fields.iter().zip(mapping_fields) {
329        if is_document(app_field) {
330            // A `#[document]`'s inner fields share its one column, so a *value*
331            // codec has nothing per-field to bind: refuse rather than hand back
332            // one column for several fields (the misbind GH #100 guards
333            // against). Leaf binding still reaches a document's column (GH #185).
334            return None;
335        }
336        match &app_field.ty {
337            toasty::schema::app::FieldTy::Primitive(_) => {
338                push_column(schema, mapping_field, out)?;
339            }
340            toasty::schema::app::FieldTy::Embedded(embedded) => {
341                match schema.app.get_model(embedded.target)? {
342                    toasty::schema::app::Model::EmbeddedStruct(e) => {
343                        let MappingField::Struct(ms) = mapping_field else {
344                            return None;
345                        };
346                        collect_columns(schema, &e.fields, &ms.fields, out)?;
347                    }
348                    toasty::schema::app::Model::EmbeddedEnum(e) => {
349                        let MappingField::Enum(me) = mapping_field else {
350                            return None;
351                        };
352                        collect_enum_columns(schema, e, me, out)?;
353                    }
354                    toasty::schema::app::Model::Root(_) => return None,
355                }
356            }
357            _ => return None,
358        }
359    }
360    Some(())
361}

363/// Collect every variant's payload columns under one embedded enum.
364fn collect_enum_columns(
365    schema: &toasty_core::Schema,
366    e: &toasty::schema::app::EmbeddedEnum,
367    me: &toasty_core::schema::mapping::FieldEnum,
368    out: &mut Vec<String>,
369) -> Option<()> {
370    // The discriminant column first: it is a form key of this value (the
371    // control that carries the variant), whether the enum is the value itself
372    // or nested inside one.
373    push_column(
374        schema,
375        &MappingField::Primitive(me.discriminant.clone()),
376        out,
377    )?;
378    for (index, variant) in me.variants.iter().enumerate() {
379        collect_columns(schema, e.variant_fields(index), &variant.fields, out)?;
380    }
381    Some(())
382}

384/// Push a column name once: a `#[shared(..)]` column is declared by several
385/// variants and is still one column.
386fn push_column(
387    schema: &toasty_core::Schema,
388    field: &MappingField,
389    out: &mut Vec<String>,
390) -> Option<()> {
391    let name = column_name(schema, field)?;
392    if !out.contains(&name) {
393        out.push(name);
394    }
395    Some(())
396}

398/// The text an embedded enum's discriminant stores (GH #191).
399///
400/// Toasty's discriminants are integers or strings, and the form carries that
401/// same text — so what a submission posts is what a row stores. Anything else
402/// is reported rather than guessed.
403fn discriminant_text(value: &toasty_core::stmt::Value) -> Option<String> {
404    use toasty_core::stmt::Value;
405    match value {
406        Value::Bool(v) => Some(v.to_string()),
407        Value::I8(v) => Some(v.to_string()),
408        Value::I16(v) => Some(v.to_string()),
409        Value::I32(v) => Some(v.to_string()),
410        Value::I64(v) => Some(v.to_string()),
411        Value::U8(v) => Some(v.to_string()),
412        Value::U16(v) => Some(v.to_string()),
413        Value::U32(v) => Some(v.to_string()),
414        Value::U64(v) => Some(v.to_string()),
415        Value::String(v) => Some(v.clone()),
416        _ => None,
417    }
418}

420/// A `mapping::Field`, aliased so the traversal signatures stay readable.
421type MappingField = toasty_core::schema::mapping::Field;
422
423/// Walk the path down to the column that stores it.
424///
425/// `app_fields` and `mapping_fields` are indexed by the same field index, so
426/// every step reads both: the app side decides *whether to descend*, the mapping
427/// side names *the column*. Neither alone is enough — the mapping cannot say a
428/// field is a `#[document]`, and the app schema cannot say which column a leaf
429/// occupies without re-deriving Toasty's naming.
430fn descend<F>(
431    schema: &toasty_core::Schema,
432    app_fields: &[F],
433    mapping_fields: &[MappingField],
434    steps: &[usize],
435) -> Option<LeafField>
436where
437    F: std::borrow::Borrow<toasty::schema::app::Field>,
438{
439    let (first, rest) = steps.split_first()?;
440    let app_field: &toasty::schema::app::Field = app_fields.get(*first)?.borrow();
441    let mapping_field = mapping_fields.get(*first)?;
442    // A `#[document]`'s inner fields share its one column, so reaching the
443    // document is reaching the leaf, however many steps remain.
444    if rest.is_empty() || is_document(app_field) {
445        return column_of(schema, mapping_field);
446    }
447    let toasty::schema::app::FieldTy::Embedded(embedded) = &app_field.ty else {
448        // A relation hop, or a primitive with steps left over: not an embedded
449        // leaf either way.
450        return None;
451    };
452    match schema.app.get_model(embedded.target)? {
453        toasty::schema::app::Model::EmbeddedStruct(e) => {
454            let MappingField::Struct(ms) = mapping_field else {
455                return None;
456            };
457            descend(schema, &e.fields, &ms.fields, rest)
458        }
459        toasty::schema::app::Model::EmbeddedEnum(e) => {
460            let MappingField::Enum(me) = mapping_field else {
461                return None;
462            };
463            // A variant consumes two steps: the variant index, then the field.
464            let (variant_index, tail) = rest.split_first()?;
465            let variant = me.variants.get(*variant_index)?;
466            let payloads: Vec<_> = e.variant_fields(*variant_index).iter().collect();
467            descend(schema, &payloads, &variant.fields, tail)
468        }
469        _ => None,
470    }
471}

473/// The column a single mapping field occupies, if it is a leaf.
474fn column_of(schema: &toasty_core::Schema, field: &MappingField) -> Option<LeafField> {
475    let MappingField::Primitive(p) = field else {
476        // A path stopping on a struct or enum names no single column, and a
477        // relation stores none.
478        return None;
479    };
480    let column = &schema.db.tables[p.column.table.0].columns[p.column.index];
481    Some(LeafField {
482        name: column.name.clone(),
483        label: capitalize(&column.name.replace('_', " ")),
484        // Reported nullable by policy, not read off the column: an embedded
485        // leaf is never required by default, because only the matching enum
486        // variant writes a variant payload's column. The compiled column can be
487        // `NOT NULL` — an embedded struct's flattened column is.
488        nullable: true,
489    })
490}

492/// Whether `field` is a `#[document]`: a *primitive* whose storage is a model,
493/// so its inner fields collapse into the one column named after it.
494fn is_document(field: &toasty::schema::app::Field) -> bool {
495    matches!(
496        &field.ty,
497        toasty::schema::app::FieldTy::Primitive(p)
498            if matches!(p.ty, toasty_core::stmt::Type::Model(_))
499    )
500}

502/// The bindable surface of one embedded **value** (GH #191): every column its
503/// fields occupy, plus — for an enum — the discriminant column and each
504/// variant's stored value.
505///
506/// [`FieldResolver::resolve`] answers "which column does this one leaf occupy";
507/// this answers "what does this whole value consist of", which is what a value
508/// codec and a generated form need. Both read the same two authorities: the app
509/// schema decides the structure, the compiled mapping names every column.
510#[derive(Debug, Clone)]
511pub(crate) struct EmbeddedValueSpec {
512    /// Every form key the value occupies, in declaration order and
513    /// deduplicated: each leaf column, and the discriminant column of every
514    /// enum the value contains — the top-level one and any nested inside a
515    /// struct or a variant. A `#[shared(..)]` column declared by several
516    /// variants appears once, because it *is* one column.
517    pub(crate) columns: Vec<String>,
518    /// `Some` for an embedded enum: its discriminant column and its variants
519    /// (each one's stored value and the name it is labelled with). The one
520    /// `EnumSpec` type is shared with the public seam.
521    pub(crate) enum_spec: Option<crate::schema::embedded::EnumSpec>,
522}

524/// The embedded model an app field targets, if it is embedded.
525fn app_embedded<'a>(
526    schema: &'a toasty_core::Schema,
527    field: &toasty::schema::app::Field,
528) -> Option<&'a toasty::schema::app::Model> {
529    let toasty::schema::app::FieldTy::Embedded(embedded) = &field.ty else {
530        return None;
531    };
532    schema.app.get_model(embedded.target)
533}

535/// The app-level field `steps` reaches, used by the variant root to find the
536/// enum's payload list before descending.
537///
538/// Recurses through embedded structs: the variant root's parent path can walk
539/// through them (an embedded struct holding the enum), not just one step.
540fn app_field_at<'a>(
541    schema: &'a toasty_core::Schema,
542    fields: &'a [toasty::schema::app::Field],
543    steps: &[usize],
544) -> Option<&'a toasty::schema::app::Field> {
545    let (first, rest) = steps.split_first()?;
546    let field = fields.get(*first)?;
547    if rest.is_empty() {
548        return Some(field);
549    }
550    match app_embedded(schema, field)? {
551        toasty::schema::app::Model::EmbeddedStruct(e) => app_field_at(schema, &e.fields, rest),
552        // An enum nested in the parent path consumes two steps per level: the
553        // variant index, then a variant-local field index. `variant_fields`
554        // returns the variant's own slice of the enum's global field list
555        // (upstream 7ff180db changed it from an iterator to that slice), so the
556        // second step indexes it directly — the generated accessor's index is
557        // variant-local, which is the same field either way.
558        //
559        // The variant step is bounds-checked here first: `variant_fields`
560        // indexes `variants[i]` and panics out of range where the old iterator
561        // returned `None`, and this walk answers `None` for a path it cannot
562        // resolve (a wrong lens must not abort a request).
563        toasty::schema::app::Model::EmbeddedEnum(e) => {
564            let (variant, tail) = rest.split_first()?;
565            e.variants.get(*variant)?;
566            let field = e.variant_fields(*variant).get(*tail.first()?)?;
567            if tail.len() == 1 {
568                Some(field)
569            } else {
570                app_field_at(schema, std::slice::from_ref(field), &tail[1..])
571            }
572        }
573        toasty::schema::app::Model::Root(_) => None,
574    }
575}

577/// The mapping field `steps` reaches, used by the variant root to find the
578/// enum's per-variant mappings before descending.
579fn mapping_field_at<'a>(fields: &'a [MappingField], steps: &[usize]) -> Option<&'a MappingField> {
580    let (first, rest) = steps.split_first()?;
581    let field = fields.get(*first)?;
582    if rest.is_empty() {
583        return Some(field);
584    }
585    match field {
586        MappingField::Struct(s) => mapping_field_at(&s.fields, rest),
587        MappingField::Enum(e) => {
588            let (variant, tail) = rest.split_first()?;
589            mapping_field_at(&e.variants.get(*variant)?.fields, tail)
590        }
591        _ => None,
592    }
593}

595/// Resolve a typed lens to the app-level [`Field`] behind it.
596///
597/// The single walk over `Path → toasty_core::stmt::Path → projection`, reading
598/// off the model's built schema (upstream issues #114/#183). Callers read
599/// name, label, nullability, storage name, `FieldTy`, `auto`, and `constraints`
600/// off the result; uniqueness is the one property a `Field` cannot answer, so
601/// it has [`lens_field_unique`] of its own.
602///
603/// `model` is the owning `Model::schema()`, passed in so one form-input
604/// construction resolves the schema once (never per row) and shares it with
605/// [`lens_field_unique`]. The returned `Field` is owned because
606/// `Model::schema()` builds the model by value with no cache, so no borrow of
607/// it can escape — but only the one field is cloned.
608///
609/// Traversal lenses are rejected (GH #100): a multi-step path has no single
610/// field name, and silently binding its first segment misbinds in release.
611/// Use [`FieldResolver`] (GH #185) to bind an embedded path instead.
612pub(crate) fn lens_field<M, T>(
613    path: FieldLens<M, T>,
614    model: &toasty::schema::app::Model,
615) -> toasty::schema::app::Field
616where
617    M: toasty::schema::Model,
618{
619    let core_path: toasty_core::stmt::Path = path.into();
620    require_single_segment(&core_path, "lens");
621    let idx = core_path
622        .projection
623        .as_slice()
624        .first()
625        .copied()
626        .expect("field lens must have a projection");
627    model
628        .as_root_unwrap()
629        .fields
630        .get(idx)
631        .cloned()
632        .unwrap_or_else(|| {
633            panic!(
634                "field index {idx} out of bounds for {}",
635                std::any::type_name::<M>()
636            )
637        })
638}

640/// The capitalized label for a field's app-level name.
641pub(crate) fn lens_label(field: &toasty::schema::app::Field) -> String {
642    capitalize(field.name.app_unwrap())
643}

645/// Whether `field` is backed by a unique index it participates in.
646///
647/// Uniqueness is not a property of a `Field`: Toasty stores it on the model's
648/// index list, so this needs the owning `ModelRoot` too. The primary key is
649/// excluded — a key column is unique by construction, not by a declared
650/// constraint.
651///
652/// A **composite** unique index counts too (GH #88). `#[unique(tenant_id,
653/// email)]` is how a tenant-scoped resource expresses "unique within the
654/// tenant", and the app-side pre-check has to recognise it or the field's
655/// `unique()` declaration would be silently dead. Recognizing the index is not
656/// the same as checking it exactly: the pre-check probes the field's value
657/// inside the tenant-scoped query's scope (GH #223), so it enforces the
658/// constraint only when that
659/// scope matches the index's remaining components — which is the arrangement
660/// `#[unique(tenant_id, ..)]` on a tenant-scoped resource produces, and which
661/// `two_tenants_may_share_an_author_email` / `duplicate_email_within_one_tenant
662/// _is_reported_inline` in the showcase pin from both sides. Declaring `unique()`
663/// on a field whose index carries components *outside* the resource's scope
664/// stays a gap, and it is not checkable here because a query's filters are not
665/// introspectable; see `Resource::query`'s note and upstream #117, whose driver
666/// predicate is the real fix.
667///
668/// This reports declared schema uniqueness, not a global uniqueness guarantee:
669/// SQL permits multiple `NULL`s in a unique index, and enum-variant columns are
670/// storage-nullable, so a nullable unique field can still repeat. That is the
671/// same caveat the app-side pre-check has always carried (`panel/forms.rs`).
672pub(crate) fn lens_field_unique(
673    field: &toasty::schema::app::Field,
674    model: &toasty::schema::app::ModelRoot,
675) -> bool {
676    model.indices.iter().any(|index| {
677        index.unique && !index.primary_key && index.fields.iter().any(|f| f.field == field.id)
678    })
679}

681/// Panic unless a lens path addresses exactly one field (GH #100).
682///
683/// A traversal lens (relation hops, embedded steps) has no single field name,
684/// nullability, or uniqueness — silently binding its first segment misbinds in
685/// release, so every lens helper rejects multi-segment paths loudly instead.
686pub(crate) fn require_single_segment(path: &toasty_core::stmt::Path, what: &str) {
687    assert_eq!(
688        path.projection.as_slice().len(),
689        1,
690        "{what} requires a single-field lens, got a {}-segment traversal path (GH #100)",
691        path.projection.as_slice().len()
692    );
693}
694
695/// A field's human label from its storage name.
696///
697/// Sentence case, with the underscores a Rust column name carries read as
698/// spaces: `word_count` is "Word count", not "Word_count". A label is the one
699/// place a column name becomes prose, so it should not leak the identifier —
700/// which is what `Media_poster_url` and `Seo_title` did everywhere an embedded
701/// or document leaf rendered its own name (GH #185/#192). An explicit
702/// [`.label(..)`](crate::schema::TextInput::label) still wins.
703pub(crate) fn capitalize(s: &str) -> String {
704    let spaced = s.replace('_', " ");
705    let mut c = spaced.chars();
706    match c.next() {
707        None => String::new(),
708        Some(f) => f.to_uppercase().collect::<String>() + c.as_str(),
709    }
710}
