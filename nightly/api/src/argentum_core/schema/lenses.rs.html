1//! Field lenses — typed Toasty paths and their app-level metadata.
2//!
3//! Base bridge layer (with `pk`): these two modules are the only schema
4//! modules that name `toasty_core` (upstream #114/#183), so upstream
5//! churn has one blast radius. Everything above reads lenses through
6//! `FieldLens` and the helpers here.
7//!
8//! [`lens_field`] hands callers the built `app::Field`, so name, label,
9//! nullability, storage name, `FieldTy`, `auto`, and `constraints` all come
10//! from one walk. [`lens_field_unique`] covers the one property `Field` does
11//! not carry, because Toasty keeps uniqueness on the model's index list.
12//!
13//! [`FieldResolver`] is the schema-aware walk (GH #185): with the app schema
14//! in hand an embedded path resolves to its **flattened storage column**, so
15//! `TextInput::r#for_post(...)`-style bindings can reach a field inside an
16//! embedded struct or a `#[document]`. Without a schema the single-segment
17//! rule (GH #100) still applies — the owned `app::Model` cannot see embedded
18//! models, which is exactly what the schema adds.
19
20use toasty_core::stmt::PathRoot;
21use topcoat::context::Cx;
22
23/// Spec alias — ADR-0001 typed lens. Currently uses `toasty::stmt::Path` directly;
24/// a richer `FieldLens` trait will replace this alias if Toasty exposes the
25/// metadata walk directly (see GH #11, upstream issue #183).
26pub type FieldLens<M, T> = toasty::stmt::Path<M, T>;
27
28/// The storage column an embedded path flattens to, plus the metadata field
29/// constructors need.
30///
31/// Owned because the schema is borrowed from the request context for the length
32/// of one call and no borrow of it can be held by a `Schema`.
33#[derive(Debug, Clone)]
34pub(crate) struct LeafField {
35    /// The flattened storage column (`application_seo_title`), or the field's
36    /// own name for a single-segment path.
37    pub(crate) name: String,
38    pub(crate) label: String,
39    /// Whether the leaf column is nullable. An embedded step makes every
40    /// column under it storage-nullable — only the matching enum variant
41    /// writes a value — so an embedded leaf is never required by default.
42    pub(crate) nullable: bool,
43}
44
45/// The compiled schema for this request, or `None` when no `Db` is in context.
46///
47/// `Db::schema()` is public and gives all three halves the walk needs: `.app`
48/// carries the embedded models the owned `Model::schema()` cannot see,
49/// `.mapping` records which physical column each field resolves to, and `.db`
50/// holds the physical table and column names. So the request path can bind an
51/// embedded lens without opening a connection.
52///
53/// Borrowed, never cloned, and optional: a bare `CxTestBuilder` has no `Db`, and
54/// then the single-segment rule (GH #100) applies, so a schema-less test fails
55/// loudly on a traversal lens rather than silently binding the first segment.
56fn request_schema(cx: &Cx) -> Option<&toasty_core::Schema> {
57    topcoat::context::try_app_context::<toasty::Db>(cx).map(|db| &**db.schema())
58}

60/// Walks a lens path against the app schema, resolving embedded steps.
61///
62/// Kept as a struct rather than free functions so the schema borrow has one
63/// owner and the constructors can share one resolution between the name, the
64/// label, and the uniqueness check.
65pub(crate) struct FieldResolver<'a> {
66    /// The **compiled** schema: its `.app` half resolves the path, its
67    /// `.mapping` half names the column, its `.db` half holds the name.
68    schema: Option<&'a toasty_core::Schema>,
69}
70
71impl<'a> FieldResolver<'a> {
72    /// Build the resolver from the app schema this request carries, if any.
73    pub(crate) fn from_cx(cx: &'a Cx) -> Self {
74        Self {
75            schema: request_schema(cx),
76        }
77    }
78
79    /// Whether this request carries an app schema at all (a `Db` in context).
80    ///
81    /// The leaf constructors fall back to the single-segment rule without one;
82    /// a *value* binding has no fallback — every key comes from the schema — so
83    /// its entry points say so rather than reporting a traversal-lens error
84    /// (GH #191).
85    pub(crate) fn has_schema(&self) -> bool {
86        self.schema.is_some()
87    }
88
89    /// Resolve a lens to its leaf field.
90    ///
91    /// With a schema this walks the whole projection, so an embedded step lands
92    /// on the flattened column. Without one it falls back to the single-segment
93    /// rule and panics on a traversal lens — loudly, because silently binding
94    /// the first segment misbinds in release (GH #100).
95    pub(crate) fn resolve<M, T>(&self, path: FieldLens<M, T>) -> LeafField
96    where
97        M: toasty::schema::Model,
98    {
99        let model = M::schema();
100        let core_path: toasty_core::stmt::Path = path.into();
101        let segments = core_path.projection.as_slice().len();
102        // A variant root is an embedded-enum payload accessor *whatever* its
103        // projection length: the generated accessor rebases onto the variant, so
104        // the steps are variant-local and a single-step projection is the
105        // payload field. Only a model root with one step is a plain field.
106        let is_embedded_path = segments > 1 || matches!(core_path.root, PathRoot::Variant { .. });
107        if is_embedded_path && let Some(schema) = self.schema {
108            return self
109                .walk_embedded(schema, &core_path, &model)
110                .unwrap_or_else(|| {
111                    panic!(
112                        "lens path {projection:?} does not resolve to a single column for {}: \
113                         only embedded steps (embedded structs, enum variant fields, and \
114                         `#[document]` fields) can be bound, not relation hops, and only when \
115                         the model is in the app schema this request carries (GH #185)",
116                        std::any::type_name::<M>(),
117                        projection = core_path.projection.as_slice(),
118                    )
119                });
120        }
121        // No schema: the single-segment rule is all an owned `app::Model` can
122        // answer, so resolve the first step against `ModelRoot.fields` and let
123        // `require_single_segment` reject a traversal lens (GH #100).
124        require_single_segment(&core_path, "lens");
125        let idx = core_path
126            .projection
127            .as_slice()
128            .first()
129            .copied()
130            .expect("field lens must have a projection");
131        let field = model
132            .as_root_unwrap()
133            .fields
134            .get(idx)
135            .cloned()
136            .unwrap_or_else(|| {
137                panic!(
138                    "field index {idx} out of bounds for {}",
139                    std::any::type_name::<M>()
140                )
141            });
142        LeafField {
143            name: field.name.app_unwrap().to_string(),
144            label: lens_label(&field),
145            nullable: field.nullable(),
146        }
147    }
148
149    /// Walk a lens path to the physical column it names.
150    ///
151    /// Two sources, each authoritative for one thing:
152    ///
153    /// - the **app schema** drives the traversal, because it is the only side
154    ///   that knows a field is a `#[document]` — a *primitive* whose storage is
155    ///   a model, so its inner fields collapse into the one column named after
156    ///   it, and a path through one has steps left over on arrival;
157    /// - the **compiled mapping** names the column. `Db::schema().mapping`
158    ///   records, per model, which column every field resolves to — flattened
159    ///   embedded structs, enum discriminant and payload columns, shared
160    ///   columns, and a document's single column alike — so this reads the name
161    ///   off `db::Table` rather than re-deriving Toasty's naming rules. An
162    ///   earlier revision accumulated names from `app::Field.name` and had to
163    ///   encode those rules itself.
164    ///
165    /// Probed against the pinned rev the two agree for every reachable shape,
166    /// including the cases that are easy to get wrong: `#[shared(..)]` payloads
167    /// collapse onto one column (`publication_timestamp`), and a `#[document]`
168    /// is a single `Primitive` column named after the field (`stats`), not one
169    /// column per inner field.
170    ///
171    /// Two roots reach here:
172    ///
173    /// - a **model** root, for a plain path (`seo.title`);
174    /// - a **variant** root, for an enum payload accessor
175    ///   (`media.video().poster().url()`): the generated accessor rebases onto
176    ///   the variant, so the root carries the parent path *to the enum field*
177    ///   while the projection's steps are **variant-local**.
178    ///
179    /// Only embedded steps are followed. A relation hop yields `None`: this
180    /// exists for embedded binding (GH #185), and binding anything else here
181    /// would reintroduce the misbind GH #100 guards against.
182    fn walk_embedded(
183        &self,
184        schema: &'a toasty_core::Schema,
185        path: &toasty_core::stmt::Path,
186        owner: &toasty::schema::app::Model,
187    ) -> Option<LeafField> {
188        match &path.root {
189            PathRoot::Model(id) => {
190                let root = schema.app.get_model(*id)?.as_root()?;
191                // The accessor's `ModelId` and the schema's may come from
192                // different `models!(..)` expansions, so an id can name a
193                // *different* model here. Verify identity by root model name
194                // before trusting any index: binding another model's column is
195                // the misbind GH #100 exists to prevent.
196                if root.name.upper_camel_case() != owner.as_root()?.name.upper_camel_case() {
197                    return None;
198                }
199                let model = schema.mapping.models.get(&root.id)?;
200                descend(
201                    schema,
202                    &root.fields,
203                    &model.fields,
204                    path.projection.as_slice(),
205                )
206            }
207            // An enum payload: resolve the parent path (it ends at the enum
208            // field), then descend into the variant the accessor selected.
209            PathRoot::Variant { parent, variant_id } => {
210                let root = schema
211                    .app
212                    .get_model(parent.root.as_model_unwrap())?
213                    .as_root()?;
214                let model = schema.mapping.models.get(&root.id)?;
215                let parent_steps = parent.projection.as_slice();
216                // The app side supplies the enum's payload field list, the
217                // mapping side its per-variant column mappings.
218                let app_field = app_field_at(schema, &root.fields, parent_steps)?;
219                let Some(toasty::schema::app::Model::EmbeddedEnum(e)) =
220                    app_embedded(schema, app_field)
221                else {
222                    return None;
223                };
224                let MappingField::Enum(me) = mapping_field_at(&model.fields, parent_steps)? else {
225                    return None;
226                };
227                let variant = me.variants.get(variant_id.index)?;
228                let payloads: Vec<_> = e.variant_fields(variant_id.index).iter().collect();
229                descend(
230                    schema,
231                    &payloads,
232                    &variant.fields,
233                    path.projection.as_slice(),
234                )
235            }
236        }
237    }

239    /// Enumerate the bindable surface of the embedded **value** at `path`
240    /// (GH #191).
241    ///
242    /// [`Self::resolve`] answers "which column does this one leaf occupy"; this
243    /// answers "what does this whole value consist of" — every leaf column,
244    /// plus the discriminant for an enum — which is what a value codec and a
245    /// generated form need.
246    ///
247    /// `path` addresses the embedded field itself (`Post::fields().seo()`,
248    /// `Post::fields().publication()`), not one of its leaves: a variant-rooted
249    /// path names a *variant* of a value, not the value, and yields `None`.
250    pub(crate) fn resolve_embedded_value<M, T>(
251        &self,
252        path: FieldLens<M, T>,
253    ) -> Option<EmbeddedValueSpec>
254    where
255        M: toasty::schema::Model,
256    {
257        let schema = self.schema?;
258        let core_path: toasty_core::stmt::Path = path.into();
259        let PathRoot::Model(id) = core_path.root else {
260            return None;
261        };
262        let owner = M::schema();
263        let root = schema.app.get_model(id)?.as_root()?;
264        // The same identity check `walk_embedded` makes: an accessor's
265        // `ModelId` and the schema's can come from different `models!(..)`
266        // expansions, so verify the root by name before trusting any index.
267        if root.name.upper_camel_case() != owner.as_root()?.name.upper_camel_case() {
268            return None;
269        }
270        let mapping = schema.mapping.models.get(&root.id)?;
271        let steps = core_path.projection.as_slice();
272        let app_field = app_field_at(schema, &root.fields, steps)?;
273        let mapping_field = mapping_field_at(&mapping.fields, steps)?;
274        let mut columns = Vec::new();
275        let enum_spec = match (app_embedded(schema, app_field)?, mapping_field) {
276            (toasty::schema::app::Model::EmbeddedStruct(e), MappingField::Struct(ms)) => {
277                collect_columns(schema, &e.fields, &ms.fields, &mut columns)?;
278                None
279            }
280            (toasty::schema::app::Model::EmbeddedEnum(e), MappingField::Enum(me)) => {
281                collect_enum_columns(schema, e, me, &mut columns)?;
282                let discriminant =
283                    column_name(schema, &MappingField::Primitive(me.discriminant.clone()))?;
284                let variants = e
285                    .variants
286                    .iter()
287                    .map(|v| {
288                        discriminant_text(&v.discriminant)
289                            .map(|value| (value, capitalize(&v.name.snake_case())))
290                    })
291                    .collect::<Option<Vec<_>>>()?;
292                Some(crate::schema::embedded::EnumSpec::new(
293                    discriminant,
294                    variants,
295                ))
296            }
297            _ => return None,
298        };
299        Some(EmbeddedValueSpec { columns, enum_spec })
300    }
301}

303/// The column one mapping field occupies, by name.
304fn column_name(schema: &toasty_core::Schema, field: &MappingField) -> Option<String> {
305    column_of(schema, field).map(|leaf| leaf.name)
306}

308/// Collect every leaf column under one embedded level.
309///
310/// A relation inside an embedded value is not bindable and yields `None`: this
311/// walk exists for value binding (GH #191), the same boundary GH #100 draws for
312/// single lenses.
313fn collect_columns(
314    schema: &toasty_core::Schema,
315    app_fields: &[toasty::schema::app::Field],
316    mapping_fields: &[MappingField],
317    out: &mut Vec<String>,
318) -> Option<()> {
319    for (app_field, mapping_field) in app_fields.iter().zip(mapping_fields) {
320        if is_document(app_field) {
321            // A `#[document]`'s inner fields share its one column, so a *value*
322            // codec has nothing per-field to bind: refuse rather than hand back
323            // one column for several fields (the misbind GH #100 guards
324            // against). Leaf binding still reaches a document's column (GH #185).
325            return None;
326        }
327        match &app_field.ty {
328            toasty::schema::app::FieldTy::Primitive(_) => {
329                push_column(schema, mapping_field, out)?;
330            }
331            toasty::schema::app::FieldTy::Embedded(embedded) => {
332                match schema.app.get_model(embedded.target)? {
333                    toasty::schema::app::Model::EmbeddedStruct(e) => {
334                        let MappingField::Struct(ms) = mapping_field else {
335                            return None;
336                        };
337                        collect_columns(schema, &e.fields, &ms.fields, out)?;
338                    }
339                    toasty::schema::app::Model::EmbeddedEnum(e) => {
340                        let MappingField::Enum(me) = mapping_field else {
341                            return None;
342                        };
343                        collect_enum_columns(schema, e, me, out)?;
344                    }
345                    toasty::schema::app::Model::Root(_) => return None,
346                }
347            }
348            _ => return None,
349        }
350    }
351    Some(())
352}

354/// Collect every variant's payload columns under one embedded enum.
355fn collect_enum_columns(
356    schema: &toasty_core::Schema,
357    e: &toasty::schema::app::EmbeddedEnum,
358    me: &toasty_core::schema::mapping::FieldEnum,
359    out: &mut Vec<String>,
360) -> Option<()> {
361    // The discriminant column first: it is a form key of this value (the
362    // control that carries the variant), whether the enum is the value itself
363    // or nested inside one.
364    push_column(
365        schema,
366        &MappingField::Primitive(me.discriminant.clone()),
367        out,
368    )?;
369    for (index, variant) in me.variants.iter().enumerate() {
370        collect_columns(schema, e.variant_fields(index), &variant.fields, out)?;
371    }
372    Some(())
373}

375/// Push a column name once: a `#[shared(..)]` column is declared by several
376/// variants and is still one column.
377fn push_column(
378    schema: &toasty_core::Schema,
379    field: &MappingField,
380    out: &mut Vec<String>,
381) -> Option<()> {
382    let name = column_name(schema, field)?;
383    if !out.contains(&name) {
384        out.push(name);
385    }
386    Some(())
387}

389/// The text an embedded enum's discriminant stores (GH #191).
390///
391/// Toasty's discriminants are integers or strings, and the form carries that
392/// same text — so what a submission posts is what a row stores. Anything else
393/// is reported rather than guessed.
394fn discriminant_text(value: &toasty_core::stmt::Value) -> Option<String> {
395    use toasty_core::stmt::Value;
396    match value {
397        Value::Bool(v) => Some(v.to_string()),
398        Value::I8(v) => Some(v.to_string()),
399        Value::I16(v) => Some(v.to_string()),
400        Value::I32(v) => Some(v.to_string()),
401        Value::I64(v) => Some(v.to_string()),
402        Value::U8(v) => Some(v.to_string()),
403        Value::U16(v) => Some(v.to_string()),
404        Value::U32(v) => Some(v.to_string()),
405        Value::U64(v) => Some(v.to_string()),
406        Value::String(v) => Some(v.clone()),
407        _ => None,
408    }
409}

411/// A `mapping::Field`, aliased so the traversal signatures stay readable.
412type MappingField = toasty_core::schema::mapping::Field;
413
414/// Walk the path down to the column that stores it.
415///
416/// `app_fields` and `mapping_fields` are indexed by the same field index, so
417/// every step reads both: the app side decides *whether to descend*, the mapping
418/// side names *the column*. Neither alone is enough — the mapping cannot say a
419/// field is a `#[document]`, and the app schema cannot say which column a leaf
420/// occupies without re-deriving Toasty's naming.
421fn descend<F>(
422    schema: &toasty_core::Schema,
423    app_fields: &[F],
424    mapping_fields: &[MappingField],
425    steps: &[usize],
426) -> Option<LeafField>
427where
428    F: std::borrow::Borrow<toasty::schema::app::Field>,
429{
430    let (first, rest) = steps.split_first()?;
431    let app_field: &toasty::schema::app::Field = app_fields.get(*first)?.borrow();
432    let mapping_field = mapping_fields.get(*first)?;
433    // A `#[document]`'s inner fields share its one column, so reaching the
434    // document is reaching the leaf, however many steps remain.
435    if rest.is_empty() || is_document(app_field) {
436        return column_of(schema, mapping_field);
437    }
438    let toasty::schema::app::FieldTy::Embedded(embedded) = &app_field.ty else {
439        // A relation hop, or a primitive with steps left over: not an embedded
440        // leaf either way.
441        return None;
442    };
443    match schema.app.get_model(embedded.target)? {
444        toasty::schema::app::Model::EmbeddedStruct(e) => {
445            let MappingField::Struct(ms) = mapping_field else {
446                return None;
447            };
448            descend(schema, &e.fields, &ms.fields, rest)
449        }
450        toasty::schema::app::Model::EmbeddedEnum(e) => {
451            let MappingField::Enum(me) = mapping_field else {
452                return None;
453            };
454            // A variant consumes two steps: the variant index, then the field.
455            let (variant_index, tail) = rest.split_first()?;
456            let variant = me.variants.get(*variant_index)?;
457            let payloads: Vec<_> = e.variant_fields(*variant_index).iter().collect();
458            descend(schema, &payloads, &variant.fields, tail)
459        }
460        _ => None,
461    }
462}

464/// The column a single mapping field occupies, if it is a leaf.
465fn column_of(schema: &toasty_core::Schema, field: &MappingField) -> Option<LeafField> {
466    let MappingField::Primitive(p) = field else {
467        // A path stopping on a struct or enum names no single column, and a
468        // relation stores none.
469        return None;
470    };
471    let column = &schema.db.tables[p.column.table.0].columns[p.column.index];
472    Some(LeafField {
473        name: column.name.clone(),
474        label: capitalize(&column.name.replace('_', " ")),
475        // Every column under an embedded step is storage-nullable: only the
476        // matching variant writes a value.
477        nullable: true,
478    })
479}

481/// Whether `field` is a `#[document]`: a *primitive* whose storage is a model,
482/// so its inner fields collapse into the one column named after it.
483fn is_document(field: &toasty::schema::app::Field) -> bool {
484    matches!(
485        &field.ty,
486        toasty::schema::app::FieldTy::Primitive(p)
487            if matches!(p.ty, toasty_core::stmt::Type::Model(_))
488    )
489}

491/// The bindable surface of one embedded **value** (GH #191): every column its
492/// fields occupy, plus — for an enum — the discriminant column and each
493/// variant's stored value.
494///
495/// [`FieldResolver::resolve`] answers "which column does this one leaf occupy";
496/// this answers "what does this whole value consist of", which is what a value
497/// codec and a generated form need. Both read the same two authorities: the app
498/// schema decides the structure, the compiled mapping names every column.
499#[derive(Debug, Clone)]
500pub(crate) struct EmbeddedValueSpec {
501    /// Every form key the value occupies, in declaration order and
502    /// deduplicated: each leaf column, and the discriminant column of every
503    /// enum the value contains — the top-level one and any nested inside a
504    /// struct or a variant. A `#[shared(..)]` column declared by several
505    /// variants appears once, because it *is* one column.
506    pub(crate) columns: Vec<String>,
507    /// `Some` for an embedded enum: its discriminant column and its variants
508    /// (each one's stored value and the name it is labelled with). The one
509    /// `EnumSpec` type is shared with the public seam.
510    pub(crate) enum_spec: Option<crate::schema::embedded::EnumSpec>,
511}

513/// The embedded model an app field targets, if it is embedded.
514fn app_embedded<'a>(
515    schema: &'a toasty_core::Schema,
516    field: &toasty::schema::app::Field,
517) -> Option<&'a toasty::schema::app::Model> {
518    let toasty::schema::app::FieldTy::Embedded(embedded) = &field.ty else {
519        return None;
520    };
521    schema.app.get_model(embedded.target)
522}

524/// The app-level field `steps` reaches, used by the variant root to find the
525/// enum's payload list before descending.
526///
527/// Recurses through embedded structs: the variant root's parent path can walk
528/// through them (an embedded struct holding the enum), not just one step.
529fn app_field_at<'a>(
530    schema: &'a toasty_core::Schema,
531    fields: &'a [toasty::schema::app::Field],
532    steps: &[usize],
533) -> Option<&'a toasty::schema::app::Field> {
534    let (first, rest) = steps.split_first()?;
535    let field = fields.get(*first)?;
536    if rest.is_empty() {
537        return Some(field);
538    }
539    match app_embedded(schema, field)? {
540        toasty::schema::app::Model::EmbeddedStruct(e) => app_field_at(schema, &e.fields, rest),
541        // An enum nested in the parent path consumes two steps per level: the
542        // variant index, then a variant-local field index. `variant_fields`
543        // returns the variant's own slice of the enum's global field list
544        // (upstream 7ff180db changed it from an iterator to that slice), so the
545        // second step indexes it directly — the generated accessor's index is
546        // variant-local, which is the same field either way.
547        //
548        // The variant step is bounds-checked here first: `variant_fields`
549        // indexes `variants[i]` and panics out of range where the old iterator
550        // returned `None`, and this walk answers `None` for a path it cannot
551        // resolve (a wrong lens must not abort a request).
552        toasty::schema::app::Model::EmbeddedEnum(e) => {
553            let (variant, tail) = rest.split_first()?;
554            e.variants.get(*variant)?;
555            let field = e.variant_fields(*variant).get(*tail.first()?)?;
556            if tail.len() == 1 {
557                Some(field)
558            } else {
559                app_field_at(schema, std::slice::from_ref(field), &tail[1..])
560            }
561        }
562        toasty::schema::app::Model::Root(_) => None,
563    }
564}

566/// The mapping field `steps` reaches, used by the variant root to find the
567/// enum's per-variant mappings before descending.
568fn mapping_field_at<'a>(fields: &'a [MappingField], steps: &[usize]) -> Option<&'a MappingField> {
569    let (first, rest) = steps.split_first()?;
570    let field = fields.get(*first)?;
571    if rest.is_empty() {
572        return Some(field);
573    }
574    match field {
575        MappingField::Struct(s) => mapping_field_at(&s.fields, rest),
576        MappingField::Enum(e) => {
577            let (variant, tail) = rest.split_first()?;
578            mapping_field_at(&e.variants.get(*variant)?.fields, tail)
579        }
580        _ => None,
581    }
582}

584/// Resolve a typed lens to the app-level [`Field`] behind it.
585///
586/// The single walk over `Path → toasty_core::stmt::Path → projection`, reading
587/// off the model's built schema (upstream issues #114/#183). Callers read
588/// name, label, nullability, storage name, `FieldTy`, `auto`, and `constraints`
589/// off the result; uniqueness is the one property a `Field` cannot answer, so
590/// it has [`lens_field_unique`] of its own.
591///
592/// `model` is the owning `Model::schema()`, passed in so one form-input
593/// construction resolves the schema once (never per row) and shares it with
594/// [`lens_field_unique`]. The returned `Field` is owned because
595/// `Model::schema()` builds the model by value with no cache, so no borrow of
596/// it can escape — but only the one field is cloned.
597///
598/// Traversal lenses are rejected (GH #100): a multi-step path has no single
599/// field name, and silently binding its first segment misbinds in release.
600/// Use [`FieldResolver`] (GH #185) to bind an embedded path instead.
601pub(crate) fn lens_field<M, T>(
602    path: FieldLens<M, T>,
603    model: &toasty::schema::app::Model,
604) -> toasty::schema::app::Field
605where
606    M: toasty::schema::Model,
607{
608    let core_path: toasty_core::stmt::Path = path.into();
609    require_single_segment(&core_path, "lens");
610    let idx = core_path
611        .projection
612        .as_slice()
613        .first()
614        .copied()
615        .expect("field lens must have a projection");
616    model
617        .as_root_unwrap()
618        .fields
619        .get(idx)
620        .cloned()
621        .unwrap_or_else(|| {
622            panic!(
623                "field index {idx} out of bounds for {}",
624                std::any::type_name::<M>()
625            )
626        })
627}
628
629/// The capitalized label for a field's app-level name.
630pub(crate) fn lens_label(field: &toasty::schema::app::Field) -> String {
631    capitalize(field.name.app_unwrap())
632}
633
634/// Whether `field` is backed by a unique index it participates in.
635///
636/// Uniqueness is not a property of a `Field`: Toasty stores it on the model's
637/// index list, so this needs the owning `ModelRoot` too. The primary key is
638/// excluded — a key column is unique by construction, not by a declared
639/// constraint.
640///
641/// A **composite** unique index counts too (GH #88). `#[unique(tenant_id,
642/// email)]` is how a tenant-scoped resource expresses "unique within the
643/// tenant", and the app-side pre-check has to recognise it or the field's
644/// `unique()` declaration would be silently dead. Recognizing the index is not
645/// the same as checking it exactly: the pre-check probes the field's value
646/// inside the tenant-scoped query's scope (GH #223), so it enforces the
647/// constraint only when that
648/// scope matches the index's remaining components — which is the arrangement
649/// `#[unique(tenant_id, ..)]` on a tenant-scoped resource produces, and which
650/// `two_tenants_may_share_an_author_email` / `duplicate_email_within_one_tenant
651/// _is_reported_inline` in the showcase pin from both sides. Declaring `unique()`
652/// on a field whose index carries components *outside* the resource's scope
653/// stays a gap, and it is not checkable here because a query's filters are not
654/// introspectable; see `Resource::query`'s note and upstream #117, whose driver
655/// predicate is the real fix.
656///
657/// This reports declared schema uniqueness, not a global uniqueness guarantee:
658/// SQL permits multiple `NULL`s in a unique index, and enum-variant columns are
659/// storage-nullable, so a nullable unique field can still repeat. That is the
660/// same caveat the app-side pre-check has always carried (`panel/forms.rs`).
661pub(crate) fn lens_field_unique(
662    field: &toasty::schema::app::Field,
663    model: &toasty::schema::app::ModelRoot,
664) -> bool {
665    model.indices.iter().any(|index| {
666        index.unique && !index.primary_key && index.fields.iter().any(|f| f.field == field.id)
667    })
668}

670/// Panic unless a lens path addresses exactly one field (GH #100).
671///
672/// A traversal lens (relation hops, embedded steps) has no single field name,
673/// nullability, or uniqueness — silently binding its first segment misbinds in
674/// release, so every lens helper rejects multi-segment paths loudly instead.
675pub(crate) fn require_single_segment(path: &toasty_core::stmt::Path, what: &str) {
676    assert_eq!(
677        path.projection.as_slice().len(),
678        1,
679        "{what} requires a single-field lens, got a {}-segment traversal path (GH #100)",
680        path.projection.as_slice().len()
681    );
682}
683
684/// A field's human label from its storage name.
685///
686/// Sentence case, with the underscores a Rust column name carries read as
687/// spaces: `word_count` is "Word count", not "Word_count". A label is the one
688/// place a column name becomes prose, so it should not leak the identifier —
689/// which is what `Media_poster_url` and `Seo_title` did everywhere an embedded
690/// or document leaf rendered its own name (GH #185/#192). An explicit
691/// [`.label(..)`](crate::schema::TextInput::label) still wins.
692pub(crate) fn capitalize(s: &str) -> String {
693    let spaced = s.replace('_', " ");
694    let mut c = spaced.chars();
695    match c.next() {
696        None => String::new(),
697        Some(f) => f.to_uppercase().collect::<String>() + c.as_str(),
698    }
699}
