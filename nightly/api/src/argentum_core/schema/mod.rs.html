1//! Unified Schema primitive — layout blocks that compose via `view!`.
2//!
3//! `Schema` is a container for `Section`, `Group`, `Grid`, `Tabs` and
4//! `Repeater` nodes. Each node renders through Topcoat's `view!` macro;
5//! `Schema::render` combines them. The API mirrors Filament's
6//! `Schema::new(( ... ))` tuple form via the `IntoSchema` trait.
7//!
8//! Bridge note: `lens_field` is the one walk reaching into `toasty_core`
9//! (upstream issue #114), alongside the `pk_*` bridge helpers and `cursor.rs`
10//! cursor values. It hands back the built `app::Field`, so field metadata no
11//! longer needs a helper per property; uniqueness comes from
12//! `lens_field_unique`, since Toasty keeps it on the model's index list rather
13//! than the field. Retire the walk when Toasty exposes it (upstream #183).

64    /// Panics on duplicate field names (GH #100): two inputs sharing one name

83    /// Render the schema read-only (GH #187): the detail page's side of the
84    /// same declaration.
85    ///
86    /// Every field shows the record's stored value in place of its control, so
87    /// a detail page reuses the field types and layout blocks a form already
88    /// declares rather than a parallel infolist vocabulary. `values` is the
89    /// record hydrated exactly as the edit form hydrates it
90    /// ([`Resource::hydrate_form_values`](crate::resource::Resource::hydrate_form_values)):
91    /// what a user reads on the page is what the form would have shown them.
92    pub async fn render_readonly<'a>(
93        &self,
94        cx: &'a Cx,
95        values: &HashMap<String, String>,
96    ) -> Result<BoxView<'a>> {
97        self.render_source(
98            cx,
99            &RenderSource {
100                values,
101                errors: &HashMap::new(),
102                mode: Mode::View,
103            },
104        )
105        .await
106    }

108    /// Rewrite submitted values into their fields' stored spelling (GH #192).
109    ///
110    /// Runs after validation and before a record fn sees the map, so a typed
111    /// field's `Display` — not the browser's spelling — is what gets written.
112    /// That is what makes an untouched edit round-trip: the form hydrates a
113    /// stored value, the browser echoes it, and this puts back the same string
114    /// the record fn would have produced rather than a re-spelling of it.
115    ///
116    /// A `Select` takes its trimmed submission (GH #297): its presence rule and
117    /// its option-existence check both read `value.trim()`, so the trimmed
118    /// value is the one that passed — writing the untrimmed spelling would
119    /// store a value no rule authorised.
120    ///
121    /// A value the caller has not validated cannot be normalised, so a parse
122    /// failure here leaves the submission untouched and reports nothing: it is
123    /// unreachable from the handlers (validation refuses it first), and a
124    /// silent rewrite would hide a bypass rather than surface it. A field with
125    /// no submission keeps its absence — an update writes only present keys.
126    ///
127    /// An **empty** submission is left empty for the same reason (GH #192): a
128    /// typed column has no spelling for "no value" — `""` is not an `i64` and
129    /// not a `Timestamp` — so inventing one here would put a value in a record
130    /// the user never gave. Empty is the presence rule's business, which is
131    /// where the panel already answers it: `.required()` refuses it inline, and
132    /// an optional typed field reaches its record fn as `""` for the record fn's
133    /// own default. The record fn therefore reads a typed field through the same
134    /// "present or absent" check it uses for any other optional column.
135    pub fn normalize_values(&self, values: &mut HashMap<String, String>) {
136        for (name, input) in self.text_inputs() {
137            let Some(submitted) = values.get(&name) else {
138                continue;
139            };
140            if submitted.trim().is_empty() {
141                continue;
142            }
143            if let Ok(normalized) = input.normalize(submitted) {
144                values.insert(name, normalized);
145            }
146        }
147        for name in self.select_inputs().keys() {
148            let Some(submitted) = values.get(name).cloned() else {
149                continue;
150            };
151            let trimmed = submitted.trim();
152            if trimmed.len() != submitted.len() {
153                values.insert(name.clone(), trimmed.to_string());
154            }
155        }
156    }

158    /// Append another schema's nodes after this one's (GH #191).
159    ///
160    /// [`Schema::new`] composes through `IntoSchema`, whose tuple form stops at
161    /// eight nodes; a derived embedded form has one control per leaf column and
162    /// composes nested values, so it builds its schema by appending instead. The
163    /// nodes keep their order, so a form reads in declaration order either way.
164    pub fn extend(mut self, other: Schema) -> Schema {
165        self.nodes.extend(other.nodes);
166        // The same guard `Schema::new` runs: a derived form is built by
167        // appending, so this is the only check for the shapes `new` cannot see.
168        self.assert_unique_field_names();
169        self
170    }

190    /// The one node walk: every node renders its static form.
191    pub(crate) async fn render_source<'a>(
192        &self,
193        cx: &'a Cx,
194        source: &RenderSource<'_>,
195    ) -> Result<BoxView<'a>> {
196        let mut views = Vec::with_capacity(self.nodes.len());
197        for node in &self.nodes {
198            views.push(Box::pin(node.render_source(cx, source)).await?.boxed());
199        }
200        Ok(view! {
201            cx =>
202            for v in views {
203                (v)
204            }
205        }
206        .boxed())
207    }

209    /// Collect field names for validation (TextInput + Textarea + Select + FileUpload).
210    pub fn field_names(&self) -> Vec<String> {
211        let mut out = Vec::new();
212        for node in &self.nodes {
213            for_each_field(node, &mut |n| match n {
214                Node::TextInput(f) => out.push(f.field_name().to_string()),
215                Node::Textarea(f) => out.push(f.field_name().to_string()),
216                Node::Select(f) => out.push(f.field_name().to_string()),
217                Node::FileUpload(f) => out.push(f.field_name().to_string()),
218                _ => {}
219            });
220        }
221        out
222    }

224    /// Keys in `values` that no declared input owns, sorted (GH #89).
225    ///
226    /// Framework-level allow-list seam, enforced by the create/edit POST
227    /// handlers (unknown keys → 400): record handlers already whitelist via
228    /// per-field `.get(..)`, but a generic impl iterating `values` would
229    /// silently promote `role`/`tenant_id`/handler keys (`csrf_token`,
230    /// `confirm`, `ids`) to client-controlled writes. The transport keys the
231    /// handlers own (`csrf_token`, `clear_<field>`, `keep_<field>`) are
232    /// stripped before the record fns run (GH #148), so a generic impl cannot
233    /// promote those either; `confirm`/`ids` are only read, never written.
234    /// Callers should reject or ignore the rest (at least `debug_assert!` in
235    /// tests); handler keys must be filtered by the caller before calling this.
236    pub fn unknown_keys(&self, values: &HashMap<String, String>) -> Vec<String> {
237        use std::collections::HashSet;
238        let known: HashSet<String> = self.field_names().into_iter().collect();
239        let mut out: Vec<String> = values
240            .keys()
241            .filter(|k| !known.contains(k.as_str()))
242            .cloned()
243            .collect();
244        out.sort();
245        out
246    }

248    fn assert_unique_field_names(&self) {
249        let names = self.field_names();
250        let mut seen = std::collections::HashSet::new();
251        for name in names {
252            assert!(
253                seen.insert(name.clone()),
254                "duplicate field name '{name}': each Schema input needs a distinct field (GH #100)"
255            );
256        }

259    /// Every leaf `pick` selects, keyed by field name — the one walk behind the
260    /// per-kind accessors (GH #209). `pick` answers a node's `(field name,
261    /// leaf)`, or `None` when the node is not that kind.
262    fn leaves<T>(&self, pick: impl Fn(&Node) -> Option<(&str, T)>) -> HashMap<String, T> {
263        let mut map = HashMap::new();
264        for node in &self.nodes {
265            for_each_field(node, &mut |n| {
266                if let Some((name, leaf)) = pick(n) {
267                    map.insert(name.to_string(), leaf);
268                }
269            });
270        }
271        map
272    }

298    /// Whether this schema (including nested Section/Group/Grid/Repeater/Tabs)
299    /// contains a [`FileUpload`]. `Panel` uses it to emit
300    /// `enctype="multipart/form-data"` only on forms that need it (GH #73).
301    pub fn has_file_upload(&self) -> bool {
302        let mut found = false;
303        for node in &self.nodes {
304            for_each_field(node, &mut |n| {
305                if matches!(n, Node::FileUpload(_)) {
306                    found = true;
307                }
308            });
309        }
310        found
311    }

313    /// Validate submitted values against declared inputs (GH #89).
314    ///
315    /// Absent keys are treated as `""` for validation; update record fns must
316    /// therefore only write keys present in the submission, or an omitted
317    /// optional field silently blanks the stored value. Use
318    /// [`Self::unknown_keys`] to allow-list POST keys.
319    ///
320    /// A field a submission hides is not validated (GH #297): an all-empty
321    /// Repeater group is absent (GH #147), and a variant group the submission's
322    /// discriminant does not name is not rendered by `variant.js`, so neither
323    /// can fail the submit for a value the user cannot see.
324    pub fn validate(&self, values: &HashMap<String, String>) -> HashMap<String, Vec<String>> {
325        // Classify the groups first (GH #147, GH #297): an all-empty group is
326        // "absent" — an untouched group submits empty strings (or omits the
327        // keys), both treated as absent — so its inner inputs must not fail
328        // the submit for any requiredness. A `required` repeater answers with
329        // its one label-keyed error instead, and required repeaters nested
330        // inside an absent group are suppressed with it. A partially filled
331        // group (any inner value non-empty) enforces inner `required` as
332        // usual. Whitespace-only values count as empty, matching the
333        // codebase-wide trim convention. A variant group the submission's
334        // discriminant does not name is hidden with its subtree, which is what
335        // makes validation agree with the render (GH #297).
336        let mut errors: HashMap<String, Vec<String>> = HashMap::new();
337        let mut skip: HashSet<String> = HashSet::new();
338        walk_absent_groups(&self.nodes, values, &mut skip, &mut errors, false);
339        // One walk, one match per node (`validate_leaf`): the single place a
340        // field kind joins validation (GH #209).
341        for node in &self.nodes {
342            for_each_field(node, &mut |n| {
343                let Some((name, errs)) = validate_leaf(n, values) else {
344                    return;
345                };
346                if !skip.contains(name) && !errs.is_empty() {
347                    errors.insert(name.to_string(), errs);
348                }
349            });
350        }
351        errors
352    }

354    /// Field names a submission leaves out of validation (GH #147, GH #297):
355    /// the same classification `validate` uses — an all-empty Repeater group
356    /// is absent, and a variant group the discriminant does not name is hidden
357    /// — minus the required-group errors, which validation already reported.
358    /// `check_unique` consults it so an untouched group is never unique-checked
359    /// while validation calls it clean, and `validate_async` so a hidden
360    /// group's select is not probed for existence.
361    pub(crate) fn absent_fields(&self, values: &HashMap<String, String>) -> HashSet<String> {
362        let mut skip = HashSet::new();
363        let mut discarded = HashMap::new();
364        walk_absent_groups(&self.nodes, values, &mut skip, &mut discarded, false);
365        skip
366    }

368    /// Async validation for Select relationship existence (tenancy-aware).
369    ///
370    /// A field `validate` skipped is skipped here too (GH #297): an absent
371    /// repeater group or a hidden variant group holds no value the user can
372    /// see, so its select must not be probed for existence.
373    pub async fn validate_async(
374        &self,
375        cx: &Cx,
376        values: &HashMap<String, String>,
377    ) -> HashMap<String, Vec<String>> {
378        let mut errors = self.validate(values);
379        let absent = self.absent_fields(values);
380        for (name, sel) in self.select_inputs() {
381            if errors.contains_key(&name) || absent.contains(&name) {
382                continue;
383            }
384            if sel.relationship.is_some() || !sel.options_static.is_empty() {
385                let val = values.get(&name).map(|s| s.as_str()).unwrap_or("");
386                if !val.trim().is_empty() {
387                    let async_errs = sel.validate_async(cx, val).await;
388                    // validate_async returns required errs plus existence; we already did required,
389                    // so filter.
390                    let existence_errs: Vec<String> = async_errs
391                        .into_iter()
392                        .filter(|e| !e.contains("is required"))
393                        .collect();
394                    if !existence_errs.is_empty() {
395                        errors.insert(name, existence_errs);
396                    }
397                }
398            }
399        }
400        errors
401    }

469    /// GH #297: a `Select`'s presence and option checks read `value.trim()`,
470    /// so the trimmed spelling is the one validation authorises. Normalisation
471    /// writes exactly that value, and a padded value no option matches is still
472    /// refused rather than trimmed into one.
473    #[tokio::test]
474    async fn a_select_stores_the_value_its_check_authorised() {
475        let cx = topcoat::context::CxTestBuilder::new().build();
476        let schema = Schema::new(
477            Select::r#for(DummyUser::fields().name())
478                .options(vec!["red".to_string(), "blue".to_string()]),
479        );
480
481        let mut values = HashMap::new();
482        values.insert("name".to_string(), "  red ".to_string());
483        assert!(
484            schema.validate_async(&cx, &values).await.is_empty(),
485            "the option check reads the trimmed value"
486        );
487        schema.normalize_values(&mut values);
488        assert_eq!(
489            values.get("name").map(String::as_str),
490            Some("red"),
491            "the stored value is the one the check authorised"
492        );
493
494        let mut invalid = HashMap::new();
495        invalid.insert("name".to_string(), "  re d ".to_string());
496        assert!(
497            !schema.validate_async(&cx, &invalid).await.is_empty(),
498            "trimming does not turn a non-option into one"
