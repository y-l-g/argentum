1//! Unified Schema primitive — layout blocks that compose via `view!`.
2//!
3//! `Schema` is a container for `Section`, `Group`, `Grid`, `Tabs` and
4//! `Repeater` nodes. Each node renders through Topcoat's `view!` macro;
5//! `Schema::render` combines them. The API mirrors Filament's
6//! `Schema::new(( ... ))` tuple form via the `IntoSchema` trait.
7//!
8//! Bridge note: `lens_field` is the one walk reaching into `toasty_core`
9//! (upstream issue #114), alongside the `pk_*` bridge helpers and `cursor.rs`
10//! cursor values. It hands back the built `app::Field`, so field metadata no
11//! longer needs a helper per property; uniqueness comes from
12//! `lens_field_unique`, since Toasty keeps it on the model's index list rather
13//! than the field. Retire the walk when Toasty exposes it (upstream #183).

63    ///
64    /// Panics on duplicate field names (GH #100): two inputs sharing one name

79    pub async fn render<'a>(&self, cx: &'a Cx) -> Result<BoxView<'a>> {
80        self.render_with(cx, &HashMap::new(), &HashMap::new()).await
81    }

83    /// Render the schema read-only (GH #187): the detail page's side of the
84    /// same declaration.
85    ///
86    /// Every field shows the record's stored value in place of its control, so
87    /// a detail page reuses the field types and layout blocks a form already
88    /// declares rather than a parallel infolist vocabulary. `values` is the
89    /// record hydrated exactly as the edit form hydrates it
90    /// ([`Resource::hydrate_form_values`](crate::resource::Resource::hydrate_form_values)):
91    /// what a user reads on the page is what the form would have shown them.
92    pub async fn render_readonly<'a>(
93        &self,
94        cx: &'a Cx,
95        values: &HashMap<String, String>,
96    ) -> Result<BoxView<'a>> {
97        self.render_source(
98            cx,
99            &RenderSource {
100                values,
101                errors: &HashMap::new(),
102                mode: Mode::View,
103            },
104        )
105        .await
106    }

108    /// Rewrite submitted values into their fields' stored spelling (GH #192).
109    ///
110    /// Runs after validation and before a record fn sees the map, so a typed
111    /// field's `Display` — not the browser's spelling — is what gets written.
112    /// That is what makes an untouched edit round-trip: the form hydrates a
113    /// stored value, the browser echoes it, and this puts back the same string
114    /// the record fn would have produced rather than a re-spelling of it.
115    ///
116    /// A value the caller has not validated cannot be normalised, so a parse
117    /// failure here leaves the submission untouched and reports nothing: it is
118    /// unreachable from the handlers (validation refuses it first), and a
119    /// silent rewrite would hide a bypass rather than surface it. A field with
120    /// no submission keeps its absence — an update writes only present keys.
121    ///
122    /// An **empty** submission is left empty for the same reason (GH #192): a
123    /// typed column has no spelling for "no value" — `""` is not an `i64` and
124    /// not a `Timestamp` — so inventing one here would put a value in a record
125    /// the user never gave. Empty is the presence rule's business, which is
126    /// where the panel already answers it: `.required()` refuses it inline, and
127    /// an optional typed field reaches its record fn as `""` for the record fn's
128    /// own default. The record fn therefore reads a typed field through the same
129    /// "present or absent" check it uses for any other optional column.
130    pub fn normalize_values(&self, values: &mut HashMap<String, String>) {
131        for (name, input) in self.text_inputs() {
132            let Some(submitted) = values.get(&name) else {
133                continue;
134            };
135            if submitted.trim().is_empty() {
136                continue;
137            }
138            if let Ok(normalized) = input.normalize(submitted) {
139                values.insert(name, normalized);
140            }
141        }
142    }

144    /// Append another schema's nodes after this one's (GH #191).
145    ///
146    /// [`Schema::new`] composes through `IntoSchema`, whose tuple form stops at
147    /// four nodes; a derived embedded form has one control per leaf column and
148    /// composes nested values, so it builds its schema by appending instead. The
149    /// nodes keep their order, so a form reads in declaration order either way.
150    pub fn extend(mut self, other: Schema) -> Schema {
151        self.nodes.extend(other.nodes);
152        // The same guard `Schema::new` runs: a derived form is built by
153        // appending, so this is the only check for the shapes `new` cannot see.
154        self.assert_unique_field_names();
155        self
156    }

176    /// The one node walk: every node renders its static form.
177    pub(crate) async fn render_source<'a>(
178        &self,
179        cx: &'a Cx,
180        source: &RenderSource<'_>,
181    ) -> Result<BoxView<'a>> {
182        let mut views = Vec::with_capacity(self.nodes.len());
183        for node in &self.nodes {
184            views.push(Box::pin(node.render_source(cx, source)).await?.boxed());
185        }
186        Ok(view! {
187            cx =>
188            for v in views {
189                (v)
190            }
191        }
192        .boxed())
193    }

195    /// Collect field names for validation (TextInput + Textarea + Select + FileUpload).
196    pub fn field_names(&self) -> Vec<String> {
197        let mut out = Vec::new();
198        for node in &self.nodes {
199            for_each_field(node, &mut |n| match n {
200                Node::TextInput(f) => out.push(f.field_name().to_string()),
201                Node::Textarea(f) => out.push(f.field_name().to_string()),
202                Node::Select(f) => out.push(f.field_name().to_string()),
203                Node::FileUpload(f) => out.push(f.field_name().to_string()),
204                _ => {}
205            });
206        }
207        out
208    }

210    /// Keys in `values` that no declared input owns, sorted (GH #89).
211    ///
212    /// Framework-level allow-list seam, enforced by the create/edit POST
213    /// handlers (unknown keys → 400): record handlers already whitelist via
214    /// per-field `.get(..)`, but a generic impl iterating `values` would
215    /// silently promote `role`/`tenant_id`/handler keys (`csrf_token`,
216    /// `confirm`, `ids`) to client-controlled writes. The transport keys the
217    /// handlers own (`csrf_token`, `clear_<field>`) are stripped before the
218    /// record fns run (GH #148), so a generic impl cannot promote those
219    /// either; `confirm`/`ids` are only read, never written. Callers should
220    /// reject or ignore the rest (at least `debug_assert!` in tests); handler
221    /// keys must be filtered by the caller before calling this.
222    pub fn unknown_keys(&self, values: &HashMap<String, String>) -> Vec<String> {
223        use std::collections::HashSet;
224        let known: HashSet<String> = self.field_names().into_iter().collect();
225        let mut out: Vec<String> = values
226            .keys()
227            .filter(|k| !known.contains(k.as_str()))
228            .cloned()
229            .collect();
230        out.sort();
231        out
232    }

234    fn assert_unique_field_names(&self) {
235        let names = self.field_names();
236        let mut seen = std::collections::HashSet::new();
237        for name in names {
238            assert!(
239                seen.insert(name.clone()),
240                "duplicate field name '{name}': each Schema input needs a distinct field (GH #100)"
241            );
242        }
243    }

245    /// Every leaf `pick` selects, keyed by field name — the one walk behind the
246    /// per-kind accessors (GH #209). `pick` answers a node's `(field name,
247    /// leaf)`, or `None` when the node is not that kind.
248    fn leaves<T>(&self, pick: impl Fn(&Node) -> Option<(&str, T)>) -> HashMap<String, T> {
249        let mut map = HashMap::new();
250        for node in &self.nodes {
251            for_each_field(node, &mut |n| {
252                if let Some((name, leaf)) = pick(n) {
253                    map.insert(name.to_string(), leaf);
254                }
255            });
256        }
257        map
258    }

284    /// Whether this schema (including nested Section/Group/Grid/Repeater/Tabs)
285    /// contains a [`FileUpload`]. `Panel` uses it to emit
286    /// `enctype="multipart/form-data"` only on forms that need it (GH #73).
287    pub fn has_file_upload(&self) -> bool {
288        let mut found = false;
289        for node in &self.nodes {
290            for_each_field(node, &mut |n| {
291                if matches!(n, Node::FileUpload(_)) {
292                    found = true;
293                }
294            });
295        }
296        found
297    }

299    /// Validate submitted values against declared inputs (GH #89).
300    ///
301    /// Absent keys are treated as `""` for validation; update record fns must
302    /// therefore only write keys present in the submission, or an omitted
303    /// optional field silently blanks the stored value. Use
304    /// [`Self::unknown_keys`] to allow-list POST keys.
305    pub fn validate(&self, values: &HashMap<String, String>) -> HashMap<String, Vec<String>> {
306        // Classify the repeaters first (GH #147): an all-empty group is
307        // "absent" — an untouched group submits empty strings (or omits the
308        // keys), both treated as absent — so its inner inputs must not fail
309        // the submit for any requiredness. A `required` group answers with
310        // its one label-keyed error instead, and required repeaters nested
311        // inside an absent group are suppressed with it. A partially filled
312        // group (any inner value non-empty) enforces inner `required` as
313        // usual. Whitespace-only values count as empty, matching the
314        // codebase-wide trim convention.
315        let mut errors: HashMap<String, Vec<String>> = HashMap::new();
316        let mut skip: HashSet<String> = HashSet::new();
317        walk_repeater_absence(&self.nodes, values, &mut skip, &mut errors, false);
318        // One walk, one match per node (`validate_leaf`): the single place a
319        // field kind joins validation (GH #209).
320        for node in &self.nodes {
321            for_each_field(node, &mut |n| {
322                let Some((name, errs)) = validate_leaf(n, values) else {
323                    return;
324                };
325                if !skip.contains(name) && !errs.is_empty() {
326                    errors.insert(name.to_string(), errs);
327                }
328            });
329        }
330        errors
331    }

333    /// Field names hidden inside absent Repeater groups for these values
334    /// (GH #147): the same classification `validate` uses — an all-empty
335    /// group is "absent" — minus the required-group errors, which validation
336    /// already reported. `check_unique` consults it so an untouched group is
337    /// never unique-checked while validation calls it clean.
338    pub(crate) fn absent_repeater_fields(
339        &self,
340        values: &HashMap<String, String>,
341    ) -> HashSet<String> {
342        let mut skip = HashSet::new();
343        let mut discarded = HashMap::new();
344        walk_repeater_absence(&self.nodes, values, &mut skip, &mut discarded, false);
345        skip
346    }

348    /// Async validation for Select relationship existence (tenancy-aware).
349    pub async fn validate_async(
350        &self,
351        cx: &Cx,
352        values: &HashMap<String, String>,
353    ) -> HashMap<String, Vec<String>> {
354        let mut errors = self.validate(values);
355        for (name, sel) in self.select_inputs() {
356            if errors.contains_key(&name) {
357                continue;
358            }
359            if sel.relationship.is_some() || !sel.options_static.is_empty() {
360                let val = values.get(&name).map(|s| s.as_str()).unwrap_or("");
361                if !val.trim().is_empty() {
362                    let async_errs = sel.validate_async(cx, val).await;
363                    // validate_async returns required errs plus existence; we already did required,
364                    // so filter.
365                    let existence_errs: Vec<String> = async_errs
366                        .into_iter()
367                        .filter(|e| !e.contains("is required"))
368                        .collect();
369                    if !existence_errs.is_empty() {
370                        errors.insert(name, existence_errs);
371                    }
372                }
373            }
374        }
375        errors
376    }

395    fn has_file_upload_detects_nested() {
396        #[derive(Debug, toasty::Model)]
397        struct Doc {
398            #[key]
399            #[auto]
400            id: uuid::Uuid,
401            path: String,
402            title: String,
403        }
404        let plain = Schema::new(TextInput::r#for(DummyUser::fields().name()));
405        assert!(!plain.has_file_upload());
406        let direct = Schema::new(FileUpload::r#for(Doc::fields().path()));
407        assert!(direct.has_file_upload());
408        // Nested inside Section/Grid/Repeater counts.
409        let nested = Schema::new(Section::new("S").schema(Grid::new(2).schema((
410            TextInput::r#for(DummyUser::fields().name()),
411            FileUpload::r#for(Doc::fields().path()),
412        ))));
413        assert!(nested.has_file_upload());
414        let in_repeater =
415            Schema::new(Repeater::new("R").schema(FileUpload::r#for(Doc::fields().path())));
416        assert!(in_repeater.has_file_upload());
417    }
