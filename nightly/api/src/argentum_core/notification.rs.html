1//! `Notification` — transient user-visible message (CONTEXT.md).
2//!
3//! Produced by an `Action`'s result and rendered in the `Panel` shell's
4//! top-level boundary so it survives `Table` swaps. Status + title (plus an
5//! optional description), auto-dismissed after ~4s by
6//! `argentum-ui/assets/notifications.js` and dismissible through the toast's
7//! close button — the shadcn/Sonner toast surface (GH #151).
8//!
9//! The flash cookie is Topcoat's `CookieStore` (serde JSON, GH #139); the jar
10//! defaults carry the hardened attributes (HttpOnly, Secure, SameSite=Lax,
11//! Path=/ — the `__Host-` name requires them, GH #149) on writes and removals
12//! alike, so set and clear cannot drift. One-time semantics ride the cookie
13//! alone: Topcoat flushes `Set-Cookie` on error responses too (topcoat#408), so
14//! the mutation `Err` redirects carry the flash in the cookie alone.
15

28/// The kind of notification (status).
29///
30/// The serde tokens are lowercase so the JSON cookie reads naturally (GH #139).
31#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
32#[serde(rename_all = "lowercase")]
33pub enum NotificationStatus {
34    Success,
35    Error,
36    Info,
37    Warning,
38}
39
40impl NotificationStatus {
41    /// The lowercase token, shared with the flash cookie and `data-type`.
42    pub fn as_str(&self) -> &'static str {
43        match self {
44            Self::Success => "success",
45            Self::Error => "error",
46            Self::Info => "info",
47            Self::Warning => "warning",
48        }
49    }
50}

52/// A transient message shown after a mutation.
53#[derive(Debug, Clone, Serialize, Deserialize)]
54pub struct Notification {
55    pub status: NotificationStatus,
56    pub title: String,
57    /// Optional supporting line under the title, rendered as the toast
58    /// description. Absent (`None`) keeps the GH #139 cookie wire format, so
59    /// cookies written before the field existed still decode.
60    #[serde(default, skip_serializing_if = "Option::is_none")]
61    pub description: Option<String>,
62}
63
64impl Notification {
65    pub fn success(title: impl Into<String>) -> Self {
66        Self::new(NotificationStatus::Success, title)
67    }
68
69    pub fn error(title: impl Into<String>) -> Self {
70        Self::new(NotificationStatus::Error, title)
71    }
72
73    pub fn info(title: impl Into<String>) -> Self {
74        Self::new(NotificationStatus::Info, title)
75    }
76
77    pub fn warning(title: impl Into<String>) -> Self {
78        Self::new(NotificationStatus::Warning, title)
79    }
80
81    fn new(status: NotificationStatus, title: impl Into<String>) -> Self {
82        Self {
83            status,
84            title: title.into(),
85            description: None,
86        }
87    }
88
89    /// Attach the supporting line under the title.
90    pub fn description(mut self, description: impl Into<String>) -> Self {
91        self.description = Some(description.into());
92        self
93    }
94}

96pub(crate) const COOKIE_NAME: &str = "__Host-argentum_notification";

98/// The jar defaults the flash cookie relies on (GH #139): `CookieStore`
99/// commits a bare cookie, so the hardened attributes live here and apply to
100/// writes *and* removals alike — the `Map` adapter transforms both. The
101/// `__Host-` name requires Secure + Path=/ + no Domain (GH #149); the
102/// session and CSRF cookies set their own attributes and are unaffected
103/// (`default_*` only fills what is unset).
104fn hardened(jar: &CookieJar) -> impl Cookies + '_ {
105    jar.default_path("/")
106        .default_http_only(true)
107        .default_secure(true)
108        .default_same_site(topcoat::cookie::SameSite::Lax)
109}
110
111/// Store a notification for the next request (flash).
112pub fn set_notification(cx: &Cx, notification: Notification) {
113    if try_request_context::<CookieJarCell>(cx).is_none() {
114        return;
115    }
116    // `commit` serializes to JSON and queues the Set-Cookie — one hand-rolled
117    // wire format less (GH #139). A failed commit must not fail the mutation
118    // it rides on, but swallowing it retries a completed write with no toast
119    // (GH #174) — log it for operators instead.
120    if let Err(error) = cookie_store::<Notification, _>(hardened(cookies(cx)), COOKIE_NAME)
121        .set(notification)
122        .commit()
123    {
124        tracing::error!(error = %error, "flash notification commit failed");
125    }
126}

128/// Flash the failure of a write that passed validation but did not land
129/// (GH #174).
130///
131/// Every mutation handler (create, update, delete, bulk delete) routes its
132/// write and commit failures through this before returning the error, so the
133/// user gets "couldn't …" instead of a bare 500 that leaves them guessing
134/// whether the write went through. The title names the operation, never the
135/// driver's text — internals stay in the server log (GH #174 §1).
136///
137/// Delivery: the flash rides a `Set-Cookie`, and Topcoat's cookie layer writes
138/// pending cookies on **both** paths — on `Err` it stashes them in
139/// `response_headers`, which the router applies once the error response exists
140/// (`topcoat-router/src/router.rs`), so the 500 response carries the cookie. Its
141/// body is Topcoat's plain-text error page, which renders no toast: the message
142/// appears on the next panel page, whose shell consumes the flash
143/// ([`take_notification`]). `a_failed_write_toasts_on_the_next_panel_page` pins
144/// that delivery.
145pub(crate) fn notify_write_failure(cx: &Cx, action: &str) {
146    set_notification(
147        cx,
148        Notification::error(format!("Couldn't {action}"))
149            .description("Nothing was changed — try again."),
150    );
151}

153/// Take the notification from the request (if present) and clear it.
154pub fn take_notification(cx: &Cx) -> Option<Notification> {
155    try_request_context::<CookieJarCell>(cx)?;
156    let unparsed = cookie_store::<Notification, _>(hardened(cookies(cx)), COOKIE_NAME);
157    match unparsed.parse() {
158        // Present and readable: hand it out, then expire it (the removal
159        // carries the same hardened attributes through the jar defaults).
160        Ok(Some(store)) => {
161            let notification = store.get();
162            store.remove();
163            Some(notification)
164        }
165        // Unreadable garbage (a corrupted or foreign value): expire it, no
166        // toast.
167        Err(_) => {
168            cookie_store::<Notification, _>(hardened(cookies(cx)), COOKIE_NAME).remove();
169            None
170        }
171        Ok(None) => None,
172    }
173}

175/// Render one notification as the shadcn/Sonner toast (GH #151).
176///
177/// The status picks Sonner's icon under shadcn's theming (`circle-check`,
178/// `info`, `triangle-alert`, `octagon-x`); the error icon reads
179/// `text-destructive` so a failure cannot look like an info toast.
180///
181/// `attrs` are merged onto the toast surface. The shell's flash stack passes
182/// an empty set; the live transport ([`live_toaster`]) adds a per-mount `id`
183/// so re-rendering the same variant replaces the mounted toast instead of
184/// re-syncing its `data-mounted` state (GH #154 §3).
185pub async fn render_notification<'a>(
186    cx: &'a Cx,
187    notification: Notification,
188    attrs: Attributes,
189) -> topcoat::Result<BoxView<'a>> {
190    let Notification {
191        status,
192        title,
193        description,
194    } = notification;
195    let icon_data = match status {
196        NotificationStatus::Success => icons::CIRCLE_CHECK,
197        NotificationStatus::Error => icons::OCTAGON_X,
198        NotificationStatus::Info => icons::INFO,
199        NotificationStatus::Warning => icons::TRIANGLE_ALERT,
200    };
201    let icon_class = if status == NotificationStatus::Error {
202        "size-4 text-destructive"
203    } else {
204        "size-4"
205    };
206    let status = status.as_str();
207    Ok(view! {
208        cx =>
209        toast(
210            attrs: attributes! { data-type=(status) (attrs) },
211            toast_icon(icon(data: icon_data, attrs: attributes! { class=(icon_class) }))
212            toast_content(
213                toast_title((title))
214                if let Some(description) = description {
215                    toast_description((description))
216                }
217            )
218            toast_close()
219        )
220    }
221    .boxed())
222}

224/// The signals a page owns to mount a [`Notification`] in place (GH #154 §3).
225///
226/// [`live_toast`] creates them; a click handler writes a procedure's returned
227/// `(status, title, description)` into them and bumps `serial` (a repeat of
228/// the same variant must still be a change), and the shell's [`live_toaster`]
229/// shard reads them and mounts the real Sonner surface — no navigation, no
230/// scroll reset. Every value is client input by the time the shard reads it.
231#[derive(Clone)]
232pub struct LiveToast {
233    /// The Sonner status token (`success`/`info`/`warning`/`error`); empty
234    /// means "no toast mounted".
235    pub status: Signal<String>,
236    /// The toast title.
237    pub title: Signal<String>,
238    /// The optional supporting line; empty renders no description.
239    pub description: Signal<String>,
240    /// Bumped on every mount so an identical repeat still re-renders.
241    pub serial: Signal<u64>,
242}
243
244/// The live toast signals for this request (call once per page).
245///
246/// The signals are created here, so both the page's handlers and the shell's
247/// [`live_toaster`] resolve the same handles in one request.
248pub fn live_toast(cx: &Cx) -> LiveToast {
249    LiveToast {
250        status: signal(cx, String::new),
251        title: signal(cx, String::new),
252        description: signal(cx, String::new),
253        serial: signal(cx, || 0u64),
254    }
255}

257/// The endpoint [`live_toaster`] is served at.
258///
259/// A shard that declares no path is served at a build-random one (topcoat#441);
260/// naming it keeps the endpoint stable across builds and legible in logs and
261/// tests. The `/argentum-` prefix separates it from a generated path, and the
262/// whole path stays under `/_topcoat/runtime`, so the panel's auth gate covers
263/// the shard and an unauthenticated rerun answers 401 rather than redirecting to
264/// the login page.
265///
266/// The literal in [`live_toaster`]'s attribute is the same path;
267/// `live_toaster_endpoint_is_the_named_path` pins the two together.
268#[cfg(test)]
269pub(crate) const LIVE_TOASTER_PATH: &str = "/_topcoat/runtime/shards/argentum-live-toaster";
270
271/// The shell's live toaster shard (GH #154 §3): reads the page's
272/// [`LiveToast`] signals and mounts the toast in place when one is set.
273///
274/// A shard rather than an eager read, so writing the signals re-renders only
275/// this stack — the page, its scroll, and its focus stay put. The mount `id`
276/// carries the serial: the browser's morph matches by id, so a new mount is a
277/// fresh toast node (with `data-mounted="false"`) that `notifications.js`
278/// arms, instead of rewriting the already-mounted one.
279///
280/// The body lives in `render_live_toaster`: the shard macro's generated
281/// handler cannot name the request lifetime its `impl View` would capture, so
282/// the helper resolves the boxed view and the shard only forwards it.
283#[shard("/_topcoat/runtime/shards/argentum-live-toaster")]
284pub async fn live_toaster(
285    cx: &Cx,
286    status: Signal<String>,
287    title: Signal<String>,
288    description: Signal<String>,
289    serial: Signal<u64>,
290) -> topcoat::Result<impl View> {
291    render_live_toaster(cx, &status, &title, &description, &serial).await
292}
293
294async fn render_live_toaster<'a>(
295    cx: &'a Cx,
296    status: &Signal<String>,
297    title: &Signal<String>,
298    description: &Signal<String>,
299    serial: &Signal<u64>,
300) -> topcoat::Result<BoxView<'a>> {
301    // Runtime endpoints bypass page guards (topcoat shard contract), so the
302    // shard restates the panel gate; a slot with no live toast needs no auth
303    // to render empty, but a direct POST must not mount toasts unauthenticated.
304    #[cfg(feature = "auth")]
305    if crate::auth::enforced(cx) {
306        crate::auth::require_authenticated(cx)?;
307    }
308    let status = status.get();
309    let mount = serial.get();
310    if status.is_empty() {
311        // No `<span>` placeholder (GH #160): the shell mounts this slot inside
312        // the toaster `<ol>`, which permits only `li`/`script`/`template`
313        // children — the empty view renders nothing.
314        return Ok(().boxed());
315    }
316    let title = title.get();
317    let description = description.get();
318    let notification = match status.as_str() {
319        "success" => Notification::success(title),
320        "warning" => Notification::warning(title),
321        "error" => Notification::error(title),
322        _ => Notification::info(title),
323    };
324    let notification = if description.is_empty() {
325        notification
326    } else {
327        notification.description(description)
328    };
329    let mount_id = format!("live-toast-{mount}");
330    render_notification(cx, notification, attributes! { cx => id=(mount_id) }).await
331}

453    /// The consumed flash cookie must be cleared with a `__Host-`-conformant
454    /// removal (GH #149): a `__Host-`-named `Set-Cookie` without `Secure` is
455    /// ignored by browsers — `Max-Age=0` deletions included — so the flash
456    /// would survive every navigation. Pinned here through topcoat's own
457    /// response finalization; the create/edit flow end-to-end is covered by
458    /// the panel test `mutation_redirect_carries_the_flash_cookie_instead_of_a_query`.
459    #[test]
460    fn notification_removal_header_carries_the_host_prefix_contract() {
461        use http::header::SET_COOKIE;
462
463        let enc = serde_json::to_string(&Notification::success("hello")).unwrap();
464        let cx = cx_with_cookie(COOKIE_NAME, Some(&enc));
465        let n = take_notification(&cx);
466        assert!(n.is_some(), "the flash cookie must decode");
467
468        let mut headers = http::HeaderMap::new();
469        topcoat::cookie::write_cookies(&cx, &mut headers);
470        let cleared = headers
471            .get_all(SET_COOKIE)
472            .iter()
473            .filter_map(|v| v.to_str().ok())
474            .find(|v| v.starts_with(&format!("{COOKIE_NAME}=")))
475            .expect("the consumed flash cookie must be cleared")
476            .to_string();
477        assert!(
478            cleared.contains("Max-Age=0") || cleared.contains("Expires=Thu, 01 Jan 1970"),
479            "the removal header must expire the cookie: {cleared}"
480        );
481        assert!(
482            cleared.contains("Secure") && cleared.contains("Path=/"),
483            "the removal header must satisfy the __Host- contract: {cleared}"
484        );
485        assert!(
486            cleared.split(';').all(|attr| {
487                let attr = attr.trim();
488                !attr.starts_with("Domain=")
489            }),
490            "a `__Host-` cookie must not carry a Domain: {cleared}"
491        );
492    }
