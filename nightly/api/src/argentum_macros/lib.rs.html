8/// Derive `EmbeddedForm` for an embedded struct or enum.
9///
10/// The generated impl converts the value to and from the panel's flat form map,
11/// answers whether a submission mentions it, and generates a `form(cx, parent)`
12/// returning the value's controls, all driven by the columns the app schema
13/// resolves for the parent path. The framework supplies the storage names, the
14/// derive supplies the Rust shape.
15///
16/// ```ignore
17/// #[derive(Debug, Clone, toasty::Embed, argentum_core::EmbeddedForm)]
18/// pub enum Publication {
19///     #[column(variant = 1)]
20///     Scheduled {
21///         #[shared(timestamp)]
22///         #[form(label = "Publication timestamp")]
23///         scheduled_at: String,
24///         scheduled_for: String,
25///     },
26///     #[column(variant = 2)]
27///     Published { #[shared(timestamp)] published_at: String, canonical_url: String },
28/// }
29///
30/// // form declaration — no field bindings written by hand
31/// Section::new("Publication").schema(Publication::form(cx, Post::fields().publication()))
32/// ```
33///
34/// # How a field is classified
35///
36/// A type this panel can spell — `String`, the integer family, `bool`,
37/// `f32`/`f64`, `Uuid`, `jiff::Timestamp` — is a **leaf**: one column, read and
38/// written as text (typed leaves parse through `TypedValue`). Any other type is
39/// another **embedded value**, delegated to that type's own `EmbeddedForm`. A
40/// relation, an `Option<T>`, a `Vec<T>` and a `#[document]` inside a value do
41/// not compile, or are refused at the schema (the `argentum-core`
42/// `schema::embedded` module docs list what is not covered).
43///
44/// # Which variant an enum reads
45///
46/// A named discriminant always wins, and an undeclared one is refused loudly;
47/// otherwise the first variant, in declaration order, with a **payload of its
48/// own** submitted — a `#[shared(..)]` column belongs to several variants and
49/// never selects one; otherwise the first variant.
50///
51/// # Per-field overrides
52///
53/// - `#[form(label = "Canonical URL")]` — the control's label (default: the field name, humanized).
54/// - `#[form(textarea)]` / `#[form(textarea, rows = 3)]` — a multi-line control for a `String`
55///   leaf, and its height.
56///
57/// Anything else in `#[form(..)]` is a compile error.
58#[proc_macro_derive(EmbeddedForm, attributes(form))]
59pub fn embedded_form(input: TokenStream) -> TokenStream {
60    let input = syn::parse_macro_input!(input as DeriveInput);
61    embedded::expand(input)
62}
