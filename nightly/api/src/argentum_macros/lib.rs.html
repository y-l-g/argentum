8/// Derive `EmbeddedForm` for an embedded struct or enum (GH #191).
9///
10/// The generated impl converts the value to and from the panel's flat form map,
11/// answers whether a submission mentions it, and generates a `form(cx, parent)`
12/// function returning the value's controls — all driven by the columns the app
13/// schema resolves for the parent path. The framework supplies the storage
14/// names, this derive supplies the Rust shape, so no flattened name is ever
15/// spelled by hand.
16///
17/// ```ignore
18/// #[derive(Debug, Clone, toasty::Embed, argentum_core::EmbeddedForm)]
19/// pub enum Publication {
20///     #[column(variant = 1)]
21///     Scheduled {
22///         #[shared(timestamp)]
23///         #[form(label = "Publication timestamp")]
24///         scheduled_at: String,
25///         scheduled_for: String,
26///     },
27///     #[column(variant = 2)]
28///     Published {
29///         #[shared(timestamp)]
30///         published_at: String,
31///         canonical_url: String,
32///     },
33/// }
34///
35/// // form declaration — no field bindings written by hand
36/// Section::new("Publication").schema(Publication::form(cx, Post::fields().publication()))
37///
38/// // hydration and the record fn
39/// write_embedded(cx, Post::fields().publication(), &record.publication, &mut values);
40/// let publication = read_embedded(cx, Post::fields().publication(), &values);
41/// ```
42///
43/// # How a field is classified
44///
45/// A field whose type is a Rust primitive this panel can spell — `String`, the
46/// integer family, `bool`, `f32`/`f64`, `Uuid`, `jiff::Timestamp` — is a
47/// **leaf**: one column, read and written as text (typed leaves parse through
48/// `TypedValue`, GH #192). Any other type is another **embedded value**,
49/// delegated to that type's own `EmbeddedForm` impl, so nesting works by
50/// deriving on each type. A relation, an `Option<T>`, a `Vec<T>`, and a
51/// `#[document]` inside a value do not compile, or are refused at the schema
52/// (see the `argentum-core` module docs).
53///
54/// # Which variant an enum reads
55///
56/// The discriminant column decides, in this order:
57///
58/// 1. a discriminant the submission **names** — always wins, and one the enum does not declare is
59///    refused loudly rather than read as some other variant;
60/// 2. otherwise (the create form, a hand-written POST) the first variant, in declaration order,
61///    with a **payload of its own** submitted — a `#[shared(..)]` column belongs to several
62///    variants and so never selects one;
63/// 3. otherwise the first variant.
64///
65/// # Per-field overrides
66///
67/// - `#[form(label = "Canonical URL")]` — the control's label (default: the field name, humanized).
68/// - `#[form(textarea)]` / `#[form(textarea, rows = 3)]` — a multi-line control for a `String`
69///   leaf, and its height.
70///
71/// Anything else in `#[form(..)]` is a compile error.
72///
73/// # Not covered
74///
75/// A `#[document]` **inside** an embedded value (its inner fields share one
76/// column, so no per-field binding exists), a relation inside one, an embedded
77/// enum nested **inside an enum variant** (value resolution starts at a model
78/// root), and a tuple or unit struct.
79#[proc_macro_derive(EmbeddedForm, attributes(form))]
80pub fn embedded_form(input: TokenStream) -> TokenStream {
81    let input = syn::parse_macro_input!(input as DeriveInput);
82    embedded::expand(input)
83}
