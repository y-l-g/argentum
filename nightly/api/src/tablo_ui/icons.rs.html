1//! The app's Lucide icons, resolved once from the staged icon set.
2//!
3//! `iconify_icon!` resolves against the **compiling crate's** staged sets
4//! (ADR-0007 stages Lucide in this crate's build script), so the consts live
5//! here and `tablo-core` renders them with `icon` instead of staging a
6//! second copy of the set.
7
8use topcoat::icon::{IconData, iconify::iconify_icon};
9
10/// Sortable column, neither direction active.
11pub const ARROW_UP_DOWN: IconData = iconify_icon!("lucide:arrow-up-down");
12/// Sortable column, sorted ascending.
13pub const ARROW_UP: IconData = iconify_icon!("lucide:arrow-up");
14/// Sortable column, sorted descending.
15pub const ARROW_DOWN: IconData = iconify_icon!("lucide:arrow-down");
16/// Toast success.
17pub const CIRCLE_CHECK: IconData = iconify_icon!("lucide:circle-check");
18/// Toast info.
19pub const INFO: IconData = iconify_icon!("lucide:info");
20/// Toast warning.
21pub const TRIANGLE_ALERT: IconData = iconify_icon!("lucide:triangle-alert");
22/// Toast error.
23pub const OCTAGON_X: IconData = iconify_icon!("lucide:octagon-x");
