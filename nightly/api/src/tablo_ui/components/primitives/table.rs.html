7/// A table inside a horizontally scrollable container.
8///
9/// Pass table sections as children. `attrs` are forwarded to the `<table>`, with extra
10/// classes added to its classes.
11///
12/// ```ignore
13/// view! {
14///     table(
15///         table_header(
16///             table_row(
17///                 table_head("Environment")
18///                 table_head("Status")
19///             )
20///         )
21///         table_body(
22///             table_row(
23///                 table_cell("production")
24///                 table_cell("Live")
25///             )
26///         )
27///     )
28/// }
29/// ```
30#[component]
31pub async fn table(
32    #[default] mut attrs: Attributes,
33    #[default] child: Child<'_>,
34) -> Result<impl View> {
35    Ok(view! {
36        <div class="w-full overflow-x-auto">
37            <table
38                class=(class!(
39                    "w-full caption-bottom border-collapse text-sm",
40                    attrs.remove("class"),
41                ))
42                (attrs)
43            >
44                (child)
45            </table>
46        </div>
47    })
48}
