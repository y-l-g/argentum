1//! xtask — repo tasks (ADR-0007).
2//!
3//! `sync-topcoat-ui` mirrors the `topcoat-ui-registry` sources into
4//! `crates/argentum-ui/src/components/primitives/` **verbatim**: every file is
5//! the registry's byte-for-byte source under a one-line SYNC header that
6//! records the registry version *and* the sha256 content hash of the source
7//! (the same hash scheme topcoat's own registry and `topcoat ui` use). Because
8//! the copy is verbatim, drift — a hand edit, a stale file, a component the
9//! registry gained or dropped — is detectable by [`verify_sync`], which the
10//! `xtask` test suite runs as a guard.
11
12use std::path::{Path, PathBuf};
13
14use topcoat_ui::Registry;

16/// The one-line header prepended to every synced file.
17///
18/// `hash` is the registry source's `sha256:` content hash (see
19/// `topcoat_ui::content_hash`), so a guard can tell a drifted file
20/// from a merely stale header without a sibling clone.
21fn sync_header(version: &str, hash: &str) -> String {
22    format!(
23        "// SYNC: topcoat-ui-registry@{version} {hash} — do not hand-edit. Sync via `cargo xtask sync-topcoat-ui` (ADR-0007).\n"
24    )
25}
26
27/// The header for the generated `mod.rs` (it is not a copy of any source, so
28/// it carries only the registry version).
29fn mod_header(version: &str) -> String {
30    format!(
31        "// SYNC: topcoat-ui-registry@{version} — generated from the registry manifest. Sync via `cargo xtask sync-topcoat-ui` (ADR-0007).\n"
32    )
33}

35/// The destination directory for synced primitives.
36pub fn primitives_dir() -> PathBuf {
37    let manifest_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
38    // xtask is at <repo>/xtask, so repo root is parent of manifest_dir
39    manifest_dir
40        .parent()
41        .unwrap_or(Path::new("."))
42        .join("crates/argentum-ui/src/components/primitives")
43}

45/// The registry Cargo resolved for this workspace, plus its crate version.
46///
47/// Located through `cargo metadata` — the same mechanism `topcoat ui` itself
48/// uses (topcoat-ui/src/manage/workspace.rs) — so the synced sources always
49/// come from the exact `topcoat-ui-registry` the workspace compiles against,
50/// pinned by `Cargo.lock`. The registry directory is read from the data
51/// crate's `[package.metadata.topcoat-ui] registry` declaration.
52fn locate_registry() -> anyhow::Result<(Registry, String)> {
53    // Anchored at xtask's own manifest (GH #175): a bare `cargo metadata`
54    // resolves the caller's CWD, so invoking from a detached workspace
55    // (e.g. benchmarks/argentum, which has no topcoat-ui-registry in its
56    // graph) failed with a misleading "must be a dependency of xtask".
57    let manifest = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("Cargo.toml");
58    let output = std::process::Command::new("cargo")
59        .args(["metadata", "--format-version", "1", "--manifest-path"])
60        .arg(&manifest)
61        .output()
62        .map_err(|error| anyhow::anyhow!("failed to run cargo metadata: {error}"))?;
63    if !output.status.success() {
64        anyhow::bail!(
65            "cargo metadata failed: {}",
66            String::from_utf8_lossy(&output.stderr).trim()
67        );
68    }
69    let metadata: serde_json::Value = serde_json::from_slice(&output.stdout)
70        .map_err(|error| anyhow::anyhow!("could not parse cargo metadata: {error}"))?;
71
72    let package = metadata["packages"]
73        .as_array()
74        .and_then(|packages| {
75            packages
76                .iter()
77                .find(|package| package["name"] == "topcoat-ui-registry")
78        })
79        .ok_or_else(|| {
80            anyhow::anyhow!(
81                "`topcoat-ui-registry` is not in the dependency graph — it must be a \
82                 dependency of xtask (see xtask/Cargo.toml)"
83            )
84        })?;
85
86    let version = package["version"]
87        .as_str()
88        .ok_or_else(|| anyhow::anyhow!("topcoat-ui-registry has no version in cargo metadata"))?
89        .to_string();
90    let manifest_path = package["manifest_path"]
91        .as_str()
92        .ok_or_else(|| anyhow::anyhow!("topcoat-ui-registry has no manifest_path"))?;
93    let relative = package["metadata"]["topcoat-ui"]["registry"]
94        .as_str()
95        .unwrap_or(".");
96    let dir = Path::new(manifest_path)
97        .parent()
98        .unwrap_or(Path::new("."))
99        .join(relative);
100
101    let registry = Registry::load(dir)?;
102    Ok((registry, version))
103}

105/// What to run when a vendored file has drifted from the registry.
106const HINT: &str = "run `cargo xtask sync-topcoat-ui` to restore the verbatim copy";
107
108/// Copy every registry component into `primitives/` **verbatim** under a SYNC
109/// header recording the registry version and the source's sha256, then
110/// regenerate `mod.rs` from the manifest. Never touches `composites/`
111/// (ADR-0007).
112///
113/// No sibling clone required — the registry comes from the same git source
114/// Cargo compiles against.
115///
116/// `prune` deletes vendored files the registry no longer owns (the orphan
117/// guard in `verify_sync` otherwise leaves `verify` red after an upstream
118/// removal with `sync` alone unable to fix it). Without it, orphans are only
119/// reported — pass `--prune` to converge.
120pub fn sync_topcoat_ui(dry_run: bool, prune: bool) -> anyhow::Result<()> {
121    let dst_dir = primitives_dir();
122    std::fs::create_dir_all(&dst_dir)?;
123
124    let (registry, version) = locate_registry()?;
125
126    // `Registry::names()` yields BTreeMap keys — already sorted.
127    let names: Vec<String> = registry.names().map(String::from).collect();
128
129    let mut count = 0;
130    for name in &names {
131        let component = registry
132            .get(name)
133            .ok_or_else(|| anyhow::anyhow!("registry name {name} vanished between load and get"))?;
134        let src = component.read_source()?;
135        let header = sync_header(&version, &topcoat_ui::content_hash(&src));
136        let dst_path = dst_dir.join(component.file_name());
137        if dry_run {
138            println!("would sync {name} -> {}", dst_path.display());
139        } else {
140            std::fs::write(&dst_path, format!("{header}{src}"))?;
141            println!("synced {name}");
142        }
143        count += 1;
144    }
145    if dry_run {
146        println!("dry-run: {count} components would be synced (topcoat-ui-registry@{version})");
147    } else {
148        println!(
149            "done: {count} components synced to {} (topcoat-ui-registry@{version})",
150            dst_dir.display()
151        );
152        println!("note: composites/ was not touched (ADR-0007)");
153    }
154    ensure_primitives_mod(&dst_dir, &version, &names, dry_run)?;
155    if prune {
156        prune_orphans(&dst_dir, &registry, dry_run)?;
157    }
158    Ok(())
159}

161/// Delete vendored files the registry manifest no longer owns (GH #175):
162/// the same expected-set as the `verify_sync` orphan guard (`mod.rs`
163/// included — it is regenerated, never pruned). Dry runs only report.
164fn prune_orphans(dst_dir: &Path, registry: &Registry, dry_run: bool) -> anyhow::Result<()> {
165    use std::collections::HashSet;
166    let mut expected: HashSet<String> = HashSet::new();
167    for name in registry.names() {
168        if let Some(component) = registry.get(name) {
169            expected.insert(component.file_name().to_string());
170        }
171    }
172    expected.insert("mod.rs".to_string());
173    let mut orphans: Vec<PathBuf> = Vec::new();
174    if let Ok(entries) = std::fs::read_dir(dst_dir) {
175        for entry in entries.flatten() {
176            let file_name = entry.file_name().to_string_lossy().to_string();
177            if file_name.starts_with('.') {
178                continue;
179            }
180            if !expected.contains(&file_name) {
181                orphans.push(entry.path());
182            }
183        }
184    }
185    orphans.sort();
186    for orphan in orphans {
187        if dry_run {
188            println!("would delete orphan {}", orphan.display());
189        } else {
190            std::fs::remove_file(&orphan)?;
191            println!("deleted orphan {}", orphan.display());
192        }
193    }
194    Ok(())
195}

197/// Regenerate `primitives/mod.rs` from the registry manifest.
198fn ensure_primitives_mod(
199    dst_dir: &Path,
200    version: &str,
201    names: &[String],
202    dry_run: bool,
203) -> anyhow::Result<()> {
204    let mod_path = dst_dir.join("mod.rs");
205    let mut content = mod_header(version);
206    for name in names {
207        content.push_str(&format!("pub mod {name};\n"));
208    }
209    if dry_run {
210        println!("would write {}", mod_path.display());
211    } else {
212        std::fs::write(&mod_path, content)?;
213        println!("wrote {}", mod_path.display());
214    }
215    Ok(())
216}

218/// Guard: every vendored primitive is still the registry's verbatim source,
219/// every SYNC header records the current version *and* the current content
220/// hash, and `mod.rs` still lists exactly the registry's components.
221///
222/// This is Argentum's counterpart of topcoat's own
223/// `examples/ui/tests/registry_sync.rs`: because the sync is byte-for-byte
224/// (no injected headers *inside* the source, no string patches), a hash
225/// comparison is meaningful and drift cannot hide.
226pub fn verify_sync() -> anyhow::Result<()> {
227    let dst_dir = primitives_dir();
228    let (registry, version) = locate_registry()?;
229
230    let names: Vec<String> = registry.names().map(String::from).collect();
231    let mut failures = Vec::new();
232
233    for name in &names {
234        let component = registry
235            .get(name)
236            .ok_or_else(|| anyhow::anyhow!("registry name {name} vanished between load and get"))?;
237        let src = component.read_source()?;
238        let expected = sync_header(&version, &topcoat_ui::content_hash(&src)) + &src;
239        let dst_path = dst_dir.join(component.file_name());
240        let installed = match std::fs::read_to_string(&dst_path) {
241            Ok(content) => content,
242            Err(error) => {
243                failures.push(format!(
244                    "{} cannot be read: {error}; {HINT}",
245                    dst_path.display()
246                ));
247                continue;
248            }
249        };
250        if installed == expected {
251            continue;
252        }
253        // Distinguish a stale/mismatched header from a hand edit of the body.
254        let header_matches = installed
255            .strip_prefix("// SYNC: topcoat-ui-registry@")
256            .is_some_and(|rest| {
257                rest.split_once(" — do not hand-edit.")
258                    .is_some_and(|(head, _)| {
259                        head.split_once(' ').is_some_and(|(ver, hash)| {
260                            ver == version && hash == topcoat_ui::content_hash(&src)
261                        })
262                    })
263            });
264        if header_matches {
265            failures.push(format!(
266                "{} was hand-edited — it no longer matches the registry source; {HINT}",
267                dst_path.display()
268            ));
269        } else {
270            failures.push(format!(
271                "{} carries a stale SYNC header or drifted content (topcoat-ui-registry@{version}); {HINT}",
272                dst_path.display()
273            ));
274        }
275    }
276
277    // mod.rs must list exactly the registry's components.
278    let mod_path = dst_dir.join("mod.rs");
279    let mut expected = mod_header(&version);
280    for name in &names {
281        expected.push_str(&format!("pub mod {name};\n"));
282    }
283    match std::fs::read_to_string(&mod_path) {
284        Ok(actual) if actual == expected => {}
285        Ok(_) => failures.push(format!(
286            "{} does not match the registry manifest; {HINT}",
287            mod_path.display()
288        )),
289        Err(error) => failures.push(format!(
290            "{} cannot be read: {error}; {HINT}",
291            mod_path.display()
292        )),
293    }
294
295    // Orphan guard (GH #103): an upstream-removed component must not linger as
296    // a stale vendored file that still compiles when referenced. Flag any file
297    // in primitives/ that the registry does not own.
298    {
299        use std::collections::HashSet;
300        let mut expected_files: HashSet<String> = HashSet::new();
301        for name in &names {
302            if let Some(component) = registry.get(name) {
303                expected_files.insert(component.file_name().to_string());
304            }
305        }
306        expected_files.insert("mod.rs".to_string());
307        if let Ok(entries) = std::fs::read_dir(&dst_dir) {
308            let mut orphans: Vec<String> = Vec::new();
309            for entry in entries.flatten() {
310                let file_name = entry.file_name().to_string_lossy().to_string();
311                if file_name.starts_with('.') {
312                    continue;
313                }
314                if !expected_files.contains(&file_name) {
315                    orphans.push(dst_dir.join(&file_name).display().to_string());
316                }
317            }
318            orphans.sort();
319            for orphan in orphans {
320                failures.push(format!(
321                    "{orphan} is not in the registry manifest (orphaned vendored file); delete it or {HINT}"
322                ));
323            }
324        }
325    }
326
327    if failures.is_empty() {
328        println!(
329            "verified: {} primitives match topcoat-ui-registry@{version} verbatim",
330            names.len()
331        );
332        Ok(())
333    } else {
334        anyhow::bail!("registry drift detected:\n{}", failures.join("\n"));
335    }
336}

591/// Whether `needle` appears in `haystack` as a hook, not as a prefix of a
592/// longer name.
593///
594/// A plain substring check misses renames by extension (`data-copy-button` →
595/// `data-copy-button-2` still contains the old string), so an occurrence only
596/// counts when neither neighbor continues the name. Still structural: any
597/// spelling (`[data-x]`, `data-x=""`, `dataset.x`) matches.
598///
599/// The Rust half of the contract is checked against the sources with test
600/// modules and comment lines removed ([`rust_sources`]). Without that, the
601/// check proves nothing about the render sites: when GH #184 retired
602/// `data-bulk-submit` from `bulk.js`, the string survived in `render.rs`'s
603/// assertions and in `panel/list.rs`'s prose, so a registry entry for it stayed
604/// green while the attribute was gone from the markup. Reading the actual
605/// rendering would be stronger still, but that means building a Db, a panel and
606/// a request per hook — the attributes are the cheaper proxy.
607fn contains_hook(haystack: &str, needle: &str) -> bool {
608    if needle.is_empty() {
609        return true;
610    }
611    haystack.match_indices(needle).any(|(i, _)| {
612        let before = haystack[..i].chars().next_back();
613        let after = haystack[i + needle.len()..].chars().next();
614        !before.is_some_and(is_hook_char) && !after.is_some_and(is_hook_char)
615    })
616}

618/// Source text with test modules and comment-only lines dropped.
619///
620/// `#[cfg(test)]` sections are cut out by brace nesting, and any line whose
621/// first non-space characters are `//` (or `//!`, `///`, `/*`, `*`) is dropped:
622/// a hook named in an assertion or a doc comment is not a hook the markup
623/// renders. Only whole-line comments are removed, so trailing `// data-x`
624/// annotations stay in the haystack — over-inclusion here only makes the check
625/// more forgiving, never falsely red.
626fn production_sources(src: &str) -> String {
627    let mut out = String::with_capacity(src.len());
628    let mut rest = src;
629    // Drop `#[cfg(test)]` modules, brace-counted from the attribute.
630    while let Some(at) = rest.find("#[cfg(test)]") {
631        out.push_str(&rest[..at]);
632        let after = &rest[at + "#[cfg(test)]".len()..];
633        let mut depth = 0usize;
634        let mut chars = after.char_indices();
635        let mut end = after.len();
636        let mut seen_open = false;
637        for (i, c) in chars.by_ref() {
638            match c {
639                '{' => {
640                    depth += 1;
641                    seen_open = true;
642                }
643                '}' => {
644                    depth -= 1;
645                    if seen_open && depth == 0 {
646                        end = i + 1;
647                        break;
648                    }
649                }
650                _ => {}
651            }
652        }
653        // No opening brace (a bare `#[cfg(test)]` on one item): drop that one
654        // item's line rather than the rest of the file.
655        rest = if seen_open {
656            &after[end..]
657        } else {
658            after.split_once('\n').map(|(_, tail)| tail).unwrap_or("")
659        };
660    }
661    out.push_str(rest);
662    out.lines()
663        .filter(|line| {
664            let t = line.trim_start();
665            !(t.starts_with("//") || t.starts_with("/*") || t.starts_with('*'))
666        })
667        .collect::<Vec<_>>()
668        .join("\n")
669}

671/// Concatenate the workspace's Rust sources with test modules and comment lines
672/// removed, for the hook contract's Rust half.
673fn rust_sources(root: &Path) -> anyhow::Result<String> {
674    let mut out = String::new();
675    for dir in ["crates/argentum-ui/src", "crates/argentum-core/src"] {
676        let mut files = Vec::new();
677        collect_rs(&root.join(dir), &mut files)?;
678        for path in files {
679            let src = std::fs::read_to_string(&path)?;
680            out.push_str(&production_sources(&src));
681            out.push('\n');
682        }
683    }
684    Ok(out)
685}

687/// Characters that continue a hook/identifier name.
688fn is_hook_char(c: char) -> bool {
689    c.is_ascii_alphanumeric() || c == '-' || c == '_'
690}

692/// What to do when the hook contract breaks.
693const HOOK_HINT: &str = "update the hook list and both sides together (GH #152, ADR-0014)";
694
695/// Collect every `.rs` file under `dir`, recursively.
696fn collect_rs(dir: &Path, out: &mut Vec<PathBuf>) -> anyhow::Result<()> {
697    for entry in std::fs::read_dir(dir)? {
698        let entry = entry?;
699        let path = entry.path();
700        if path.is_dir() {
701            collect_rs(&path, out)?;
702        } else if path.extension().is_some_and(|ext| ext == "rs") {
703            out.push(path);
704        }
705    }
706    Ok(())
707}

709/// Guard: every shell JS asset still exists and stays wired to its `lib.rs`
710/// constant, and every hook in [`ASSET_HOOKS`] still appears in both its JS
711/// asset and the Rust render sources.
712///
713/// This is the hook-contract counterpart of [`verify_sync`]: `asset!` does
714/// not stat its source at compile time, so a deleted/renamed `.js` passes
715/// `cargo test`, and a rename on either side of a string-selector coupling is
716/// otherwise silent. The check is structural on purpose (hook-name presence
717/// with identifier-boundary matching, never classes or pixel markup) so
718/// restyles cannot fail it.
719pub fn verify_asset_hooks() -> anyhow::Result<()> {
720    let manifest_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
721    let root = manifest_dir.parent().unwrap_or(Path::new("."));
722    let assets = assets_dir();
723    let mut failures = Vec::new();
724
725    let lib_rs = std::fs::read_to_string(root.join("crates/argentum-ui/src/lib.rs"))
726        .map_err(|error| anyhow::anyhow!("cannot read argentum-ui/src/lib.rs: {error}"))?;
727
728    // Every asset file exists and stays wired to its constant.
729    let mut sources: std::collections::HashMap<&str, String> = std::collections::HashMap::new();
730    for (file, constant) in ASSET_FILES {
731        let path = assets.join(file);
732        match std::fs::read_to_string(&path) {
733            Ok(src) if !src.trim().is_empty() => {
734                sources.insert(file, src);
735            }
736            Ok(_) => failures.push(format!("{file} is empty; {HOOK_HINT}")),
737            Err(error) => failures.push(format!(
738                "{} cannot be read: {error}; {HOOK_HINT}",
739                path.display()
740            )),
741        }
742        if !contains_hook(&lib_rs, constant) {
743            failures.push(format!(
744                "{constant} is gone from argentum-ui/src/lib.rs, so {file} is no longer wired into the document; {HOOK_HINT}"
745            ));
746        }
747    }
748
749    // Every hook appears in both its JS asset and the Rust sources — the
750    // latter with test modules and comment lines removed, so an assertion or a
751    // doc comment cannot stand in for the markup (see `contains_hook`).
752    let rust_src = match rust_sources(root) {
753        Ok(sources) => sources,
754        Err(error) => {
755            failures.push(format!(
756                "cannot read the Rust sources: {error}; {HOOK_HINT}"
757            ));
758            String::new()
759        }
760    };
761    for hook in ASSET_HOOKS {
762        match sources.get(hook.asset) {
763            Some(src) if contains_hook(src, hook.js) => {}
764            Some(_) => failures.push(format!(
765                "{} no longer contains `{}`; {HOOK_HINT}",
766                hook.asset, hook.js
767            )),
768            None => failures.push(format!(
769                "{} is missing, so its `{}` hook cannot be checked; {HOOK_HINT}",
770                hook.asset, hook.js
771            )),
772        }
773        if !contains_hook(&rust_src, hook.rust) {
774            failures.push(format!(
775                "`{}` (consumed by {}) is gone from the Rust sources; {HOOK_HINT}",
776                hook.rust, hook.asset
777            ));
778        }
779    }
780
781    if failures.is_empty() {
782        println!(
783            "verified: {} assets and {} hooks match the hook contract",
784            ASSET_FILES.len(),
785            ASSET_HOOKS.len()
786        );
787        Ok(())
788    } else {
789        anyhow::bail!("asset hook drift detected:\n{}", failures.join("\n"));
790    }
791}
