1//! The live feed: a Topcoat shard whose WebSocket connection keeps every open
2//! page current.
3//!
4//! The page renders `live_feed`, a shard, so a write re-renders the list
5//! without touching the page around it. The shard's body is a `live!` region
6//! that calls `connected`: during the HTTP render it emits the newest users and
7//! finishes, and the browser then opens a WebSocket and renders the shard
8//! again. On that render the body emits the current rows and waits on the
9//! board's wake channel, re-reading and emitting again after every write. The
10//! connection belongs to the shard, so the rest of the shell stays on HTTP and
11//! no polling script runs.
12//!
13//! The rows come from the request's own database, so one router's feed never
14//! shows another router's writes. The wake channel is a process-global because
15//! `Panel::app_context` carries only the `Db`; an app that owns a context of
16//! its own keeps the sender beside it instead.
17
18use std::sync::LazyLock;

22    Result,
23    context::Cx,
24    router::page,
25    runtime::{connected, shard},
26    view::{View, emit, live, view},
27};

31/// Where the live feed lives.
32pub const LIVE_PATH: &str = "/admin/live";
33
34/// How many of the newest users the feed shows.
35const FEED_ROWS: usize = 20;
36
37/// The process-wide wake channel behind every open feed.
38static BOARD: LazyLock<Board> = LazyLock::new(Board::default);
39
40/// Wakes every connected feed after a committed write.
41///
42/// The board carries no event: a wake only tells each feed to re-read the rows
43/// it shows, so the rows stay per-request and the channel stays small.
44struct Board {
45    changed: tokio::sync::broadcast::Sender<()>,
46}
47
48impl Default for Board {
49    fn default() -> Self {
50        Self {
51            changed: tokio::sync::broadcast::channel(16).0,
52        }
53    }
54}
55
56impl Board {
57    /// Wake every connected feed.
58    fn notify(&self) {
59        // No connection listening is not a failure: the next HTTP render reads
60        // the rows.
61        let _ = self.changed.send(());
62    }
63
64    /// Subscribe before reading the rows, so a write between the read and the
65    /// wait is not missed.
66    fn subscribe(&self) -> tokio::sync::broadcast::Receiver<()> {
67        self.changed.subscribe()
68    }
69}
70
71/// Wake every connected feed the process holds.
72pub(crate) fn notify() {
73    BOARD.notify();
74}
75
76/// Subscribe to the process-wide board.
77fn subscribe() -> tokio::sync::broadcast::Receiver<()> {
78    BOARD.subscribe()
79}

81/// `GET /admin/live` — the panel page holding the live feed.
82///
83/// The attribute spells the path [`LIVE_PATH`] names; the integration test
84/// requests `LIVE_PATH`, so a drift between the two fails the test.
85#[page("/admin/live")]
86async fn live_page(cx: &Cx) -> Result<impl View> {
87    Ok(view! {
88        cx =>
89        tablo_ui::page(
90            tablo_ui::page_header(
91                tablo_ui::page_title("Live activity")
92                tablo_ui::page_description(
93                    "The newest users, re-read after every committed write and pushed over a WebSocket."
94                )
95            )
96            tablo_ui::page_content(
97                tablo_ui::card(
98                    tablo_ui::card_header(tablo_ui::card_title("Newest users"))
99                    tablo_ui::card_content(live_feed())
100                )
101            )
102        )
103    })
104}

106/// The feed itself: a shard whose live region streams the newest users to every
107/// open page.
108///
109/// The first emission is what the HTTP render sends; `connected` then returns
110/// false and asks the browser for a connection. The connected render runs the
111/// body from the top, so it emits the current rows again before it waits — a
112/// write between the two renders cannot be missed.
113#[shard]
114async fn live_feed(cx: &Cx) -> Result<impl View> {
115    // Runtime endpoints bypass page guards (Topcoat's shard contract), so the
116    // shard restates the panel gate: a request without a permitted user must
117    // not read the rows, over HTTP or over the connection.
118    if tablo_core::auth::enforced(cx) {
119        tablo_core::auth::require_authenticated(cx)?;
120    }
121    Ok(live! {
122        let mut changed = subscribe();
123        loop {
124            let users = newest_users(cx).await?;
125            // The emptiness check borrows `users`, and the `else` arm then moves
126            // it into the view; the bool keeps the two uses apart.
127            let empty = users.is_empty();
128            let token = emit! {
129                if empty {
130                    <p class="text-sm text-muted-foreground" data-live-feed="">
131                        "No users yet."
132                    </p>
133                } else {
134                    <ul class="flex flex-col gap-2" data-live-feed="">
135                        for user in users {
136                            <li class="text-sm">(user.name)</li>
137                        }
138                    </ul>
139                }
140            }?;
141            if !connected(cx) {
142                break Ok(token);
143            }
144            match changed.recv().await {
145                // Re-emit immediately: the next pass re-reads the rows.
146                Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {}
147                // The board lives for the process, so this arm is unreachable;
148                // ending the region is the safe answer if that changes.
149                Err(tokio::sync::broadcast::error::RecvError::Closed) => break Ok(token),
150                Ok(()) => {}
151            }
152        }
153    })
154}

156/// The newest users the panel serves, newest first.
157async fn newest_users(cx: &Cx) -> Result<Vec<User>> {
158    let mut db = db(cx);
159    let users = UserResource::query(cx)
160        .order_by(User::fields().created_at().desc())
161        .limit(FEED_ROWS)
162        .exec(&mut db)
163        .await?;
164    Ok(users)
165}

167#[cfg(test)]
168mod tests {
169    use std::time::Duration;
170
171    use tablo_core::Committed;
172
173    use super::*;
174
175    /// A wake reaches a subscriber: a feed waiting on a board resumes when the
176    /// board is notified.
177    #[tokio::test]
178    async fn notify_wakes_a_subscriber() {
179        let board = Board::default();
180        let mut changed = board.subscribe();
181        board.notify();
182
183        tokio::time::timeout(Duration::from_secs(1), changed.recv())
184            .await
185            .expect("notify must wake a subscriber")
186            .expect("the sender outlives the receiver");
187    }
188
189    /// The resource hook wakes the process-wide board, so a committed write
190    /// re-runs every connected feed.
191    #[tokio::test]
192    async fn a_committed_write_wakes_the_feed() {
193        let mut changed = subscribe();
194        let user = User {
195            id: uuid::Uuid::new_v4(),
196            name: "Ada".to_string(),
197            email: "ada@example.com".to_string(),
198            role: "admin".to_string(),
199            active: true,
200            created_at: jiff::Timestamp::now(),
201        };
202        UserResource::after_commit(&Cx::default(), Committed::created(user))
203            .await
204            .expect("the hook notifies the board");
205
206        tokio::time::timeout(Duration::from_secs(1), changed.recv())
207            .await
208            .expect("a committed write must wake the feed")
209            .expect("the sender outlives the receiver");
210    }
