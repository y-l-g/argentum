60/// A post's lifecycle — an embedded enum whose **timestamps are shared**
61/// (GH #185).
62///
63/// Every variant declares a timestamp under the same `#[shared(timestamp)]`
64/// identifier, so the three coalesce into one `publication_timestamp` column
65/// instead of one column per variant. The rest of each variant is its own
66/// nullable column (`publication_scheduled_for`, `publication_canonical_url`,
67/// `publication_reason`).
68///
69/// The timestamps are `String` rather than `jiff::Timestamp` because a bound
70/// field is a `String` lens: the Schema's text fields accept `Path<M, String>`,
71/// and a typed leaf (a timestamp, an integer) cannot bind as text yet. The
72/// shared column's *coalescing* is what this demonstrates.
73#[derive(Debug, Clone, PartialEq, toasty::Embed, argentum_core::EmbeddedForm)]
74pub enum Publication {
75    #[column(variant = 1)]
76    Scheduled {
77        /// The shared column's one control renders from the first variant that
78        /// declares it, so its label is written there (GH #191).
79        #[shared(timestamp)]
80        #[form(label = "Publication timestamp")]
81        scheduled_at: String,
82        scheduled_for: String,
83    },
84    #[column(variant = 2)]
85    Published {
86        #[shared(timestamp)]
87        published_at: String,
88        #[form(label = "Canonical URL")]
89        canonical_url: String,
90    },
91    #[column(variant = 3)]
92    Archived {
93        #[shared(timestamp)]
94        archived_at: String,
95        #[form(label = "Archive reason")]
96        reason: String,
97    },
98}

414/// The Argon2id PHC hash of a demo password, computed once per process.
415///
416/// Argon2id at the shipped parameters costs ~0.4s in a debug build *by design*,
417/// and the integration suite seats a fresh database for almost every test —
418/// so the same two demo passwords were being hashed ~260 times per run, which
419/// was the suite's single largest setup cost (GH #218). The hash is a pure
420/// function of the password, so compute it once and hand the same PHC string to
421/// every caller.
422///
423/// Verification is untouched: a login still runs a real Argon2id verify against
424/// this string, at the shipped parameters, and the login tests still exercise
425/// that. Only the *hashing* is memoised, and only in this demo seeder.
426fn memoized_password_hash(password: &str) -> String {
427    use std::collections::HashMap;
428    use std::sync::{Mutex, OnceLock};
429
430    static HASHES: OnceLock<Mutex<HashMap<String, String>>> = OnceLock::new();
431    HASHES
432        .get_or_init(|| Mutex::new(HashMap::new()))
433        .lock()
434        .expect("the demo hash cache is never poisoned")
435        .entry(password.to_string())
436        .or_insert_with(|| hash_password(password).expect("hash a demo password"))
437        .clone()
438}

650                .checked_add(jiff::Span::new().hours(index as i64 * 6))
651                .expect("a representable timestamp")
652                .timestamp();
653            let author_id = match index % 4 {
654                0 => ada_author.id,
655                1 => alan_author.id,
656                2 => june_writer.id,
657                _ => rosa_editor.id,
658            };
659            let (seo, publication, media, post_stats) = filler_embedded(index);
660            toasty::create!(Post {
661                id: uuid::Uuid::from_u128(FILLER_ID_BASE + index as u128),
662                tenant_id: tenant,
663                title: *title,
664                body: "Backlog draft kept for pagination coverage.",
665                status: "draft".to_string(),
666                featured: false,
667                created_at: created_at,
668                image_path: "draft-cover.jpg".to_string(),
669                tags: "backlog,draft".to_string(),
670                seo: seo,
671                publication: publication,
672                media: media,
673                post_stats: post_stats,
674                author_id: author_id,
675            })
676            .exec(db)
677            .await?;
678        }
679        let first_post = Post::filter(Post::fields().title().eq("Hello Toasty".to_string()))
680            .first()
681            .exec(db)
682            .await?
683            .expect("post");
684        for body in [
685            "Clear write-up — the include strategy finally clicked.",
686            "Tried this on our staging data, pagination stayed stable.",
687            "Small nit: the CSV export section deserves its own post.",
688        ] {
689            toasty::create!(Comment {
690                body: body,
691                post_id: first_post.id,
692            })
693            .exec(db)
694            .await?;
695        }
696    }
697    Ok(())
