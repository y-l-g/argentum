12/// User shown in the admin list — the realistic spec model (US16, GH #13):
13/// role/active/created_at plus `#[index]` on the searchable `name` column.
14/// `email` keeps only `#[unique]` — a unique constraint already implies an
15/// index, and stacking `#[index]` on top would double it.
16#[derive(Debug, Clone, toasty::Model)]
17pub struct User {
18    #[key]
19    #[auto]
20    pub id: uuid::Uuid,
21    #[index]
22    pub name: String,
23    #[unique]
24    pub email: String,
25    /// "admin" or "member" — the form renders them as a static-options Select
26    /// (GH #13).
27    pub role: String,
28    pub active: bool,
29    pub created_at: Timestamp,
30}

33// Scoped, not global (GH #88): the form's unique probe runs through the
34// tenant-scoped query (`scoped_query`, GH #223), so a *global* unique index
35// on `email` would be rejected by the database for an email another tenant
36// already owns — after the probe passed — and surface as a 500. Constraining
37// `(tenant_id, email)` makes the constraint say what the probe enforces, so two
38// tenants may share an email.
39#[unique(tenant_id, email)]
40pub struct Author {
41    #[key]
42    #[auto]
43    pub id: uuid::Uuid,
44    #[index]
45    pub tenant_id: uuid::Uuid,
46    /// Display name.
47    pub name: String,
48    pub email: String,
49    #[has_many]
50    pub posts: Deferred<Vec<Post>>,
51}

66/// A post's lifecycle — an embedded enum whose **timestamps are shared**
67/// (GH #185).
68///
69/// Every variant declares a timestamp under the same `#[shared(timestamp)]`
70/// identifier, so the three coalesce into one `publication_timestamp` column
71/// instead of one column per variant. The rest of each variant is its own
72/// nullable column (`publication_scheduled_for`, `publication_canonical_url`,
73/// `publication_reason`).
74///
75/// The timestamps are `String` rather than `jiff::Timestamp` because a bound
76/// field is a `String` lens: the Schema's text fields accept `Path<M, String>`,
77/// and a typed leaf (a timestamp, an integer) cannot bind as text yet. The
78/// shared column's *coalescing* is what this demonstrates.
79#[derive(Debug, Clone, PartialEq, toasty::Embed, argentum_core::EmbeddedForm)]
80pub enum Publication {
81    #[column(variant = 1)]
82    Scheduled {
83        /// The shared column's one control renders from the first variant that
84        /// declares it, so its label is written there (GH #191).
85        #[shared(timestamp)]
86        #[form(label = "Publication timestamp")]
87        scheduled_at: String,
88        scheduled_for: String,
89    },
90    #[column(variant = 2)]
91    Published {
92        #[shared(timestamp)]
93        published_at: String,
94        #[form(label = "Canonical URL")]
95        canonical_url: String,
96    },
97    #[column(variant = 3)]
98    Archived {
99        #[shared(timestamp)]
100        archived_at: String,
101        #[form(label = "Archive reason")]
102        reason: String,
103    },
104}

125#[derive(Debug, Clone, toasty::Embed, argentum_core::EmbeddedForm)]
126pub enum Media {
127    #[column(variant = 1)]
128    Image {
129        #[form(label = "Image URL")]
130        url: String,
131        #[form(label = "Image alt")]
132        alt: String,
133    },
134    #[column(variant = 2)]
135    Video {
136        // Distinct from `Image::url` on purpose: two variant fields mapping to
137        // one column is a schema error unless they declare `#[shared(..)]` —
138        // which is the right answer only when they mean the same thing.
139        #[form(label = "Video URL")]
140        video_url: String,
141        poster: Poster,
142    },
143}
