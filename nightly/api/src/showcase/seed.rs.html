116/// A deterministic id for the `index`th seeded post (GH #184).

210/// Create an active admin (or another app user) with an Argon2id-hashed
211/// password. Used by the showcase seed and the tenancy test fixtures.
212pub async fn create_admin(
213    db: &mut Db,
214    email: &str,
215    display_name: &str,
216    password: &str,
217    tenant_id: Option<uuid::Uuid>,
218) -> toasty::Result<AdminUser> {
219    toasty::create!(AdminUser {
220        email: email.to_string(),
221        password_hash: memoized_password_hash(password),
222        display_name: display_name.to_string(),
223        active: true,
224        tenant_id,
225        created_at: Timestamp::now(),
226    })
227    .exec(db)
228    .await
229}

231/// The Argon2id PHC hash of a demo password, computed once per process.
232///
233/// Argon2id at the shipped parameters costs ~0.4s in a debug build *by design*,
234/// and the integration suite seats a fresh database for almost every test —
235/// so the same two demo passwords were being hashed ~260 times per run, which
236/// was the suite's single largest setup cost (GH #218). The hash is a pure
237/// function of the password, so compute it once and hand the same PHC string to
238/// every caller.
239///
240/// Verification is untouched: a login still runs a real Argon2id verify against
241/// this string, at the shipped parameters, and the login tests still exercise
242/// that. Only the *hashing* is memoised, and only in this demo seeder.
243fn memoized_password_hash(password: &str) -> String {
244    use std::collections::HashMap;
245    use std::sync::{Mutex, OnceLock};
246
247    static HASHES: OnceLock<Mutex<HashMap<String, String>>> = OnceLock::new();
248    HASHES
249        .get_or_init(|| Mutex::new(HashMap::new()))
250        .lock()
251        .expect("the demo hash cache is never poisoned")
252        .entry(password.to_string())
253        .or_insert_with(|| hash_password(password).expect("hash a demo password"))
254        .clone()
255}
256/// The embedded shapes a filler/backlog row carries (GH #185).

259/// nested literals sixty times. The narrative rows below spell theirs out, so

261fn filler_embedded(index: usize) -> (Seo, Publication, Media, PostStats) {
262    (

264            title: String::new(),
265            description: String::new(),
266        },
267        Publication::Scheduled {
268            scheduled_at: String::new(),
269            scheduled_for: String::new(),

272            url: format!("backlog-{index:02}.jpg"),
273            alt: String::new(),
274        },
275        PostStats {
276            word_count: 0,
277            read_minutes: 0,
278        },
279    )
280}
