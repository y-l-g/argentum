115/// A deterministic id for the `index`th seeded post.

217/// Create an active admin (or another app user) with an Argon2id-hashed
218/// password. Used by the showcase seed and the tenancy test fixtures.
219pub async fn create_admin(
220    db: &mut Db,
221    email: &str,
222    display_name: &str,
223    password: &str,
224    tenant_id: Option<uuid::Uuid>,
225) -> toasty::Result<AdminUser> {
226    toasty::create!(AdminUser {
227        email: email.to_string(),
228        password_hash: memoized_password_hash(password),
229        display_name: display_name.to_string(),
230        active: true,
231        tenant_id,
232        created_at: Timestamp::now(),
233    })
234    .exec(db)
235    .await
236}

238/// The Argon2id PHC hash of a demo password, computed once per process.
239///
240/// Argon2id at the shipped parameters costs ~0.4s in a debug build *by design*,
241/// and the integration suite seats a fresh database for almost every test —
242/// so the same two demo passwords were being hashed ~260 times per run, which
243/// was the suite's single largest setup cost. The hash is a pure
244/// function of the password, so compute it once and hand the same PHC string to
245/// every caller.
246///
247/// Verification is untouched: a login still runs a real Argon2id verify against
248/// this string, at the shipped parameters, and the login tests still exercise
249/// that. Only the *hashing* is memoised, and only in this demo seeder.
250fn memoized_password_hash(password: &str) -> String {
251    use std::{
252        collections::HashMap,
253        sync::{Mutex, OnceLock},
254    };
255
256    static HASHES: OnceLock<Mutex<HashMap<String, String>>> = OnceLock::new();
257    HASHES
258        .get_or_init(|| Mutex::new(HashMap::new()))
259        .lock()
260        .expect("the demo hash cache is never poisoned")
261        .entry(password.to_string())
262        .or_insert_with(|| hash_password(password).expect("hash a demo password"))
263        .clone()
264}
265/// The embedded shapes a filler/backlog row carries.

271    (

288    )
